## 1. 背景与目标

### 1.1 背景

本项目技能系统已实现 AI 辅助创建和炼化优化，但缺少**自动从用户经验中学习**的能力。用户每天重复大量操作（截图→分析→保存、搜索→复制→粘贴等），这些重复模式本可以自动识别并转化为技能。

硬件约束：AMD 3800 CPU + 30GB 内存，无 GPU。所有分析必须轻量、本地完成。

### 1.2 目标

1. **行为记录**：轻量记录用户操作日志，不记录敏感内容
2. **模式检测**：自动发现重复的操作序列，CPU 零压力
3. **技能建议**：检测到模式后自动生成技能建议，用户确认后创建
4. **闭环学习**：用户反馈（接受/拒绝/修改）反哺检测参数，越用越准

### 1.3 与现有模块的关系

```
意图识别系统（00-意图识别系统设计.md）
    │ 输出：用户当前意图
    ▼
行为分析系统（本文档）        ◀── 核心新增
    │ 输出：重复操作模式
    ▼
技能系统（modules/skill/）    ◀── 已有，增强
    │ 输出：自动化技能
    ▼
工作流引擎（SkillChain.ts）   ◀── 已有，联动
    │ 输出：多技能编排执行
```

## 4. 行为记录层（ActionTracker）

### 4.1 设计原则

| 原则 | 说明 |
|------|------|
| **零侵入** | 通过模块管理器拦截，不修改各模块代码 |
| **异步写入** | 行为记录在后台线程，不阻塞主流程 |
| **自动脱敏** | 参数在记录前自动处理，源头就不存敏感数据 |
| **环形缓冲** | 超过保留天数的记录自动清理 |

### 4.2 拦截机制

```typescript
/**
 * 在 ModuleManager 的模块调用链中插入追踪点
 * 伪代码示意
 */
class ActionTracker {
  /**
   * 模块调用前：记录开始时间
   */
  beforeCall(module: string, action: string, params: unknown): string {
    const callId = generateId()
    this.pendingCalls.set(callId, {
      module,
      action,
      params: this.sanitizeParams(params),
      startTime: Date.now()
    })
    return callId
  }

  /**
   * 模块调用后：完成记录并写入存储
   */
  afterCall(callId: string, success: boolean): void {
    const pending = this.pendingCalls.get(callId)
    if (!pending) return
    this.pendingCalls.delete(callId)

    const record: ActionRecord = {
      id: callId,
      timestamp: pending.startTime,
      module: pending.module,
      action: pending.action,
      paramSummary: pending.params,
      success,
      sessionId: this.currentSessionId,
      durationMs: Date.now() - pending.startTime
    }

    // 异步写入，不阻塞
    this.store.enqueueWrite(record)
  }

  /**
   * 参数脱敏
   */
  private sanitizeParams(params: unknown): Record<string, string> {
    if (!params || typeof params !== 'object') return {}
    const result: Record<string, string> = {}
    for (const [key, value] of Object.entries(params as Record<string, unknown>)) {
      result[key] = this.sanitizeValue(key, value)
    }
    return result
  }

  private sanitizeValue(key: string, value: unknown): string {
    // 敏感字段直接丢弃
    const sensitiveKeys = ['password', 'secret', 'token', 'apiKey', 'key', 'credential']
    if (sensitiveKeys.some(k => key.toLowerCase().includes(k))) return '<redacted>'

    if (typeof value === 'string') {
      // 文件路径：保留扩展名
      if (value.includes('/') || value.includes('\\')) {
        const ext = value.split('.').pop()
        return ext ? `*.${ext}` : '<path>'
      }
      // URL：保留域名
      if (value.startsWith('http')) {
        try { return new URL(value).origin + '/*' } catch { return '<url>' }
      }
      // 普通字符串：保留长度
      return `<string:${value.length}>`
    }
    if (typeof value === 'number') return String(value)
    if (typeof value === 'boolean') return String(value)
    if (Array.isArray(value)) return `<array:${value.length}>`
    return `<${typeof value}>`
  }
}
```

### 4.3 会话管理

```typescript
/**
 * 会话 ID 生成规则
 * - 应用启动 → 新会话
 * - 用户空闲超过 30 分钟 → 下次操作开始新会话
 * - 手动重置 → 新会话
 */
function getSessionId(): string {
  // 格式：yyyyMMdd-HHmmss-随机4位
  const now = new Date()
  const date = now.toISOString().slice(0, 10).replace(/-/g, '')
  const time = now.toTimeString().slice(0, 8).replace(/:/g, '')
  const rand = Math.random().toString(36).slice(2, 6)
  return `${date}-${time}-${rand}`
}
```

### 5.2 异步写入队列

```typescript
/**
 * 写入队列
 * 批量异步写入，避免高频操作影响性能
 */
class WriteQueue {
  private queue: ActionRecord[] = []
  private flushTimer: ReturnType<typeof setInterval> | null = null
  private readonly batchSize = 50
  private readonly flushIntervalMs = 5000  // 5 秒刷一次

  enqueue(record: ActionRecord): void {
    this.queue.push(record)
    if (this.queue.length >= this.batchSize) {
      this.flush()
    }
  }

  private async flush(): Promise<void> {
    if (this.queue.length === 0) return
    const batch = this.queue.splice(0, this.batchSize)
    // 批量 INSERT
    await this.db.batchInsert('action_log', batch)
  }

  start(): void {
    this.flushTimer = setInterval(() => this.flush(), this.flushIntervalMs)
  }

  stop(): void {
    if (this.flushTimer) clearInterval(this.flushTimer)
    this.flush()  // 最后刷一次
  }
}
```

## 6. 模式检测层（PatternDetector）

### 6.1 检测算法：滑动窗口 + 频繁序列

```typescript
/**
 * 检测频繁操作序列
 *
 * 时间复杂度：O(n * w)，n = 日志条数，w = 最大窗口长度
 * 空间复杂度：O(n) 用于存储候选模式
 *
 * @param actions 按时间排序的行为记录
 * @param options 检测参数
 * @returns 检测到的模式列表
 */
function detectPatterns(
  actions: ActionRecord[],
  options: DetectionOptions = {}
): DetectedPattern[] {
  const {
    minSupport = 3,        // 最少出现次数
    minWindowLen = 2,      // 最短模式长度
    maxWindowLen = 5,      // 最长模式长度
    maxTimeGapMs = 5 * 60 * 1000  // 同一序列内最大时间间隔 5 分钟
  } = options

  // 1. 符号化：把操作转为 "模块:操作" 符号
  const symbols: Array<{ symbol: string; timestamp: number; index: number }> =
    actions.map((a, i) => ({
      symbol: `${a.module}:${a.action}`,
      timestamp: a.timestamp,
      index: i
    }))

  // 2. 按时间间隔切分会话段
  //    如果两次操作间隔超过 maxTimeGapMs，认为是不同操作序列
  const segments = splitByTimeGap(symbols, maxTimeGapMs)

  // 3. 在每个段内提取子序列并计数
  const candidateCounts = new Map<string, {
    steps: string[]
    count: number
    firstSeen: number
    lastSeen: number
    intervals: number[]
  }>()

  for (const segment of segments) {
    // 用 Set 去重（同一段内同一模式只计一次）
    const seenInSegment = new Set<string>()

    for (let len = minWindowLen; len <= maxWindowLen; len++) {
      for (let i = 0; i <= segment.length - len; i++) {
        const window = segment.slice(i, i + len)
        const key = window.map(w => w.symbol).join('→')

        if (seenInSegment.has(key)) continue
        seenInSegment.add(key)

        const existing = candidateCounts.get(key)
        if (existing) {
          existing.count++
          existing.lastSeen = window[window.length - 1].timestamp
          if (len >= 2) {
            const interval = window[len - 1].timestamp - window[0].timestamp
            existing.intervals.push(interval)
          }
        } else {
          candidateCounts.set(key, {
            steps: window.map(w => w.symbol),
            count: 1,
            firstSeen: window[0].timestamp,
            lastSeen: window[len - 1].timestamp,
            intervals: len >= 2 ? [window[len - 1].timestamp - window[0].timestamp] : []
          })
        }
      }
    }
  }

  // 4. 过滤：保留出现次数 >= minSupport 的
  const frequent = Array.from(candidateCounts.entries())
    .filter(([_, data]) => data.count >= minSupport)

  // 5. 去重：移除被更长模式包含的短模式
  const patterns = removeSubPatterns(frequent)

  // 6. 计算置信度和平均间隔
  return patterns.map(([key, data]) => ({
    id: generateId(),
    steps: data.steps.map(s => parseSymbol(s)),
    supportCount: data.count,
    firstSeen: data.firstSeen,
    lastSeen: data.lastSeen,
    avgIntervalMs: data.intervals.length > 0
      ? data.intervals.reduce((a, b) => a + b, 0) / data.intervals.length
      : 0,
    confidence: calculateConfidence(data.count, data.steps.length),
    status: 'pending' as const
  }))
}
```

### 6.2 子模式去重

```typescript
/**
 * 移除被更长模式包含的短模式
 * 例如：A→B→C 出现 3 次，A→B 出现 5 次
 *       保留 A→B→C（更完整），丢弃 A→B
 */
function removeSubPatterns(
  candidates: Array<[string, CandidateData]>
): Array<[string, CandidateData]> {
  // 按长度降序排列
  candidates.sort((a, b) => b[1].steps.length - a[1].steps.length)

  const kept: Array<[string, CandidateData]> = []
  const dominatedKeys = new Set<string>()

  for (const candidate of candidates) {
    const [key, data] = candidate
    if (dominatedKeys.has(key)) continue

    kept.push(candidate)

    // 标记所有被此模式包含的子模式
    for (const [otherKey, otherData] of candidates) {
      if (otherKey === key) continue
      if (otherData.steps.length >= data.steps.length) continue
      if (isSubSequence(otherData.steps, data.steps)) {
        dominatedKeys.add(otherKey)
      }
    }
  }

  return kept
}

/** 判断 A 是否是 B 的子序列 */
function isSubSequence(a: string[], b: string[]): boolean {
  let i = 0
  for (let j = 0; j < b.length && i < a.length; j++) {
    if (a[i] === b[j]) i++
  }
  return i === a.length
}
```

### 6.3 置信度计算

```typescript
/**
 * 计算模式置信度
 * 考虑因素：出现次数、模式长度、时间规律性
 */
function calculateConfidence(
  supportCount: number,
  patternLength: number,
  intervals: number[]
): number {
  // 基础分：出现次数的对数（避免 100 次和 101 次差距过大）
  const baseScore = Math.min(Math.log2(supportCount) / Math.log2(20), 1) // 20次为满分

  // 长度加分：越长的模式越有价值
  const lengthBonus = Math.min((patternLength - 2) * 0.1, 0.3)

  // 时间规律性加分：间隔越稳定，越像是有意为之
  let regularityBonus = 0
  if (intervals.length >= 2) {
    const mean = intervals.reduce((a, b) => a + b, 0) / intervals.length
    const variance = intervals.reduce((sum, v) => sum + (v - mean) ** 2, 0) / intervals.length
    const cv = Math.sqrt(variance) / (mean || 1)  // 变异系数
    // CV 越小越规律：< 0.3 加 0.15，< 0.5 加 0.08
    regularityBonus = cv < 0.3 ? 0.15 : cv < 0.5 ? 0.08 : 0
  }

  return Math.min(baseScore + lengthBonus + regularityBonus, 1)
}
```

### 6.4 模糊匹配：相似序列合并

用户操作可能有细微变化（中间多一步或少一步），需要模糊匹配：

```typescript
/**
 * 模糊序列相似度
 * 基于最长公共子序列（LCS）计算相似度
 *
 * 示例：
 *   A→B→C→D 与 A→B→E→D 的相似度 = 3/4 = 0.75（A、B、D 匹配）
 *   A→B→C 与 A→B→C→D 的相似度 = 3/4 = 0.75（完全包含）
 */
function sequenceSimilarity(a: string[], b: string[]): number {
  const m = a.length, n = b.length
  // LCS 动态规划
  const dp: number[][] = Array.from({ length: m + 1 }, () => Array(n + 1).fill(0))
  for (let i = 1; i <= m; i++) {
    for (let j = 1; j <= n; j++) {
      dp[i][j] = a[i - 1] === b[j - 1]
        ? dp[i - 1][j - 1] + 1
        : Math.max(dp[i - 1][j], dp[i][j - 1])
    }
  }
  const lcsLen = dp[m][n]
  return lcsLen / Math.max(m, n)
}

/**
 * 合并相似模式
 * 如果两个模式相似度 > 阈值，保留出现次数更高的那个
 */
function mergeSimilarPatterns(
  patterns: DetectedPattern[],
  similarityThreshold = 0.8
): DetectedPattern[] {
  const merged: DetectedPattern[] = []
  const used = new Set<number>()

  // 按出现次数降序
  const sorted = [...patterns].sort((a, b) => b.supportCount - a.supportCount)

  for (let i = 0; i < sorted.length; i++) {
    if (used.has(i)) continue
    merged.push(sorted[i])
    used.add(i)

    // 检查后续模式是否与此相似
    for (let j = i + 1; j < sorted.length; j++) {
      if (used.has(j)) continue
      const sim = sequenceSimilarity(
        sorted[i].steps.map(s => s.symbol),
        sorted[j].steps.map(s => s.symbol)
      )
      if (sim >= similarityThreshold) {
        // 合并：累加计数，保留更长的步骤
        sorted[i].supportCount += sorted[j].supportCount
        if (sorted[j].steps.length > sorted[i].steps.length) {
          sorted[i].steps = sorted[j].steps
        }
        used.add(j)
      }
    }
  }

  return merged
}
```

### 6.5 通知冷却机制

避免同一个模式反复提醒用户：

```typescript
/**
 * 通知冷却管理
 * - 同一模式被忽略后，7 天内不再提醒
 * - 同一模式被拒绝后，永久不再提醒
 * - 全局冷却：两次通知间隔至少 30 分钟
 */
class NotificationCooldown {
  private globalLastNotify = 0
  private readonly globalCooldownMs = 30 * 60 * 1000  // 30 分钟
  private readonly dismissCooldownMs = 7 * 24 * 60 * 60 * 1000  // 7 天

  /** 检查是否可以发送通知 */
  canNotify(pattern: DetectedPattern): boolean {
    // 全局冷却
    if (Date.now() - this.globalLastNotify < this.globalCooldownMs) return false

    // 已拒绝 → 永不提醒
    if (pattern.status === 'rejected') return false

    // 已忽略 → 7 天后可以再提醒
    if (pattern.status === 'dismissed') {
      if (Date.now() - pattern.lastSeen < this.dismissCooldownMs) return false
    }

    return true
  }

  markNotified(): void {
    this.globalLastNotify = Date.now()
  }
}
```

### 6.6 跨会话模式检测

某些模式可能跨会话出现（比如每次打开应用都先搜索再截图）：

```typescript
/**
 * 跨会话模式检测策略
 *
 * 1. 会话内模式：同一 sessionId 内的连续操作（主要检测对象）
 * 2. 会话首尾模式：会话最后一步 → 下个会话第一步（跨会话衔接）
 * 3. 会话开头模式：每个会话的前 N 步（习惯性操作）
 */
function detectCrossSessionPatterns(actions: ActionRecord[]): DetectedPattern[] {
  // 按 sessionId 分组
  const sessions = groupBySession(actions)
  const sessionStarts: string[] = []
  const sessionEnds: string[] = []

  for (const [, sessionActions] of sessions) {
    // 取每个会话的前 3 步
    const firstSteps = sessionActions.slice(0, 3).map(a => `${a.module}:${a.action}`)
    sessionStarts.push(...firstSteps)

    // 取每个会话的最后 1 步
    const lastStep = sessionActions[sessionActions.length - 1]
    if (lastStep) sessionEnds.push(`${lastStep.module}:${lastStep.action}`)
  }

  // 检测会话开头的高频操作
  const startCounts = countFrequent(sessionStarts, 1)
  // 出现 5 次以上的会话首步 → 候选模式
  return startCounts
    .filter(([_, count]) => count >= 5)
    .map(([action, count]) => ({
      steps: [parseSymbol(action)],
      supportCount: count,
      type: 'session_start'  // 标记为会话开头习惯
    }))
}
```

### 6.7 检测调度

```typescript
/**
 * 定时检测调度器
 * 默认每小时运行一次，扫描最近 7 天的行为日志
 */
class PatternScheduler {
  private timer: ReturnType<typeof setInterval> | null = null
  private readonly intervalMs = 60 * 60 * 1000  // 1 小时
  private readonly lookbackDays = 7

  start(): void {
    // 启动后 5 分钟执行第一次（避免启动时太忙）
    setTimeout(() => this.runDetection(), 5 * 60 * 1000)
    this.timer = setInterval(() => this.runDetection(), this.intervalMs)
  }

  stop(): void {
    if (this.timer) clearInterval(this.timer)
  }

  private async runDetection(): Promise<void> {
    const lookbackMs = this.lookbackDays * 24 * 60 * 60 * 1000
    const since = Date.now() - lookbackMs

    // 1. 加载行为日志
    const actions = await this.store.getActionsSince(since)
    if (actions.length < 10) return  // 数据太少，跳过

    // 2. 检测模式
    const patterns = detectPatterns(actions, {
      minSupport: 3,
      minWindowLen: 2,
      maxWindowLen: 5
    })

    // 3. 与已有模式合并
    for (const pattern of patterns) {
      const existing = await this.store.findSimilarPattern(pattern.steps)
      if (existing) {
        // 已有模式：更新计数和时间
        await this.store.updatePattern(existing.id, {
          supportCount: existing.supportCount + 1,
          lastSeen: pattern.lastSeen
        })
      } else {
        // 新模式：保存并通知用户
        await this.store.savePattern(pattern)
        await this.notifier.notifyNewPattern(pattern)
      }
    }
  }
}
```

### 7.1 模式到技能的转换

```typescript
/**
 * 把检测到的模式转化为技能建议
 */
async function generateSuggestion(pattern: DetectedPattern): Promise<SkillSuggestion> {
  const steps = pattern.steps

  // 1. 生成技能名称
  const name = generateSkillName(steps)

  // 2. 生成技能描述
  const description = generateSkillDescription(steps, pattern.supportCount)

  // 3. 构建工作流步骤
  const workflowSteps = steps.map(step => ({
    module: step.module,
    action: step.action,
    description: getActionDescription(step.module, step.action)
  }))

  return {
    patternId: pattern.id,
    name,
    description,
    workflowSteps,
    createdAt: Date.now()
  }
}

/**
 * 自动生成技能名称
 * 规则：取各步骤的中文动词，组合成有意义的名称
 */
function generateSkillName(steps: ActionStep[]): string {
  const verbMap: Record<string, string> = {
    'screen:capture': '截图',
    'vision:analyze': '分析图片',
    'file:save': '保存文件',
    'file:read': '读取文件',
    'knowledge:search': '搜索文档',
    'memory:save': '保存记忆',
    'clipboard:copy': '复制',
    'clipboard:paste': '粘贴',
    'translator:translate': '翻译',
    'code:execute': '运行代码'
  }

  const verbs = steps
    .map(s => verbMap[`${s.module}:${s.action}`] ?? `${s.module}:${s.action}`)
    .slice(0, 3)  // 最多取前 3 步

  return verbs.join('→') + ' 自动化'
}

/**
 * 自动生成技能描述
 */
function generateSkillDescription(steps: ActionStep[], count: number): string {
  const stepDesc = steps
    .map((s, i) => `${i + 1}. ${getActionDescription(s.module, s.action)}`)
    .join('\n')

  return `检测到您最近重复执行以下操作 ${count} 次：\n${stepDesc}\n\n已自动封装为技能，可一键执行。`
}
```

## 13. 风险与对策

| 风险 | 影响 | 对策 |
|------|------|------|
| 误检测伪模式 | 生成无用技能建议 | 提高 minSupport 阈值；用户可拒绝 |
| 高频操作产生大量日志 | 磁盘占用 | 写入队列批量处理；环形缓冲自动清理 |
| 用户隐私担忧 | 不信任 | 透明展示记录内容；一键清空；源头脱敏 |
| 检测到的模式过于碎片化 | 建议质量低 | 子模式去重；置信度排序；只推高置信度模式 |
| 与现有技能重复 | 重复创建 | 检查已有技能列表，避免重复 |
