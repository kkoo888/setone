> **为什么使用 `@nut-tree-fork/nut-js` 而非原版 `@nut-tree/nut-js`？**
>
> `@nut-tree-fork/nut-js` 是社区维护的 fork 版本，选择它的原因：
> 1. **原版维护停滞**：`@nut-tree/nut-js` 上游仓库更新频率降低，PR 积压严重，关键 Bug 修复无法及时合入
> 2. **兼容性修复**：fork 版本修复了 Electron 高版本中的兼容性问题（如 `screen.find` 在 Electron ≥ 28 中的崩溃）、macOS ARM64 原生支持、以及 Node.js ≥ 18 的 ABI 兼容
> 3. **活跃维护**：fork 仓库持续接收社区 PR，发布节奏更稳定，安全漏洞响应更及时
> 4. **API 一致**：fork 保持与原版完全兼容的 API 接口，切换成本为零（仅需替换包名）
>
> 如原上游恢复活跃维护且解决上述问题，可评估迁移回 `@nut-tree/nut-js`。

**modules/control/services/mouse-service.ts**：

```typescript
import { mouse, Point, Button, straightTo, screen } from '@nut-tree-fork/nut-js'
import type { Logger } from '../../../src/main/types/logger'

export interface MouseServiceConfig {
  /** 鼠标移动速度（像素/毫秒） */
  mouseSpeed?: number
  /** 单次操作超时时间（毫秒），超时后抛出 TimeoutError。默认 10000ms */
  timeoutMs?: number
}

export class MouseService {
  private logger: Logger
  private timeoutMs: number

  constructor(logger: Logger, config?: MouseServiceConfig) {
    this.logger = logger
    this.timeoutMs = config?.timeoutMs ?? 10_000
    mouse.config.mouseSpeed = config?.mouseSpeed ?? 1000
  }

  /**
   * 带超时保护的 nut-js 调用包装
   * 当底层调用卡死时，超过 timeoutMs 后自动中止并记录警告日志
   */
  private async withTimeout<T>(operation: string, fn: () => Promise<T>): Promise<T> {
    const controller = new AbortController()
    const timer = setTimeout(() => controller.abort(), this.timeoutMs)

    try {
      // 使用 Promise.race 实现超时：nut-js 不原生支持 AbortSignal，
      // 因此通过 race 与一个基于 AbortSignal 的超时 Promise 竞争
      const result = await Promise.race([
        fn(),
        new Promise<never>((_, reject) => {
          controller.signal.addEventListener('abort', () => {
            reject(new DOMException(`操作 "${operation}" 超时 (${this.timeoutMs}ms)`, 'TimeoutError'))
          })
        })
      ])
      return result
    } catch (err) {
      if ((err as Error).name === 'TimeoutError' || controller.signal.aborted) {
        this.logger.warn(`[MouseService] 操作 "${operation}" 超时 (${this.timeoutMs}ms)，nut-js 底层可能卡死`)
        throw new DOMException(`操作 "${operation}" 超时 (${this.timeoutMs}ms)`, 'TimeoutError')
      }
      throw err
    } finally {
      clearTimeout(timer)
    }
  }

  /**
   * 将坐标限制在屏幕边界内（Issue#289: 屏幕边界检查）
   * 获取屏幕宽高后 clamp 坐标，防止鼠标移出可视区域
   */
  private async clampToScreen(x: number, y: number): Promise<{ x: number; y: number }> {
    const screenWidth = await this.withTimeout('screenWidth', () => screen.width())
    const screenHeight = await this.withTimeout('screenHeight', () => screen.height())
    const clampedX = Math.max(0, Math.min(x, screenWidth - 1))
    const clampedY = Math.max(0, Math.min(y, screenHeight - 1))
    if (clampedX !== x || clampedY !== y) {
      this.logger.debug(`[MouseService] 坐标超出屏幕边界，已修正: (${x}, ${y}) → (${clampedX}, ${clampedY})`)
    }
    return { x: clampedX, y: clampedY }
  }

  /** 移动鼠标到指定位置（含屏幕边界检查） */
  async moveTo(x: number, y: number): Promise<void> {
    const pos = await this.clampToScreen(x, y)
    await this.withTimeout('setPosition', () => mouse.setPosition(new Point(pos.x, pos.y)))
    this.logger.debug(`鼠标移动到: (${pos.x}, ${pos.y})`)
  }

  /** 平滑移动鼠标到指定位置（含屏幕边界检查） */
  async smoothMoveTo(x: number, y: number): Promise<void> {
    const pos = await this.clampToScreen(x, y)
    await this.withTimeout('smoothMove', () => mouse.move(straightTo(new Point(pos.x, pos.y))))
    this.logger.debug(`鼠标平滑移动到: (${pos.x}, ${pos.y})`)
  }

  /** 点击 */
  async click(button: Button = Button.LEFT): Promise<void> {
    await this.withTimeout('click', () => mouse.click(button))
    this.logger.debug(`鼠标点击: ${button}`)
  }

  /** 双击 */
  async doubleClick(): Promise<void> {
    await this.withTimeout('doubleClick', () => mouse.doubleClick(Button.LEFT))
    this.logger.debug('鼠标双击')
  }

  /** 右键点击 */
  async rightClick(): Promise<void> {
    await this.withTimeout('rightClick', () => mouse.click(Button.RIGHT))
    this.logger.debug('鼠标右键')
  }

  /** 按下鼠标 */
  async press(button: Button = Button.LEFT): Promise<void> {
    await this.withTimeout('press', () => mouse.pressButton(button))
  }

  /** 释放鼠标 */
  async release(button: Button = Button.LEFT): Promise<void> {
    await this.withTimeout('release', () => mouse.releaseButton(button))
  }

  /** 拖拽（含屏幕边界检查 + try-finally 确保按键释放） */
  async drag(fromX: number, fromY: number, toX: number, toY: number): Promise<void> {
    const from = await this.clampToScreen(fromX, fromY)
    const to = await this.clampToScreen(toX, toY)
    await this.withTimeout('drag:setPosition', () => mouse.setPosition(new Point(from.x, from.y)))
    await this.withTimeout('drag:press', () => mouse.pressButton(Button.LEFT))
    try {
      await this.withTimeout('drag:move', () => mouse.move(straightTo(new Point(to.x, to.y))))
      this.logger.debug(`拖拽: (${from.x},${from.y}) → (${to.x},${to.y})`)
    } finally {
      await this.withTimeout('drag:release', () => mouse.releaseButton(Button.LEFT))
    }
  }

  /**
   * 滚动（支持上下左右四个方向）
   * @param amount - 滚动量（正数表示正方向，负数表示反方向）
   * @param direction - 滚动方向：'down'(默认) | 'up' | 'left' | 'right'
   */
  async scroll(amount: number, direction: 'up' | 'down' | 'left' | 'right' = 'down'): Promise<void> {
    const absAmount = Math.abs(amount)
    // 负数 amount 翻转方向
    const effectiveDir = amount < 0
      ? ({ up: 'down', down: 'up', left: 'right', right: 'left' } as const)[direction]
      : direction

    switch (effectiveDir) {
      case 'down':
        await this.withTimeout('scrollDown', () => mouse.scrollDown(absAmount))
        break
      case 'up':
        await this.withTimeout('scrollUp', () => mouse.scrollUp(absAmount))
        break
      case 'right':
        await this.withTimeout('scrollRight', () => mouse.scrollRight(absAmount))
        break
      case 'left':
        await this.withTimeout('scrollLeft', () => mouse.scrollLeft(absAmount))
        break
    }
    this.logger.debug(`滚动: direction=${effectiveDir}, amount=${absAmount}`)
  }

  /** 获取当前位置 */
  async getPosition(): Promise<{ x: number; y: number }> {
    const pos = await this.withTimeout('getPosition', () => mouse.getPosition())
    return { x: pos.x, y: pos.y }
  }
}
```

```typescript
import { keyboard, Key } from '@nut-tree-fork/nut-js'
import type { Logger } from '../../../src/main/types/logger'

/** 平台感知的修饰键 */
const isMac = process.platform === 'darwin'
const ModKey = isMac ? Key.LeftSuper : Key.LeftControl

export interface KeyboardServiceConfig {
  /** 按键间自动延迟（毫秒） */
  autoDelayMs?: number
  /** 单次操作超时时间（毫秒），超时后抛出 TimeoutError。默认 10000ms */
  timeoutMs?: number
}

export class KeyboardService {
  private logger: Logger
  private timeoutMs: number

  constructor(logger: Logger, config?: KeyboardServiceConfig) {
    this.logger = logger
    this.timeoutMs = config?.timeoutMs ?? 10_000
    keyboard.config.autoDelayMs = config?.autoDelayMs ?? 50
  }

  /**
   * 带超时保护的 nut-js 调用包装
   * 当底层调用卡死时，超过 timeoutMs 后自动中止并记录警告日志
   */
  private async withTimeout<T>(operation: string, fn: () => Promise<T>): Promise<T> {
    const controller = new AbortController()
    const timer = setTimeout(() => controller.abort(), this.timeoutMs)

    try {
      const result = await Promise.race([
        fn(),
        new Promise<never>((_, reject) => {
          controller.signal.addEventListener('abort', () => {
            reject(new DOMException(`操作 "${operation}" 超时 (${this.timeoutMs}ms)`, 'TimeoutError'))
          })
        })
      ])
      return result
    } catch (err) {
      if ((err as Error).name === 'TimeoutError' || controller.signal.aborted) {
        this.logger.warn(`[KeyboardService] 操作 "${operation}" 超时 (${this.timeoutMs}ms)，nut-js 底层可能卡死`)
        throw new DOMException(`操作 "${operation}" 超时 (${this.timeoutMs}ms)`, 'TimeoutError')
      }
      throw err
    } finally {
      clearTimeout(timer)
    }
  }

  /** 输入文本（日志中不记录内容，防止泄露密码、token 等敏感信息） */
  async type(text: string): Promise<void> {
    await this.withTimeout('type', () => keyboard.type(text))
    // 安全：仅记录操作类型和字符长度，不记录任何实际输入内容
    this.logger.debug(`[keyboard] type (长度: ${text.length})`)
  }

  /** 按下单个键 */
  async pressKey(key: Key): Promise<void> {
    await this.withTimeout('pressKey', () => keyboard.pressKey(key))
    this.logger.debug(`按下键: ${key}`)
  }

  /** 释放单个键 */
  async releaseKey(key: Key): Promise<void> {
    await this.withTimeout('releaseKey', () => keyboard.releaseKey(key))
  }

  /** 快捷键组合（修复 reverse 原地修改参数数组的 bug） */
  async hotkey(...keys: Key[]): Promise<void> {
    // 创建副本再反转，避免原地修改调用方的参数数组
    const reversed = [...keys].reverse()
    await this.withTimeout('hotkey:press', () => keyboard.pressKey(...keys))
    await this.withTimeout('hotkey:release', () => keyboard.releaseKey(...reversed))
    this.logger.debug(`快捷键: ${keys.join('+')}`)
  }

  /** 常用快捷键（平台自适应） */
  async copy(): Promise<void> {
    await this.hotkey(ModKey, Key.C)
  }

  async paste(): Promise<void> {
    await this.hotkey(ModKey, Key.V)
  }

  async selectAll(): Promise<void> {
    await this.hotkey(ModKey, Key.A)
  }

  async undo(): Promise<void> {
    await this.hotkey(ModKey, Key.Z)
  }

  async save(): Promise<void> {
    await this.hotkey(ModKey, Key.S)
  }
}
```

**modules/control/services/screen-match.ts**：

```typescript
import type { Logger } from '../../../src/main/types/logger'

/** 匹配结果 */
export interface MatchResult {
  x: number
  y: number
  width: number
  height: number
  confidence: number
}

/** 匹配配置 */
export interface MatchConfig {
  /** 置信度阈值 (0-1)，低于此值的结果被过滤 */
  confidenceThreshold?: number
  /** 最大返回结果数 */
  maxResults?: number
  /** AI 视觉服务端点 */
  visionEndpoint?: string
  /** 视觉模型名称 */
  visionModel?: string
}

/**
 * 屏幕图像匹配服务
 * 在屏幕截图中查找目标元素的位置
 * 与版块 17（视觉感知模块）的 screen-capture-service 配合使用：
 *   - vision 模块负责截图捕获
 *   - 本服务负责在截图中定位目标元素
 */
export class ScreenMatchService {
  private logger: Logger
  private config: Required<MatchConfig>

  constructor(logger: Logger, config?: MatchConfig) {
    this.logger = logger
    this.config = {
      confidenceThreshold: config?.confidenceThreshold ?? 0.7,
      maxResults: config?.maxResults ?? 10,
      visionEndpoint: config?.visionEndpoint ?? 'http://localhost:11434',
      visionModel: config?.visionModel ?? 'llava',
    }
  }

  /** 在屏幕中查找指定文本的位置 */
  async findTextPosition(
    screenshot: string,
    targetText: string,
    options?: { threshold?: number }
  ): Promise<MatchResult | null> {
    const threshold = options?.threshold ?? this.config.confidenceThreshold

    try {
      const base64Image = screenshot.replace(/^data:image\/\w+;base64,/, '')
      const response = await fetch(`${this.config.visionEndpoint}/api/generate`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          model: this.config.visionModel,
          prompt: `在截图中定位 "${targetText}" 的位置，返回 JSON: {"x": 0, "y": 0, "width": 0, "height": 0, "confidence": 0.9}，坐标基于图片像素，confidence 为 0-1 的置信度。如果找不到返回 null。仅返回 JSON，不要其他文字。`,
          images: [base64Image],
          stream: false
        }),
        signal: AbortSignal.timeout(15000)
      })

      if (!response.ok) {
        throw new Error(`视觉服务响应异常: ${response.status}`)
      }

      const data = await response.json() as { response: string }
      const jsonMatch = data.response.match(/\{[^}]+\}/)
      if (jsonMatch) {
        const pos = JSON.parse(jsonMatch[0]) as MatchResult
        if (pos.x !== undefined && pos.y !== undefined && pos.confidence >= threshold) {
          this.logger.debug(`找到元素 "${targetText}": (${pos.x}, ${pos.y}), 置信度: ${pos.confidence}`)
          return pos
        }
        if (pos.confidence < threshold) {
          this.logger.debug(`元素 "${targetText}" 置信度 ${pos.confidence} 低于阈值 ${threshold}`)
        }
      }
    } catch (err) {
      if ((err as Error).name === 'TimeoutError') {
        this.logger.error('文本匹配超时 (15s)', err as Error)
      } else {
        this.logger.error('文本匹配失败', err as Error)
      }
    }

    return null
  }

  /** 在屏幕中查找多个文本元素的位置 */
  async findMultipleTexts(
    screenshot: string,
    targets: string[]
  ): Promise<Map<string, MatchResult>> {
    const results = new Map<string, MatchResult>()
    // 并行查找所有目标
    const promises = targets.map(async (target) => {
      const result = await this.findTextPosition(screenshot, target)
      if (result) results.set(target, result)
    })
    await Promise.allSettled(promises)
    this.logger.debug(`批量查找: ${results.size}/${targets.length} 个元素已定位`)
    return results
  }

  /** 在屏幕中查找指定图标的近似位置（基于 AI 视觉模型） */
  async findImagePosition(
    screenshot: string,
    templateDescription: string,
    options?: { threshold?: number }
  ): Promise<MatchResult | null> {
    const threshold = options?.threshold ?? this.config.confidenceThreshold

    try {
      const base64Image = screenshot.replace(/^data:image\/\w+;base64,/, '')
      const response = await fetch(`${this.config.visionEndpoint}/api/generate`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          model: this.config.visionModel,
          prompt: `在截图中找到 "${templateDescription}" 对应的 UI 元素位置。返回 JSON: {"x": 0, "y": 0, "width": 0, "height": 0, "confidence": 0.9}，坐标基于图片像素，confidence 为 0-1 的置信度。如果找不到返回 null。仅返回 JSON。`,
          images: [base64Image],
          stream: false
        }),
        signal: AbortSignal.timeout(15000)
      })

      if (!response.ok) {
        throw new Error(`视觉服务响应异常: ${response.status}`)
      }

      const data = await response.json() as { response: string }
      const jsonMatch = data.response.match(/\{[^}]+\}/)
      if (jsonMatch) {
        const pos = JSON.parse(jsonMatch[0]) as MatchResult
        if (pos.x !== undefined && pos.y !== undefined && pos.confidence >= threshold) {
          this.logger.debug(`找到图标 "${templateDescription}": (${pos.x}, ${pos.y}), 置信度: ${pos.confidence}`)
          return pos
        }
      }
    } catch (err) {
      this.logger.error('图像匹配失败', err as Error)
    }

    return null
  }

  /** 检查视觉服务是否可用 */
  async checkServiceHealth(): Promise<boolean> {
    try {
      const response = await fetch(`${this.config.visionEndpoint}/api/tags`, {
        signal: AbortSignal.timeout(5000)
      })
      return response.ok
    } catch {
      return false
    }
  }
}
```

```typescript
import type { Module, ModuleContext, Capability } from '../../src/main/types/module'
import { MouseService } from './services/mouse-service'
import { KeyboardService } from './services/keyboard-service'
import { ScreenMatchService } from './services/screen-match'

export default class ControlModule implements Module {
  id = 'control'
  meta!: import('../../src/main/types/module').ModuleMeta

  private mouse!: MouseService
  private keyboard!: KeyboardService
  private screenMatch!: ScreenMatchService
  private context!: ModuleContext

  async activate(context: ModuleContext): Promise<void> {
    this.context = context

    // 可通过模块配置自定义超时时间，默认 10000ms
    const timeoutMs = context.config?.timeoutMs ?? 10_000

    this.mouse = new MouseService(context.logger, {
      mouseSpeed: context.config?.mouseSpeed,
      timeoutMs,
    })
    this.keyboard = new KeyboardService(context.logger, {
      autoDelayMs: context.config?.autoDelayMs,
      timeoutMs,
    })
    this.screenMatch = new ScreenMatchService(context.logger, {
      visionEndpoint: context.config?.visionEndpoint,
      visionModel: context.config?.visionModel,
    })

    // macOS 辅助功能权限检查
    if (process.platform === 'darwin') {
      context.logger.info('macOS: 请确保已在系统偏好设置 > 隐私与安全性 > 辅助功能 中授权本应用')
    }

    // 健康检查：视觉服务可用性
    const visionHealthy = await this.screenMatch.checkServiceHealth()
    if (!visionHealthy) {
      context.logger.warn('视觉服务不可用，screen_find 能力将受限')
    }

    context.logger.info('鼠标键盘控制模块已激活')
  }

  async deactivate(): Promise<void> {
    this.context.logger.info('鼠标键盘控制模块已停用')
  }

  /** 处理来自沙箱 worker 或其他模块的 IPC 消息 */
  handleMessage(channel: string, data: any): void {
    switch (channel) {
      case 'control:mouse:move':
        this.mouse.moveTo(data.x, data.y).catch(err =>
          this.context.logger.error('鼠标移动失败', err)
        )
        break
      case 'control:mouse:click':
        this.mouse.click().catch(err =>
          this.context.logger.error('鼠标点击失败', err)
        )
        break
      case 'control:keyboard:type':
        this.keyboard.type(data.text).catch(err =>
          this.context.logger.error('键盘输入失败', err)
        )
        break
      case 'control:health':
        // 返回模块健康状态
        this.screenMatch.checkServiceHealth().then(ok => {
          this.context.logger.debug(`健康检查: 视觉服务 ${ok ? '可用' : '不可用'}`)
        })
        break
      default:
        this.context.logger.debug(`未知消息通道: ${channel}`)
    }
  }

  getCapabilities(): Capability[] {
    return [
      {
        type: 'tool',
        name: 'mouse_move',
        description: '移动鼠标到指定位置',
        priority: 10,
        moduleId: this.id,
        handler: {
          execute: async (params) => {
            const { x, y, smooth } = params as { x: number; y: number; smooth?: boolean }
            if (smooth) {
              await this.mouse.smoothMoveTo(x, y)
            } else {
              await this.mouse.moveTo(x, y)
            }
            return { success: true }
          }
        }
      },
      {
        type: 'tool',
        name: 'mouse_click',
        description: '鼠标点击',
        priority: 10,
        moduleId: this.id,
        handler: {
          execute: async (params) => {
            const { button, doubleClick } = params as { button?: string; doubleClick?: boolean }
            if (doubleClick) {
              await this.mouse.doubleClick()
            } else if (button === 'right') {
              await this.mouse.rightClick()
            } else {
              await this.mouse.click()
            }
            return { success: true }
          }
        }
      },
      {
        type: 'tool',
        name: 'mouse_drag',
        description: '鼠标拖拽',
        priority: 10,
        moduleId: this.id,
        handler: {
          execute: async (params) => {
            const { fromX, fromY, toX, toY } = params as { fromX: number; fromY: number; toX: number; toY: number }
            await this.mouse.drag(fromX, fromY, toX, toY)
            return { success: true }
          }
        }
      },
      {
        type: 'tool',
        name: 'mouse_scroll',
        description: '鼠标滚动（支持上下左右四个方向，amount 负数表示反方向）',
        priority: 10,
        moduleId: this.id,
        handler: {
          execute: async (params) => {
            const { amount, direction } = params as { amount: number; direction?: 'up' | 'down' | 'left' | 'right' }
            await this.mouse.scroll(amount, direction)
            return { success: true }
          }
        }
      },
      {
        type: 'tool',
        name: 'keyboard_type',
        description: '键盘输入文本',
        priority: 10,
        moduleId: this.id,
        handler: {
          execute: async (params) => {
            const { text } = params as { text: string }
            await this.keyboard.type(text)
            return { success: true }
          }
        }
      },
      {
        type: 'tool',
        name: 'keyboard_hotkey',
        description: '键盘快捷键组合',
        priority: 10,
        moduleId: this.id,
        handler: {
          execute: async (params) => {
            const { keys } = params as { keys: string[] }
            const KeyMod = await import('@nut-tree-fork/nut-js')
            // macOS 修饰键映射：LeftControl → LeftSuper (Cmd)
            const isMac = process.platform === 'darwin'
            const MODIFIER_MAP: Record<string, string> = isMac
              ? { LeftControl: 'LeftSuper', RightControl: 'RightSuper', Control: 'Super' }
              : {}
            const resolvedKeys = keys.map(k => MODIFIER_MAP[k] || k)
            const keyEnums = resolvedKeys.map((k) => (KeyMod.Key as any)[k]).filter(Boolean)
            if (keyEnums.length > 0) {
              await this.keyboard.hotkey(...keyEnums)
            }
            return { success: true }
          }
        }
      },
      {
        type: 'tool',
        name: 'screen_find',
        description: '在屏幕上查找元素位置',
        priority: 10,
        moduleId: this.id,
        handler: {
          execute: async (params) => {
            const { screenshot, target } = params as { screenshot: string; target: string }
            return await this.screenMatch.findTextPosition(screenshot, target)
          }
        }
      }
    ]
  }
}
```

### 18.3 代码规范

- **权限检查**：macOS 需要辅助功能权限，Windows 需要管理员权限
- **操作延迟**：`autoDelayMs = 50`，避免操作过快导致丢失
- **安全边界**：限制鼠标可移动的屏幕范围
- **错误处理**：操作失败时记录日志，不抛出异常
- **平台适配**：快捷键修饰键自动适配——macOS 使用 `Key.LeftSuper` (Cmd)，Windows/Linux 使用 `Key.LeftControl` (Ctrl)；`keyboard_hotkey` 处理器内置 `MODIFIER_MAP` 映射表，外部传入的 `LeftControl` 等键名在 macOS 上自动转换为 `LeftSuper`
