```typescript
// src/main/core/types/performance.ts

/**
 * 系统资源快照
 */
export interface SystemResourceSnapshot {
  /** 时间戳（毫秒） */
  readonly timestamp: number;
  /** CPU 使用率（0-100） */
  readonly cpuPercent: number;
  /** 内存使用率（0-100） */
  readonly memoryPercent: number;
  /** 内存总量（MB） */
  readonly totalMemoryMB: number;
  /** 内存已用（MB） */
  readonly usedMemoryMB: number;
  /** 磁盘使用率（0-100） */
  readonly diskPercent: number;
  /** 磁盘读取速率（bytes/s），-1 表示不可用 */
  readonly diskReadBytesPerSec: number;
  /** 磁盘写入速率（bytes/s），-1 表示不可用 */
  readonly diskWriteBytesPerSec: number;
  /** GPU 使用率（0-100），-1 表示不可用 */
  readonly gpuPercent: number;
  /** GPU 显存使用率（0-100），-1 表示不可用 */
  readonly gpuMemoryPercent: number;
}

/**
 * 降频策略等级
 */
export enum ThrottleLevel {
  /** 全功能模式：< 60% */
  FULL = 'full',
  /** 感知模式：60-80%，感知间隔翻倍 */
  AWARE = 'aware',
  /** 降级模式：> 80%，暂停感知模块 */
  DEGRADED = 'degraded',
  /** 低功耗模式：> 90%，仅保留核心功能 */
  LOW_POWER = 'low_power',
}

/**
 * 降频策略配置
 */
export interface ThrottleStrategyConfig {
  /** 触发感知模式的阈值（%） */
  readonly awareThreshold: number;
  /** 触发降级模式的阈值（%） */
  readonly degradedThreshold: number;
  /** 触发低功耗模式的阈值（%） */
  readonly lowPowerThreshold: number;
  /** 感知模式下采集间隔倍数 */
  readonly awareIntervalMultiplier: number;
  /** 恢复到全功能模式的冷却时间（毫秒） */
  readonly cooldownMs: number;
}

/**
 * 模块资源声明
 */
export interface ModuleResourceDeclaration {
  /** 模块 ID */
  readonly moduleId: string;
  /** 最大内存（MB） */
  readonly maxMemoryMB: number;
  /** 最大 CPU（%） */
  readonly maxCpuPercent: number;
  /** 优先级（1-10，越高越优先保留） */
  readonly priority: number;
  /** 是否为感知模块（可被暂停） */
  readonly isPerceptual: boolean;
}

/**
 * 模块运行时资源状态
 */
export interface ModuleResourceState {
  readonly moduleId: string;
  readonly currentMemoryMB: number;
  readonly currentCpuPercent: number;
  readonly uptimeSeconds: number;
  readonly isPaused: boolean;
  readonly pauseReason: string | null;
}

/**
 * 资源趋势方向
 */
export enum TrendDirection {
  /** 资源使用率上升 */
  RISING = 'rising',
  /** 资源使用率稳定 */
  STABLE = 'stable',
  /** 资源使用率下降 */
  DECLINING = 'declining',
}

/**
 * 资源趋势预测结果
 */
export interface TrendPrediction {
  /** 资源类型（cpu / memory / disk / gpu） */
  readonly resource: string;
  /** 趋势方向 */
  readonly direction: TrendDirection;
  /** 线性回归斜率（每秒变化百分比） */
  readonly slopePerSecond: number;
  /** 当前使用率（%） */
  readonly currentValue: number;
  /** 预测 N 秒后的使用率（%） */
  readonly predictedValueAtSeconds: number;
  /** 预测时间窗口（秒） */
  readonly predictionWindowSeconds: number;
  /** 预估资源耗尽剩余时间（秒），-1 表示不会耗尽或趋势下降 */
  readonly estimatedExhaustionSeconds: number;
  /** R² 决定系数（0-1），越高表示拟合越可靠 */
  readonly rSquared: number;
  /** 用于回归的采样点数量 */
  readonly sampleCount: number;
}

/**
 * 性能监控事件
 */
export type PerformanceEvent =
  | { type: 'snapshot'; data: SystemResourceSnapshot }
  | { type: 'throttle-change'; data: { from: ThrottleLevel; to: ThrottleLevel } }
  | { type: 'module-paused'; data: { moduleId: string; reason: string } }
  | { type: 'module-resumed'; data: { moduleId: string } }
  | { type: 'resource-warning'; data: { resource: string; value: number; threshold: number } }
  | { type: 'trend-alert'; data: TrendPrediction };

/**
 * 性能监控配置

export interface PerformanceMonitorConfig {
  /** 采集间隔（毫秒） */
  readonly collectIntervalMs: number;
  /** 降频策略配置 */
  readonly throttle: ThrottleStrategyConfig;
  /** 是否启用 GPU 监控 */
  readonly gpuEnabled: boolean;
  /** 历史数据保留条数 */
  readonly historySize: number;
  /** 资源警告阈值 */
  readonly warningThresholds: {
    readonly cpu: number;
    readonly memory: number;
    readonly disk: number;
    readonly gpu: number;
  };
}

/**
 * 性能监控器接口
 */
export interface IPerformanceMonitor {
  start(): void;
  stop(): void;
  getLatestSnapshot(): SystemResourceSnapshot | null;
  getHistory(): readonly SystemResourceSnapshot[];
  getCurrentThrottleLevel(): ThrottleLevel;
  registerModule(declaration: ModuleResourceDeclaration): void;
  unregisterModule(moduleId: string): void;
  getModuleStates(): Readonly<Record<string, ModuleResourceState>>;
  onEvent(callback: (event: PerformanceEvent) => void): () => void;
  /** 运行时更新配置（由设置面板通过 IPC 触发，联动版块25） */
  updateConfig(config: Partial<PerformanceMonitorConfig>): void;
  /**
   * 将历史数据导出为文件
   * @param format - 导出格式：'json' | 'csv'
   * @param filePath - 导出文件路径
   * @param limit - 可选，导出最近 N 条数据
   * @returns 导出的记录数
   */
  exportHistory(format: 'json' | 'csv', filePath: string, limit?: number): number;
  /**
   * 基于最近 N 个采样点进行线性回归，预测资源使用趋势
   * @param resource - 资源类型：'cpu' | 'memory' | 'disk' | 'gpu'
   * @param sampleCount - 使用最近多少个采样点（默认 30）
   * @param predictionWindowSeconds - 预测未来多少秒后的值（默认 300）
   * @returns 趋势预测结果，数据不足时返回 null
   */
  predictTrend(
    resource: 'cpu' | 'memory' | 'disk' | 'gpu',
    sampleCount?: number,
    predictionWindowSeconds?: number
  ): TrendPrediction | null;
}
```

#### 步骤 2：实现资源采集器

```typescript
// src/main/core/resource-collector.ts

import * as os from 'os';
import { exec } from 'child_process';
import { promisify } from 'util';
import type { SystemResourceSnapshot } from './types/performance';

const execAsync = promisify(exec);

/**
 * 资源采集器
 * 负责采集 CPU、内存、磁盘、GPU 系统资源数据
 */
export class ResourceCollector {
  private lastCpuIdle = 0;
  private lastCpuTotal = 0;
  /**
   * GPU 厂商类型（首次采集时自动检测）
   * - 'nvidia': NVIDIA GPU，使用 nvidia-smi
   * - 'amd': AMD GPU，使用 rocm-smi 或 /sys/class/drm 降级读取
   * - 'intel': Intel GPU，暂不支持直接采集，返回 -1
   * - 'unknown': 检测到 GPU 设备但无法读取指标
   * - null: 尚未检测
   */
  private gpuVendor: 'nvidia' | 'amd' | 'intel' | 'unknown' | null = null;
  private lastDiskIo: { readBytes: number; writeBytes: number; timestamp: number } | null = null;

  /**
   * 采集完整的系统资源快照
   */
  async collect(): Promise<SystemResourceSnapshot> {
    const [cpu, memory, disk, diskIo, gpu] = await Promise.all([
      this.collectCpu(),
      this.collectMemory(),
      this.collectDisk(),
      this.collectDiskIo(),
      this.collectGpu(),
    ]);

    return {
      timestamp: Date.now(),
      cpuPercent: cpu,
      ...memory,
      diskPercent: disk,
      ...diskIo,
      ...gpu,
    };
  }

  /**
   * 采集 CPU 使用率
   * 基于 /proc/stat 差值计算，跨平台兼容
   */
  private async collectCpu(): Promise<number> {
    const cpus = os.cpus();
    let idle = 0;
    let total = 0;

    for (const cpu of cpus) {
      const times = cpu.times;
      idle += times.idle;
      total +=
        times.user +
        times.nice +
        times.sys +
        times.idle +
        times.irq;
    }

    // 首次采集：记录基线值，无法计算差值，返回 0
    if (this.lastCpuTotal === 0) {
      this.lastCpuIdle = idle;
      this.lastCpuTotal = total;
      return 0;
    }

    const idleDiff = idle - this.lastCpuIdle;
    const totalDiff = total - this.lastCpuTotal;

    this.lastCpuIdle = idle;
    this.lastCpuTotal = total;

    if (totalDiff === 0) return 0;
    return Math.round(((totalDiff - idleDiff) / totalDiff) * 100 * 10) / 10;
  }

  /**
   * 采集内存使用情况
   */
  private async collectMemory(): Promise<{
    memoryPercent: number;
    totalMemoryMB: number;
    usedMemoryMB: number;
  }> {
    const totalBytes = os.totalmem();
    const freeBytes = os.freemem();
    const usedBytes = totalBytes - freeBytes;

    const totalMemoryMB = Math.round(totalBytes / (1024 * 1024));
    const usedMemoryMB = Math.round(usedBytes / (1024 * 1024));
    const memoryPercent =
      Math.round((usedBytes / totalBytes) * 100 * 10) / 10;

    return { memoryPercent, totalMemoryMB, usedMemoryMB };
  }

  /**
   * 采集磁盘使用率
   * Linux:   df -h / → 第 5 列为 Use%（如 "22%"）
   * macOS:   df -h / → 第 5 列为 Capacity（如 "86%"），列名与 Linux 不同但位置一致
   * Windows: wmic 获取 FreeSpace 和 Size 后计算
   */
  private async collectDisk(): Promise<number> {
    try {
      const platform = os.platform();

      let cmd: string;
      if (platform === 'win32') {
        cmd = 'wmic logicaldisk where "DeviceID=\'C:\'" get FreeSpace,Size /format:csv';
      } else if (platform === 'darwin') {
        // macOS: df -h / 输出格式
        //   Filesystem      Size   Used  Avail Capacity  Mounted on
        //   /dev/disk1s1s1  233Gi  195Gi   34Gi    86%    /
        // Capacity 为第 5 列；macOS 使用 IEC 单位后缀（Gi/Mi），
        // 但百分比格式与 Linux 一致，仍可用 awk '{print $5}' 提取。
        // 使用 grep 精确匹配根挂载点，避免 APFS 快照等额外行干扰。
        cmd = "df -h / | grep ' /$' | awk '{print $5}'";
      } else {
        // Linux: df -h / 输出格式
        //   Filesystem      Size  Used Avail Use% Mounted on
        //   /dev/nvme0n1p3   40G  8.1G   30G  22% /
        // Use% 为第 5 列
        cmd = "df -h / | tail -1 | awk '{print $5}'";
      }

      const { stdout } = await execAsync(cmd, { timeout: 5000 });

      if (platform === 'win32') {
        // 解析 Windows wmic CSV 输出
        // wmic 列顺序因系统语言而异，通过表头动态查找列索引
        const lines = stdout.trim().split('\n').filter(Boolean);
        if (lines.length >= 2) {
          const headers = lines[0].split(',').map(h => h.trim().toLowerCase());
          const freeIdx = headers.indexOf('freespace');
          const sizeIdx = headers.indexOf('size');
          if (freeIdx !== -1 && sizeIdx !== -1) {
            const parts = lines[lines.length - 1].split(',');
            const free = parseInt(parts[freeIdx], 10);
            const size = parseInt(parts[sizeIdx], 10);
            if (size > 0) {
              return Math.round(((size - free) / size) * 100 * 10) / 10;
            }
          }
        }
        return 0;
      }

      // 解析 Linux/macOS 输出（百分比格式如 "22%" 或 "86%"）
      // macOS 列名为 "Capacity"，Linux 列名为 "Use%"，但输出格式相同
      const match = stdout.trim().match(/(\d+)%/);
      return match ? parseInt(match[1], 10) : 0;
    } catch {
      return 0;
    }
  }

  /**
   * 采集磁盘 I/O 速率（bytes/s）
   * Linux: 解析 /proc/diskstats 计算读写字节数差值
   * macOS: 使用 iostat -d -c 1 获取 KB/s 转换为 bytes/s
   * Windows: 使用 wmic path Win32_PerfFormattedData_PerfDisk_LogicalDisk 获取读写字节速率
   */
  private async collectDiskIo(): Promise<{
    diskReadBytesPerSec: number;
    diskWriteBytesPerSec: number;
  }> {
    try {
      const platform = os.platform();

      if (platform === 'win32') {
        // Windows（主平台）：优先使用 PowerShell Get-CimInstance，
        // 兼容新版 Windows 11（wmic 已弃用）；回退 wmic
        try {
          const { stdout } = await execAsync(
            'powershell -NoProfile -Command "Get-CimInstance Win32_PerfFormattedData_PerfDisk_LogicalDisk | Where-Object { $_.Name -eq \'_Total\' } | Select-Object -ExpandProperty DiskReadBytesPersec, DiskWriteBytesPersec"',
            { timeout: 8000 }
          );
          const lines = stdout.trim().split('\n').map(l => l.trim()).filter(Boolean);
          if (lines.length >= 2) {
            return {
              diskReadBytesPerSec: parseInt(lines[0], 10) || 0,
              diskWriteBytesPerSec: parseInt(lines[1], 10) || 0,
            };
          }
        } catch {
          // PowerShell 不可用或超时，回退 wmic（兼容旧版 Windows）
        }

        try {
          const { stdout } = await execAsync(
            'wmic path Win32_PerfFormattedData_PerfDisk_LogicalDisk where "Name=\'_Total\'" get DiskReadBytesPersec,DiskWriteBytesPersec /format:csv',
            { timeout: 5000 }
          );
          const lines = stdout.trim().split('\n').filter(Boolean);
          if (lines.length >= 2) {
            const headers = lines[0].split(',').map(h => h.trim().toLowerCase());
            const readIdx = headers.indexOf('diskreadbytespersec');
            const writeIdx = headers.indexOf('diskwritebytespersec');
            if (readIdx !== -1 && writeIdx !== -1) {
              const parts = lines[lines.length - 1].split(',');
              return {
                diskReadBytesPerSec: parseInt(parts[readIdx], 10) || 0,
                diskWriteBytesPerSec: parseInt(parts[writeIdx], 10) || 0,
              };
            }
          }
        } catch {
          // wmic 也不可用
        }

        return { diskReadBytesPerSec: 0, diskWriteBytesPerSec: 0 };
      }

      if (platform === 'linux') {
        // Linux：解析 /proc/diskstats，汇总所有主磁盘（排除分区）的读写字节数
        // 字段说明：第 5 列为读取扇区数，第 9 列为写入扇区数（每扇区 512 字节）
        const { stdout } = await execAsync(
          "cat /proc/diskstats | awk '$3 ~ /^(sd|vd|nvme|hd)[a-z]$/ {reads+=$6; writes+=$10} END {print reads*512, writes*512}'",
          { timeout: 5000 }
        );
        const parts = stdout.trim().split(/\s+/);
        const readBytes = parseInt(parts[0], 10) || 0;
        const writeBytes = parseInt(parts[1], 10) || 0;
        const now = Date.now();

        // 首次采集：记录基线，返回 0
        if (!this.lastDiskIo) {
          this.lastDiskIo = { readBytes, writeBytes, timestamp: now };
          return { diskReadBytesPerSec: 0, diskWriteBytesPerSec: 0 };
        }

        const elapsedSec = (now - this.lastDiskIo.timestamp) / 1000;
        if (elapsedSec <= 0) {
          return { diskReadBytesPerSec: 0, diskWriteBytesPerSec: 0 };
        }

        const readPerSec = Math.round((readBytes - this.lastDiskIo.readBytes) / elapsedSec);
        const writePerSec = Math.round((writeBytes - this.lastDiskIo.writeBytes) / elapsedSec);

        this.lastDiskIo = { readBytes, writeBytes, timestamp: now };

        return {
          diskReadBytesPerSec: Math.max(0, readPerSec),
          diskWriteBytesPerSec: Math.max(0, writePerSec),
        };
      }

      if (platform === 'darwin') {
        // macOS：iostat -d -c 1 输出两次采样，取最后一次（即 1 秒间隔后的差值）
        const { stdout } = await execAsync('iostat -d -c 1', { timeout: 10000 });
        const lines = stdout.trim().split('\n').filter(Boolean);
        const lastLine = lines[lines.length - 1];
        const cols = lastLine.trim().split(/\s+/);
        const numericCols = cols.filter(c => /^[\d.]+$/.test(c));
        if (numericCols.length >= 2) {
          const val1 = parseFloat(numericCols[numericCols.length - 2]) || 0;
          const val2 = parseFloat(numericCols[numericCols.length - 1]) || 0;
          if (numericCols.length >= 3 && val1 !== val2) {
            return {
              diskReadBytesPerSec: Math.round(val1 * 1024 * 1024),
              diskWriteBytesPerSec: Math.round(val2 * 1024 * 1024),
            };
          }
          const totalBytesPerSec = Math.round(val2 * 1024 * 1024);
          return {
            diskReadBytesPerSec: Math.round(totalBytesPerSec / 2),
            diskWriteBytesPerSec: Math.round(totalBytesPerSec / 2),
          };
        }
        return { diskReadBytesPerSec: 0, diskWriteBytesPerSec: 0 };
      }

      return { diskReadBytesPerSec: -1, diskWriteBytesPerSec: -1 };
    } catch {
      return { diskReadBytesPerSec: -1, diskWriteBytesPerSec: -1 };
    }
  }

  /**
   * 采集 GPU 使用率
   * 支持的 GPU 类型及检测优先级：
   *   1. NVIDIA — 通过 nvidia-smi 采集利用率和显存
   *   2. AMD    — 优先使用 rocm-smi，降级读取 /sys/class/drm 节点
   *   3. 其他   — 返回 -1 表示不可用，不抛出异常
   */
  private async collectGpu(): Promise<{
    gpuPercent: number;
    gpuMemoryPercent: number;
  }> {
    // 首次检测 GPU 厂商
    if (this.gpuVendor === null) {
      this.gpuVendor = await this.detectGpuVendor();
    }

    // 不支持的 GPU 类型返回空数据（-1），不报错
    if (this.gpuVendor === 'unknown' || this.gpuVendor === 'intel') {
      return { gpuPercent: -1, gpuMemoryPercent: -1 };
    }

    try {
      switch (this.gpuVendor) {
        case 'nvidia':
          return await this.collectNvidiaGpu();
        case 'amd':
          return await this.collectAmdGpu();
        default:
          return { gpuPercent: -1, gpuMemoryPercent: -1 };
      }
    } catch {
      return { gpuPercent: -1, gpuMemoryPercent: -1 };
    }
  }

  /**
   * 检测 GPU 厂商
   * 检测顺序：nvidia-smi → rocm-smi → /sys/class/drm → 未知
   */
  private async detectGpuVendor(): Promise<'nvidia' | 'amd' | 'intel' | 'unknown'> {
    // 1. 检测 NVIDIA
    try {
      await execAsync('nvidia-smi --query-gpu=utilization.gpu --format=csv,noheader,nounits', {
        timeout: 5000,
      });
      return 'nvidia';
    } catch {
      // 不是 NVIDIA，继续检测
    }

    // 2. 检测 AMD（rocm-smi）
    try {
      await execAsync('rocm-smi --showuse', { timeout: 5000 });
      return 'amd';
    } catch {
      // 没有 rocm-smi，尝试 sysfs
    }

    // 3. 检测 AMD/Intel（通过 /sys/class/drm 存在 GPU 设备）
    try {
      const { stdout } = await execAsync(
        'ls /sys/class/drm/card*/device/gpu_busy_percent 2>/dev/null || echo ""',
        { timeout: 5000 }
      );
      if (stdout.trim().length > 0) {
        // 有 GPU 设备暴露了 gpu_busy_percent，可能是 AMD 或 Intel
        return 'amd'; // AMD 较常见此接口，优先按 AMD 处理
      }
    } catch {
      // 忽略
    }

    // 4. 尝试检测 Intel GPU（通过 intel_gpu_top）
    try {
      await execAsync('which intel_gpu_top', { timeout: 5000 });
      return 'intel';
    } catch {
      // 未找到
    }

    return 'unknown';
  }

  /**
   * NVIDIA GPU 采集（通过 nvidia-smi）
   */
  private async collectNvidiaGpu(): Promise<{
    gpuPercent: number;
    gpuMemoryPercent: number;
  }> {
    const { stdout } = await execAsync(
      'nvidia-smi --query-gpu=utilization.gpu,utilization.memory --format=csv,noheader,nounits',
      { timeout: 5000 }
    );

    const parts = stdout.trim().split(',').map((s) => parseInt(s.trim(), 10));

    return {
      gpuPercent: isNaN(parts[0]) ? -1 : parts[0],
      gpuMemoryPercent: isNaN(parts[1]) ? -1 : parts[1],
    };
  }

  /**
   * AMD GPU 采集
   * 优先使用 rocm-smi，降级读取 /sys/class/drm 节点
   */
  private async collectAmdGpu(): Promise<{
    gpuPercent: number;
    gpuMemoryPercent: number;
  }> {
    // 方式 1：rocm-smi
    try {
      const { stdout } = await execAsync(
        'rocm-smi --showuse --showmemuse --csv',
        { timeout: 5000 }
      );
      // CSV 输出格式：device, GPU use%, Memory use%, ...
      const lines = stdout.trim().split('\n').filter(Boolean);
      if (lines.length >= 2) {
        const headers = lines[0].split(',').map(h => h.trim().toLowerCase());
        const gpuUseIdx = headers.findIndex(h => h.includes('gpu') && h.includes('use'));
        const memUseIdx = headers.findIndex(h => h.includes('memory') && h.includes('use'));
        const values = lines[1].split(',').map(v => parseInt(v.trim(), 10));
        return {
          gpuPercent: gpuUseIdx >= 0 && !isNaN(values[gpuUseIdx]) ? values[gpuUseIdx] : -1,
          gpuMemoryPercent: memUseIdx >= 0 && !isNaN(values[memUseIdx]) ? values[memUseIdx] : -1,
        };
      }
    } catch {
      // rocm-smi 失败，降级到 sysfs
    }

    // 方式 2：/sys/class/drm/card*/device/gpu_busy_percent
    try {
      const { stdout } = await execAsync(
        'cat /sys/class/drm/card0/device/gpu_busy_percent 2>/dev/null',
        { timeout: 5000 }
      );
      const gpuPercent = parseInt(stdout.trim(), 10);
      if (!isNaN(gpuPercent)) {
        // 尝试读取显存信息
        let gpuMemoryPercent = -1;
        try {
          const { stdout: memStdout } = await execAsync(
            'cat /sys/class/drm/card0/device/mem_info_vram_used 2>/dev/null && cat /sys/class/drm/card0/device/mem_info_vram_total 2>/dev/null',
            { timeout: 5000 }
          );
          const memLines = memStdout.trim().split('\n');
          if (memLines.length >= 2) {
            const used = parseInt(memLines[0], 10);
            const total = parseInt(memLines[1], 10);
            if (!isNaN(used) && !isNaN(total) && total > 0) {
              gpuMemoryPercent = Math.round((used / total) * 100);
            }
          }
        } catch {
          // 显存读取失败，保持 -1
        }
        return { gpuPercent, gpuMemoryPercent };
      }
    } catch {
      // sysfs 读取失败
    }

    return { gpuPercent: -1, gpuMemoryPercent: -1 };
  }
}
```

#### 步骤 3：实现降频策略引擎

```typescript
// src/main/core/throttle-strategy.ts

import type {
  SystemResourceSnapshot,
  ThrottleStrategyConfig,
} from './types/performance';
import { ThrottleLevel } from './types/performance';

/**
 * 降频策略引擎
 * 根据系统资源使用情况动态调整运行等级
 */
export class ThrottleStrategy {
  private currentLevel: ThrottleLevel = ThrottleLevel.FULL;
  private lastDowngradeTime = 0;
  private config: ThrottleStrategyConfig;

  constructor(config?: Partial<ThrottleStrategyConfig>) {
    this.config = {
      awareThreshold: 60,
      degradedThreshold: 80,
      lowPowerThreshold: 90,
      awareIntervalMultiplier: 2,
      cooldownMs: 30000,
      ...config,
    };
  }

  /**
   * 根据资源快照计算推荐的降频等级
   */
  evaluate(snapshot: SystemResourceSnapshot): {
    level: ThrottleLevel;
    changed: boolean;
    reason: string;
  } {
    // 取所有资源指标的最大值作为决策依据
    const maxUsage = Math.max(
      snapshot.cpuPercent,
      snapshot.memoryPercent,
      snapshot.gpuPercent >= 0 ? snapshot.gpuPercent : 0,
      snapshot.gpuMemoryPercent >= 0 ? snapshot.gpuMemoryPercent : 0
    );

    let targetLevel: ThrottleLevel;
    let reason: string;

    if (maxUsage > this.config.lowPowerThreshold) {
      targetLevel = ThrottleLevel.LOW_POWER;
      reason = `资源使用率 ${maxUsage.toFixed(1)}% 超过低功耗阈值 ${this.config.lowPowerThreshold}%`;
    } else if (maxUsage > this.config.degradedThreshold) {
      targetLevel = ThrottleLevel.DEGRADED;
      reason = `资源使用率 ${maxUsage.toFixed(1)}% 超过降级阈值 ${this.config.degradedThreshold}%`;
    } else if (maxUsage > this.config.awareThreshold) {
      targetLevel = ThrottleLevel.AWARE;
      reason = `资源使用率 ${maxUsage.toFixed(1)}% 超过感知阈值 ${this.config.awareThreshold}%`;
    } else {
      targetLevel = ThrottleLevel.FULL;
      reason = `资源使用率 ${maxUsage.toFixed(1)}% 处于正常范围`;
    }

    // 冷却检查：降级需要冷却时间才能恢复
    if (
      this.isHigherLevel(targetLevel) &&
      !this.isHigherLevel(this.currentLevel) &&
      Date.now() - this.lastDowngradeTime < this.config.cooldownMs
    ) {
      return {
        level: this.currentLevel,
        changed: false,
        reason: `冷却中，距离上次降级 ${Math.round(
          (Date.now() - this.lastDowngradeTime) / 1000
        )}秒`,
      };
    }

    const changed = targetLevel !== this.currentLevel;

    if (changed && this.isLowerLevel(targetLevel)) {
      this.lastDowngradeTime = Date.now();
    }

    if (changed) {
      this.currentLevel = targetLevel;
    }

    return { level: this.currentLevel, changed, reason };
  }

  /**
   * 获取当前等级
   */
  getCurrentLevel(): ThrottleLevel {
    return this.currentLevel;
  }

  /**
   * 获取当前等级对应的采集间隔（毫秒）
   */
  getCollectInterval(baseIntervalMs: number): number {
    switch (this.currentLevel) {
      case ThrottleLevel.AWARE:
        return baseIntervalMs * this.config.awareIntervalMultiplier;
      case ThrottleLevel.DEGRADED:
        return baseIntervalMs * this.config.awareIntervalMultiplier * 2;
      case ThrottleLevel.LOW_POWER:
        return baseIntervalMs * this.config.awareIntervalMultiplier * 4;
      default:
        return baseIntervalMs;
    }
  }

  /**
   * 判断是否为更低等级（资源更紧张）
   */
  private isLowerLevel(level: ThrottleLevel): boolean {
    const order = [
      ThrottleLevel.FULL,
      ThrottleLevel.AWARE,
      ThrottleLevel.DEGRADED,
      ThrottleLevel.LOW_POWER,
    ];
    return order.indexOf(level) > order.indexOf(this.currentLevel);
  }

  /**
   * 判断是否为更高等级（资源更充裕）
   */
  private isHigherLevel(level: ThrottleLevel): boolean {
    const order = [
      ThrottleLevel.FULL,
      ThrottleLevel.AWARE,
      ThrottleLevel.DEGRADED,
      ThrottleLevel.LOW_POWER,
    ];
    return order.indexOf(level) < order.indexOf(this.currentLevel);
  }

  /**
   * 重置到全功能模式
   */
  reset(): void {
    this.currentLevel = ThrottleLevel.FULL;
    this.lastDowngradeTime = 0;
  }
}
```

```typescript
// src/main/core/module-resource-controller.ts

import { ThrottleLevel } from './types/performance';
import type {
  ModuleResourceDeclaration,
  ModuleResourceState,
  SystemResourceSnapshot,
} from './types/performance';

/**
 * 模块资源控制器
 * 根据降频等级自动暂停/恢复低优先级模块
 */
export class ModuleResourceController {
  private declarations = new Map<string, ModuleResourceDeclaration>();
  private states = new Map<string, ModuleResourceState>();
  private consecutiveViolationCount = new Map<string, number>();
  private readonly MAX_CONSECUTIVE_VIOLATIONS = 3;
  private onModulePause: ((moduleId: string, reason: string) => void) | null =
    null;
  private onModuleResume: ((moduleId: string) => void) | null = null;
  private onModuleRestart: ((moduleId: string, reason: string) => void) | null =
    null;
  private onWarning: ((event: { moduleId: string; resource: string; current: number; limit: number }) => void) | null =
    null;

  /**
   * 设置模块暂停/恢复/重启及警告回调
   */
  setCallbacks(
    onPause: (moduleId: string, reason: string) => void,
    onResume: (moduleId: string) => void,
    onRestart?: (moduleId: string, reason: string) => void,
    onWarning?: (event: { moduleId: string; resource: string; current: number; limit: number }) => void
  ): void {
    this.onModulePause = onPause;
    this.onModuleResume = onResume;
    if (onRestart) this.onModuleRestart = onRestart;
    if (onWarning) this.onWarning = onWarning;
  }

  /**
   * 注册模块资源声明
   */
  register(declaration: ModuleResourceDeclaration): void {
    this.declarations.set(declaration.moduleId, declaration);
    if (!this.states.has(declaration.moduleId)) {
      this.states.set(declaration.moduleId, {
        moduleId: declaration.moduleId,
        currentMemoryMB: 0,
        currentCpuPercent: 0,
        uptimeSeconds: 0,
        isPaused: false,
        pauseReason: null,
      });
    }
  }

  /**
   * 注销模块
   */
  unregister(moduleId: string): void {
    this.declarations.delete(moduleId);
    this.states.delete(moduleId);
  }

  /**
   * 更新模块资源使用数据
   */
  updateModuleUsage(
    moduleId: string,
    usage: { memoryMB: number; cpuPercent: number; uptimeSeconds: number }
  ): void {
    const state = this.states.get(moduleId);
    if (!state) return;

    this.states.set(moduleId, {
      ...state,
      currentMemoryMB: usage.memoryMB,
      currentCpuPercent: usage.cpuPercent,
      uptimeSeconds: usage.uptimeSeconds,
    });
  }

  /**
   * 根据降频等级执行模块控制策略
   */
  applyThrottleLevel(
    level: ThrottleLevel,
    _snapshot: SystemResourceSnapshot
  ): void {
    switch (level) {
      case ThrottleLevel.FULL:
        this.resumeAll();
        break;

      case ThrottleLevel.AWARE:
        // 感知模式下不停止模块，仅增加采集间隔
        break;

      case ThrottleLevel.DEGRADED:
        this.pausePerceptualModules('系统资源紧张，已暂停感知模块');
        break;

      case ThrottleLevel.LOW_POWER:
        this.pauseLowPriorityModules(5, '低功耗模式，仅保留高优先级模块');
        break;
    }
  }

  /**
   * 暂停所有感知模块
   */
  private pausePerceptualModules(reason: string): void {
    for (const [id, declaration] of this.declarations) {
      if (declaration.isPerceptual) {
        this.pauseModule(id, reason);
      }
    }
  }

  /**
   * 暂停优先级低于阈值的模块
   */
  private pauseLowPriorityModules(
    priorityThreshold: number,
    reason: string
  ): void {
    for (const [id, declaration] of this.declarations) {
      if (declaration.priority < priorityThreshold) {
        this.pauseModule(id, reason);
      }
    }
  }

  /**
   * 暂停指定模块
   */
  private pauseModule(moduleId: string, reason: string): void {
    const state = this.states.get(moduleId);
    if (!state || state.isPaused) return;

    this.states.set(moduleId, {
      ...state,
      isPaused: true,
      pauseReason: reason,
    });

    this.onModulePause?.(moduleId, reason);
  }

  /**
   * 恢复所有暂停的模块
   */
  private resumeAll(): void {
    for (const [id, state] of this.states) {
      if (state.isPaused) {
        this.states.set(id, {
          ...state,
          isPaused: false,
          pauseReason: null,
        });
        this.onModuleResume?.(id);
      }
    }
  }

  /**
   * 获取所有模块状态
   */
  getStates(): Readonly<Record<string, ModuleResourceState>> {
    const result: Record<string, ModuleResourceState> = {};
    for (const [id, state] of this.states) {
      result[id] = { ...state };
    }
    return result;
  }

  /**
   * 检查模块是否超过资源上限，并自动处理违规情况
   * - 超过内存限制时发送警告事件
   * - 超过 CPU 限制时降低模块优先级（暂停低优先级模块）
   * - 持续超限达到阈值时触发模块重启
   */
  checkResourceLimits(): Array<{
    moduleId: string;
    resource: string;
    current: number;
    limit: number;
  }> {
    const violations: Array<{
      moduleId: string;
      resource: string;
      current: number;
      limit: number;
    }> = [];

    for (const [id, declaration] of this.declarations) {
      const state = this.states.get(id);
      if (!state) continue;

      const moduleViolations: Array<{ resource: string; current: number; limit: number }> = [];

      if (state.currentMemoryMB > declaration.maxMemoryMB) {
        moduleViolations.push({
          resource: 'memory',
          current: state.currentMemoryMB,
          limit: declaration.maxMemoryMB,
        });
      }

      if (state.currentCpuPercent > declaration.maxCpuPercent) {
        moduleViolations.push({
          resource: 'cpu',
          current: state.currentCpuPercent,
          limit: declaration.maxCpuPercent,
        });
      }

      if (moduleViolations.length === 0) {
        // 无违规，重置连续计数
        this.consecutiveViolationCount.delete(id);
        continue;
      }

      // 记录违规
      for (const v of moduleViolations) {
        violations.push({ moduleId: id, ...v });
      }

      // 累加连续违规计数
      const prevCount = this.consecutiveViolationCount.get(id) ?? 0;
      const newCount = prevCount + 1;
      this.consecutiveViolationCount.set(id, newCount);

      // === 自动处理逻辑 ===

      // 1. 内存超限：发送警告事件
      const memoryViolation = moduleViolations.find(v => v.resource === 'memory');
      if (memoryViolation && this.onWarning) {
        this.onWarning({
          moduleId: id,
          resource: 'memory',
          current: memoryViolation.current,
          limit: memoryViolation.limit,
        });
      }

      // 2. CPU 超限：降低模块优先级（暂停该模块以释放 CPU）
      const cpuViolation = moduleViolations.find(v => v.resource === 'cpu');
      if (cpuViolation && this.onModulePause) {
        this.onModulePause(
          id,
          `[ResourceController] CPU 超限 ${cpuViolation.current.toFixed(1)}% > ${cpuViolation.limit}%，暂停模块以降低优先级`
        );
      }

      // 3. 持续超限：触发模块重启
      if (newCount >= this.MAX_CONSECUTIVE_VIOLATIONS && this.onModuleRestart) {
        const resources = moduleViolations.map(v => `${v.resource}(${v.current.toFixed(1)}>${v.limit})`).join(', ');
        this.onModuleRestart(
          id,
          `[ResourceController] 连续 ${newCount} 次超限 [${resources}]，触发重启`
        );
        // 重启后重置计数
        this.consecutiveViolationCount.delete(id);
      }
    }

    return violations;
  }
}
```

```typescript
import { EventEmitter } from 'events';
import type {

const DEFAULT_CONFIG: PerformanceMonitorConfig = {

    degradedThreshold: 80,
    lowPowerThreshold: 90,
    awareIntervalMultiplier: 2,

/**

 */
export class PerformanceMonitor
  extends EventEmitter
  implements IPerformanceMonitor
{
  private readonly config: PerformanceMonitorConfig;
  private readonly collector: ResourceCollector;
  private readonly strategy: ThrottleStrategy;
  private readonly moduleController: ModuleResourceController;

  private history: SystemResourceSnapshot[] = [];
  private collectTimer: ReturnType<typeof setInterval> | null = null;
  private currentIntervalMs: number;
  private isRunning = false;

  constructor(config?: Partial<PerformanceMonitorConfig>) {
    super();

    this.config = { ...DEFAULT_CONFIG, ...config };
    this.currentIntervalMs = this.config.collectIntervalMs;

    this.collector = new ResourceCollector();
    this.strategy = new ThrottleStrategy(this.config.throttle);
    this.moduleController = new ModuleResourceController();

    // 设置模块控制回调（含资源超限自动处理：警告、降优先级、重启）
    this.moduleController.setCallbacks(
      (moduleId, reason) => {
        this.emitEvent({
          type: 'module-paused',
          data: { moduleId, reason },
        });
      },
      (moduleId) => {
        this.emitEvent({
          type: 'module-resumed',
          data: { moduleId },
        });
      },
      (moduleId, reason) => {
        this.emitEvent({
          type: 'module-restart',
          data: { moduleId, reason },
        });
        console.warn(`[PerformanceMonitor] 模块 ${moduleId} 触发重启: ${reason}`);
      },
      (event) => {
        this.emitEvent({
          type: 'module-memory-warning',
          data: event,
        });
        console.warn(
          `[PerformanceMonitor] 模块 ${event.moduleId} 内存超限警告: ${event.current.toFixed(1)}MB > ${event.limit}MB`
        );
      }
    );
  }

  /**
   * 启动性能监控
   */
  start(): void {
    if (this.isRunning) return;
    this.isRunning = true;

    // 立即执行一次采集
    void this.collect();

    // 启动定时采集
    this.scheduleNextCollection();

    console.log(
      `[PerformanceMonitor] 已启动，采集间隔 ${this.currentIntervalMs}ms`
    );
  }

  /**
   * 停止性能监控
   */
  stop(): void {
    if (!this.isRunning) return;
    this.isRunning = false;

    if (this.collectTimer) {
      clearTimeout(this.collectTimer);
      this.collectTimer = null;
    }

    console.log('[PerformanceMonitor] 已停止');
  }

  /**
   * 获取最新的资源快照
   */
  getLatestSnapshot(): SystemResourceSnapshot | null {
    return this.history.length > 0
      ? this.history[this.history.length - 1]
      : null;
  }

  /**
   * 获取历史数据
   */
  getHistory(): readonly SystemResourceSnapshot[] {
    return [...this.history];
  }

  /**
   * 将历史数据导出为文件
   * @param format - 导出格式：'json' | 'csv'
   * @param filePath - 导出文件路径
   * @param limit - 可选，导出最近 N 条数据
   * @returns 导出的记录数

    const path = require('path');
    const data = limit && limit > 0 ? this.history.slice(-limit) : [...this.history];

    // 确保输出目录存在
    const dir = path.dirname(filePath);
    if (!fs.existsSync(dir)) {
      fs.mkdirSync(dir, { recursive: true });
    }

    if (format === 'json') {
      const output = {
        exportedAt: new Date().toISOString(),
        recordCount: data.length,
        snapshots: data,
      };
      fs.writeFileSync(filePath, JSON.stringify(output, null, 2), 'utf-8');
    } else {
      // CSV 格式
      const headers = [
        'timestamp',
        'cpuPercent',
        'memoryPercent',
        'totalMemoryMB',
        'usedMemoryMB',
        'diskPercent',
        'gpuPercent',
        'gpuMemoryPercent',
      ];
      const rows = data.map((s) =>
        headers.map((h) => String(s[h as keyof SystemResourceSnapshot])).join(',')
      );
      const csv = [headers.join(','), ...rows].join('\n');
      fs.writeFileSync(filePath, csv, 'utf-8');
    }

    console.log(`[PerformanceMonitor] 已导出 ${data.length} 条历史数据到 ${filePath} (${format})`);
    return data.length;
  }

  /**
   * 获取当前降频等级
   */
  getCurrentThrottleLevel(): ThrottleLevel {
    return this.strategy.getCurrentLevel();
  }

  /**
   * 注册模块资源声明
   */
  registerModule(declaration: ModuleResourceDeclaration): void {
    this.moduleController.register(declaration);
    console.log(
      `[PerformanceMonitor] 注册模块: ${declaration.moduleId} (优先级: ${declaration.priority})`
    );
  }

  /**
   * 注销模块
   */
  unregisterModule(moduleId: string): void {
    this.moduleController.unregister(moduleId);
  }

  /**
   * 获取所有模块状态
   */
  getModuleStates(): Readonly<Record<string, ModuleResourceState>> {
    return this.moduleController.getStates();
  }

  /**
   * 监听性能事件
   */
  onEvent(callback: (event: PerformanceEvent) => void): () => void {
    this.on('performance-event', callback);
    return () => {
      this.off('performance-event', callback);
    };
  }

  /**
   * 执行一次资源采集
   */
  private async collect(): Promise<void> {
    try {
      const snapshot = await this.collector.collect();

      // 保存历史
      this.history.push(snapshot);
      if (this.history.length > this.config.historySize) {
        this.history = this.history.slice(-this.config.historySize);
      }

      // 发出快照事件
      this.emitEvent({ type: 'snapshot', data: snapshot });

      // 检查警告阈值
      this.checkWarningThresholds(snapshot);

      // 评估降频策略
      const oldLevel = this.strategy.getCurrentLevel();
      const result = this.strategy.evaluate(snapshot);
      if (result.changed) {
        this.emitEvent({
          type: 'throttle-change',
          data: {
            from: oldLevel,
            to: result.level,
          },
        });
        console.log(
          `[PerformanceMonitor] 降频等级变更: ${result.level} (${result.reason})`
        );

        // 更新采集间隔
        this.currentIntervalMs = this.strategy.getCollectInterval(
          this.config.collectIntervalMs
        );
      }

      // 应用模块控制策略
      this.moduleController.applyThrottleLevel(result.level, snapshot);

      // 检查模块资源超限（checkResourceLimits 内部已自动处理：警告事件、降低优先级、持续超限重启）
      const violations = this.moduleController.checkResourceLimits();
      for (const violation of violations) {
        this.emitEvent({
          type: 'module-resource-violation',
          moduleId: violation.moduleId,
          resource: violation.resource,
          current: violation.current,
          limit: violation.limit,
          timestamp: Date.now(),
        } as PerformanceEvent);
      }
    } catch (error) {
      console.error('[PerformanceMonitor] 采集失败:', error);
    }
  }

  /**
   * 安排下一次采集
   */
  private scheduleNextCollection(): void {
    if (!this.isRunning) return;

    this.collectTimer = setTimeout(() => {
      void this.collect().then(() => {
        this.scheduleNextCollection();
      });
    }, this.currentIntervalMs);
  }

  /**
   * 检查资源警告阈值
   */
  private checkWarningThresholds(snapshot: SystemResourceSnapshot): void {
    const { warningThresholds } = this.config;

    if (snapshot.cpuPercent > warningThresholds.cpu) {
      this.emitEvent({
        type: 'resource-warning',
        data: {
          resource: 'cpu',
          value: snapshot.cpuPercent,
          threshold: warningThresholds.cpu,
        },
      });
    }

    if (snapshot.memoryPercent > warningThresholds.memory) {
      this.emitEvent({
        type: 'resource-warning',
        data: {
          resource: 'memory',
          value: snapshot.memoryPercent,
          threshold: warningThresholds.memory,
        },
      });
    }

    if (snapshot.diskPercent > warningThresholds.disk) {
      this.emitEvent({
        type: 'resource-warning',
        data: {
          resource: 'disk',
          value: snapshot.diskPercent,
          threshold: warningThresholds.disk,
        },
      });
    }

    if (
      snapshot.gpuPercent > 0 &&
      snapshot.gpuPercent > warningThresholds.gpu
    ) {
      this.emitEvent({
        type: 'resource-warning',
        data: {
          resource: 'gpu',
          value: snapshot.gpuPercent,
          threshold: warningThresholds.gpu,
        },
      });
    }
  }

  /**
   * 发出事件
   */
  private emitEvent(event: PerformanceEvent): void {
    this.emit('performance-event', event);
  }

  /**
   * 基于最近 N 个采样点进行线性回归，预测资源使用趋势
   *
   * 使用最小二乘法对历史采样数据进行线性拟合，计算斜率和 R² 决定系数，
   * 从而判断资源使用趋势方向，并预估资源耗尽时间。
   *
   * @param resource - 资源类型
   * @param sampleCount - 回归采样点数量（默认 30）
   * @param predictionWindowSeconds - 预测窗口（秒，默认 300）
   * @returns 趋势预测结果，数据不足时返回 null
   */
  predictTrend(
    resource: 'cpu' | 'memory' | 'disk' | 'gpu',
    sampleCount = 30,
    predictionWindowSeconds = 300
  ): TrendPrediction | null {
    const history = this.history;
    if (history.length < 2) return null;

    // 取最近 N 个采样点
    const samples = history.slice(-Math.min(sampleCount, history.length));
    const n = samples.length;
    if (n < 2) return null;

    // 提取资源值序列
    const getResourceValue = (s: SystemResourceSnapshot): number => {
      switch (resource) {
        case 'cpu': return s.cpuPercent;
        case 'memory': return s.memoryPercent;
        case 'disk': return s.diskPercent;
        case 'gpu': return s.gpuPercent >= 0 ? s.gpuPercent : 0;
      }
    };

    // 构建时间-值序列（时间转为相对秒数）
    const t0 = samples[0].timestamp;
    const xs: number[] = samples.map((s) => (s.timestamp - t0) / 1000);
    const ys: number[] = samples.map(getResourceValue);

    // 最小二乘法线性回归: y = slope * x + intercept
    let sumX = 0, sumY = 0, sumXY = 0, sumXX = 0;
    for (let i = 0; i < n; i++) {
      sumX += xs[i];
      sumY += ys[i];
      sumXY += xs[i] * ys[i];
      sumXX += xs[i] * xs[i];
    }

    const denominator = n * sumXX - sumX * sumX;
    // 所有采样点时间戳相同（极端情况），无法拟合
    if (denominator === 0) return null;

    const slope = (n * sumXY - sumX * sumY) / denominator;
    const intercept = (sumY - slope * sumX) / n;

    // 计算 R² 决定系数
    const yMean = sumY / n;
    let ssTot = 0, ssRes = 0;
    for (let i = 0; i < n; i++) {
      const predicted = slope * xs[i] + intercept;
      ssTot += (ys[i] - yMean) ** 2;
      ssRes += (ys[i] - predicted) ** 2;
    }
    const rSquared = ssTot > 0 ? 1 - ssRes / ssTot : 1;

    // 当前值与预测值
    const currentValue = ys[n - 1];
    const lastRelativeTime = xs[n - 1];
    const predictedValueAtSeconds =
      slope * (lastRelativeTime + predictionWindowSeconds) + intercept;

    // 判断趋势方向（斜率阈值：每秒变化 < 0.01% 视为稳定）
    const SLOPE_THRESHOLD = 0.01;
    let direction: TrendDirection;
    if (slope > SLOPE_THRESHOLD) {
      direction = TrendDirection.RISING;
    } else if (slope < -SLOPE_THRESHOLD) {
      direction = TrendDirection.DECLINING;
    } else {
      direction = TrendDirection.STABLE;
    }

    // 预估资源耗尽时间（斜率 > 0 时，计算到达 100% 的时间）
    let estimatedExhaustionSeconds = -1;
    if (slope > SLOPE_THRESHOLD && currentValue < 100) {
      const secondsToFull = (100 - currentValue) / slope;
      // 转换为从"现在"起算的秒数（而非从第一个采样点起算）
      estimatedExhaustionSeconds = Math.max(
        0,
        secondsToFull - lastRelativeTime
      );
    }

    const prediction: TrendPrediction = {
      resource,
      direction,
      slopePerSecond: Math.round(slope * 10000) / 10000,
      currentValue: Math.round(currentValue * 10) / 10,
      predictedValueAtSeconds:
        Math.round(Math.max(0, Math.min(100, predictedValueAtSeconds)) * 10) / 10,
      predictionWindowSeconds,
      estimatedExhaustionSeconds: Math.round(estimatedExhaustionSeconds),
      rSquared: Math.round(rSquared * 10000) / 10000,
      sampleCount: n,
    };

    // 趋势上升且 R² 较高时，发出趋势预警事件
    if (direction === TrendDirection.RISING && rSquared > 0.7) {
      this.emitEvent({ type: 'trend-alert', data: prediction });
    }

    return prediction;
  }
}
```

#### 步骤 6：注册 IPC 接口

```typescript
// src/main/ipc/performance-ipc.ts

import { ipcMain } from 'electron';
import type { PerformanceMonitor } from '../core/performance-monitor';
import type { ThrottleLevel } from '../core/types/performance';

/**
 * 注册性能监控相关的 IPC 处理器
 */
export function registerPerformanceIPC(monitor: PerformanceMonitor): void {
  /**
   * 获取最新资源快照
   */
  ipcMain.handle('performance:snapshot', () => {
    const snapshot = monitor.getLatestSnapshot();
    const moduleStates = monitor.getModuleStates();

    if (!snapshot) return null;

    return {
      ...snapshot,
      modules: moduleStates,
    };
  });

  /**
   * 获取历史数据
   */
  ipcMain.handle('performance:history', (_event, limit?: number) => {
    const history = monitor.getHistory();
    if (limit && limit > 0) {
      return history.slice(-limit);
    }
    return history;
  });

  /**
   * 获取当前降频等级
   */
  ipcMain.handle('performance:throttle-level', (): ThrottleLevel => {
    return monitor.getCurrentThrottleLevel();
  });

  /**
   * 获取模块状态
   */
  ipcMain.handle('performance:module-states', () => {
    return monitor.getModuleStates();
  });

  /**
   * 注册模块资源声明
   */
  ipcMain.handle(
    'performance:register-module',
    (_event, declaration) => {
      monitor.registerModule(declaration);
      return true;
    }
  );

  /**
   * 注销模块
   */
  ipcMain.handle(
    'performance:unregister-module',
    (_event, moduleId: string) => {
      monitor.unregisterModule(moduleId);
      return true;
    }
  );

  /**
   * 导出历史数据为文件
   * 支持 JSON 和 CSV 两种格式
   * @param format - 导出格式：'json' | 'csv'
   * @param filePath - 导出文件的完整路径
   * @param limit - 可选，导出最近 N 条数据
   * @returns { success: boolean; recordCount: number; filePath: string }
   *
   * 渲染进程调用示例:
   * window.electronAPI.invoke('performance:export-history', {
   *   format: 'json',
   *   filePath: '/path/to/export.json',
   *   limit: 100
   * })
   */
  ipcMain.handle(
    'performance:export-history',
    async (
      _event,
      params: { format: 'json' | 'csv'; filePath: string; limit?: number }
    ): Promise<{ success: boolean; recordCount: number; filePath: string }> => {
      try {
        const { format, filePath, limit } = params;

        // 验证格式
        if (format !== 'json' && format !== 'csv') {
          return { success: false, recordCount: 0, filePath };
        }

        // 验证路径安全性：不允许覆盖关键系统文件
        const path = require('path');
        const resolvedPath = path.resolve(filePath);
        const forbidden = ['/etc', '/usr', '/bin', '/sbin', '/boot'];
        if (forbidden.some((p) => resolvedPath.startsWith(p))) {
          console.warn(`[PerformanceIPC] 拒绝导出到系统目录: ${resolvedPath}`);
          return { success: false, recordCount: 0, filePath };
        }

        const recordCount = monitor.exportHistory(format, resolvedPath, limit);
        return { success: true, recordCount, filePath: resolvedPath };
      } catch (err) {
        console.error('[PerformanceIPC] 导出历史数据失败:', err);
        return { success: false, recordCount: 0, filePath: params.filePath };
      }
    }
  );

  /**
   * 资源趋势预测
   * 基于最近 N 个采样点的线性回归，预测资源使用趋势和预估耗尽时间
   *
   * @param resource - 资源类型：'cpu' | 'memory' | 'disk' | 'gpu'
   * @param sampleCount - 采样点数量（默认 30）
   * @param predictionWindowSeconds - 预测窗口秒数（默认 300）
   * @returns 趋势预测结果或 null（数据不足时）
   *
   * 渲染进程调用示例:
   * const trend = await window.electronAPI.invoke(
   *   'performance:predict-trend',
   *   { resource: 'memory', sampleCount: 60, predictionWindowSeconds: 600 }
   * );
   * if (trend && trend.direction === 'rising' && trend.estimatedExhaustionSeconds < 3600) {
   *   showWarning(`内存将在 ${Math.round(trend.estimatedExhaustionSeconds / 60)} 分钟后耗尽`);
   * }
   */
  ipcMain.handle(
    'performance:predict-trend',
    (
      _event,
      params: {
        resource: 'cpu' | 'memory' | 'disk' | 'gpu';
        sampleCount?: number;
        predictionWindowSeconds?: number;
      }
    ) => {
      return monitor.predictTrend(
        params.resource,
        params.sampleCount,
        params.predictionWindowSeconds
      );
    }
  );

  console.log('[PerformanceIPC] 已注册性能监控 IPC 接口');
}
```

### 26.3 代码规范

1. **采集间隔**：默认每 5 秒采集一次，降频时自动延长间隔
2. **四级策略**：
   - `< 60%`：全功能模式，所有模块正常运行
   - `60-80%`：感知模式，采集间隔翻倍
   - `> 80%`：降级模式，暂停感知模块
   - `> 90%`：低功耗模式，仅保留高优先级模块
3. **冷却机制**：降级后需经过 30 秒冷却时间才能恢复到更高等级
4. **模块资源声明**：所有模块必须声明最大内存、CPU 和优先级
5. **GPU 检测**：首次运行自动检测 GPU 厂商（NVIDIA → AMD → Intel），不可用时返回 -1 而非报错
6. **历史保留**：默认保留 30 分钟历史数据（360 条 × 5 秒）
7. **事件驱动**：使用 EventEmitter 发布性能事件，渲染进程通过 IPC 订阅
8. **错误隔离**：采集失败不中断监控循环，仅记录错误日志
9. **类型安全**：所有 IPC 接口使用 TypeScript 类型声明，事件类型使用联合类型
10. **资源超限检测**：模块超过声明上限时记录警告，由模块管理器决定是否降级
11. **趋势预测**：`predictTrend` 方法基于最近 N 个采样点的最小二乘线性回归，预测资源使用趋势方向（上升/稳定/下降）和预估耗尽时间；当趋势上升且 R² > 0.7 时自动发出 `trend-alert` 事件；默认采样 30 个点，预测窗口 300 秒
