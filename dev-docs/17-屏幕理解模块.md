export class ScreenCaptureService {
  private logger: Logger
  /** Issue#275: 权限状态缓存，避免每次截图都重复检测系统权限 */
  private permissionGranted: boolean | null = null

  constructor(logger: Logger) {
    this.logger = logger
  }

  /**
   * Issue#275: 截屏前的权限预检
   * 首次调用时探测并缓存权限状态；权限被拒绝时抛出明确错误而非静默失败
   */
  private async ensurePermission(): Promise<void> {
    if (this.permissionGranted === true) return

    if (process.platform === 'darwin') {
      const { systemPreferences } = await import('electron')
      const status = systemPreferences.getMediaAccessStatus('screen')
      if (status === 'granted') {
        this.permissionGranted = true
        return
      }
      this.permissionGranted = false
      throw new Error(
        `屏幕录制权限未授予（状态: ${status}）。请前往「系统偏好设置 → 隐私与安全性 → 屏幕录制」中授权。`
      )
    }

    // Windows / Linux：通过 desktopCapturer 探测（不抛错，降级处理）
    this.permissionGranted = true
  }

  /** 截取整个屏幕 */
  async captureScreen(): Promise<CaptureResult> {
    // Issue#275: 截屏前权限预检
    await this.ensurePermission()

    const sources = await desktopCapturer.getSources({
      types: ['screen'],
      thumbnailSize: { width: 1920, height: 1080 }
    })

    if (sources.length === 0) {
      this.permissionGranted = null // 重置缓存，下次重新检测
      throw new Error('未找到可用的屏幕源。请检查屏幕录制权限是否已授予。')
    }

    const source = sources[0]
    const thumbnail = source.thumbnail

    return {
      dataUrl: thumbnail.toDataURL(),
      width: thumbnail.getSize().width,
      height: thumbnail.getSize().height,
      timestamp: Date.now()
    }
  }

  /** 截取指定窗口 */
  async captureWindow(windowId: number): Promise<CaptureResult> {
    // Issue#275: 截屏前权限预检
    await this.ensurePermission()

    const sources = await desktopCapturer.getSources({
      types: ['window'],
      thumbnailSize: { width: 1920, height: 1080 }
    })

    const source = sources.find((s) => parseInt(s.id) === windowId)
    if (!source) {
      throw new Error(`未找到窗口 ID: ${windowId}`)
    }

    const thumbnail = source.thumbnail

    return {
      dataUrl: thumbnail.toDataURL(),
      width: thumbnail.getSize().width,
      height: thumbnail.getSize().height,
      timestamp: Date.now()
    }
  }

  /** 获取可用的屏幕源列表 */
  async getSources(): Promise<Array<{ id: string; name: string; type: string }>> {
    const sources = await desktopCapturer.getSources({
      types: ['screen', 'window'],
      thumbnailSize: { width: 320, height: 240 }
    })

    return sources.map((s) => ({
      id: s.id,
      name: s.name,
      type: s.display_id ? 'screen' : 'window'
    }))
  }
}

#### 步骤 3：实现 OCR 服务

**modules/screen/services/ocr-service.ts**：

> **⚠️ tesseract.js 版本要求：`^5.0.0`（v5.x）**
>
> 本模块基于 **tesseract.js v5** API 编写。v4 与 v5 存在以下关键差异，**不可混用**：
>
> | 特性 | v4.x | v5.x（本项目使用） |
> |------|------|---------------------|
> | 创建 Worker | `Tesseract.createWorker()` → `worker.loadLanguage()` → `worker.initialize()`（三步） | `Tesseract.createWorker('chi_sim+eng')`（一步完成） |
> | 识别调用 | `worker.recognize(image)` 返回 `Promise<Result>` | 同 v4，但 `worker` 创建方式不同 |
> | 终止 Worker | `worker.terminate()` | 同 v4 |
> | 语言数据 | 需手动 `loadLanguage` + `initialize` | 创建时自动加载 |
> | 安装命令 | `npm install tesseract.js@4` | `npm install tesseract.js@5` |
>
> 如果项目中已使用 v4，请参照 [v4→v5 迁移指南](https://github.com/naptha/tesseract.js/blob/main/docs/migration-v5.md) 升级。

```typescript
import Tesseract from 'tesseract.js' // 要求 tesseract.js ^5.0.0
import type { Logger } from '../../../src/main/types/logger'

/** Tesseract.js v5 API 兼容 */
export interface OCRResult {
  text: string
  confidence: number
  words: Array<{ text: string; confidence: number; bbox: { x0: number; y0: number; x1: number; y1: number } }>
}

export class OCRService {
  private worker: Tesseract.Worker | null = null
  private logger: Logger
  /** Issue#280: Worker 预初始化 Promise，支持模块启动时提前加载语言数据 */
  private initPromise: Promise<void> | null = null

  constructor(logger: Logger) {
    this.logger = logger
  }

  async initialize(): Promise<void> {
    if (this.worker) return // 已初始化，跳过
    if (this.initPromise) return this.initPromise // 防止重复初始化
    this.initPromise = this.doInitialize()
    return this.initPromise
  }

  private async doInitialize(): Promise<void> {
    this.worker = await Tesseract.createWorker('chi_sim+eng')
    this.logger.info('OCR 引擎已初始化')
  }

  /**
   * Issue#280: 预热 Worker —— 在模块激活时调用，提前下载语言数据并初始化 Worker，
   * 避免首次 recognize() 调用时的冷启动延迟。
   * 不阻塞模块激活流程，失败时仅 warn 日志降级。
   */
  async warmup(): Promise<void> {
    try {
      await this.initialize()
      this.logger.info('OCR Worker 预热完成')
    } catch (err) {
      this.logger.warn('OCR Worker 预热失败，将在首次识别时重试', { error: err })
    }
  }

  async recognize(imageBase64: string): Promise<OCRResult> {
    if (!this.worker) {
      await this.initialize()
    }

    const result = await this.worker!.recognize(imageBase64)

    return {
      text: result.data.text,
      confidence: result.data.confidence,
      words: result.data.words.map((w) => ({
        text: w.text,
        confidence: w.confidence,
        bbox: w.bbox
      }))
    }
  }

  async terminate(): Promise<void> {
    await this.worker?.terminate()
    this.worker = null
    this.initPromise = null // Issue#280: 重置初始化状态，支持重新初始化
  }
}
```

export class ScreenAnalyzer {
  /** 分析结果缓存（避免短时间内重复分析同一画面） */
  private lastResult: AnalysisResult | null = null
  private lastCaptureTimestamp = 0
  /** 缓存有效期（毫秒），同一秒内的截图直接返回缓存 */
  private readonly CACHE_TTL_MS = 1000

  constructor(
    private capture: ScreenCaptureService,
    private ocr: OCRService,
    private ai: AIService,
    private logger: Logger
  ) {}

  /**
   * 分析屏幕内容（核心方法）
   *
   * 执行流程：
   * 1. 截取屏幕 → 2. 图片压缩 → 3. OCR 文字提取
   * → 4. 隐私脱敏 → 5. AI 视觉分析 → 6. UI 元素提取
   * → 7. 布局分析 → 8. 组装结果
   */
  async analyze(options: AnalysisOptions = {}): Promise<AnalysisResult> {
    const startTime = Date.now()
    const {
      question,
      skipOCR = false,
      skipAI = false,
      maxImageDimension = 1024,
      timeoutMs = 30000,
      region,
      includeWordLevelOCR = false
    } = options

    // 超时保护
    const timeoutPromise = new Promise<never>((_, reject) => {
      setTimeout(() => reject(new Error(`屏幕分析超时 (${timeoutMs}ms)`)), timeoutMs)
    })

    try {
      return await Promise.race([
        this.doAnalyze({ question, skipOCR, skipAI, maxImageDimension, region, includeWordLevelOCR, startTime }),
        timeoutPromise
      ])
    } catch (err) {
      this.logger.error('屏幕分析失败', err as Error)
      // 返回最小可用结果，避免上层调用崩溃
      return {
        summary: `分析失败: ${(err as Error).message}`,
        elements: [],
        layout: [],
        resolution: { width: 0, height: 0 },
        duration: Date.now() - startTime,
        timestamp: Date.now(),
        hasSensitiveContent: false
      }
    }
  }

  /** 内部分析实现 */
  private async doAnalyze(params: {
    question?: string
    skipOCR: boolean
    skipAI: boolean
    maxImageDimension: number
    region?: { x: number; y: number; width: number; height: number }
    includeWordLevelOCR: boolean
    startTime: number
  }): Promise<AnalysisResult> {
    const { question, skipOCR, skipAI, maxImageDimension, region, includeWordLevelOCR, startTime } = params

    // 1. 截取屏幕
    this.logger.debug('开始屏幕分析: 截取屏幕')
    const screenshot = await this.capture.captureScreen()

    // 缓存检查：同一秒内的相同截图直接返回缓存
    if (this.lastResult && screenshot.timestamp - this.lastCaptureTimestamp < this.CACHE_TTL_MS) {
      this.logger.debug('命中分析缓存，跳过重复分析')
      return { ...this.lastResult, duration: Date.now() - startTime }
    }

    // 2. 图片压缩（减少 AI 传输大小，提升分析速度）
    const compressedDataUrl = await this.compressImage(screenshot.dataUrl, maxImageDimension)

    // 3. 并行执行 OCR 和 AI 分析（提高效率）
    let ocrResult: OCRResult | null = null
    let aiResponse = ''

    const tasks: Promise<void>[] = []

    if (!skipOCR) {
      tasks.push(
        this.ocr.recognize(compressedDataUrl)
          .then(result => { ocrResult = result })
          .catch(err => {
            this.logger.warn('OCR 识别失败，继续进行 AI 分析', { error: err })
          })
      )
    }

    if (!skipAI) {
      const prompt = this.buildPrompt(question, region)
      tasks.push(
        this.ai.analyzeImage(compressedDataUrl, prompt)
          .then(response => { aiResponse = response })
          .catch(err => {
            this.logger.warn('AI 视觉分析失败', { error: err })
            aiResponse = 'AI 分析不可用'
          })
      )
    }

    await Promise.all(tasks)

    // 4. 隐私脱敏
    const rawOcrText = ocrResult?.text ?? ''
    const hasSensitive = detectSensitiveContent(rawOcrText) || detectSensitiveContent(aiResponse)
    const sanitizedOcrText = rawOcrText ? this.sanitizeText(rawOcrText) : undefined
    const sanitizedSummary = aiResponse ? this.sanitizeText(aiResponse) : '未进行 AI 分析'

    // 5. 提取 UI 元素（结合 OCR 位置信息和 AI 分析结果）
    const elements = this.extractElements(sanitizedSummary, ocrResult)

    // 6. 布局分析
    const layout = this.analyzeLayout(sanitizedSummary, elements)

    // 7. 检测活跃应用
    const activeApp = this.detectActiveApp(sanitizedOcrText, sanitizedSummary)

    const result: AnalysisResult = {
      summary: sanitizedSummary,
      activeApp,
      ocrText: sanitizedOcrText,
      ocrWords: includeWordLevelOCR ? ocrResult?.words : undefined,
      elements,
      layout,
      resolution: { width: screenshot.width, height: screenshot.height },
      duration: Date.now() - startTime,
      timestamp: Date.now(),
      hasSensitiveContent: hasSensitive
    }

    // 更新缓存
    this.lastResult = result
    this.lastCaptureTimestamp = screenshot.timestamp

    this.logger.debug(`屏幕分析完成: ${result.duration}ms, 元素${elements.length}个, 布局区域${layout.length}个`)
    return result
  }

  /** 构建 AI 分析提示词 */
  private buildPrompt(question?: string, region?: { x: number; y: number; width: number; height: number }): string {
    const basePrompt = question || '请详细描述屏幕上的内容'
    const regionHint = region
      ? `（请重点关注屏幕坐标 (${Math.round(region.x * 100)}%, ${Math.round(region.y * 100)}%) 到 (${Math.round((region.x + region.width) * 100)}%, ${Math.round((region.y + region.height) * 100)}%) 区域）`
      : ''

    return `${basePrompt}${regionHint}

请按以下格式输出：
1. 【摘要】简要描述屏幕内容（1-2句话）
2. 【应用】当前活跃的应用或窗口名称
3. 【元素】列出可见的 UI 元素，格式为：
   - 按钮：xxx
   - 输入框：xxx
   - 链接：xxx
   - 菜单：xxx
   - 图标：xxx
   - 表格：xxx
   - 标签页：xxx
4. 【布局】描述页面布局结构（如：顶部标题栏、左侧导航、中间主内容区、底部状态栏）`
  }

  /** 图片压缩：使用 sharp 缩放到指定最大尺寸 */
  private async compressImage(dataUrl: string, maxDimension: number): Promise<string> {
    const base64Data = dataUrl.replace(/^data:image\/\w+;base64,/, '')
    const buffer = Buffer.from(base64Data, 'base64')

    // 小于 500KB 不压缩
    if (buffer.length < 500 * 1024) {
      return dataUrl
    }

    try {
      const sharp = await import('sharp')
      const metadata = await sharp.default(buffer).metadata()

      if (!metadata.width || !metadata.height) {
        return dataUrl
      }

      const scale = Math.min(maxDimension / metadata.width, maxDimension / metadata.height, 1)
      if (scale >= 1) return dataUrl

      this.logger.debug(`压缩图片: ${metadata.width}x${metadata.height} → ${Math.round(metadata.width * scale)}x${Math.round(metadata.height * scale)}`)

      const resized = await sharp.default(buffer)
        .resize(Math.round(metadata.width * scale), Math.round(metadata.height * scale), {
          fit: 'inside',
          withoutEnlargement: true
        })
        .jpeg({ quality: 80 })
        .toBuffer()

      return `data:image/jpeg;base64,${resized.toString('base64')}`
    } catch {
      this.logger.warn('图片压缩失败（sharp 不可用），使用原始图片')
      return dataUrl
    }
  }

  /** 文本脱敏：替换敏感信息 */
  private sanitizeText(text: string): string {
    let sanitized = text
    for (const { pattern, replacement } of SENSITIVE_PATTERNS) {
      sanitized = sanitized.replace(pattern, replacement)
    }
    return sanitized
  }

  /**
   * 从 AI 分析结果中提取 UI 元素信息
   * 结合 OCR 的位置信息，为元素提供更精确的定位
   */
  private extractElements(
    summary: string,
    ocrResult: OCRResult | null
  ): UIElement[] {
    const elements: UIElement[] = []

    // UI 元素模式匹配（中英文双语支持）
    const patterns: Array<{ type: UIElementType; regex: RegExp; interactive: boolean }> = [
      { type: 'button',   regex: /(?:按钮|Button)[：:]\s*([^\n,，。.]+)/gi, interactive: true },
      { type: 'input',    regex: /(?:输入框|输入字段|Input|Text Field)[：:]\s*([^\n,，。.]+)/gi, interactive: true },
      { type: 'link',     regex: /(?:链接|超链接|Link)[：:]\s*([^\n,，。.]+)/gi, interactive: true },
      { type: 'menu',     regex: /(?:菜单|下拉菜单|Menu|Dropdown)[：:]\s*([^\n,，。.]+)/gi, interactive: true },
      { type: 'checkbox', regex: /(?:复选框|勾选框|Checkbox)[：:]\s*([^\n,，。.]+)/gi, interactive: true },
      { type: 'radio',    regex: /(?:单选框|单选按钮|Radio)[：:]\s*([^\n,，。.]+)/gi, interactive: true },
      { type: 'tab',      regex: /(?:标签页|选项卡|Tab)[：:]\s*([^\n,，。.]+)/gi, interactive: true },
      { type: 'icon',     regex: /(?:图标|Icon)[：:]\s*([^\n,，。.]+)/gi, interactive: false },
      { type: 'image',    regex: /(?:图片|图像|Image)[：:]\s*([^\n,，。.]+)/gi, interactive: false },
      { type: 'heading',  regex: /(?:标题|Heading|Title)[：:]\s*([^\n,，。.]+)/gi, interactive: false },
      { type: 'toolbar',  regex: /(?:工具栏|Toolbar)[：:]\s*([^\n,，。.]+)/gi, interactive: false },
      { type: 'dialog',   regex: /(?:对话框|弹窗|Dialog|Modal|Popup)[：:]\s*([^\n,，。.]+)/gi, interactive: false },
      { type: 'table',    regex: /(?:表格|Table)[：:]\s*([^\n,，。.]+)/gi, interactive: false },
      { type: 'toggle',   regex: /(?:开关|切换|Toggle|Switch)[：:]\s*([^\n,，。.]+)/gi, interactive: true },
      { type: 'slider',   regex: /(?:滑块|滑动条|Slider)[：:]\s*([^\n,，。.]+)/gi, interactive: true },
    ]

    for (const { type, regex, interactive } of patterns) {
      let match
      while ((match = regex.exec(summary)) !== null) {
        const desc = match[1].trim()
        const element: UIElement = {
          type,
          description: desc,
          confidence: 0.7,
          interactive
        }

        // 尝试从 OCR 结果中匹配元素位置
        if (ocrResult?.words) {
          const position = this.findElementPosition(desc, ocrResult.words)
          if (position) {
            element.position = position
            element.confidence = 0.85 // 有位置匹配的元素置信度更高
          }
        }

        elements.push(element)
      }
    }

    return elements
  }

  /**
   * 根据元素描述文字，在 OCR 结果中查找对应位置
   * 使用模糊匹配处理 OCR 可能的识别误差
   */
  private findElementPosition(
    description: string,
    ocrWords: OCRResult['words']
  ): { x: number; y: number; width: number; height: number } | null {
    // 提取描述中的关键词（取前 10 个字符作为匹配目标）
    const keywords = description.replace(/[^\u4e00-\u9fa5a-zA-Z0-9]/g, '').slice(0, 10)
    if (!keywords) return null

    for (const word of ocrWords) {
      const wordText = word.text.replace(/[^\u4e00-\u9fa5a-zA-Z0-9]/g, '')
      if (wordText && (wordText.includes(keywords) || keywords.includes(wordText))) {
        // 找到匹配的 OCR 单词，返回归一化坐标
        // 注意：这里需要知道屏幕尺寸来归一化，使用 bbox 原始值
        return {
          x: word.bbox.x0,
          y: word.bbox.y0,
          width: word.bbox.x1 - word.bbox.x0,
          height: word.bbox.y1 - word.bbox.y0
        }
      }
    }
    return null
  }

  /** 布局分析：从 AI 描述和元素分布推断页面结构 */
  private analyzeLayout(summary: string, elements: UIElement[]): LayoutRegion[] {
    const layout: LayoutRegion[] = []

    // 从 AI 分析结果中提取布局描述
    const layoutPatterns: Array<{ name: string; type: LayoutRegion['type']; keywords: string[] }> = [
      { name: '标题栏', type: 'header', keywords: ['标题栏', 'title bar', 'header', '标题', '顶部'] },
      { name: '底部栏', type: 'footer', keywords: ['底部栏', '状态栏', 'footer', 'status bar', '底部'] },
      { name: '侧边栏', type: 'sidebar', keywords: ['侧边栏', '导航栏', 'sidebar', 'navigation', '侧边', '左边栏', '右边栏'] },
      { name: '工具栏', type: 'toolbar', keywords: ['工具栏', 'toolbar', '操作栏'] },
      { name: '主内容区', type: 'main', keywords: ['主内容', '正文', 'main content', '内容区', '中央'] },
      { name: '弹窗', type: 'modal', keywords: ['弹窗', '对话框', 'modal', 'dialog', 'popup'] },
      { name: '导航区', type: 'navigation', keywords: ['导航', 'navigation', '菜单栏', 'menu bar'] },
    ]

    const lowerSummary = summary.toLowerCase()
    for (const { name, type, keywords } of layoutPatterns) {
      if (keywords.some(kw => lowerSummary.includes(kw))) {
        // 统计该区域类型的元素数量
        const regionElements = elements.filter(() => true) // 简化：所有元素计入
        layout.push({
          name,
          type,
          textSummary: this.extractRegionText(summary, name),
          elementCount: regionElements.length
        })
      }
    }

    // 如果没有匹配到任何布局区域，至少返回一个"主内容区"
    if (layout.length === 0) {
      layout.push({
        name: '主内容区',
        type: 'main',
        textSummary: summary.slice(0, 200),
        elementCount: elements.length
      })
    }

    return layout
  }

  /** 提取与特定区域相关的文本片段 */
  private extractRegionText(summary: string, regionName: string): string {
    const lines = summary.split('\n')
    const relevantLines = lines.filter(line =>
      line.toLowerCase().includes(regionName.toLowerCase()) ||
      line.includes('区域') || line.includes('部分')
    )
    return relevantLines.join(' ').slice(0, 150) || summary.slice(0, 150)
  }

  /** 从 OCR 文本和 AI 描述中检测当前活跃应用 */
  private detectActiveApp(ocrText?: string, aiSummary?: string): string | undefined {
    // 常见应用标题栏特征

  }

  /**
   * 对比两次屏幕截图的变化差异
   * 用于视觉感知模块的帧变化检测
   */
  async diffWithPrevious(): Promise<{ changed: boolean; diffSummary: string }> {
    const currentScreenshot = await this.capture.captureScreen()

    if (!this.lastResult) {
      return { changed: true, diffSummary: '无历史数据，首次分析' }
    }

    // 简单的文件大小对比判断是否有变化
    const currentBase64 = currentScreenshot.dataUrl.replace(/^data:image\/\w+;base64,/, '')
    const currentSize = Buffer.from(currentBase64, 'base64').length

    // 使用 AI 进行语义级别的变化对比
    try {
      const diffPrompt = `对比当前屏幕与上次分析结果的差异。
上次分析摘要：${this.lastResult.summary}
请描述屏幕是否发生了显著变化，如果有的话具体变化是什么。`
      const diffResponse = await this.ai.analyzeImage(
        await this.compressImage(currentScreenshot.dataUrl, 1024),
        diffPrompt
      )
      const changed = !diffResponse.includes('没有变化') && !diffResponse.includes('未发生变化')
      return { changed, diffSummary: this.sanitizeText(diffResponse) }
    } catch (err) {
      this.logger.warn('屏幕差异分析失败', { error: err })
      return { changed: true, diffSummary: '差异分析失败，假设已变化' }
    }
  }

  /** 清除分析缓存 */
  clearCache(): void {
    this.lastResult = null
    this.lastCaptureTimestamp = 0
  }

  /**
   * 从 AI 分析文本中解析元素定位坐标
   * 支持从 JSON 代码块或自由文本中提取坐标信息
   */
  private parseLocations(
    text: string,
    target: string
  ): Array<{ element: string; x: number; y: number; width: number; height: number; confidence: number }> {
    const locations: Array<{ element: string; x: number; y: number; width: number; height: number; confidence: number }> = []

    // 策略 1：从 JSON 代码块中解析
    const jsonBlockMatch = text.match(/```(?:json)?\s*(\[[\s\S]*?\])\s*```/)
    if (jsonBlockMatch) {
      try {
        const parsed = JSON.parse(jsonBlockMatch[1])
        if (Array.isArray(parsed)) {
          for (const item of parsed) {
            if (this.isValidLocation(item)) {
              locations.push({
                element: item.element || target,
                x: Number(item.x),
                y: Number(item.y),
                width: Number(item.width),
                height: Number(item.height),
                confidence: Number(item.confidence) || 0.5
              })
            }
          }
          return locations
        }
      } catch { /* JSON 解析失败，继续其他策略 */ }
    }

    // 策略 2：从行内 JSON 数组中解析
    const inlineJsonMatch = text.match(/\[\s*\{[\s\S]*?\}\s*\]/)
    if (inlineJsonMatch) {
      try {
        const parsed = JSON.parse(inlineJsonMatch[0])
        if (Array.isArray(parsed)) {
          for (const item of parsed) {
            if (this.isValidLocation(item)) {
              locations.push({
                element: item.element || target,
                x: Number(item.x),
                y: Number(item.y),
                width: Number(item.width),
                height: Number(item.height),
                confidence: Number(item.confidence) || 0.5
              })
            }
          }
          return locations
        }
      } catch { /* JSON 解析失败，继续正则策略 */ }
    }

    // 策略 3：正则提取坐标模式（兜底）
    const coordPattern = /x[：:=]\s*(\d+)[,，\s]+y[：:=]\s*(\d+)[,，\s]+width[：:=]\s*(\d+)[,，\s]+height[：:=]\s*(\d+)/gi
    let match
    while ((match = coordPattern.exec(text)) !== null) {
      locations.push({
        element: target,
        x: parseInt(match[1]),
        y: parseInt(match[2]),
        width: parseInt(match[3]),
        height: parseInt(match[4]),
        confidence: 0.3
      })
    }

    return locations
  }

  /** 验证坐标对象结构是否合法 */
  private isValidLocation(item: any): boolean {
    return (
      typeof item === 'object' &&
      item !== null &&
      'x' in item && 'y' in item &&
      'width' in item && 'height' in item &&
      !isNaN(Number(item.x)) && !isNaN(Number(item.y)) &&
      !isNaN(Number(item.width)) && !isNaN(Number(item.height)) &&
      Number(item.width) > 0 && Number(item.height) > 0
    )
  }
}

### 17.4 隐私保护措施

屏幕理解模块涉及用户屏幕内容采集，必须严格执行以下隐私保护规范：

#### 17.4.1 截图脱敏

- **截图不持久化**：屏幕截图仅在内存中临时持有，分析完成后立即释放，禁止写入磁盘或日志
- **截图缓存限时**：同一秒内的重复截图使用内存缓存，缓存有效期极短（≤1秒），过期自动清除
- **截图分辨率限制**：默认最大分辨率 1920×1080，避免捕获超高清屏幕细节

#### 17.4.2 敏感信息过滤

- **OCR 文本脱敏**：OCR 识别结果经过 `sanitizeText()` 处理后才返回，自动替换以下敏感模式：
  - 邮箱地址 → `[邮箱已隐藏]`
  - 手机号码 → `[手机号已隐藏]`
  - 身份证号 → `[身份证已隐藏]`
  - 银行卡号 → `[银行卡号已隐藏]`
  - 密码/token/API Key 等字段值 → `[敏感信息已隐藏]`
- **AI 分析结果脱敏**：AI 视觉分析的返回文本同样经过 `sanitizeText()` 处理
- **敏感信息检测**：`detectSensitiveInfo()` 在脱敏前检测原始文本是否包含敏感信息，将结果标记在 `AnalysisResult.containsSensitiveInfo` 字段中，供上层决策

#### 17.4.3 数据传输安全

- 截图数据以 Data URL 形式仅在进程内传递，不通过网络发送
- AI 视觉分析请求中的图片数据，仅发送至用户配置的本地或可信 AI 服务端点

#### 17.4.4 用户知情与控制

- 首次使用屏幕理解功能时，需获取系统屏幕录制权限，操作系统会弹出授权对话框
- 用户可随时通过模块设置禁用屏幕理解模块
- 分析结果中应明确标注 `containsSensitiveInfo` 状态，便于上层 UI 展示隐私提示

import { describe, it, expect, vi, beforeEach } from 'vitest'
import { ScreenCaptureService } from '../../modules/screen/services/screen-capture'

  })

  it('应成功截取屏幕并返回 Data URL', async () => {
    const { desktopCapturer } = await import('electron')
    const mockThumbnail = {
      toDataURL: () => 'data:image/png;base64,iVBORw0KGgo=',
      getSize: () => ({ width: 1920, height: 1080 })
    }
    vi.mocked(desktopCapturer.getSources).mockResolvedValue([
      { id: 'screen:0', name: 'Screen 1', display_id: '0', thumbnail: mockThumbnail } as any
    ])

    const result = await service.captureScreen()

    expect(result.dataUrl).toBe('data:image/png;base64,iVBORw0KGgo=')
    expect(result.width).toBe(1920)
    expect(result.height).toBe(1080)
    expect(result.timestamp).toBeGreaterThan(0)
  })

    const { desktopCapturer } = await import('electron')
    vi.mocked(desktopCapturer.getSources).mockResolvedValue([])

    await expect(service.captureScreen()).rejects.toThrow('未找到可用的屏幕源')
  })

  it('macOS 权限未授予时应抛出明确错误', async () => {
    const { systemPreferences } = await import('electron')
    vi.mocked(systemPreferences.getMediaAccessStatus).mockReturnValue('denied' as any)

    expect(result.width).toBe(800)

    vi.clearAllMocks()
    analyzer = new ScreenAnalyzer(mockLogger)

    expect(withSensitive).toBe(true)
    expect(withoutSensitive).toBe(false)
  })
