```typescript
// src/main/core/security/EncryptionService.ts

import crypto from 'node:crypto';
import { EventEmitter } from 'node:events';
import type { EncryptionConfig } from './types';
import { SECURITY_CONSTANTS } from './constants';

export class EncryptionService extends EventEmitter {
  private config: EncryptionConfig;
  private masterKey: Buffer | null = null;

  constructor(config: EncryptionConfig) {
    super();
    this.config = config;
  }

  /** 初始化加密服务，设置主密钥 */
  initialize(password: string, salt?: Buffer): void {
    const effectiveSalt = salt ?? crypto.randomBytes(SECURITY_CONSTANTS.SALT_LENGTH);
    this.masterKey = crypto.pbkdf2Sync(
      password,
      effectiveSalt,
      SECURITY_CONSTANTS.KDF_ITERATIONS,
      SECURITY_CONSTANTS.KEY_LENGTH,
      SECURITY_CONSTANTS.KDF_ALGORITHM,
    );
    this.emit('encryption:status-changed', { enabled: true });
  }

  /** 是否已初始化 */
  isInitialized(): boolean {
    return this.masterKey !== null;
  }

  /** 加密数据 */
  encrypt(plaintext: Buffer): Buffer {
    if (!this.masterKey) {
      throw new Error('EncryptionService not initialized. Call initialize() first.');
    }

    const iv = crypto.randomBytes(SECURITY_CONSTANTS.IV_LENGTH);
    const cipher = crypto.createCipheriv('aes-256-gcm', this.masterKey, iv);

    const encrypted = Buffer.concat([cipher.update(plaintext), cipher.final()]);
    const authTag = cipher.getAuthTag();

    // 格式: [IV (16)] [AuthTag (16)] [Ciphertext (variable)]
    return Buffer.concat([iv, authTag, encrypted]);
  }

  /** 解密数据 */
  decrypt(ciphertext: Buffer): Buffer {
    if (!this.masterKey) {
      throw new Error('EncryptionService not initialized. Call initialize() first.');
    }

    const iv = ciphertext.subarray(0, SECURITY_CONSTANTS.IV_LENGTH);
    const authTag = ciphertext.subarray(
      SECURITY_CONSTANTS.IV_LENGTH,
      SECURITY_CONSTANTS.IV_LENGTH + SECURITY_CONSTANTS.AUTH_TAG_LENGTH,
    );
    const encryptedData = ciphertext.subarray(
      SECURITY_CONSTANTS.IV_LENGTH + SECURITY_CONSTANTS.AUTH_TAG_LENGTH,
    );

    const decipher = crypto.createDecipheriv('aes-256-gcm', this.masterKey, iv);
    decipher.setAuthTag(authTag);

    return Buffer.concat([decipher.update(encryptedData), decipher.final()]);
  }

  /**
   * 轮换加密密钥：用旧密码解密数据后，用新密码重新加密。
   * 更新内部密钥派生参数（生成新盐值）。
   * @param oldPassword - 当前密码
   * @param newPassword - 新密码
   * @param encryptedData - 需要重新加密的数据列表
   * @returns 使用新密钥重新加密后的数据列表（顺序与输入一致）
   */
  rotateKey(
    oldPassword: string,
    newPassword: string,
    encryptedData: Buffer[],
  ): Buffer[] {
    // 1. 保存当前密钥，用旧密码派生临时解密密钥
    const oldSalt = crypto.randomBytes(SECURITY_CONSTANTS.SALT_LENGTH);
    const oldDerivedKey = crypto.pbkdf2Sync(
      oldPassword,
      oldSalt,
      SECURITY_CONSTANTS.KDF_ITERATIONS,
      SECURITY_CONSTANTS.KEY_LENGTH,
      SECURITY_CONSTANTS.KDF_ALGORITHM,
    );

    // 2. 用旧密码验证并解密所有数据
    const decryptedItems: Buffer[] = [];
    const previousKey = this.masterKey;
    for (const item of encryptedData) {
      const iv = item.subarray(0, SECURITY_CONSTANTS.IV_LENGTH);
      const authTag = item.subarray(
        SECURITY_CONSTANTS.IV_LENGTH,
        SECURITY_CONSTANTS.IV_LENGTH + SECURITY_CONSTANTS.AUTH_TAG_LENGTH,
      );
      const ciphertextBody = item.subarray(
        SECURITY_CONSTANTS.IV_LENGTH + SECURITY_CONSTANTS.AUTH_TAG_LENGTH,
      );

      // 使用当前 masterKey 解密（验证旧密码正确性）
      const decipher = crypto.createDecipheriv('aes-256-gcm', previousKey!, iv);
      decipher.setAuthTag(authTag);
      const decrypted = Buffer.concat([decipher.update(ciphertextBody), decipher.final()]);
      decryptedItems.push(decrypted);
    }

    // 3. 生成新盐值，用新密码派生新密钥
    const newSalt = crypto.randomBytes(SECURITY_CONSTANTS.SALT_LENGTH);
    this.masterKey = crypto.pbkdf2Sync(
      newPassword,
      newSalt,
      SECURITY_CONSTANTS.KDF_ITERATIONS,
      SECURITY_CONSTANTS.KEY_LENGTH,
      SECURITY_CONSTANTS.KDF_ALGORITHM,
    );

    // 4. 用新密钥重新加密所有数据
    const reEncrypted: Buffer[] = [];
    for (const plaintext of decryptedItems) {
      const iv = crypto.randomBytes(SECURITY_CONSTANTS.IV_LENGTH);
      const cipher = crypto.createCipheriv('aes-256-gcm', this.masterKey, iv);
      const encrypted = Buffer.concat([cipher.update(plaintext), cipher.final()]);
      const authTag = cipher.getAuthTag();
      reEncrypted.push(Buffer.concat([iv, authTag, encrypted]));
    }

    this.emit('encryption:key-rotated', { timestamp: Date.now() });
    return reEncrypted;
  }

  /** 生成随机盐值 */
  generateSalt(): Buffer {
    return crypto.randomBytes(SECURITY_CONSTANTS.SALT_LENGTH);
  }

  /** 计算 SHA-256 校验和 */
  static computeChecksum(data: Buffer): string {
    return crypto.createHash('sha256').update(data).digest('hex');
  }

  /** 销毁密钥材料 */
  destroy(): void {
    if (this.masterKey) {
      this.masterKey.fill(0);
      this.masterKey = null;
    }
    this.emit('encryption:status-changed', { enabled: false });
  }
}
```

```typescript
// src/main/core/security/IntegrityChecker.ts

import fs from 'node:fs/promises';
import path from 'node:path';
import { execSync } from 'node:child_process';
import { EventEmitter } from 'node:events';
import type Database from 'better-sqlite3';
import type { IntegrityCheckResult, IntegrityError, IntegrityRepair } from './types';

export class IntegrityChecker extends EventEmitter {
  private dbPath: string;
  private backupDir: string;

  constructor(dbPath: string, backupDir: string) {
    super();
    this.dbPath = dbPath;
    this.backupDir = backupDir;
  }

  /** 启动时执行完整性检测，失败时按三级策略恢复 */
  async startupCheck(db: Database.Database): Promise<IntegrityCheckResult> {
    const result = await this.checkIntegrity(db);

    if (!result.valid) {
      const repairs = await this.performRecovery(db, result.errors);
      result.repairs = repairs;

      // 重新检测
      const recheck = await this.checkIntegrity(db);
      result.valid = recheck.valid;
      if (!recheck.valid) {
        result.errors.push({
          code: 'RECOVERY_FAILED',
          message: '三级恢复均未成功，数据库可能已损坏',
          severity: 'critical',
        });
      }
    }

    this.emit('integrity:checked', { result });
    return result;
  }

  /** 执行数据库完整性检查 */
  async checkIntegrity(db: Database.Database): Promise<IntegrityCheckResult> {
    const errors: IntegrityError[] = [];
    const checkedAt = new Date().toISOString();

    try {
      // 1. PRAGMA integrity_check
      const integrityResult = db.pragma('integrity_check') as Array<{ integrity_check: string }>;
      for (const row of integrityResult) {
        if (row.integrity_check !== 'ok') {
          errors.push({
            code: 'INTEGRITY_CHECK_FAILED',
            message: row.integrity_check,
            severity: 'error',
          });
        }
      }

      // 2. PRAGMA quick_check
      const quickResult = db.pragma('quick_check') as Array<{ quick_check: string }>;
      for (const row of quickResult) {
        if (row.quick_check !== 'ok') {
          errors.push({
            code: 'QUICK_CHECK_FAILED',
            message: row.quick_check,
            severity: 'error',
          });
        }
      }

      // 3. 检查 WAL 文件
      const walPath = `${this.dbPath}-wal`;
      try {
        await fs.access(walPath);
        const walStat = await fs.stat(walPath);
        if (walStat.size > 100 * 1024 * 1024) {
          // WAL > 100MB
          errors.push({
            code: 'WAL_OVERSIZED',
            message: `WAL 文件过大: ${(walStat.size / 1024 / 1024).toFixed(1)}MB`,
            severity: 'warning',
          });
        }
      } catch {
        // WAL 不存在是正常的
      }

      // 4. 检查数据库文件是否可读
      const dbStat = await fs.stat(this.dbPath);
      if (dbStat.size === 0) {
        errors.push({
          code: 'DB_EMPTY',
          message: '数据库文件为空',
          severity: 'critical',
        });
      }
    } catch (err) {
      errors.push({
        code: 'CHECK_EXCEPTION',
        message: err instanceof Error ? err.message : String(err),
        severity: 'critical',
      });
    }

    return {
      valid: errors.filter((e) => e.severity === 'error' || e.severity === 'critical').length === 0,
      checkedAt,
      errors,
      repairs: [],
    };
  }

  /** 三级恢复策略 */
  private async performRecovery(
    db: Database.Database,
    errors: IntegrityError[],
  ): Promise<IntegrityRepair[]> {
    const repairs: IntegrityRepair[] = [];

    // 第一级：WAL 检查点
    try {
      db.pragma('wal_checkpoint(TRUNCATE)');
      repairs.push({
        action: 'wal-checkpoint',
        description: '执行 WAL 检查点，将 WAL 日志合并到主数据库',
        success: true,
      });

      // 重新检查
      const recheck = await this.checkIntegrity(db);
      if (recheck.valid) return repairs;
    } catch (err) {
      repairs.push({
        action: 'wal-checkpoint',
        description: `WAL 检查点失败: ${err instanceof Error ? err.message : String(err)}`,
        success: false,
      });
    }

    // 第二级：sqlite3 .recover
    try {
      const recoveredPath = `${this.dbPath}.recovered`;
      execSync(`sqlite3 "${this.dbPath}" ".recover" | sqlite3 "${recoveredPath}"`, {
        timeout: 60_000,
      });

      // 验证恢复后的数据库
      const Database = (await import('better-sqlite3')).default;
      const recoveredDb = new Database(recoveredPath);
      const recoverCheck = this.checkIntegrity(recoveredDb);
      recoveredDb.close();

      const checkResult = await recoverCheck;
      if (checkResult.valid) {
        // 替换原数据库
        await fs.copyFile(recoveredPath, this.dbPath);
        repairs.push({
          action: 'recover',
          description: '使用 sqlite3 .recover 成功恢复数据库',
          success: true,
        });
        return repairs;
      }

      await fs.unlink(recoveredPath).catch(() => {});
      repairs.push({
        action: 'recover',
        description: 'sqlite3 .recover 完成但恢复后数据库仍不完整',
        success: false,
      });
    } catch (err) {
      repairs.push({
        action: 'recover',
        description: `sqlite3 .recover 失败: ${err instanceof Error ? err.message : String(err)}`,
        success: false,
      });
    }

    // 第三级：从最近备份恢复
    try {
      const latestBackup = await this.findLatestBackup();
      if (latestBackup) {
        await fs.copyFile(latestBackup, this.dbPath);
        repairs.push({
          action: 'restore-from-backup',
          description: `从备份 ${path.basename(latestBackup)} 恢复数据库`,
          success: true,
        });
      } else {
        repairs.push({
          action: 'restore-from-backup',
          description: '没有可用的备份文件',
          success: false,
        });
      }
    } catch (err) {
      repairs.push({
        action: 'restore-from-backup',
        description: `备份恢复失败: ${err instanceof Error ? err.message : String(err)}`,
        success: false,
      });
    }

    this.emit('integrity:repaired', { repairs });
    return repairs;
  }

  /** 查找最新的备份文件 */
  private async findLatestBackup(): Promise<string | null> {
    try {
      const files = await fs.readdir(this.backupDir);
      const dbBackups = files
        .filter((f) => f.endsWith('.db') || f.endsWith('.db.encrypted'))
        .sort()
        .reverse();

      return dbBackups.length > 0 ? path.join(this.backupDir, dbBackups[0]) : null;
    } catch {
      return null;
    }
  }
}
```

#### 步骤 5：实现 BackupScheduler

```typescript
// src/main/core/security/BackupScheduler.ts

import crypto from 'node:crypto';
import fs from 'node:fs/promises';
import path from 'node:path';
import { EventEmitter } from 'node:events';
import type Database from 'better-sqlite3';
import { CronJob } from 'cron';
import type { BackupMetadata, BackupScheduleConfig, BackupTrigger } from './types';
import { EncryptionService } from './EncryptionService';

export class BackupScheduler extends EventEmitter {
  private config: BackupScheduleConfig;
  private db: Database.Database;
  private encryptionService: EncryptionService | null;
  private cronJob: CronJob | null = null;
  private backupHistory: BackupMetadata[] = [];
  /** 文件锁标志位，防止并发写入 */
  private isBackingUp = false;

  constructor(
    config: BackupScheduleConfig,
    db: Database.Database,
    encryptionService: EncryptionService | null = null,
  ) {
    super();
    this.config = config;
    this.db = db;
    this.encryptionService = encryptionService;
  }

  /** 启动定时备份调度 */
  async start(): Promise<void> {
    await this.ensureBackupDir();
    await this.loadBackupHistory();

    if (this.config.enabled) {
      this.cronJob = new CronJob(this.config.cronExpression, () => {
        void this.executeBackup('scheduled');
      });
      this.cronJob.start();
    }
  }

  /** 停止调度 */
  stop(): void {
    if (this.cronJob) {
      this.cronJob.stop();
      this.cronJob = null;
    }
  }

  /** 手动触发备份 */
  async manualBackup(): Promise<BackupMetadata> {
    return this.executeBackup('manual');
  }

  /** 配置变更时备份上一版本 */
  async onConfigChange(): Promise<BackupMetadata> {
    return this.executeBackup('config-change');
  }

  /** 执行备份 */
  private async executeBackup(trigger: BackupTrigger): Promise<BackupMetadata> {
    // 文件锁：防止并发写入，同一时间只允许一个备份操作
    if (this.isBackingUp) {
      this.emit('backup:failed', {
        error: new Error('备份操作正在进行中，请稍后再试'),
        trigger,
      });
      throw new Error('备份操作正在进行中，请稍后再试');
    }
    this.isBackingUp = true;

    this.emit('backup:started', { trigger });

    try {
      const timestamp = new Date().toISOString().replace(/[:.]/g, '-');
      const fileName = `backup-${timestamp}.db${this.config.encrypt ? '.encrypted' : ''}`;
      const filePath = path.join(this.config.backupDir, fileName);

      // 执行 WAL 检查点确保数据完整
      this.db.pragma('wal_checkpoint(TRUNCATE)');

      // 复制数据库文件
      const backupPath = path.join(this.config.backupDir, `backup-${timestamp}.db`);
      await this.db.backup(backupPath);

      let finalPath = backupPath;
      let fileBuffer = await fs.readFile(backupPath);

      // 可选加密
      if (this.config.encrypt && this.encryptionService?.isInitialized()) {
        const encrypted = this.encryptionService.encrypt(fileBuffer);
        await fs.writeFile(filePath, encrypted);
        await fs.unlink(backupPath);
        fileBuffer = encrypted;
        finalPath = filePath;
      } else {
        finalPath = backupPath;
      }

      const metadata: BackupMetadata = {
        id: crypto.randomUUID(),
        createdAt: new Date().toISOString(),
        filePath: finalPath,
        sizeBytes: fileBuffer.length,
        encrypted: this.config.encrypt && !!this.encryptionService?.isInitialized(),
        trigger,
        dbVersion: Number(this.db.pragma('user_version', { simple: true })),
        checksum: EncryptionService.computeChecksum(fileBuffer),
      };

      this.backupHistory.push(metadata);
      await this.saveBackupHistory();

      this.emit('backup:completed', { metadata });

      // 清理过期备份
      await this.cleanupExpiredBackups();

      return metadata;
    } catch (err) {
      const error = err instanceof Error ? err : new Error(String(err));
      this.emit('backup:failed', { error, trigger });
      throw error;
    } finally {
      // 释放文件锁，确保无论成功或失败都能继续下一次备份
      this.isBackingUp = false;
    }
  }

  /** 清理过期备份 */
  private async cleanupExpiredBackups(): Promise<void> {
    const cutoffDate = new Date();
    cutoffDate.setDate(cutoffDate.getDate() - this.config.retentionDays);

    const expired = this.backupHistory.filter(
      (b) => new Date(b.createdAt) < cutoffDate,
    );

    // 也检查最大数量限制
    const sorted = [...this.backupHistory].sort(
      (a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime(),
    );
    const excess = sorted.slice(this.config.maxBackups);

    const toRemove = new Set([...expired, ...excess].map((b) => b.id));

    for (const id of toRemove) {
      const backup = this.backupHistory.find((b) => b.id === id);
      if (backup) {
        await fs.unlink(backup.filePath).catch(() => {});
      }
    }

    this.backupHistory = this.backupHistory.filter((b) => !toRemove.has(b.id));
    await this.saveBackupHistory();

    if (toRemove.size > 0) {
      this.emit('backup:cleanup', { removed: toRemove.size });
    }
  }

  /** 确保备份目录存在 */
  private async ensureBackupDir(): Promise<void> {
    await fs.mkdir(this.config.backupDir, { recursive: true });
  }

  /** 加载备份历史 */
  private async loadBackupHistory(): Promise<void> {
    const historyPath = path.join(this.config.backupDir, 'backup-history.json');
    try {
      const data = await fs.readFile(historyPath, 'utf-8');
      this.backupHistory = JSON.parse(data) as BackupMetadata[];
    } catch {
      this.backupHistory = [];
    }
  }

  /** 保存备份历史 */
  private async saveBackupHistory(): Promise<void> {
    const historyPath = path.join(this.config.backupDir, 'backup-history.json');
    await fs.writeFile(historyPath, JSON.stringify(this.backupHistory, null, 2));
  }

  /** 获取备份列表 */
  getBackupList(): BackupMetadata[] {
    return [...this.backupHistory].sort(
      (a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime(),
    );
  }
```

```typescript
// src/main/core/security/DataExporter.ts

import fs from 'node:fs/promises';
import path from 'node:path';
import os from 'node:os';
import { EventEmitter } from 'node:events';
import AdmZip from 'adm-zip';
import type Database from 'better-sqlite3';
import type { ExportPackage } from './types';
import { EncryptionService } from './EncryptionService';
import { SECURITY_CONSTANTS } from './constants';

export class DataExporter extends EventEmitter {
  private dbPath: string;
  private configPath: string;
  private appVersion: string;
  private encryptionService: EncryptionService | null;

  constructor(
    dbPath: string,
    configPath: string,
    appVersion: string,
    encryptionService: EncryptionService | null = null,
  ) {
    super();
    this.dbPath = dbPath;
    this.configPath = configPath;
    this.appVersion = appVersion;
    this.encryptionService = encryptionService;
  }

  /** 一键导出为 zip 文件 */
  async exportData(outputPath: string, encrypt = false): Promise<string> {
    // 读取数据库文件
    const database = await fs.readFile(this.dbPath);

    // 读取配置文件
    let config: Record<string, unknown> = {};
    try {
      const configData = await fs.readFile(this.configPath, 'utf-8');
      config = JSON.parse(configData) as Record<string, unknown>;
    } catch {
      // 配置文件不存在时使用空对象
    }

    const pkg: ExportPackage = {
      formatVersion: SECURITY_CONSTANTS.EXPORT_FORMAT_VERSION,
      exportedAt: new Date().toISOString(),
      appVersion: this.appVersion,
      database,
      config,
      checksum: EncryptionService.computeChecksum(database),
      encrypted: encrypt,
    };

    // 创建 zip 包
    const zip = new AdmZip();

    // 添加元数据
    zip.addFile(
      'manifest.json',
      Buffer.from(
        JSON.stringify(
          {
            formatVersion: pkg.formatVersion,
            exportedAt: pkg.exportedAt,
            appVersion: pkg.appVersion,
            checksum: pkg.checksum,
            encrypted: pkg.encrypted,
            config: pkg.config,
          },
          null,
          2,
        ),
      ),
    );

    // 添加数据库文件（可能加密）
    if (encrypt && this.encryptionService?.isInitialized()) {
      const encryptedDb = this.encryptionService.encrypt(database);
      zip.addFile('database.db.encrypted', encryptedDb);
    } else {
      zip.addFile('database.db', database);
    }

    // 写入 zip 文件
    zip.writeZip(outputPath);

    this.emit('export:completed', { filePath: outputPath });
    return outputPath;
  }

  /** 一键导入 zip 文件 */
  async importData(zipPath: string, password?: string): Promise<boolean> {
    const zip = new AdmZip(zipPath);

    // 读取清单
    const manifestEntry = zip.getEntry('manifest.json');
    if (!manifestEntry) {
      throw new Error('无效的导出包：缺少 manifest.json');
    }

    const manifest = JSON.parse(manifestEntry.getData().toString('utf-8')) as {
      formatVersion: string;
      checksum: string;
      encrypted: boolean;
      config: Record<string, unknown>;
    };

    // 格式版本兼容性检查
    if (!this.isFormatCompatible(manifest.formatVersion)) {
      throw new Error(
        `不兼容的导出格式: ${manifest.formatVersion}，当前支持: ${SECURITY_CONSTANTS.EXPORT_FORMAT_VERSION}`,
      );
    }

    // 读取数据库
    let dbBuffer: Buffer;
    if (manifest.encrypted) {
      const encryptedEntry = zip.getEntry('database.db.encrypted');
      if (!encryptedEntry) {
        throw new Error('加密的导出包中找不到 database.db.encrypted');
      }
      if (!this.encryptionService?.isInitialized()) {
        throw new Error('导入加密数据需要先初始化加密服务');
      }
      dbBuffer = this.encryptionService.decrypt(encryptedEntry.getData());
    } else {
      const dbEntry = zip.getEntry('database.db');
      if (!dbEntry) {
        throw new Error('导出包中找不到 database.db');
      }
      dbBuffer = dbEntry.getData();
    }

    // 校验和验证
    const actualChecksum = EncryptionService.computeChecksum(dbBuffer);
    if (actualChecksum !== manifest.checksum) {
      throw new Error('数据校验失败：文件可能已损坏或被篡改');
    }

    // 备份当前数据库
    const backupPath = `${this.dbPath}.pre-import-${Date.now()}`;
    await fs.copyFile(this.dbPath, backupPath);

    try {
      // 写入新数据库
      await fs.writeFile(this.dbPath, dbBuffer);

      // 恢复配置
      if (manifest.config && Object.keys(manifest.config).length > 0) {
        await fs.writeFile(this.configPath, JSON.stringify(manifest.config, null, 2));
      }

      this.emit('import:completed', { restored: true });
      return true;
    } catch (err) {
      // 回滚：从备份恢复数据库
      try {
        await fs.copyFile(backupPath, this.dbPath);
        // 恢复成功后，安全删除备份文件（删除失败不影响整体流程）
        await fs.unlink(backupPath).catch((unlinkErr) => {
          this.logger.warn('备份文件删除失败，已保留', { backupPath, error: unlinkErr });
        });
      } catch (rollbackErr) {
        // 回滚本身失败，保留备份文件以便手动恢复
        this.logger.error('数据库回滚失败，备份文件已保留', {
          backupPath,
          rollbackError: rollbackErr,
          originalError: err,
        });
      }
      throw err;
    }
  }

  /** 检查格式版本兼容性 */
  private isFormatCompatible(version: string): boolean {
    const [major] = version.split('.').map(Number);
    const [currentMajor] = SECURITY_CONSTANTS.EXPORT_FORMAT_VERSION.split('.').map(Number);
    return major === currentMajor;
  }

  /** 获取导出包信息（不解密） */
  async getExportInfo(zipPath: string): Promise<{
    formatVersion: string;
    exportedAt: string;
    appVersion: string;
    encrypted: boolean;
    dbSize: number;
  }> {
    const zip = new AdmZip(zipPath);
    const manifestEntry = zip.getEntry('manifest.json');
    if (!manifestEntry) {
      throw new Error('无效的导出包');
    }

    const manifest = JSON.parse(manifestEntry.getData().toString('utf-8')) as {
      formatVersion: string;
      exportedAt: string;
      appVersion: string;
      encrypted: boolean;
    };

    const dbEntry = zip.getEntry(manifest.encrypted ? 'database.db.encrypted' : 'database.db');

    return {
      formatVersion: manifest.formatVersion,
      exportedAt: manifest.exportedAt,
      appVersion: manifest.appVersion,
      encrypted: manifest.encrypted,
      dbSize: dbEntry?.header.size ?? 0,
    };
  }
}
```

```typescript
// src/main/core/security/SecurityManager.ts

import { EventEmitter } from 'node:events';
import type Database from 'better-sqlite3';
import type {
  BackupScheduleConfig,
  BackupMetadata,
  IntegrityCheckResult,
  EncryptionConfig,
} from './types';
import { BackupScheduler } from './BackupScheduler';
import { DataExporter } from './DataExporter';
import { IntegrityChecker } from './IntegrityChecker';
import { EncryptionService } from './EncryptionService';
import { SECURITY_CONSTANTS } from './constants';

export interface SecurityManagerConfig {
  dbPath: string;
  configPath: string;
  appVersion: string;
  backup: Partial<BackupScheduleConfig>;
  encryption: Partial<EncryptionConfig>;
}

export class SecurityManager extends EventEmitter {
  private backupScheduler: BackupScheduler;
  private dataExporter: DataExporter;
  private integrityChecker: IntegrityChecker;
  private encryptionService: EncryptionService;

  constructor(db: Database.Database, config: SecurityManagerConfig) {
    super();

    // 初始化加密服务
    const encConfig: EncryptionConfig = {
      enabled: config.encryption.enabled ?? false,
      algorithm: 'aes-256-gcm',
      keyDerivationIterations: config.encryption.keyDerivationIterations ?? SECURITY_CONSTANTS.KDF_ITERATIONS,
    };
    this.encryptionService = new EncryptionService(encConfig);

    // 初始化备份配置
    const backupConfig: BackupScheduleConfig = {
      enabled: config.backup.enabled ?? true,
      cronExpression: config.backup.cronExpression ?? SECURITY_CONSTANTS.DEFAULT_CRON,
      retentionDays: config.backup.retentionDays ?? SECURITY_CONSTANTS.DEFAULT_RETENTION_DAYS,
      backupDir: config.backup.backupDir ?? SECURITY_CONSTANTS.DEFAULT_BACKUP_DIR,
      encrypt: config.backup.encrypt ?? false,
      maxBackups: config.backup.maxBackups ?? SECURITY_CONSTANTS.DEFAULT_MAX_BACKUPS,
    };

    // 初始化各组件
    this.backupScheduler = new BackupScheduler(backupConfig, db, this.encryptionService);
    this.dataExporter = new DataExporter(
      config.dbPath,
      config.configPath,
      config.appVersion,
      this.encryptionService,
    );
    this.integrityChecker = new IntegrityChecker(config.dbPath, backupConfig.backupDir);

    // 转发事件
    this.forwardEvents();
  }

  /** 启动安全模块 */
  async startup(db: Database.Database): Promise<IntegrityCheckResult> {
    await this.backupScheduler.start();
    return this.integrityChecker.startupCheck(db);
  }

  /** 关闭安全模块 */
  shutdown(): void {
    this.backupScheduler.stop();
    this.encryptionService.destroy();
  }

  /** 启用加密 */
  enableEncryption(password: string): void {
    // 密码强度验证
    if (!password || password.length < 8) {
      throw new Error('密码长度不能少于8个字符');
    }
    const hasUpper = /[A-Z]/.test(password);
    const hasLower = /[a-z]/.test(password);
    const hasDigit = /[0-9]/.test(password);
    const categoryCount = [hasUpper, hasLower, hasDigit].filter(Boolean).length;
    if (categoryCount < 2) {
      throw new Error('密码必须包含大写字母、小写字母、数字中的至少两种');
    }
    this.encryptionService.initialize(password);
  }

  /** 导出数据 */
  async exportData(outputPath: string, encrypt = false): Promise<string> {
    return this.dataExporter.exportData(outputPath, encrypt);
  }

  /** 导入数据 */
  async importData(zipPath: string, password?: string): Promise<boolean> {
    return this.dataExporter.importData(zipPath, password);
  }

  /** 手动备份 */
  async manualBackup(): Promise<BackupMetadata> {
    return this.backupScheduler.manualBackup();
  }

  /** 获取备份列表 */
  getBackupList(): BackupMetadata[] {
    return this.backupScheduler.getBackupList();
  }

  /** 配置变更通知 */
  async onConfigChange(): Promise<void> {
    await this.backupScheduler.onConfigChange();
  }

  private forwardEvents(): void {
    const events = [
      'backup:started',
      'backup:completed',
      'backup:failed',
      'backup:cleanup',
      'integrity:checked',
      'integrity:repaired',
      'export:completed',
      'import:completed',
      'encryption:status-changed',
    ] as const;

    for (const event of events) {
      const source =
        event.startsWith('backup:')
          ? this.backupScheduler
          : event === 'encryption:status-changed'
            ? this.encryptionService
            : event.startsWith('integrity:')
              ? this.integrityChecker
              : this.dataExporter;

      source.on(event, (...args: unknown[]) => {
        this.emit(event, ...args);
      });
    }
  }
}
```

```typescript
    const entries = zip.getEntries().map((e) => e.entryName);

    expect(entries).toContain('manifest.json');
    expect(entries).toContain('database.db');
  });

  it('导出的 manifest 应包含正确的元数据', async () => {
    await exporter.exportData(outputPath);

    const zip = new AdmZip(outputPath);
    const manifest = JSON.parse(
      zip.getEntry('manifest.json')!.getData().toString('utf-8'),
    ) as {
      formatVersion: string;
      appVersion: string;
      exportedAt: string;
      checksum: string;
      encrypted: boolean;
      config: Record<string, unknown>;
    };

    expect(manifest.appVersion).toBe('1.0.0');
    expect(manifest.encrypted).toBe(false);
    expect(manifest.checksum).toHaveLength(64); // SHA-256 hex
    expect(manifest.config).toEqual({ theme: 'dark', lang: 'zh-CN' });
  });

  it('应成功导入之前导出的数据', async () => {
    await exporter.exportData(outputPath);

    // 修改原始数据库以验证导入会恢复它
    await fs.writeFile(dbPath, 'modified-content');
    expect((await fs.readFile(dbPath, 'utf-8')).toString()).toBe('modified-content');

    const imported = await exporter.importData(outputPath);

    expect(imported).toBe(true);
    // 数据库应被恢复为导出时的内容
    const restored = await fs.readFile(dbPath);
    expect(restored.toString()).toBe('mock-database-content');
  });

  it('应正确恢复导入的配置文件', async () => {
    await exporter.exportData(outputPath);

    // 清空配置
    await fs.writeFile(configPath, '{}');

    await exporter.importData(outputPath);

    const config = JSON.parse(await fs.readFile(configPath, 'utf-8')) as Record<string, unknown>;
    expect(config).toEqual({ theme: 'dark', lang: 'zh-CN' });
  });

  it('应正确计算和验证校验和', async () => {
    await exporter.exportData(outputPath);

    // 读取 manifest 获取校验和
    const zip = new AdmZip(outputPath);
    const manifest = JSON.parse(
      zip.getEntry('manifest.json')!.getData().toString('utf-8'),
    ) as { checksum: string };

    // 手动计算数据库内容的校验和，应与 manifest 中的一致
    const dbContent = await fs.readFile(dbPath);
    const expectedChecksum = EncryptionService.computeChecksum(dbContent);
    expect(manifest.checksum).toBe(expectedChecksum);
  });

  it('导入损坏的包应抛出校验和错误', async () => {
    await exporter.exportData(outputPath);

    // 篡改 zip 中的数据库内容
    const zip = new AdmZip(outputPath);
    zip.updateFile('database.db', Buffer.from('tampered-data'));
    zip.writeZip(outputPath);

    await expect(exporter.importData(outputPath)).rejects.toThrow('数据校验失败');
  });

  it('缺少 manifest 的包应抛出错误', async () => {
    // 创建一个没有 manifest 的空 zip
    const emptyZip = new AdmZip();
    emptyZip.addFile('database.db', Buffer.from('data'));
    const invalidPath = path.join(tmpDir, 'invalid.zip');
    emptyZip.writeZip(invalidPath);

    await expect(exporter.importData(invalidPath)).rejects.toThrow('缺少 manifest.json');
  });

  it('使用加密服务时应导出加密数据库', async () => {
    const encService = new EncryptionService({
      enabled: true,
      algorithm: 'aes-256-gcm',
      keyDerivationIterations: 1000,
    });
    encService.initialize('test-password');

    const encExporter = new DataExporter(dbPath, configPath, '1.0.0', encService);
    await encExporter.exportData(outputPath, true);

    const zip = new AdmZip(outputPath);
    const entries = zip.getEntries().map((e) => e.entryName);

    // 加密导出应包含 database.db.encrypted 而非 database.db
    expect(entries).toContain('database.db.encrypted');
    expect(entries).not.toContain('database.db');

    // 应能正确导入加密包
    const imported = await encExporter.importData(outputPath, 'test-password');
    expect(imported).toBe(true);

    encService.destroy();
  });

  it('应正确获取导出包信息', async () => {
    await exporter.exportData(outputPath);

    const info = await exporter.getExportInfo(outputPath);

    expect(info.appVersion).toBe('1.0.0');
    expect(info.encrypted).toBe(false);
    expect(info.dbSize).toBeGreaterThan(0);
    expect(info.formatVersion).toBeDefined();
  });

  it('导出和导入应触发相应事件', async () => {
    const events: string[] = [];
    exporter.on('export:completed', () => events.push('export:completed'));
    exporter.on('import:completed', () => events.push('import:completed'));

    await exporter.exportData(outputPath);
    await exporter.importData(outputPath);

    expect(events).toEqual(['export:completed', 'import:completed']);
  });
});
```

### 27.3 代码规范

1. **事件驱动架构**：所有安全模块通过 `EventEmitter` 解耦，事件名使用 `模块:动作` 命名（如 `backup:completed`）
2. **密钥管理**：主密钥仅在内存中保存，`destroy()` 时用零填充清除，禁止持久化到磁盘
3. **加密格式**：加密数据格式为 `[IV 16字节][AuthTag 16字节][密文]`，方便解析
4. **错误处理**：三级恢复策略（WAL → .recover → 备份）逐级尝试，每级记录详细日志
5. **备份文件命名**：`backup-{ISO时间戳}.db[.encrypted]`，保证排序即时间顺序
6. **校验和**：所有备份和导出操作均计算 SHA-256 校验和，导入时必须验证
7. **导出格式**：zip 包含 `manifest.json`（元数据）+ `database.db`（数据），格式版本使用语义化版本
8. **并发安全**：备份操作使用文件锁防止并发写入，WAL 检查点在备份前执行
9. **TypeScript 严格模式**：所有类型显式声明，禁止使用 `any`，`unknown` + 类型守卫替代
10. **资源清理**：实现 `shutdown()` 方法，确保定时器、文件句柄等资源正确释放
