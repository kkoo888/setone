```typescript
import { Tray, Menu, nativeImage, BrowserWindow, app } from 'electron'
import { join } from 'path'
import type { Logger } from '../../../src/main/types/logger'

export class TrayService {
  private tray: Tray | null = null
  private mainWindow: BrowserWindow | null = null
  private logger: Logger
  private iconPath: string

  /**
   * @param logger   日志实例
   * @param iconPath 可选：自定义托盘图标路径（绝对路径）。
   *                 未提供时自动根据运行环境解析：
   *                 - 打包后：process.resourcesPath/resources/tray-icon.{ico|png}
   *                 - 开发中：__dirname/../../resources/tray-icon.{ico|png}
   *                 Windows 使用 .ico 格式，macOS/Linux 使用 .png
   */
  constructor(logger: Logger, iconPath?: string) {
    this.logger = logger
    // Issue#235: 图标路径支持配置注入，避免硬编码；同时兼容打包/开发两种环境
    // Issue#234: Windows 使用 .ico 格式图标
    const iconFile = process.platform === 'win32' ? 'tray-icon.ico' : 'tray-icon.png'
    this.iconPath = iconPath ?? (app.isPackaged
      ? join(process.resourcesPath, 'resources', iconFile)
      : join(__dirname, '../../resources', iconFile))
  }

  /** 创建托盘图标 */
  create(mainWindow: BrowserWindow): void {
    this.mainWindow = mainWindow

    const icon = nativeImage.createFromPath(this.iconPath)
    if (icon.isEmpty()) {
      this.logger.warn(`托盘图标加载失败，路径: ${this.iconPath}`)
    }

    // macOS: 设置 TemplateImage，使图标自动适配深色/浅色菜单栏
    // 要求图标文件为单色（黑色主体 + alpha 通道），文件名建议包含 "Template" 后缀
    if (process.platform === 'darwin') {
      icon.setTemplateImage(true)
    }

    this.tray = new Tray(icon)
    this.tray.setToolTip('智能桌面助手')

    this.updateContextMenu()

    if (process.platform === 'darwin') {
      // macOS: 左键点击弹出上下文菜单（macOS 惯例）
      // macOS 菜单栏图标默认行为是弹出菜单，而非 toggle 窗口
      this.tray.on('click', () => {
        this.tray?.popUpContextMenu()
      })
    } else {
      // Windows/Linux: 左键点击切换窗口显示/隐藏
      this.tray.on('click', () => {
        if (this.mainWindow) {
          if (this.mainWindow.isVisible()) {
            this.mainWindow.hide()
          } else {
            this.mainWindow.show()
            this.mainWindow.focus()
          }
        }
      })
    }

    this.logger.info('系统托盘已创建')
  }

  /** 更新右键菜单 */
  updateContextMenu(): void {
    if (!this.tray) return

    const isVisible = this.mainWindow?.isVisible() ?? false

    const contextMenu = Menu.buildFromTemplate([
      {
        label: isVisible ? '隐藏主窗口' : '显示主窗口',
        click: () => {
          if (this.mainWindow) {
            if (this.mainWindow.isVisible()) {
              this.mainWindow.hide()
            } else {
              this.mainWindow.show()
              this.mainWindow.focus()
            }
          }
        }
      },
      { type: 'separator' },
      {
        label: '设置',
        click: () => {
          this.mainWindow?.webContents.send('navigate', 'settings')
          this.mainWindow?.show()
        }
      },
      {
        label: '模块管理',
        click: () => {
          this.mainWindow?.webContents.send('navigate', 'modules')
          this.mainWindow?.show()
        }
      },
      { type: 'separator' },
      {
        label: '退出',
        click: () => {
          this.destroy()
          app.quit()  // Issue#232: 使用 app.quit() 替代 process.exit(0)，确保触发 will-quit 事件
        }
      }
    ])

    this.tray.setContextMenu(contextMenu)
  }

  /** 销毁托盘 */
  destroy(): void {
    this.tray?.destroy()
    this.tray = null
  }
}
```

```typescript
import { globalShortcut, BrowserWindow } from 'electron'
import type { Logger } from '../../../src/main/types/logger'
import { NotificationService } from './notification'

export interface HotkeyRegistration {
  accelerator: string
  description: string
  callback: () => void
}

export class HotkeyService {
  private registrations = new Map<string, HotkeyRegistration>()
  private logger: Logger
  private notificationService: NotificationService

  constructor(logger: Logger, notificationService: NotificationService) {
    this.logger = logger
    this.notificationService = notificationService
  }

  /**
   * 注册全局快捷键
   *
   * 冲突处理策略（Issue#236）：
   * - 内部冲突（已在本应用注册）：发送系统通知告知用户，返回 false
   * - 外部冲突（被其他应用占用）：发送系统通知告知用户具体冲突信息，返回 false
   * - 注册成功：记录日志，返回 true
   */
  register(accelerator: string, description: string, callback: () => void): boolean {
    // 检查内部冲突：同一 accelerator 是否已被本应用注册
    if (this.registrations.has(accelerator)) {
      const existing = this.registrations.get(accelerator)!
      const msg = `快捷键 "${accelerator}" 已被占用（功能：${existing.description}），无法绑定到 "${description}"`
      this.logger.warn(msg)
      this.notificationService.send('快捷键冲突', {
        body: msg,
      })
      return false
    }

    try {
      const success = globalShortcut.register(accelerator, callback)
      if (success) {
        this.registrations.set(accelerator, { accelerator, description, callback })
        this.logger.info(`快捷键已注册: ${accelerator} → ${description}`)
      } else {
        // 注册失败：通常被其他应用占用
        const msg = `快捷键 "${accelerator}" 注册失败，可能已被其他应用占用。功能 "${description}" 将无法使用快捷键触发。`
        this.logger.error(msg)
        this.notificationService.send('快捷键注册失败', {
          body: msg,
        })
      }
      return success
    } catch (err) {
      const errMsg = err instanceof Error ? err.message : String(err)
      const msg = `快捷键 "${accelerator}" 注册异常：${errMsg}。功能 "${description}" 将无法使用快捷键触发。`
      this.logger.error(msg, err as Error)
      this.notificationService.send('快捷键注册异常', {
        body: msg,
      })
      return false
    }
  }

  /** 注销快捷键 */
  unregister(accelerator: string): void {
    globalShortcut.unregister(accelerator)
    this.registrations.delete(accelerator)
    this.logger.info(`快捷键已注销: ${accelerator}`)
  }

  /** 注销所有快捷键 */
  unregisterAll(): void {
    globalShortcut.unregisterAll()
    this.registrations.clear()
    this.logger.info('所有快捷键已注销')
  }

  /** 获取已注册的快捷键列表 */
  getRegistrations(): HotkeyRegistration[] {
    return Array.from(this.registrations.values())
  }
}
```

```typescript
import { Notification } from 'electron'
import { execSync } from 'child_process'
import type { Logger } from '../../../src/main/types/logger'

/** 通知选项 */
export interface NotificationOptions {
  /** 通知正文 */
  body?: string
  /** 图标路径（可选） */
  icon?: string
  /** 是否静音 */
  silent?: boolean
  /** 点击通知后的回调 */
  onClick?: () => void
}

/**
 * 系统通知服务
 * 封装 Electron Notification API，提供跨平台通知能力
 *
 * Linux 兼容策略：
 * 1. 优先使用 Electron Notification（依赖 libnotify / dbus）
 * 2. 若 libnotify 不可用（Notification.isSupported() 返回 false），
 *    降级使用 notify-send 命令行工具发送通知
 * 3. 图标自动适配：Linux 使用 .png，Windows 使用 .ico
 */
export class NotificationService {
  private logger: Logger
  /** Linux 降级：是否可用 notify-send 命令 */
  private notifySendAvailable: boolean | null = null

  constructor(logger: Logger) {
    this.logger = logger
    // 启动时异步检测 Linux notify-send 可用性（仅 Linux 平台）
    if (process.platform === 'linux') {
      this.detectNotifySend()
    }
  }

  /**
   * 检测 Linux 系统是否安装了 notify-send（来自 libnotify-bin）
   */
  private detectNotifySend(): void {
    try {
      execSync('which notify-send', { stdio: 'ignore' })
      this.notifySendAvailable = true
      this.logger.debug('检测到 notify-send 工具，可用作通知降级方案')
    } catch {
      this.notifySendAvailable = false
      this.logger.warn('notify-send 未安装（apt install libnotify-bin），Linux 通知降级方案不可用')
    }
  }

  /**
   * Linux 降级方案：通过 notify-send 命令行发送通知
   * 支持标题、正文、图标
   */
  private sendViaNotifySend(title: string, opts: NotificationOptions): boolean {
    if (!this.notifySendAvailable) {
      this.logger.warn('notify-send 不可用，无法发送 Linux 通知')
      return false
    }

    try {
      // 构建 notify-send 命令参数
      const args: string[] = ['notify-send']
      // 转义 shell 特殊字符，防止命令注入
      const escapedTitle = title.replace(/'/g, "'\\''")
      const escapedBody = (opts.body ?? '').replace(/'/g, "'\\''")

      if (opts.icon) {
        // Linux 图标：优先使用 .png 格式
        args.push(`-i '${opts.icon.replace(/'/g, "'\\''")}'`)
      }
      // 设置通知类别和超时（-t 单位毫秒，0 表示由服务器决定）
      args.push(`-t 5000`)
      args.push(`'${escapedTitle}'`)
      if (opts.body) {
        args.push(`'${escapedBody}'`)
      }

      execSync(args.join(' '), {
        stdio: 'ignore',
        timeout: 5000, // 防止命令挂起
      })
      this.logger.info(`通知已通过 notify-send 发送: ${title}`)
      return true
    } catch (err) {
      this.logger.error(`notify-send 发送通知失败: ${title}`, err as Error)
      return false
    }
  }

  /**
   * 获取 Linux 通知图标路径
   * - Linux 使用 .png 格式（.ico 不被支持）
   * - 自动将 .ico 后缀替换为 .png
   */
  private normalizeIconForLinux(iconPath?: string): string | undefined {
    if (!iconPath) return undefined
    // Linux 不支持 .ico 格式，尝试替换为 .png
    if (iconPath.endsWith('.ico')) {
      const pngPath = iconPath.replace(/\.ico$/, '.png')
      this.logger.debug(`Linux 平台：图标格式转换 ${iconPath} → ${pngPath}`)
      return pngPath
    }
    return iconPath
  }

  /**
   * 发送系统通知
   * @param title 通知标题
   * @param options 通知选项（body / icon / silent / onClick）
   * @returns 是否成功发送
   */
  send(title: string, options: NotificationOptions | string = {}): boolean {
    // 兼容直接传 body 字符串的情况
    const opts: NotificationOptions =
      typeof options === 'string' ? { body: options } : options

    // Linux 平台：自动转换图标格式
    const icon =
      process.platform === 'linux'
        ? this.normalizeIconForLinux(opts.icon)
        : opts.icon

    // 检查 Electron Notification 是否支持
    if (!Notification.isSupported()) {
      this.logger.warn('Electron Notification 不支持当前系统')

      // Linux 降级：尝试 notify-send
      if (process.platform === 'linux') {
        this.logger.info('尝试 Linux 降级方案：notify-send')
        return this.sendViaNotifySend(title, { ...opts, icon })
      }

      return false
    }

    try {
      const notification = new Notification({
        title,
        body: opts.body ?? '',
        icon,
        silent: opts.silent ?? false,
      })

      // 注册点击回调
      if (opts.onClick) {
        notification.on('click', () => {
          try {
            opts.onClick!()
          } catch (err) {
            this.logger.error('通知点击回调执行异常', err as Error)
          }
        })
      }

      // 记录通知关闭事件（调试用）
      notification.on('close', () => {
        this.logger.debug(`通知已关闭: ${title}`)
      })

      notification.show()
      this.logger.info(`通知已发送: ${title}`)
      return true
    } catch (err) {
      this.logger.error(`发送通知失败: ${title}`, err as Error)

      // Linux 降级：Electron Notification 异常时尝试 notify-send
      if (process.platform === 'linux') {
        this.logger.info('Electron Notification 异常，尝试 notify-send 降级')
        return this.sendViaNotifySend(title, { ...opts, icon })
      }

      return false
    }
  }

  /**
   * 快捷方法：发送带 body 的通知
   */
  notify(title: string, body: string, onClick?: () => void): boolean {
    return this.send(title, { body, onClick })
  }

  /**
   * 检查当前平台是否支持系统通知
   * Linux：检查 Electron Notification 或 notify-send 是否可用
   */
  isSupported(): boolean {
    if (Notification.isSupported()) return true
    if (process.platform === 'linux' && this.notifySendAvailable) return true
    return false
  }
}
```

```typescript
import { app } from 'electron'
import type { Logger } from '../../../src/main/types/logger'

/**
 * 开机自启服务
 * 封装 Electron app.setLoginItemSettings，提供跨平台开机自启管理
 *
 * - Windows: 通过注册表实现
 * - macOS:   通过 Login Items 实现
 * - Linux:   通过 ~/.config/autostart/*.desktop 文件实现
 */
export class AutostartService {
  private logger: Logger

  constructor(logger: Logger) {
    this.logger = logger
  }

  /**
   * 检查当前是否已启用开机自启
   * @returns 是否启用
   */
  isEnabled(): boolean {
    try {
      const settings = app.getLoginItemSettings()
      return settings.openAtLogin
    } catch (err) {
      this.logger.error('检查开机自启状态失败', err as Error)
      return false
    }
  }

  /**
   * 设置开机自启
   * @param enabled 是否启用
   * @returns 是否设置成功
   */
  setEnabled(enabled: boolean): boolean {
    try {
      app.setLoginItemSettings({
        openAtLogin: enabled,
        // macOS 专属：以隐藏模式启动（不显示窗口）
        ...(process.platform === 'darwin' && { openAsHidden: true }),
        // Windows 专属：指定启动参数
        ...(process.platform === 'win32' && { args: ['--hidden'] }),
      })

      this.logger.info(`开机自启已${enabled ? '启用' : '禁用'}`)
      return true
    } catch (err) {
      this.logger.error(`设置开机自启失败: ${enabled}`, err as Error)
      return false
    }
  }

  /**
   * 切换开机自启状态
   * @returns 切换后是否启用
   */
  toggle(): boolean {
    const current = this.isEnabled()
    const success = this.setEnabled(!current)
    return success ? !current : current
  }
}
```

```typescript
import { useEffect, useState, useCallback } from 'react'
import { ipcRenderer } from 'electron'

/** 托盘菜单项 */
interface MenuItem {
  id: string
  label: string
  icon?: string
  shortcut?: string
  divider?: boolean
  action: () => void
}

/**
 * 系统托盘菜单 UI 组件
 *
 * 在渲染进程中提供托盘菜单的可视化交互界面，
 * 与主进程的 TrayService 配合，处理菜单点击事件。
 *
 * 功能：
 * - 显示/隐藏主窗口
 * - 快速跳转设置页
 * - 快速跳转模块管理页
 * - 查看当前运行状态
 * - 退出应用
 */
export default function TrayMenu() {
  const [visible, setVisible] = useState(false)
  const [position, setPosition] = useState({ x: 0, y: 0 })
  const [isWindowVisible, setIsWindowVisible] = useState(true)
  const [moduleCount, setModuleCount] = useState(0)

  /** 监听主进程发来的托盘菜单打开事件 */
  useEffect(() => {
    const handleOpen = (_event: unknown, data: { x: number; y: number }) => {
      setPosition(data)
      setVisible(true)
    }

    const handleClose = () => setVisible(false)

    const handleWindowState = (_event: unknown, state: { visible: boolean }) => {
      setIsWindowVisible(state.visible)
    }

    const handleModuleCount = (_event: unknown, count: number) => {
      setModuleCount(count)
    }

    ipcRenderer.on('tray-menu:open', handleOpen)
    ipcRenderer.on('tray-menu:close', handleClose)
    ipcRenderer.on('tray:window-state', handleWindowState)
    ipcRenderer.on('tray:module-count', handleModuleCount)

    // 请求初始状态
    ipcRenderer.send('tray:request-state')

    return () => {
      ipcRenderer.removeListener('tray-menu:open', handleOpen)
      ipcRenderer.removeListener('tray-menu:close', handleClose)
      ipcRenderer.removeListener('tray:window-state', handleWindowState)
      ipcRenderer.removeListener('tray:module-count', handleModuleCount)
    }
  }, [])

  /** 点击菜单项后关闭菜单 */
  const handleAction = useCallback((action: () => void) => {
    setVisible(false)
    action()
  }, [])

  /** 点击外部关闭菜单 */
  useEffect(() => {
    if (!visible) return

    const handleClickOutside = (e: MouseEvent) => {
      const target = e.target as HTMLElement
      if (!target.closest('.tray-menu')) {
        setVisible(false)
      }
    }

    document.addEventListener('mousedown', handleClickOutside)
    return () => document.removeEventListener('mousedown', handleClickOutside)
  }, [visible])

  /** 快捷键关闭 */
  useEffect(() => {
    if (!visible) return

    const handleKeyDown = (e: KeyboardEvent) => {
      if (e.key === 'Escape') setVisible(false)
    }

    document.addEventListener('keydown', handleKeyDown)
    return () => document.removeEventListener('keydown', handleKeyDown)
  }, [visible])

  /** 菜单项定义 */
  const menuItems: MenuItem[] = [
    {
      id: 'toggle-window',
      label: isWindowVisible ? '隐藏主窗口' : '显示主窗口',
      icon: isWindowVisible ? '👁️' : '👁️‍🗨️',
      shortcut: 'Ctrl+Shift+A',
      action: () => ipcRenderer.send('tray:toggle-window'),
    },
    { id: 'divider-1', label: '', divider: true, action: () => {} },
    {
      id: 'settings',
      label: '设置',
      icon: '⚙️',
      shortcut: 'Ctrl+,',
      action: () => {
        ipcRenderer.send('tray:navigate', 'settings')
        ipcRenderer.send('tray:show-window')
      },
    },
    {
      id: 'modules',
      label: `模块管理${moduleCount > 0 ? ` (${moduleCount})` : ''}`,
      icon: '🧩',
      action: () => {
        ipcRenderer.send('tray:navigate', 'modules')
        ipcRenderer.send('tray:show-window')
      },
    },
    { id: 'divider-2', label: '', divider: true, action: () => {} },
    {
      id: 'quit',
      label: '退出',
      icon: '🚪',
      shortcut: 'Ctrl+Q',
      action: () => ipcRenderer.send('tray:quit'),
    },
  ]

  if (!visible) return null

  return (
    <div

        position: 'fixed',
        left: position.x,
        top: position.y,
        zIndex: 99999,
        minWidth: 220,

        border: '1px solid var(--border-color, #313244)',
        borderRadius: 8,
        boxShadow: '0 8px 32px rgba(0, 0, 0, 0.4)',
        padding: '6px 0',
        animation: 'tray-menu-enter 0.15s ease-out',
      }}
    >
      {menuItems.map((item) =>
        item.divider ? (
          <div
            key={item.id}
            style={{
              height: 1,
              background: 'var(--border-color, #313244)',
              margin: '4px 12px',
            }}
          />
        ) : (
          <button
            key={item.id}
            onClick={() => handleAction(item.action)}
            style={{
              display: 'flex',
              alignItems: 'center',
              width: '100%',
              padding: '8px 16px',
              border: 'none',
              background: 'transparent',
              color: item.id === 'quit' ? 'var(--color-danger, #f38ba8)' : 'var(--text-primary, #cdd6f4)',
              fontSize: 13,
              cursor: 'pointer',
              textAlign: 'left',
              transition: 'background 0.15s',
            }}
            onMouseEnter={(e) => {
              e.currentTarget.style.background = 'var(--bg-hover, #313244)'
            }}
            onMouseLeave={(e) => {
              e.currentTarget.style.background = 'transparent'
            }}
          >
            {item.icon && (
              <span style={{ marginRight: 10, fontSize: 14, width: 20, textAlign: 'center' }}>
                {item.icon}
              </span>
            )}
            <span style={{ flex: 1 }}>{item.label}</span>
            {item.shortcut && (
              <span style={{ fontSize: 11, color: 'var(--text-secondary, #6c7086)', marginLeft: 12 }}>
                {item.shortcut}
              </span>
            )}
          </button>
        )
      )}

      {/* 注入动画样式 */}
      <style>{`
        @keyframes tray-menu-enter {
          from { opacity: 0; transform: translateY(-4px) scale(0.98); }
          to { opacity: 1; transform: translateY(0) scale(1); }
        }
      `}</style>
    </div>
  )
}
```

#### 步骤 4：实现模块入口

**modules/desktop-integration/index.ts**：

```typescript
import type { Module, ModuleContext, Capability } from '../../src/main/types/module'
import { TrayService } from './services/tray'
import { HotkeyService } from './services/hotkey'
import { NotificationService } from './services/notification'
import { AutostartService } from './services/autostart'

export default class DesktopIntegrationModule implements Module {
  id = 'desktop'
  meta!: import('../../src/main/types/module').ModuleMeta

  private tray!: TrayService
  private hotkey!: HotkeyService
  private notification!: NotificationService
  private autostart!: AutostartService
  private context!: ModuleContext

  async activate(context: ModuleContext): Promise<void> {
    this.context = context

    this.tray = new TrayService(context.logger)
    this.hotkey = new HotkeyService(context.logger)
    this.notification = new NotificationService(context.logger)
    this.autostart = new AutostartService(context.logger)

    // 注册默认快捷键
    this.hotkey.register('CommandOrControl+Shift+A', '显示/隐藏助手', () => {
      // 通过事件总线通知主窗口
      context.eventBus.emit('on_toggle_window', {})
    })

    context.logger.info('桌面集成模块已激活')
  }

  async deactivate(): Promise<void> {
    this.hotkey.unregisterAll()
    this.tray.destroy()
    this.context.logger.info('桌面集成模块已停用')
  }

  getCapabilities(): Capability[] {
    return [
      {
        type: 'tool',
        name: 'notify',
        description: '发送系统通知',
        priority: 5,
        moduleId: this.id,
        handler: {
          execute: async (params) => {
            const { title, body } = params as { title: string; body: string }
            this.notification.send(title, body)
            return { success: true }
          }
        }
      },
      {
        type: 'tool',
        name: 'hotkey_register',
        description: '注册全局快捷键',
        priority: 5,
        moduleId: this.id,
        handler: {
          execute: async (params) => {
            const { accelerator, description, action } = params as {
              accelerator: string
              description: string
              action: string
            }
            const success = this.hotkey.register(accelerator, description, () => {
              this.context.eventBus.emit('on_hotkey_triggered', { accelerator, action })
            })
            return { success, accelerator }
          }
        }
      }
    ]
  }
}
```
