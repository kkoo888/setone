#### 步骤 1：实现工具注册表

**modules/tools/services/tool-registry.ts**：

```typescript
import type { ToolDefinition } from '../../../src/main/types/ai'
import type { Logger } from '../../../src/main/types/logger'

export interface RegisteredTool {
  definition: ToolDefinition
  moduleId: string
  handler: (params: Record<string, unknown>) => Promise<unknown>
}

export class ToolRegistry {
  private tools = new Map<string, RegisteredTool>()
  private logger: Logger

  constructor(logger: Logger) {
    this.logger = logger
  }

  register(tool: RegisteredTool): void {
    if (this.tools.has(tool.definition.function.name)) {
      this.logger.warn(`工具 "${tool.definition.function.name}" 已存在，将被覆盖`)
    }
    this.tools.set(tool.definition.function.name, tool)
    this.logger.debug(`工具已注册: ${tool.definition.function.name}`)
  }

  unregister(name: string): void {
    this.tools.delete(name)
  }

  unregisterByModule(moduleId: string): void {
    // <!-- ✅ Issue#311: 已修复 — 遍历中不直接删除 Map 条目，先收集 key 再批量删除 -->
    const keysToDelete: string[] = []
    for (const [name, tool] of this.tools) {
      if (tool.moduleId === moduleId) {
        keysToDelete.push(name)
      }
    }
    for (const name of keysToDelete) {
      this.tools.delete(name)
    }
  }

  get(name: string): RegisteredTool | undefined {
    return this.tools.get(name)
  }

  getAll(): RegisteredTool[] {
    return Array.from(this.tools.values())
  }

  getDefinitions(): ToolDefinition[] {
    return this.getAll().map((t) => t.definition)
  }
}

#### 步骤 2：实现工具路由器

**modules/tools/services/tool-router.ts**：

```typescript
import type { ToolDefinition } from '../../../src/main/types/ai'
import type { AIService } from '../../../src/main/types/ai'
import type { ToolRegistry } from './tool-registry'
import type { ToolExecutor } from './tool-executor'
import type { ToolExecutionContext, ToolExecutionResult } from './tool-executor'
import type { Logger } from '../../../src/main/types/logger'

export interface RouteResult {
  toolName: string
  params: Record<string, unknown>
  confidence: number
}

export class ToolRouter {
  // ─── 路由缓存层（Issue#313） ─────────────────────────────
  /** LRU 路由缓存：input → RouteResult，避免重复 AI 调用 */
  private routeCache = new Map<string, { result: RouteResult | null; timestamp: number }>()
  /** 缓存最大条目数 */
  private static readonly CACHE_MAX_SIZE = 256
  /** 缓存 TTL（毫秒），默认 5 分钟 */
  private static readonly CACHE_TTL_MS = 5 * 60 * 1000

  // ─── 快速路径配置（Issue#313） ───────────────────────────
  /**
   * 快速路径关键词映射表：当用户输入匹配某个模式时，直接路由到对应工具，
   * 跳过 AI 调用，大幅降低延迟。由模块初始化时从已注册工具的 description
   * 中自动提取高频关键词，也可通过 configureFastPath() 手动配置。
   */
  private fastPathRules = new Map<string, { toolName: string; extractParams?: (input: string) => Record<string, unknown> }>()

  // ─── 超时保护配置（Issue#313） ───────────────────────────
  /** AI 路由请求超时时间（毫秒），默认 15 秒 */
  private static readonly ROUTE_TIMEOUT_MS = 15_000

  // ─── 置信度降级策略（Issue#314） ─────────────────────────
  /** 路由置信度阈值：低于此值触发降级策略（默认 0.6，可由模块配置覆盖） */
  private confidenceThreshold: number

  constructor(
    private ai: AIService,
    private registry: ToolRegistry,
    private executor: ToolExecutor,
    private logger: Logger,
    options?: { confidenceThreshold?: number }
  ) {
    this.confidenceThreshold = options?.confidenceThreshold ?? 0.6
  }

  /**
   * 配置快速路径规则
   * 在模块加载完成后调用，为常见工具注册关键词快速匹配规则，
   * 使这些请求完全跳过 AI 调用，响应时间从 ~2s 降至 ~1ms。
   *
   * @example
   * router.configureFastPath([
   *   { pattern: /天气|weather/i, toolName: 'weather_query' },
   *   { pattern: /截图|screenshot/i, toolName: 'screen_capture' },
   *   { pattern: /提醒|remind/i, toolName: 'reminder_create' },
   * ])
   */
  configureFastPath(rules: Array<{
    pattern: RegExp
    toolName: string
    extractParams?: (input: string) => Record<string, unknown>
  }>): void {
    for (const rule of rules) {
      this.fastPathRules.set(rule.pattern.source, {
        toolName: rule.toolName,
        extractParams: rule.extractParams
      })
    }
    this.logger.info(`快速路径已配置 ${rules.length} 条规则`)
  }

  /**
   * 尝试快速路径匹配（Issue#313）
   * 遍历快速路径规则表，首条匹配即返回结果，完全跳过 AI 调用。
   * @returns 匹配到的 RouteResult，未匹配返回 null
   */
  private tryFastPath(input: string): RouteResult | null {
    const normalized = input.trim().toLowerCase()
    for (const [_patternSource, rule] of this.fastPathRules) {
      // 重建正则（Map key 是 source 字符串）
      const regex = new RegExp(_patternSource, 'i')
      if (regex.test(normalized)) {
        const tool = this.registry.get(rule.toolName)
        if (!tool) continue // 工具未注册，跳过此规则
        const params = rule.extractParams ? rule.extractParams(input) : {}
        this.logger.debug(`快速路径命中: "${rule.toolName}"`, { input: input.slice(0, 50) })
        return { toolName: rule.toolName, params, confidence: 1.0 }
      }
    }
    return null
  }

  /**
   * 路由用户输入到合适的工具（Issue#313：已添加缓存 + 快速路径 + 超时保护）
   *
   * 执行优先级：
   * 1. 缓存命中 → 直接返回（~0ms）
   * 2. 快速路径命中 → 直接返回（~1ms，跳过 AI）
   * 3. AI 路由（带超时保护） → 缓存结果后返回（~1-15s）
   */
  async route(input: string): Promise<RouteResult | null> {
    const tools = this.registry.getDefinitions()
    if (tools.length === 0) return null

    // ── 1. 缓存查找（Issue#313） ──
    const cacheKey = input.trim().toLowerCase()
    const cached = this.routeCache.get(cacheKey)
    if (cached && (Date.now() - cached.timestamp) < ToolRouter.CACHE_TTL_MS) {
      this.logger.debug('路由缓存命中', { input: input.slice(0, 50) })
      return cached.result
    }
    // 清理过期缓存
    if (cached) {
      this.routeCache.delete(cacheKey)
    }

    // ── 2. 快速路径匹配（Issue#313） ──
    const fastResult = this.tryFastPath(input)
    if (fastResult) {
      this.routeCache.set(cacheKey, { result: fastResult, timestamp: Date.now() })
      this.pruneCache()
      return fastResult
    }

    // ── 3. AI 路由（带超时保护，Issue#313） ──
    const systemPrompt = `你是一个工具路由引擎。根据用户输入，判断应该使用哪个工具。

可用工具：
${tools.map((t) => `- ${t.function.name}: ${t.function.description}`).join('\n')}

返回 JSON 格式：
{
  "toolName": "工具名称",
  "params": { ... },
  "confidence": 0.0-1.0
}

如果不需要使用工具，返回 null。`

    let result: RouteResult | null = null
    try {
      const response = await this.ai.chat(
        [
          { role: 'system', content: systemPrompt },
          { role: 'user', content: input }
        ],
        { temperature: 0.1 }
      )

      const content = response.message.content
      result = this.extractRouteResult(content)

      // ── 置信度校验与降级策略（Issue#314） ──
      // 低于阈值时，不执行工具路由，降级为通用对话（返回 null）
      // 快速路径命中（confidence=1.0）不受此限制
      if (result && result.confidence < this.confidenceThreshold) {
        this.logger.warn(
          `工具路由置信度过低（${result.confidence.toFixed(2)} < ${this.confidenceThreshold}），` +
          `工具 "${result.toolName}" 不执行，降级为通用对话`,
          { input: input.slice(0, 100), toolName: result.toolName, confidence: result.confidence }
        )
        result = null
      }
    } catch (err) {
      // 超时保护：AbortSignal.timeout() 触发的 AbortError 或网络错误
      const isTimeout = (err as Error).name === 'AbortError' || (err as Error).message?.includes('timeout')
      if (isTimeout) {
        this.logger.warn(`路由 AI 请求超时（${ToolRouter.ROUTE_TIMEOUT_MS}ms），回退到通用对话`, { input: input.slice(0, 50) })
      } else {
        this.logger.error('工具路由结果解析失败', err as Error)
      }
      result = null
    }

    // ── 4. 缓存结果（Issue#313） ──
    this.routeCache.set(cacheKey, { result, timestamp: Date.now() })
    this.pruneCache()

    return result
  }

  /**
   * 清理过期缓存并淘汰最老条目（LRU 策略）
   */
  private pruneCache(): void {
    // 清理过期条目
    const now = Date.now()
    for (const [key, entry] of this.routeCache) {
      if (now - entry.timestamp >= ToolRouter.CACHE_TTL_MS) {
        this.routeCache.delete(key)
      }
    }
    // 淘汰最老条目（Map 插入顺序即时间顺序，删除头部即可）
    while (this.routeCache.size > ToolRouter.CACHE_MAX_SIZE) {
      const oldest = this.routeCache.keys().next().value
      if (oldest !== undefined) this.routeCache.delete(oldest)
      else break
    }
  }

  /** 清空路由缓存（工具注册变更时调用） */
  clearCache(): void {
    this.routeCache.clear()
    this.logger.debug('路由缓存已清空')
  }

  /**
   * 从 AI 回复文本中安全提取 RouteResult
   * 优先级：```json 代码块 > 花括号配对提取 > 非贪婪回退
   * 提取后校验必要字段（toolName），缺失则返回 null
   */
  private extractRouteResult(text: string): RouteResult | null {
    // 策略 1：从 ```json ... ``` 代码块中提取
    const codeBlockMatch = text.match(/```(?:json)?\s*\n?([\s\S]*?)\n?\s*```/)
    if (codeBlockMatch) {
      const parsed = this.safeParseRouteResult(codeBlockMatch[1].trim())
      if (parsed) return parsed
    }

    // 策略 2：花括号配对提取（正确处理字符串内的转义花括号）
    const firstBrace = text.indexOf('{')
    if (firstBrace !== -1) {
      let depth = 0
      let inString = false
      let escapeNext = false
      for (let i = firstBrace; i < text.length; i++) {
        const ch = text[i]
        if (escapeNext) { escapeNext = false; continue }
        if (ch === '\\' && inString) { escapeNext = true; continue }
        if (ch === '"') { inString = !inString; continue }
        if (inString) continue
        if (ch === '{') depth++
        else if (ch === '}') depth--
        if (depth === 0) {
          const parsed = this.safeParseRouteResult(text.slice(firstBrace, i + 1))
          if (parsed) return parsed
          break
        }
      }
    }

    // 策略 3：非贪婪匹配回退
    const conservativeMatch = text.match(/\{[^{}]*(?:\{[^{}]*\}[^{}]*)*\}/)
    if (conservativeMatch) {
      const parsed = this.safeParseRouteResult(conservativeMatch[0])
      if (parsed) return parsed
    }

    return null
  }

  /**
   * 安全解析并校验 RouteResult 结构
   * - JSON.parse 失败返回 null（不抛异常）
   * - 缺少必要字段 toolName 时返回 null
   * - confidence 超出 [0,1] 范围时修正
   */
  private safeParseRouteResult(raw: string): RouteResult | null {
    try {
      const parsed: unknown = JSON.parse(raw)
      if (parsed === null || typeof parsed !== 'object' || Array.isArray(parsed)) return null
      const obj = parsed as Record<string, unknown>
      // 必要字段校验
      if (typeof obj.toolName !== 'string' || obj.toolName.length === 0) return null
      // 可选字段类型修正
      const confidence = typeof obj.confidence === 'number'
        ? Math.max(0, Math.min(1, obj.confidence))
        : 0.5
      const params = (typeof obj.params === 'object' && obj.params !== null && !Array.isArray(obj.params))
        ? obj.params as Record<string, unknown>
        : {}
      return { toolName: obj.toolName, params, confidence }
    } catch { return null }
  }

  /**
   * 执行指定工具
   * 委托给 ToolExecutor，获得超时控制（30s）、重试和统一错误处理
   * 被版块21 TaskPlanner/TaskExecutor 调用，提供统一的工具执行入口
   * @param toolName 工具名称
   * @param params 工具参数
   * @returns 工具执行结果
   * @throws 工具未找到或执行失败时抛出异常
   */
  async execute(toolName: string, params: Record<string, unknown>): Promise<unknown> {
    return this.executor.executeOrThrow(toolName, params, { source: 'user' })
  }

  /**
   * 执行工具并返回详细结果（含成功/失败状态、耗时等）
   * 供 TaskExecutor 等需要细粒度控制的调用方使用
   * @param toolName 工具名称
   * @param params 工具参数
   * @param context 执行上下文（来源、会话ID、超时覆盖等）
   * @returns ToolExecutionResult 统一结果对象
   */
  async executeDetailed(
    toolName: string,
    params: Record<string, unknown>,
    context?: ToolExecutionContext
  ): Promise<ToolExecutionResult> {
    return this.executor.execute(toolName, params, context)
  }

  /**
   * 列出所有已注册工具（供 TaskPlanner 获取可用工具列表）
   */
  listTools(): Array<{ name: string; description: string }> {
    return this.registry.getAll().map(t => ({
      name: t.definition.function.name,
      description: t.definition.function.description
    }))
  }
}
```

```typescript
import type { Logger } from '../../../src/main/types/logger'
import type { ToolRegistry, RegisteredTool } from './tool-registry'

// ─── 类型定义 ───────────────────────────────────────────────

/** 工具执行上下文 */
export interface ToolExecutionContext {
  /** 触发来源：用户直接输入 / 任务计划 / 定时触发 */
  source: 'user' | 'task' | 'schedule'
  /** 会话 ID，用于日志追踪 */
  sessionId?: string
  /** 超时覆盖（毫秒），优先级高于默认值 */
  timeoutMs?: number
}

/** 单次工具执行结果 */
export interface ToolExecutionResult {
  /** 是否成功 */
  success: boolean
  /** 工具返回值（成功时） */
  data?: unknown
  /** 错误信息（失败时） */
  error?: string
  /** 错误码 */
  errorCode?: ToolErrorCode
  /** 执行耗时（毫秒） */
  durationMs: number
  /** 工具名称 */
  toolName: string
  /** 执行时间戳 */
  timestamp: number
}

/** 工具错误码枚举 */
export const ToolErrorCode = {
  TOOL_NOT_FOUND: 'TOOL_NOT_FOUND',
  EXECUTION_TIMEOUT: 'EXECUTION_TIMEOUT',
  EXECUTION_ERROR: 'EXECUTION_ERROR',
  INVALID_PARAMS: 'INVALID_PARAMS'
} as const

export type ToolErrorCode = (typeof ToolErrorCode)[keyof typeof ToolErrorCode]

// ─── 常量 ───────────────────────────────────────────────────

/** 全局默认工具执行超时（毫秒） */
const DEFAULT_TIMEOUT_MS = 30_000

/** 最大重试次数 */
const MAX_RETRIES = 1

/** 可重试的错误模式 */
const RETRYABLE_PATTERNS = [
  /ECONNRESET/i,
  /ETIMEDOUT/i,
  /socket hang up/i,
  /429/
]

// ─── ToolExecutor 类 ────────────────────────────────────────

/**
 * 模块级工具执行器
 *
 * 职责：
 * 1. 接收 ToolRouter 的工具调用请求
 * 2. 从 ToolRegistry 查找已注册的工具处理器
 * 3. 参数校验 → 超时控制 → 执行 → 重试 → 结果格式化
 * 4. 统一错误处理与日志记录
 *
 * 与 ToolRouter 的关系：
 * - ToolRouter 负责「决定调用哪个工具」（AI 意图路由）
 * - ToolExecutor 负责「安全地执行工具调用」（运行时执行引擎）
 * - ToolRouter.execute() 内部委托给 ToolExecutor.execute()
 *
 * 与版块10 src/main/core/tool-executor.ts 的关系：
 * - 版块10是核心层 ToolExecutor，供 AIDispatcher 直接使用
 * - 本模块是模块层 ToolExecutor，通过 ToolRegistry 获取工具并执行
 * - 两者共享相同的执行策略（超时、重试、错误码），职责分层不同
 */
export class ToolExecutor {
  private defaultTimeout: number
  private logger: Logger
  private registry: ToolRegistry

  constructor(registry: ToolRegistry, logger: Logger, defaultTimeoutMs = DEFAULT_TIMEOUT_MS) {
    this.registry = registry
    this.logger = logger
    this.defaultTimeout = defaultTimeoutMs
  }

  // ─── 核心执行 ──────────────────────────────────────────

  /**
   * 执行工具调用
   *
   * 完整流程：查找工具 → 参数校验 → 超时执行 → 重试 → 结果格式化
   *
   * @param toolName 工具名称
   * @param params 工具参数
   * @param context 执行上下文
   * @returns ToolExecutionResult 统一结果对象
   */
  async execute(
    toolName: string,
    params: Record<string, unknown>,
    context: ToolExecutionContext = { source: 'user' }
  ): Promise<ToolExecutionResult> {
    const startTime = Date.now()
    const timestamp = startTime

    // 1. 从注册表查找工具
    const tool = this.registry.get(toolName)
    if (!tool) {
      this.logger.warn(`[ToolExecutor] 工具未找到: ${toolName}`)
      return {
        success: false,
        error: `工具 "${toolName}" 未注册`,
        errorCode: ToolErrorCode.TOOL_NOT_FOUND,
        durationMs: Date.now() - startTime,
        toolName,
        timestamp
      }
    }

    // 2. 参数校验（基础检查）
    const paramError = this.validateParams(toolName, params)
    if (paramError) {
      return {
        success: false,
        error: paramError,
        errorCode: ToolErrorCode.INVALID_PARAMS,
        durationMs: Date.now() - startTime,
        toolName,
        timestamp
      }
    }

    // 3. 确定超时时间（优先级：context > 全局默认）
    const timeoutMs = context.timeoutMs ?? this.defaultTimeout

    // 4. 带重试的执行
    let lastError: Error | undefined
    const maxAttempts = MAX_RETRIES + 1

    for (let attempt = 1; attempt <= maxAttempts; attempt++) {
      try {
        this.logger.info(`[ToolExecutor] 执行工具: ${toolName}`, {
          params,
          source: context.source,
          sessionId: context.sessionId,
          attempt,
          timeoutMs
        })

        const result = await this.executeWithTimeout(tool.handler, params, timeoutMs)

        const durationMs = Date.now() - startTime
        this.logger.info(`[ToolExecutor] 工具 "${toolName}" 执行成功`, {
          durationMs,
          attempt
        })

        return {
          success: true,
          data: result,
          durationMs,
          toolName,
          timestamp
        }
      } catch (err) {
        lastError = err as Error
        const isRetryable = this.isRetryableError(lastError)

        this.logger.warn(`[ToolExecutor] 工具 "${toolName}" 执行失败 (尝试 ${attempt}/${maxAttempts})`, {
          error: lastError.message,
          isRetryable
        })

        // 不可重试的错误或已用完重试次数，直接跳出
        if (!isRetryable || attempt >= maxAttempts) break

        // 重试前短暂等待（指数退避）
        await this.delay(Math.min(1000 * attempt, 3000))
      }
    }

    // 5. 所有重试均失败，返回错误结果
    const durationMs = Date.now() - startTime
    const isTimeout = lastError?.message?.includes('执行超时')

    return {
      success: false,
      error: lastError?.message ?? '未知错误',
      errorCode: isTimeout ? ToolErrorCode.EXECUTION_TIMEOUT : ToolErrorCode.EXECUTION_ERROR,
      durationMs,
      toolName,
      timestamp
    }
  }

  /**
   * 执行工具并返回结果（抛异常版本，供 ToolRouter.execute() 调用）
   *
   * @param toolName 工具名称
   * @param params 工具参数
   * @param context 执行上下文
   * @returns 工具返回值
   * @throws 工具未找到或执行失败时抛出异常
   */
  async executeOrThrow(
    toolName: string,
    params: Record<string, unknown>,
    context?: ToolExecutionContext
  ): Promise<unknown> {
    const result = await this.execute(toolName, params, context)
    if (!result.success) {
      throw new Error(result.error ?? `工具 "${toolName}" 执行失败`)
    }
    return result.data
  }

  // ─── 内部方法 ──────────────────────────────────────────

  /** 带超时的工具执行 */
  private executeWithTimeout(
    handler: RegisteredTool['handler'],
    params: Record<string, unknown>,
    timeoutMs: number
  ): Promise<unknown> {
    return Promise.race([
      handler(params),
      new Promise<never>((_, reject) =>
        setTimeout(() => reject(new Error(`工具执行超时 (${timeoutMs}ms)`)), timeoutMs)
      )
    ])
  }

  /** 基础参数校验 */
  private validateParams(toolName: string, params: Record<string, unknown>): string | null {
    if (params === null || params === undefined || typeof params !== 'object') {
      return `工具 "${toolName}" 的参数必须是对象`
    }
    return null
  }

  /** 判断错误是否可重试 */
  private isRetryableError(err: Error): boolean {
    return RETRYABLE_PATTERNS.some((pattern) => pattern.test(err.message))
  }

  /** 延迟 */
  private delay(ms: number): Promise<void> {
    return new Promise((resolve) => setTimeout(resolve, ms))
  }

  /**
   * 格式化工具结果为可读字符串（供 AI 消息组装使用）
   */
  static formatResult(toolName: string, result: unknown): string {
    if (result === null || result === undefined) {
      return `工具 "${toolName}" 执行完成，无返回结果`
    }
    if (typeof result === 'string') return result
    try {
      return JSON.stringify(result, null, 2)
    } catch {
      return String(result)
    }
  }
}
```

> **与版块10 ToolExecutor 的区别**：版块10的 `src/main/core/tool-executor.ts` 是核心层实现，维护自己的工具注册表（`Map<string, ToolEntry>`），供 `AIDispatcher` 直接使用。本模块的 `modules/tools/services/tool-executor.ts` 是模块层实现，通过注入 `ToolRegistry` 获取工具，与 `ToolRouter` 配合工作。两者共享相同的执行策略（超时控制、重试逻辑、错误码体系），但职责分层不同——核心层面向全局调度，模块层面向模块内工具执行。

**modules/tools/index.ts**：

```typescript
import type { ModuleContext, ModuleLifecycle } from '../../src/main/types/module'
import type { AIService } from '../../src/main/types/ai'
import type { Logger } from '../../src/main/types/logger'
import { ToolRegistry } from './services/tool-registry'
import { ToolRouter } from './services/tool-router'
import { ToolExecutor } from './services/tool-executor'

/**
 * 工具路由模块入口
 *
 * 职责：
 * 1. 创建并组装 ToolRegistry → ToolExecutor → ToolRouter 依赖链
 * 2. 注册到模块生命周期，提供 init / dispose 完整生命周期管理
 * 3. 将 ToolRouter 和 ToolRegistry 暴露给模块系统，供其他模块（如版块21任务规划）使用
 *
 * 模块生命周期：
 * - onLoad → init()：创建实例、注入依赖、注册到服务容器
 * - onUnload → dispose()：清理注册表、释放资源
 */
export class ToolsModule implements ModuleLifecycle {
  private registry!: ToolRegistry
  private router!: ToolRouter
  private executor!: ToolExecutor
  private logger!: Logger

  /**
   * 模块初始化
   * 创建 ToolRegistry → ToolExecutor → ToolRouter 完整依赖链，
   * 并将 ToolRouter 注册到模块上下文，供外部模块通过 context.getService() 获取。
   */
  async init(context: ModuleContext): Promise<void> {
    this.logger = context.getLogger('tools-module')

    // 从模块上下文获取 AI 服务（由版块10 AI 调度器提供）
    const aiService = context.getService<AIService>('ai-service')

    // 1. 创建工具注册表
    this.registry = new ToolRegistry(this.logger)

    // 2. 创建工具执行器（注入注册表，获得超时控制和重试能力）
    const moduleConfig = context.getConfig<{ defaultTimeoutMs?: number; routingConfidenceThreshold?: number }>()
    this.executor = new ToolExecutor(
      this.registry,
      this.logger,
      moduleConfig?.defaultTimeoutMs
    )

    // 3. 创建工具路由器（注入 AI 服务、注册表、执行器、置信度阈值）
    this.router = new ToolRouter(aiService, this.registry, this.executor, this.logger, {
      confidenceThreshold: moduleConfig?.routingConfidenceThreshold
    })

    // 4. 将 ToolRouter 和 ToolRegistry 注册到服务容器，
    //    供版块21 TaskPlanner / TaskExecutor 通过 context.getService() 获取
    context.registerService('tools', this.router)
    context.registerService('tool-registry', this.registry)
    context.registerService('tool-executor', this.executor)

    this.logger.info('工具路由模块初始化完成', {
      registeredTools: this.registry.getAll().length
    })
  }

  /**
   * 模块销毁
   * 清空工具注册表，释放资源。
   */
  async dispose(): Promise<void> {
    // 按模块注销所有工具（防止内存泄漏）
    const allTools = this.registry.getAll()
    for (const tool of allTools) {
      this.registry.unregister(tool.definition.function.name)
    }

    this.logger.info('工具路由模块已销毁', {
      cleanedTools: allTools.length
    })
  }

  /** 获取工具路由器实例（供模块内部调试使用） */
  getRouter(): ToolRouter {
    return this.router
  }

  /** 获取工具注册表实例（供模块内部调试使用） */
  getRegistry(): ToolRegistry {
    return this.registry
  }

  /** 获取工具执行器实例（供模块内部调试使用） */
  getExecutor(): ToolExecutor {
    return this.executor
  }
}

export default ToolsModule

// 重导出服务类，供外部直接导入使用
export { ToolRegistry } from './services/tool-registry'
export { ToolRouter } from './services/tool-router'
export { ToolExecutor, ToolErrorCode } from './services/tool-executor'
export type { RegisteredTool } from './services/tool-registry'
export type { RouteResult } from './services/tool-router'
export type { ToolExecutionContext, ToolExecutionResult } from './services/tool-executor'

### 20.3 代码规范 <!-- ✅ Issue#315: 已修复 -->

- **工具定义**：遵循 Ollama Tool Calling 规范，`ToolDefinition` 包含 `type: 'function'` + `function.name` / `description` / `parameters`；参数 schema 使用 JSON Schema 格式
- **注册表（ToolRegistry）**：`Map<string, RegisteredTool>` 按工具名索引；同名工具覆盖并记录 warn 日志；支持 `unregisterByModule(moduleId)` 按模块批量注销（先收集 key 再删除，避免遍历中修改 Map）；模块卸载时自动清理
- **路由器（ToolRouter）**：使用 AI 进行意图识别，返回 `RouteResult`（toolName + params + confidence）
  - **路由优先级**：① LRU 缓存命中 → ② 快速路径关键词匹配 → ③ AI 路由（带超时保护）→ ④ 返回 null（无匹配）
  - **LRU 路由缓存**：`routeCache` 最大 256 条，TTL 5 分钟；`pruneCache()` 清理过期条目 + 淘汰最老条目（Map 插入顺序）；工具注册变更时调用 `clearCache()` 清空缓存
  - **快速路径（Fast Path）**：通过 `configureFastPath(rules)` 注册正则关键词规则，命中时直接路由（~1ms），完全跳过 AI 调用（~1-2s）；适用于高频场景（天气、截图、提醒等）
  - **AI 路由超时**：`ROUTE_TIMEOUT_MS = 15s`，超时后回退到通用对话，不阻塞用户
  - **路由结果安全解析**：`extractRouteResult()` 三级策略——```json 代码块 > 花括号配对提取 > 非贪婪回退；`safeParseRouteResult()` 校验 toolName 必填、confidence 限幅 [0,1]、params 类型修正
- **执行器（ToolExecutor）**：超时控制 + 指数退避重试 + 统一错误码
  - **执行超时**：默认 30s，支持 `context.timeoutMs` 覆盖（优先级：context > 全局默认）
  - **重试策略**：仅网络类错误（`ECONNRESET`、`ETIMEDOUT`、`socket hang up`、`429`）自动重试 1 次，重试前指数退避（1s/2s/3s cap）
  - **错误码**：`TOOL_NOT_FOUND`（工具未注册）/ `EXECUTION_TIMEOUT`（执行超时）/ `EXECUTION_ERROR`（执行异常）/ `INVALID_PARAMS`（参数校验失败）
  - **参数校验**：基础检查（params 必须为非 null 对象），模块可扩展更细粒度的 schema 校验
  - **formatResult()**：静态方法，null → "无返回结果"，string → 原样，object → JSON.stringify
- **委托模式**：`ToolRouter.execute()` 委托 `ToolExecutor.executeOrThrow()`（抛异常版本）；`ToolRouter.executeDetailed()` 委托 `ToolExecutor.execute()`（返回 ToolExecutionResult 对象）；职责分离——Router 决定调用谁，Executor 安全执行
- **与版块10核心层 ToolExecutor 的分层**：版块10 `src/main/core/tool-executor.ts` 是核心层，维护自己的工具注册表，供 AIDispatcher 直接使用；本模块 `modules/tools/services/tool-executor.ts` 是模块层，通过注入 ToolRegistry 获取工具；两者共享相同的执行策略（超时、重试、错误码），但职责分层不同
- **模块入口（ToolsModule）**：实现 `ModuleLifecycle`，`init()` 按顺序组装依赖链（ToolRegistry → ToolExecutor → ToolRouter）并通过 `context.registerService()` 暴露给其他模块；`dispose()` 按模块注销所有工具释放资源
- **模块配置**：`module.json` 声明模块元数据（id: "tools"）、能力列表（tool-routing / tool-execution）、默认配置（defaultTimeoutMs / maxRetries / enableAIRouting / routingConfidenceThreshold）和生命周期钩子
- **测试覆盖**：`tools.test.ts` 覆盖 ToolRegistry（注册/注销/按模块注销/查询）、ToolExecutor（执行成功/超时/重试/executeOrThrow/formatResult）、ToolRouter（路由缓存/委托执行/listTools）
