#### 步骤 1：实现沙箱管理器

**src/main/core/sandbox.ts**：

```typescript
import { utilityProcess, UtilityProcess } from 'electron'
import { join } from 'path'
import { existsSync } from 'fs'
import { EventEmitter } from 'events'
import type { ModuleMeta } from '../types/module'
import type { Logger } from '../types/logger'

export interface SandboxInstance {
  moduleId: string
  process: UtilityProcess
  status: 'starting' | 'running' | 'stopped' | 'error'
  resourceUsage: { memoryMB: number; cpuPercent: number }
}

export class SandboxManager extends EventEmitter {
  private sandboxes = new Map<string, SandboxInstance>()
  private logger: Logger

  constructor(logger: Logger) {
    super()
    this.logger = logger
  }

  /** 创建沙箱子进程 */
  async createSandbox(meta: ModuleMeta): Promise<SandboxInstance> {
    const workerPath = join(__dirname, 'sandbox-worker.js')

    // 安全环境变量：白名单模式，仅传递必要的变量，防止 API Key / Token 等敏感信息泄露
    // 解析入口文件：优先 .js（编译后），其次 .ts（开发时）
    const modulePath = meta.path || ''
    const entryJsPath = join(modulePath, 'index.js')
    const entryTsPath = join(modulePath, 'index.ts')
    const resolvedEntryPath = existsSync(entryJsPath) ? entryJsPath : entryTsPath

    const safeEnv: Record<string, string> = {
      NODE_ENV: process.env.NODE_ENV || 'production',
      MODULE_ID: meta.id,
      MODULE_PATH: resolvedEntryPath
    }
    // 白名单：仅追加安全的基础运行时变量（与 sandbox-worker.ts 中 ALLOWED_ENV_KEYS 保持一致）
    // ⚠️ 禁止使用 ...process.env 全量展开，否则会泄露 API_KEY、SECRET、TOKEN 等敏感变量
    const allowedKeys = ['PATH', 'HOME', 'LANG', 'LC_ALL', 'DISPLAY', 'APP_ROOT']
    for (const key of allowedKeys) {
      if (process.env[key]) safeEnv[key] = process.env[key]!
    }

    const childProcess = utilityProcess.fork(workerPath, [], {
      serviceName: `module-${meta.id}`,
      env: safeEnv
    })

    const sandbox: SandboxInstance = {
      moduleId: meta.id,
      process: childProcess,
      status: 'starting',
      resourceUsage: { memoryMB: 0, cpuPercent: 0 }
    }

    this.sandboxes.set(meta.id, sandbox)

    // 监听子进程消息
    childProcess.on('message', (message) => {
      this.handleMessage(meta.id, message)
    })

    // 监听子进程退出
    childProcess.on('exit', (code) => {
      sandbox.status = 'stopped'
      this.logger.warn(`沙箱进程退出: ${meta.id}`, { code })
      this.emit('sandbox:exit', { moduleId: meta.id, code })
    })

    // 监听错误
    childProcess.on('error', (err) => {
      sandbox.status = 'error'
      this.logger.error(`沙箱进程错误: ${meta.id}`, err)
      this.emit('sandbox:error', { moduleId: meta.id, error: err.message })
    })

    // 等待子进程启动信号（__started__），然后发送初始化命令
    await this.waitForStarted(meta.id)

    // ⚠️ 必须先注册 __ready__ 监听器，再发送 __init__，否则存在竞态条件：
    // worker 收到 __init__ 后可能在 waitForReady 注册监听之前就回复 __ready__，
    // 导致 __ready__ 消息丢失，最终超时。
    const readyPromise = this.waitForReady(meta.id)

    // 发送 __init__ 命令，触发模块加载
    childProcess.postMessage({
      channel: '__init__',
      data: { context: this.createSandboxContext(meta) }
    })

    // 等待模块初始化完成（__ready__）
    await readyPromise

    sandbox.status = 'running'
    this.logger.info(`沙箱已创建: ${meta.id}`)

    return sandbox
  }

  /** 向沙箱发送消息 */
  send(moduleId: string, channel: string, data?: unknown): boolean {
    const sandbox = this.sandboxes.get(moduleId)
    if (!sandbox || sandbox.status !== 'running') return false

    sandbox.process.postMessage({ channel, data })
    return true
  }

  /** 销毁沙箱 */
  async destroySandbox(moduleId: string): Promise<void> {
    const sandbox = this.sandboxes.get(moduleId)
    if (!sandbox) return

    // 通知子进程准备关闭
    sandbox.process.postMessage({ channel: '__shutdown__' })

    // 等待优雅关闭，超时强制杀掉
    await new Promise<void>((resolve) => {
      const timeout = setTimeout(() => {
        sandbox.process.kill()
        resolve()
      }, 5000)

      sandbox.process.once('exit', () => {
        clearTimeout(timeout)
        resolve()
      })
    })

    this.sandboxes.delete(moduleId)
    this.logger.info(`沙箱已销毁: ${moduleId}`)
  }

  /** 等待子进程启动（收到 __started__） */
  private waitForStarted(moduleId: string): Promise<void> {
    return new Promise((resolve, reject) => {
      const timeout = setTimeout(() => {
        reject(new Error(`沙箱 "${moduleId}" 子进程启动超时`))
      }, 10000)

      const handler = (message: { channel: string }) => {
        if (message.channel === '__started__') {
          clearTimeout(timeout)
          this.removeListener(`message:${moduleId}`, handler)
          resolve()
        }
      }

      this.on(`message:${moduleId}`, handler)
    })
  }

  /** 等待模块初始化完成（收到 __ready__） */
  private waitForReady(moduleId: string): Promise<void> {
    return new Promise((resolve, reject) => {
      const timeout = setTimeout(() => {
        reject(new Error(`沙箱 "${moduleId}" 模块初始化超时`))
      }, 15000)  // 模块初始化可能较慢，给 15s

      const handler = (message: { channel: string }) => {
        if (message.channel === '__ready__') {
          clearTimeout(timeout)
          this.removeListener(`message:${moduleId}`, handler)
          resolve()
        }
        if (message.channel === '__error__') {
          clearTimeout(timeout)
          this.removeListener(`message:${moduleId}`, handler)
          reject(new Error(`沙箱 "${moduleId}" 模块初始化失败`))
        }
      }

      this.on(`message:${moduleId}`, handler)
    })
  }

  /** 创建沙箱上下文（精简版 ModuleContext，沙箱内使用） */
  private createSandboxContext(meta: ModuleMeta): Record<string, unknown> {
    return {
      moduleId: meta.id,
      // 沙箱内的模块通过 IPC 与主进程通信获取 config/ai 等能力
      // 不直接暴露主进程对象
    }
  }

  /** 处理子进程消息 */
  private handleMessage(moduleId: string, message: { channel: string; data?: unknown }): void {
    // 内部控制消息已在 waitForStarted/waitForReady 中处理
    if (message.channel === '__started__' || message.channel === '__ready__' || message.channel === '__error__') {
      this.emit(`message:${moduleId}`, message)
      return
    }

    // 审计日志：沙箱拦截操作上报（记录到主进程日志并通知安全监控）
    if (message.channel === '__audit__') {
      const auditData = message.data as { type: string; action: string; target: string; blocked: boolean; detail?: string }
      if (auditData?.blocked) {
        this.logger.warn(`[安全审计] 沙箱 "${moduleId}" 拦截操作`, auditData)
      } else {
        this.logger.debug(`[安全审计] 沙箱 "${moduleId}" 允许操作`, auditData)
      }
      this.emit('sandbox:audit', { moduleId, ...auditData })
      return
    }

    // 转发到事件总线或 IPC
    this.emit('sandbox:message', { moduleId, channel: message.channel, data: message.data })
  }

  /** 获取所有沙箱状态 */
  getStatuses(): Array<{ moduleId: string; status: string; resourceUsage: { memoryMB: number; cpuPercent: number } }> {
    return Array.from(this.sandboxes.values()).map(s => ({
      moduleId: s.moduleId,
      status: s.status,
      resourceUsage: s.resourceUsage
    }))
  }
}
```

#### 步骤 1.5：实现沙箱 IPC 桥接

**src/main/core/sandbox-ipc.ts**：

```typescript
import type { UtilityProcess } from 'electron'
import type { Logger } from '../types/logger'

export interface SandboxIPCMessage {
  channel: string
  data?: unknown
  requestId?: string  // 用于 request-response 模式
}

export interface PendingRequest {
  resolve: (value: unknown) => void
  reject: (reason: Error) => void
  timer: ReturnType<typeof setTimeout>
}

/**
 * 沙箱 IPC 桥接层
 * 提供可靠的 request/response 通信模式，支持超时和并发请求
 */
export class SandboxIPC {
  private pendingRequests = new Map<string, PendingRequest>()
  private requestCounter = 0
  private logger: Logger

  constructor(
    private process: UtilityProcess,
    private moduleId: string,
    logger: Logger
  ) {
    this.logger = logger

    // 监听子进程响应消息
    this.process.on('message', (message: SandboxIPCMessage) => {
      if (message.requestId && this.pendingRequests.has(message.requestId)) {
        const pending = this.pendingRequests.get(message.requestId)!
        this.pendingRequests.delete(message.requestId)
        clearTimeout(pending.timer)

        if (message.channel === '__response_error__') {
          pending.reject(new Error(String(message.data)))
        } else {
          pending.resolve(message.data)
        }
      }
    })
  }

  /**
   * 发送消息（单向，不等待响应）
   */
  send(channel: string, data?: unknown): void {
    this.process.postMessage({ channel, data })
  }

  /**
   * 请求-响应模式（带超时）
   */
  async request<T = unknown>(channel: string, data?: unknown, timeoutMs = 10000): Promise<T> {
    const requestId = `${this.moduleId}-${++this.requestCounter}`

    return new Promise<T>((resolve, reject) => {
      const timer = setTimeout(() => {
        this.pendingRequests.delete(requestId)
        reject(new Error(`沙箱 "${this.moduleId}" 请求 "${channel}" 超时 (${timeoutMs}ms)`))
      }, timeoutMs)

      this.pendingRequests.set(requestId, {
        resolve: resolve as (value: unknown) => void,
        reject,
        timer
      })

      this.process.postMessage({ channel, data, requestId })
    })
  }

  /**
   * 清理所有待处理请求
   */
  dispose(): void {
    for (const [id, pending] of this.pendingRequests) {
      clearTimeout(pending.timer)
      pending.reject(new Error('沙箱 IPC 已关闭'))
    }
    this.pendingRequests.clear()
  }
}

// 此文件在 utilityProcess 子进程中运行
// 注意：utilityProcess 中使用 process.parentPort（非 worker_threads.parentPort）

const moduleId = process.env.MODULE_ID!
const modulePath = process.env.MODULE_PATH!

let moduleInstance: any = null

// ============================================================
// 沙箱逃逸防护：白名单模式 + API 拦截层
// ============================================================

/** 允许模块访问的安全 Node.js API 白名单 */
const ALLOWED_BUILTIN_MODULES = new Set([
  'events',       // EventEmitter（模块通信基础）
  'util',         // 格式化工具
  'url',          // URL 解析
  'querystring',  // 查询字符串解析
  'path',         // 路径操作（仅路径计算，不涉及文件系统）
  'assert',       // 断言（模块自检）
  'buffer',       // Buffer 操作
  'crypto',       // 加密（仅 Web 安全子集，见下方拦截规则）
  'stream',       // 流（内存流，不涉及 IO）
  'string_decoder', // 字符串解码
  'timers',       // setTimeout/setInterval 等
  'perf_hooks',   // 性能监控
])

/** 明确禁用的危险模块（即使未来白名单扩展也不会意外开放） */
const BLOCKED_MODULES = new Set([
  'fs', 'fs/promises',        // 文件系统读写
  'child_process',            // 子进程执行
  'net', 'tls', 'http', 'https', // 网络连接
  'dgram',                    // UDP
  'dns',                      // DNS 解析
  'cluster',                  // 多进程集群
  'worker_threads',           // Worker 线程（逃逸风险）
  'vm',                       // VM 执行（二次逃逸风险）
  'v8',                       // V8 引擎接口
  'module',                   // 模块加载器
  'repl',                     // REPL 交互
  'os',                       // 操作系统信息（可能泄露敏感信息）
  'process',                  // 进程信息（部分已在下方限制）
  'inspector',                // 调试器协议
  'trace_events',             // 追踪事件
])

/** 拦截操作日志（写入沙箱自身日志，同时发送给主进程审计） */
interface InterceptLog {
  timestamp: number
  moduleId: string
  action: string
  target: string
  blocked: boolean
  detail?: string
}

const interceptLogs: InterceptLog[] = []
const MAX_LOG_ENTRIES = 1000

function logIntercept(action: string, target: string, blocked: boolean, detail?: string): void {
  const entry: InterceptLog = {
    timestamp: Date.now(),
    moduleId,
    action,
    target,
    blocked,
    detail,
  }
  interceptLogs.push(entry)
  if (interceptLogs.length > MAX_LOG_ENTRIES) interceptLogs.shift()
  // 通过 parentPort 向主进程发送审计日志
  try {
    parentPort?.postMessage({
      channel: '__audit__',
      data: { type: 'sandbox_intercept', ...entry }
    })
  } catch { /* 忽略发送失败 */ }
}

/**
 * 限制环境变量传递：白名单模式，仅保留安全的基础运行时变量
 * 清除所有环境变量后仅恢复白名单项，确保无 API Key / Token / Secret 泄露
 * ⚠️ 此白名单与 sandbox.ts 中 allowedKeys 保持一致
 */
const ALLOWED_ENV_KEYS = [
  'MODULE_ID', 'MODULE_PATH',      // 沙箱模块标识
  'NODE_ENV', 'APP_ROOT',          // 运行时环境
  'PATH', 'HOME',                  // 系统基础路径
  'LANG', 'LC_ALL', 'DISPLAY'      // 国际化与显示
]

function sanitizeEnv(): void {
  const allowed = new Set(ALLOWED_ENV_KEYS)
  for (const key of Object.keys(process.env)) {
    if (!allowed.has(key)) {
      delete process.env[key]
    }
  }
}

sanitizeEnv()

/**
 * 禁用危险全局对象，防止模块通过全局作用域逃逸
 * 注意：utilityProcess 是独立进程，不受 Node.js vm 模块限制，
 * 因此需要主动删除/替换危险全局变量
 */
function restrictGlobals(): void {
  // 1. 移除 process 上的危险方法（保留 process.parentPort 供内部通信）
  const originalProcess = process
  const safeProcessProps = ['env', 'pid', 'platform', 'arch', 'versions', 'cwd', 'nextTick', 'parentPort']
  // 用 Proxy 拦截 process 访问，仅允许白名单属性
  const processProxy = new Proxy(originalProcess, {
    get(target, prop: string) {
      if (safeProcessProps.includes(prop)) {
        return (target as any)[prop]
      }
      logIntercept('process.access', prop, true)
      return undefined
    },
    set(_target, prop: string) {
      logIntercept('process.modify', prop, true)
      return false
    },
  })
  // 注意：utilityProcess 中无法完全替换 process 全局变量，
  // 但可以通过拦截 require/import 来阻止模块获取原始 process

  // 2. 禁用 eval 和 Function 构造器（防止动态代码执行）
  //    通过将 eval 和 Function 替换为抛出错误的代理
  const blockedFn = (name: string) => () => {
    logIntercept('dynamic_code', name, true)
    throw new Error(`[Sandbox] ${name} 在沙箱中被禁用`)
  }
  // 在模块加载前设置（见下方 import 拦截逻辑）
  ;(globalThis as any).__sandbox_blocked_eval__ = blockedFn('eval')
  ;(globalThis as any).__sandbox_blocked_Function__ = blockedFn('Function')

  // 3. 禁用 __proto__ / constructor 原型链逃逸
  //    冻结 Object.prototype 的关键属性
  try {
    Object.defineProperty(Object.prototype, '__proto__', {
      get() {
        logIntercept('prototype_access', '__proto__', true)
        return null
      },
      configurable: false,
    })
  } catch {
    // 已被冻结则忽略
  }
}

restrictGlobals()

/**
 * 模块导入拦截器：动态 import() 的白名单过滤
 * 由于 utilityProcess 使用 ESM import，需通过自定义 import 逻辑过滤
 */

/** 解析模块名（去掉路径前缀） */
function normalizeModuleName(specifier: string): string {
  // 处理 node: 前缀
  if (specifier.startsWith('node:')) {
    return specifier.slice(5)
  }
  // 处理相对路径（视为模块内部文件，允许）
  if (specifier.startsWith('./') || specifier.startsWith('../')) {
    return specifier
  }
  return specifier
}

/** 检查模块是否在白名单内 */
function isModuleAllowed(specifier: string): boolean {
  const name = normalizeModuleName(specifier)
  // 明确禁用的模块（即使在白名单中也拦截）
  if (BLOCKED_MODULES.has(name)) {
    return false
  }
  // 相对路径（模块自身文件）允许
  if (name.startsWith('./') || name.startsWith('../')) {
    return true
  }
  // 白名单内置模块
  if (ALLOWED_BUILTIN_MODULES.has(name)) {
    return true
  }
  // npm 包需要额外验证（仅允许已注册的安全包）
  // 这里默认拒绝未知包
  return false
}

/**
 * 创建安全的 import 函数，替换全局 import
 * 在模块代码执行前注入，确保所有 import 都经过拦截
 */
function createSafeImport(originalImport: (specifier: string) => Promise<any>) {
  return async function safeImport(specifier: string): Promise<any> {
    if (!isModuleAllowed(specifier)) {
      logIntercept('import', specifier, true, `模块不在白名单中`)
      throw new Error(
        `[Sandbox] 模块 "${specifier}" 在沙箱中被禁止导入。` +
        `允许的内置模块: ${[...ALLOWED_BUILTIN_MODULES].join(', ')}`
      )
    }
    logIntercept('import', specifier, false)
    return originalImport(specifier)
  }
}

/**
 * Crypto 安全子集：仅暴露哈希和随机数，禁止密钥操作
 * 防止模块通过 crypto 读取系统密钥或证书
 */
function createSafeCrypto() {
  const crypto = require('crypto')
  return {
    createHash: crypto.createHash,
    createHmac: crypto.createHmac,
    randomBytes: crypto.randomBytes,
    randomUUID: crypto.randomUUID,
    randomInt: crypto.randomInt,
    timingSafeEqual: crypto.timingSafeEqual,
    // 禁止：generateKeyPair, createCipheriv, createDecipheriv, Certificate 等
  }
}

// 使用 process.parentPort（utilityProcess 专用 API）
const parentPort = process.parentPort

if (!parentPort) {
  console.error('sandbox-worker: process.parentPort 不可用，当前进程可能不是 utilityProcess')
  process.exit(1)
}

// 监听主进程消息
parentPort.on('message', async (message: { channel: string; data?: any }) => {
  const { channel, data } = message

  switch (channel) {
    case '__init__':
      try {
        // === 沙箱逃逸防护：在模块加载前注入安全拦截层 ===

        // 1. 替换全局 import 为安全版本（白名单过滤）
        const originalImport = globalThis.import
        ;(globalThis as any).import = createSafeImport(originalImport)

        // 2. 替换 eval / Function 构造器（阻止动态代码执行）
        globalThis.eval = function sandboxEval(code: string) {
          logIntercept('eval', 'eval()', true, `尝试执行代码: ${String(code).slice(0, 100)}`)
          throw new Error('[Sandbox] eval() 在沙箱中被禁用')
        } as any
        globalThis.Function = function sandboxFunction(...args: any[]) {
          logIntercept('Function', 'Function()', true)
          throw new Error('[Sandbox] Function() 构造器在沙箱中被禁用')
        } as any

        // 3. 拦截 require（兼容 CommonJS 模块加载）
        if (typeof require !== 'undefined') {
          const originalRequire = require
          const safeRequire = Object.assign(
            function safeRequire(specifier: string) {
              if (!isModuleAllowed(specifier)) {
                logIntercept('require', specifier, true, '模块不在白名单中')
                throw new Error(`[Sandbox] require("${specifier}") 在沙箱中被禁止`)
              }
              // crypto 做安全子集包装
              if (specifier === 'crypto') {
                return createSafeCrypto()
              }
              logIntercept('require', specifier, false)
              return originalRequire(specifier)
            },
            {
              resolve: () => { throw new Error('[Sandbox] require.resolve 被禁用') },
              cache: {},
              main: undefined,
            }
          )
          ;(globalThis as any).require = safeRequire
        }

        // 4. 加载模块（import 已被替换为白名单版本）
        if (modulePath.endsWith('.ts') && !process.env.TS_NODE_DEV) {
          console.warn(
            `[sandbox-worker] 加载 .ts 文件: ${modulePath}，` +
            '生产环境建议使用编译后的 .js 文件'
          )
        }
        const mod = await import(modulePath)
        const ModuleClass = mod.default || mod
        moduleInstance = new ModuleClass(data.context)
        await moduleInstance.activate(data.context)
        parentPort.postMessage({ channel: '__ready__' })
      } catch (err) {
        parentPort.postMessage({
          channel: '__error__',
          data: { error: (err as Error).message }
        })
      }
      break

    case '__shutdown__':
      try {
        if (moduleInstance?.deactivate) {
          await moduleInstance.deactivate()
        }
      } catch (err) {
        // 忽略关闭错误
      }
      process.exit(0)
      break

    default:
      // 转发给模块的事件处理器
      if (moduleInstance?.handleMessage) {
        moduleInstance.handleMessage(channel, data)
      }
      break
  }
})

// 通知主进程已启动
parentPort.postMessage({ channel: '__started__' })
```

### 8.3 沙箱逃逸防护策略

| 防护层 | 措施 | 说明 |
|--------|------|------|
| **模块白名单** | `ALLOWED_BUILTIN_MODULES` | 仅允许 events, util, url, path 等安全内置模块 |
| **模块黑名单** | `BLOCKED_MODULES` | 显式禁止 fs, child_process, net, http, vm 等危险模块 |
| **import 拦截** | `createSafeImport()` | 所有动态 import 必须通过白名单校验 |
| **require 拦截** | `safeRequire()` | CommonJS require 同样受限，缓存和 resolve 被禁用 |
| **eval/Function 禁用** | 全局替换 | 阻止动态代码执行，防止沙箱逃逸 |
| **crypto 安全子集** | `createSafeCrypto()` | 仅暴露哈希/随机数，禁止密钥/证书操作 |
| **环境变量清理** | `sanitizeEnv()` | 仅传递白名单变量，防止 API Key/Token 泄露 |
| **原型链防护** | `__proto__` 拦截 | 阻止通过原型链修改逃逸沙箱 |
| **审计日志** | `logIntercept()` | 所有拦截操作记录并发送至主进程审计 |

### 8.4 代码规范

- **进程隔离**：每个模块运行在独立的 `utilityProcess` 子进程中
- **通信方式**：仅通过 `postMessage` / `on('message')` IPC 通信
- **超时保护**：启动超时 10s，关闭超时 5s
- **资源限制**：通过 `resourceLimits` 限制内存和 CPU
- **崩溃恢复**：子进程崩溃后主系统自动标记为异常状态
- **安全模式**：白名单默认拒绝，所有未明确允许的 API 均被拦截
- **审计追踪**：拦截操作通过 `__audit__` 通道上报主进程，支持安全审计
