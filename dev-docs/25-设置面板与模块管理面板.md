// src/renderer/src/stores/settingsStore.ts

import { create } from 'zustand';
import { persist, createJSONStorage } from 'zustand/middleware';
import type {
  AppSettings,
  OllamaConfig,
  ThemeMode,
  Language,
} from '../types/settings';

/**
 * 设置 Store 状态
 */
interface SettingsState {
  settings: AppSettings;
  isLoading: boolean;
  error: string | null;
}

/**
 * 设置 Store Actions
 */
interface SettingsActions {
  updateTheme: (theme: ThemeMode) => void;
  updateLanguage: (language: Language) => void;
  updateOllamaConfig: (config: Partial<OllamaConfig>) => void;
  updateSettings: (partial: Partial<AppSettings>) => void;
  resetToDefaults: () => void;
  loadFromMainProcess: () => Promise<void>;
  saveToMainProcess: () => Promise<void>;
  setError: (error: string | null) => void;
}

/**
 * 默认设置
 */
const defaultSettings: AppSettings = {
  theme: 'system',
  language: 'zh-CN',
  ollama: {
    baseUrl: 'http://localhost:11434',
    defaultModel: 'qwen2.5:7b',
    timeoutSeconds: 120,
    contextWindowSize: 4096,
    temperature: 0.7,
    streamEnabled: true,
  },
  autoStart: false,
  minimizeToTray: true,
  notificationsEnabled: true,
  logLevel: 'info',
  performanceMonitor: {
    collectIntervalMs: 5000,
    gpuEnabled: true,
    historySize: 360,
    throttle: {
      awareThreshold: 60,
      degradedThreshold: 80,
      lowPowerThreshold: 90,
      cooldownSeconds: 30,
    },
    warningThresholds: {
      cpu: 85,
      memory: 85,
      disk: 90,
      gpu: 90,
    },
  },
};

/**
 * 设置 Store
 */
export const useSettingsStore = create<SettingsState & SettingsActions>()(
  persist(
    (set, get) => ({
      settings: defaultSettings,
      isLoading: false,
      error: null,

      updateTheme: (theme) => {
        set((state) => ({
          settings: { ...state.settings, theme },
        }));
        // 同步到主进程
        void get().saveToMainProcess();
      },

      updateLanguage: (language) => {
        set((state) => ({
          settings: { ...state.settings, language },
        }));
        void get().saveToMainProcess();
      },

      updateOllamaConfig: (config) => {
        set((state) => ({
          settings: {
            ...state.settings,
            ollama: { ...state.settings.ollama, ...config },
          },
        }));
        void get().saveToMainProcess();
      },

      updateSettings: (partial) => {
        set((state) => ({
          settings: { ...state.settings, ...partial },
        }));
        void get().saveToMainProcess();
      },

      resetToDefaults: () => {
        set({ settings: defaultSettings, error: null });
        void get().saveToMainProcess();
      },

      loadFromMainProcess: async () => {
        set({ isLoading: true, error: null });
        try {
          const result = await window.electronAPI.invoke(
            'settings:get'
          );
          if (result) {
            set({ settings: { ...defaultSettings, ...result } });
          }
        } catch (error) {
          const message =
            error instanceof Error ? error.message : '加载设置失败';
          set({ error: message });
        } finally {
          set({ isLoading: false });
        }
      },

      saveToMainProcess: async () => {
        try {
          const { settings } = get();
          await window.electronAPI.invoke(
            'settings:save',
            settings
          );
        } catch (error) {
          const message =
            error instanceof Error ? error.message : '保存设置失败';
          set({ error: message });
        }
      },

      setError: (error) => set({ error }),
    }),
    {
      name: 'app-settings',
      storage: createJSONStorage(() => localStorage),
      partialize: (state) => ({ settings: state.settings }),
    }
  )
);

// src/renderer/src/stores/useModulesStore.ts

import { create } from 'zustand';
import type {
  ModuleInfo,
  ModuleStatus,
  SystemResourceSnapshot,
} from '../types/settings';

/**
 * 模块 Store 状态
 */
interface ModulesState {
  modules: ModuleInfo[];
  selectedModuleId: string | null;
  resourceSnapshot: SystemResourceSnapshot | null;
  isLoading: boolean;
  error: string | null;
}

/**
 * 模块 Store Actions
 */
interface ModulesActions {
  fetchModules: () => Promise<void>;
  toggleModule: (moduleId: string) => Promise<void>;
  selectModule: (moduleId: string | null) => void;
  startModule: (moduleId: string) => Promise<void>;
  stopModule: (moduleId: string) => Promise<void>;
  refreshResources: () => Promise<void>;
  setError: (error: string | null) => void;
}

/**
 * 模块 Store
 */
export const useModulesStore = create<ModulesState & ModulesActions>(
  (set, get) => ({
    modules: [],
    selectedModuleId: null,
    resourceSnapshot: null,
    isLoading: false,
    error: null,

    fetchModules: async () => {
      set({ isLoading: true, error: null });
      try {
        const modules = await window.electronAPI.invoke(
          'modules:list'
        ) as ModuleInfo[];
        set({ modules });
      } catch (error) {
        const message =
          error instanceof Error ? error.message : '获取模块列表失败';
        set({ error: message });
      } finally {
        set({ isLoading: false });
      }
    },

    toggleModule: async (moduleId) => {
      const { modules } = get();
      const target = modules.find((m) => m.id === moduleId);
      if (!target) return;

      try {
        if (target.enabled) {
          await window.electronAPI.invoke(
            'modules:disable',
            moduleId
          );
        } else {
          await window.electronAPI.invoke(
            'modules:enable',
            moduleId
          );
        }

        set({
          modules: modules.map((m) =>
            m.id === moduleId ? { ...m, enabled: !m.enabled } : m
          ),
        });
      } catch (error) {
        const message =
          error instanceof Error ? error.message : '操作失败';
        set({ error: message });
      }
    },

    selectModule: (moduleId) => {
      set({ selectedModuleId: moduleId });
    },

    startModule: async (moduleId) => {
      try {
        await window.electronAPI.invoke(
          'modules:start',
          moduleId
        );
        set((state) => ({
          modules: state.modules.map((m) =>
            m.id === moduleId
              ? { ...m, status: 'running' as ModuleStatus }
              : m
          ),
        }));
      } catch (error) {
        const message =
          error instanceof Error ? error.message : '启动模块失败';
        set({ error: message });
      }
    },

    stopModule: async (moduleId) => {
      try {
        await window.electronAPI.invoke(
          'modules:stop',
          moduleId
        );
        set((state) => ({
          modules: state.modules.map((m) =>
            m.id === moduleId
              ? { ...m, status: 'stopped' as ModuleStatus }
              : m
          ),
        }));
      } catch (error) {
        const message =
          error instanceof Error ? error.message : '停止模块失败';
        set({ error: message });
      }
    },

    refreshResources: async () => {
      try {
        const snapshot = await window.electronAPI.invoke(
          'performance:snapshot'
        ) as SystemResourceSnapshot;
        set({ resourceSnapshot: snapshot });

        // 同时更新模块的当前资源使用
        const { modules } = get();
        const updatedModules = modules.map((m) => ({
          ...m,
          currentUsage: snapshot.modules[m.id],
        }));
        set({ modules: updatedModules });
      } catch (error) {
        console.error('[ModulesStore] 刷新资源失败:', error);
      }
    },

    setError: (error) => set({ error }),
  })
);

// src/renderer/src/pages/SettingsPage.tsx

import React, { useEffect } from 'react';
import { useSettingsStore } from '../stores/settingsStore';
import { OllamaConfig } from '../components/settings/OllamaConfig';
import { ThemeSwitcher } from '../components/settings/ThemeSwitcher';
import { LanguageSelector } from '../components/settings/LanguageSelector';
import { PerformanceMonitorConfig } from '../components/settings/PerformanceMonitorConfig';
import { SettingsSection } from '../components/settings/SettingsSection';
import type { AppSettings } from '../types/settings';

/**
 * 设置页面
 */
export const SettingsPage: React.FC = () => {
  const {
    settings,
    isLoading,
    error,
    updateSettings,
    resetToDefaults,
    loadFromMainProcess,
    setError,
  } = useSettingsStore();

  useEffect(() => {
    void loadFromMainProcess();
  }, [loadFromMainProcess]);

  const handleToggle = (key: keyof AppSettings) => {
    updateSettings({ [key]: !settings[key] } as Partial<AppSettings>);
  };

  const handleLogLevelChange = (level: AppSettings['logLevel']) => {
    updateSettings({ logLevel: level });
  };

  if (isLoading) {
    return <div className="settings-loading">加载中...</div>;
  }

  return (
    <div className="settings-page">
      <header className="settings-header">
        <h1>设置</h1>
        {error && (
          <div className="settings-error" role="alert">
            <span>{error}</span>
            <button onClick={() => setError(null)}>关闭</button>
          </div>
        )}
      </header>

      <main className="settings-content">
        {/* Ollama 配置 */}
        <SettingsSection
          title="Ollama 配置"
          description="配置本地 AI 模型连接参数"
          icon="🤖"
        >
          <OllamaConfig
            config={settings.ollama}
            onChange={(config) =>
              updateSettings({ ollama: { ...settings.ollama, ...config } })
            }
          />
        </SettingsSection>

        {/* 外观设置 */}
        <SettingsSection
          title="外观"
          description="自定义应用主题和语言"
          icon="🎨"
        >
          <ThemeSwitcher
            value={settings.theme}
            onChange={(theme) => updateSettings({ theme })}
          />
          <LanguageSelector
            value={settings.language}
            onChange={(language) => updateSettings({ language })}
          />
        </SettingsSection>

        {/* 通用设置 */}
        <SettingsSection
          title="通用"
          description="应用基本行为设置"
          icon="⚙️"
        >
          <div className="settings-row">
            <label htmlFor="auto-start">开机自启</label>
            <Toggle
              id="auto-start"
              checked={settings.autoStart}
              onChange={() => handleToggle('autoStart')}
            />
          </div>

          <div className="settings-row">
            <label htmlFor="minimize-tray">最小化到托盘</label>
            <Toggle
              id="minimize-tray"
              checked={settings.minimizeToTray}
              onChange={() => handleToggle('minimizeToTray')}
            />
          </div>

          <div className="settings-row">
            <label htmlFor="notifications">通知</label>
            <Toggle
              id="notifications"
              checked={settings.notificationsEnabled}
              onChange={() => handleToggle('notificationsEnabled')}
            />
          </div>

          <div className="settings-row">
            <label htmlFor="log-level">日志级别</label>
            <select
              id="log-level"
              value={settings.logLevel}
              onChange={(e) =>
                handleLogLevelChange(e.target.value as AppSettings['logLevel'])
              }
            >
              <option value="debug">Debug</option>
              <option value="info">Info</option>
              <option value="warn">Warning</option>
              <option value="error">Error</option>
            </select>
          </div>
        </SettingsSection>

        {/* 性能监控配置（联动版块26 PerformanceMonitor） */}
        <SettingsSection
          title="性能监控"
          description="配置资源采集频率、降频策略与警告阈值，修改后实时同步到主进程 PerformanceMonitor"

            className="btn-reset"

                resetToDefaults();
              }
            }}
          >
            恢复默认设置
          </button>
        </div>
      </main>
    </div>
  );
};

export default SettingsPage;

import React, { useState, useCallback } from 'react';
import type { OllamaConfig as OllamaConfigType } from '../../types/settings';

  config: OllamaConfigType;
  onChange: (config: Partial<OllamaConfigType>) => void;
}

/**
 * Ollama 配置组件
 */
export const OllamaConfig: React.FC<OllamaConfigProps> = ({
  config,
  onChange,
}) => {
  const [testStatus, setTestStatus] = useState<
    'idle' | 'testing' | 'success' | 'error'
  >('idle');
  const [testMessage, setTestMessage] = useState('');

  const handleTestConnection = useCallback(async () => {
    setTestStatus('testing');
    setTestMessage('');
    try {
      const result = await window.electronAPI.invoke(
        'ollama:test-connection',
        config.baseUrl
      ) as { success: boolean; message: string };
      setTestStatus(result.success ? 'success' : 'error');
      setTestMessage(result.message);
    } catch (error) {
      setTestStatus('error');
      setTestMessage(error instanceof Error ? error.message : '连接失败');
    }
  }, [config.baseUrl]);

  return (
    <div className="ollama-config">
      <div className="config-field">
        <label htmlFor="ollama-url">服务地址</label>
        <div className="input-with-action">
          <input

            type="url"
            value={config.baseUrl}
            onChange={(e) => onChange({ baseUrl: e.target.value })}
            placeholder="http://localhost:11434"
          />

            className="btn-test"
            onClick={() => void handleTestConnection()}
            disabled={testStatus === 'testing'}
          >
            {testStatus === 'testing' ? '测试中...' : '测试连接'}
          </button>
        </div>

// src/renderer/src/components/common/Slider.tsx

import React, { useCallback, useRef } from 'react';

interface SliderProps {
  /** 当前值 */
  value: number;
  /** 值变化回调 */
  onChange: (value: number) => void;
  /** 最小值 */
  min?: number;
  /** 最大值 */
  max?: number;
  /** 步长 */
  step?: number;
  /** 标签文本 */
  label?: string;
  /** 是否禁用 */
  disabled?: boolean;
}

/**
 * 滑块组件
 *
 * 用于设置面板中的数值调节场景（如字体大小、音量、透明度等）。
 * 支持键盘操作（方向键微调），与 Toggle 等通用组件风格保持一致。
 */
export const Slider: React.FC<SliderProps> = ({
  value,
  onChange,
  min = 0,
  max = 100,
  step = 1,
  label,
  disabled = false,
}) => {
  const inputRef = useRef<HTMLInputElement>(null);

  const percentage = ((value - min) / (max - min)) * 100;

  const handleChange = useCallback(
    (e: React.ChangeEvent<HTMLInputElement>) => {
      const newValue = parseFloat(e.target.value);
      if (!isNaN(newValue)) {
        onChange(newValue);
      }
    },
    [onChange],
  );

  const handleKeyDown = useCallback(
    (e: React.KeyboardEvent<HTMLInputElement>) => {
      let newValue = value;
      switch (e.key) {
        case 'ArrowRight':
        case 'ArrowUp':
          e.preventDefault();
          newValue = Math.min(value + step, max);
          break;
        case 'ArrowLeft':
        case 'ArrowDown':
          e.preventDefault();
          newValue = Math.max(value - step, min);
          break;
        case 'Home':
          e.preventDefault();
          newValue = min;
          break;
        case 'End':
          e.preventDefault();
          newValue = max;
          break;
        default:
          return;
      }
      onChange(newValue);
    },
    [value, step, min, max, onChange],
  );

  return (
    <div className={`slider-wrapper ${disabled ? 'disabled' : ''}`}>
      {label && (
        <label className="slider-label" htmlFor={label}>
          {label}
        </label>
      )}
      <div className="slider-container">
        <input
          ref={inputRef}
          id={label}
          type="range"
          role="slider"
          className="slider-input"
          min={min}
          max={max}
          step={step}
          value={value}
          disabled={disabled}
          aria-valuemin={min}
          aria-valuemax={max}
          aria-valuenow={value}
          aria-label={label}
          onChange={handleChange}
          onKeyDown={handleKeyDown}
          style={
            {
              '--slider-percentage': `${percentage}%`,
            } as React.CSSProperties
          }
        />
        <span className="slider-value" aria-live="polite">
          {value}
        </span>
      </div>
    </div>
  );
};

function formatUptime(seconds: number): string {
  const h = Math.floor(seconds / 3600);
  const m = Math.floor((seconds % 3600) / 60);
  const s = Math.floor(seconds % 60);

#### 步骤 5：实现主进程 IPC Handler

渲染进程通过 `ipcRenderer.invoke` 调用了 `settings:get`、`settings:save`、`modules:list`、`modules:enable`、`modules:disable`、`modules:start`、`modules:stop` 共 7 个 IPC 通道，需要在主进程侧注册对应的 handler。

```typescript
// src/main/ipc/settings-handlers.ts

import { ipcMain } from 'electron';
import { readFile, writeFile, mkdir } from 'node:fs/promises';
import { join } from 'node:path';
import { app } from 'electron';
import type { AppSettings } from '../../renderer/src/types/settings';
import type { ModuleInfo, ModuleStatus } from '../../renderer/src/types/module';
import type { ModuleManager } from '../core/module-manager';

/**
 * 设置文件路径
 */
function getSettingsPath(): string {
  return join(app.getPath('userData'), 'settings.json');
}

/**
 * 从磁盘加载设置
 */
async function loadSettings(): Promise<Partial<AppSettings>> {
  try {
    const data = await readFile(getSettingsPath(), 'utf-8');
    return JSON.parse(data) as Partial<AppSettings>;
  } catch {
    // 文件不存在或解析失败，返回空对象（渲染进程会合并默认值）
    return {};
  }
}

/**
 * 将设置写入磁盘
 */
async function saveSettingsToFile(settings: AppSettings): Promise<void> {
  const dir = join(app.getPath('userData'));
  await mkdir(dir, { recursive: true });
  await writeFile(getSettingsPath(), JSON.stringify(settings, null, 2), 'utf-8');
}

/**
 * 注册设置 & 模块管理相关的 IPC 处理器
 *
 * @param moduleManager - 模块管理器实例，负责模块的生命周期管理
 * @param performanceMonitorRef - 性能监控器实例引用（版块26），用于将设置面板的配置变更同步到监控器
 */
export function registerSettingsIPC(
  moduleManager: ModuleManager,
  performanceMonitorRef?: { updateConfig: (config: unknown) => void }
): void {
  // ── 设置相关 ──────────────────────────────────────────────

  /**
   * 获取应用设置
   * 渲染进程调用: window.electronAPI.invoke('settings:get')
   */
  ipcMain.handle('settings:get', async () => {
    const settings = await loadSettings();
    return settings;
  });

  /**
   * 保存应用设置
   * 渲染进程调用: window.electronAPI.invoke('settings:save', settings)
   */
  ipcMain.handle('settings:save', async (_event, settings: AppSettings) => {
    await saveSettingsToFile(settings);

    // 联动版块26：设置保存时同步更新 PerformanceMonitor 配置
    if (settings.performanceMonitor && performanceMonitorRef) {
      performanceMonitorRef.updateConfig({
        collectIntervalMs: settings.performanceMonitor.collectIntervalMs,
        gpuEnabled: settings.performanceMonitor.gpuEnabled,
        historySize: settings.performanceMonitor.historySize,
        throttle: {
          awareThreshold: settings.performanceMonitor.throttle.awareThreshold,
          degradedThreshold: settings.performanceMonitor.throttle.degradedThreshold,
          lowPowerThreshold: settings.performanceMonitor.throttle.lowPowerThreshold,
          awareIntervalMultiplier: 2, // 保持默认，UI 不暴露
          cooldownMs: settings.performanceMonitor.throttle.cooldownSeconds * 1000,
        },
        warningThresholds: settings.performanceMonitor.warningThresholds,
      });
    }

    return true;
  });

  /**
   * 独立更新性能监控配置（可选，支持实时调整不触发全量保存）
   * 渲染进程调用: window.electronAPI.invoke('performance:config-update', config)
   */
  ipcMain.handle(
    'performance:config-update',
    async (_event, config: AppSettings['performanceMonitor']) => {
      if (performanceMonitorRef) {
        performanceMonitorRef.updateConfig({
          collectIntervalMs: config.collectIntervalMs,
          gpuEnabled: config.gpuEnabled,
          historySize: config.historySize,
          throttle: {
            awareThreshold: config.throttle.awareThreshold,
            degradedThreshold: config.throttle.degradedThreshold,
            lowPowerThreshold: config.throttle.lowPowerThreshold,
            awareIntervalMultiplier: 2,
            cooldownMs: config.throttle.cooldownSeconds * 1000,
          },
          warningThresholds: config.warningThresholds,
        });
      }
      return true;
    }
  );

  // ── 模块管理相关 ──────────────────────────────────────────

  /**
   * 获取模块列表
   * 渲染进程调用: window.electronAPI.invoke('modules:list')
   */
  ipcMain.handle('modules:list', async (): Promise<ModuleInfo[]> => {
    return moduleManager.listModules();
  });

  /**
   * 启用模块
   * 渲染进程调用: window.electronAPI.invoke('modules:enable', moduleId)
   */
  ipcMain.handle('modules:enable', async (_event, moduleId: string): Promise<boolean> => {
    await moduleManager.enableModule(moduleId);
    return true;
  });

  /**
   * 禁用模块
   * 渲染进程调用: window.electronAPI.invoke('modules:disable', moduleId)
   */
  ipcMain.handle('modules:disable', async (_event, moduleId: string): Promise<boolean> => {
    await moduleManager.disableModule(moduleId);
    return true;
  });

  /**
   * 启动模块
   * 渲染进程调用: window.electronAPI.invoke('modules:start', moduleId)
   */
  ipcMain.handle('modules:start', async (_event, moduleId: string): Promise<boolean> => {
    await moduleManager.startModule(moduleId);
    return true;
  });

  /**
   * 停止模块
   * 渲染进程调用: window.electronAPI.invoke('modules:stop', moduleId)
   */
  ipcMain.handle('modules:stop', async (_event, moduleId: string): Promise<boolean> => {
    await moduleManager.stopModule(moduleId);
    return true;
  });

  console.log('[SettingsIPC] 已注册设置 & 模块管理 IPC 接口');
}
```

### 25.3 代码规范

1. **状态管理**：使用 Zustand 进行全局状态管理，避免 Props Drilling
2. **持久化**：设置数据通过 `zustand/middleware/persist` 持久化到 localStorage
3. **IPC 通信**：所有主进程交互通过 `window.electronAPI.invoke`（由 preload 脚本通过 contextBridge 暴露），返回类型需显式声明
4. **组件拆分**：每个设置区块独立组件，单一职责
5. **无障碍**：所有交互元素提供 `aria-label`、`role`、`aria-checked` 等属性
6. **键盘导航**：模块卡片支持 Tab 和 Enter/Space 操作
7. **错误处理**：所有异步操作统一 `try/catch`，错误状态统一展示
8. **防抖保存**：设置变更即时保存到主进程，避免丢失
9. **类型安全**：所有 IPC invoke 调用使用 `as` 进行类型断言
10. **命名规范**：页面组件使用 `Page` 后缀，Store 使用 `use...Store` 命名
11. **跨版块联动**：性能监控配置（`PerformanceMonitorSettings`）通过 `settings:save` IPC 自动同步到主进程 `PerformanceMonitor`（版块26），也可通过 `performance:config-update` IPC 独立更新，无需重启应用
