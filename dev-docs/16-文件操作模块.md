#### 步骤 2：实现文件读取服务

**modules/file/services/file-reader.ts**：

```typescript
import { readFile, stat, access, realpath } from 'fs/promises'
import { extname, resolve, normalize, isAbsolute, sep } from 'path'
import { createReadStream } from 'fs'
import type { Logger } from '../../../src/main/types/logger'

export interface ReadResult {
  content: string
  encoding: string
  size: number
  language?: string
}

/** 流式读取阈值：超过此大小使用流式读取（10MB） */
const STREAMING_THRESHOLD = 10 * 1024 * 1024 // 10MB
/** 流式读取时每块大小（64KB） */
const CHUNK_SIZE = 64 * 1024

export class FileReaderService {
  private maxFileSize: number
  private allowedExtensions: Set<string>
  private allowedRoots: string[]
  private resolvedRoots: string[]
  private logger: Logger

  constructor(
    maxFileSize: number,
    logger: Logger,
    allowedExtensions: string[] = ['.txt', '.md', '.json', '.ts', '.tsx', '.js', '.jsx', '.css', '.html'],
    allowedRoots: string[] = []
  ) {
    this.maxFileSize = maxFileSize
    this.logger = logger
    this.allowedExtensions = new Set(allowedExtensions)
    this.allowedRoots = allowedRoots
    // 预解析白名单根目录，确保尾部分隔符一致
    this.resolvedRoots = allowedRoots.map((r) => {
      const resolved = resolve(r)
      return resolved.endsWith(sep) ? resolved : resolved + sep
    })
  }

  /**
   * 验证路径安全性，防止路径遍历攻击
   * 1. 原始输入中检测 '..' 路径遍历
   * 2. path.resolve 规范化路径
   * 3. 白名单目录范围校验（带尾部分隔符防前缀误匹配）
   * 4. realpath 解析符号链接后二次校验，拒绝符号链接逃逸
   */
  private async validatePath(filePath: string): Promise<string> {
    // 在规范化之前检测路径遍历（normalize 会解析掉 ..，导致后续检查失效）
    if (filePath.includes('..')) {
      throw new Error(`[安全] 路径遍历不允许: "${filePath}"`)
    }

    // 规范化并转为绝对路径
    const resolved = resolve(normalize(filePath))

    // 白名单目录校验
    if (this.resolvedRoots.length > 0) {
      const isInAllowedRoot = this.resolvedRoots.some(
        (root) => resolved === root.slice(0, -1) || resolved.startsWith(root)
      )
      if (!isInAllowedRoot) {
        throw new Error(`[安全] 路径不在允许的目录范围内: "${filePath}"`)
      }
    }

    // 解析符号链接，防止通过 symlink 绕过路径检查
    try {
      const realPath = await realpath(resolved)
      if (realPath !== resolved) {
        // 符号链接目标也必须在白名单内
        if (this.resolvedRoots.length > 0) {
          const isRealInRoot = this.resolvedRoots.some(
            (root) => realPath === root.slice(0, -1) || realPath.startsWith(root)
          )
          if (!isRealInRoot) {
            throw new Error(`[安全] 符号链接目标超出允许范围: "${filePath}" -> "${realPath}"`)
          }
        }
        return realPath
      }
    } catch (err: any) {
      // 文件不存在时 realpath 会抛错，允许继续（后续 access 检查会捕获）
      if (err.code !== 'ENOENT') throw err
    }

    return resolved
  }

  /** 验证文件扩展名是否在白名单中 */
  private validateExtension(filePath: string): void {
    const ext = extname(filePath).toLowerCase()
    if (this.allowedExtensions.size > 0 && !this.allowedExtensions.has(ext)) {
      throw new Error(`不允许的文件类型: ${ext}（允许: ${[...this.allowedExtensions].join(', ')}）`)
    }
  }

  async read(filePath: string, encoding: BufferEncoding = 'utf-8'): Promise<ReadResult> {
    // 安全验证（路径遍历防护 + 符号链接检查）
    const safePath = await this.validatePath(filePath)
    this.validateExtension(safePath)

    // 检查文件存在性
    await access(safePath)

    // 检查文件大小
    const fileStats = await stat(safePath)
    if (fileStats.size > this.maxFileSize) {
      throw new Error(`文件大小 ${(fileStats.size / 1024 / 1024).toFixed(1)}MB 超过限制 ${(this.maxFileSize / 1024 / 1024).toFixed(1)}MB`)
    }

    // 大文件使用流式读取，避免一次性加载到内存
    let content: string
    if (fileStats.size > STREAMING_THRESHOLD) {
      content = await this.readStream(safePath, encoding)
    } else {
      content = await readFile(safePath, encoding)
    }

    const ext = extname(safePath).toLowerCase()

    return {
      content,
      encoding,
      size: fileStats.size,
      language: this.detectLanguage(ext)
    }
  }

  /**
   * 流式读取大文件，使用分块处理防止内存溢出
   * - 通过 highWaterMark 控制每块大小（64KB）
   * - 分块拼接字符串，及时释放已处理的 Buffer
   * - 适用于 >=10MB 的大文件
   */
  private readStream(filePath: string, encoding: BufferEncoding): Promise<string> {
    return new Promise((resolve, reject) => {
      const chunks: string[] = []
      const stream = createReadStream(filePath, {
        encoding,
        highWaterMark: CHUNK_SIZE
      })

      stream.on('data', (chunk: string) => {
        chunks.push(chunk)
      })

      stream.on('end', () => {
        resolve(chunks.join(''))
      })

      stream.on('error', (err) => {
        reject(err)
      })
    })
  }

  private detectLanguage(ext: string): string | undefined {
    const map: Record<string, string> = {
      '.ts': 'typescript',
      '.tsx': 'tsx',
      '.js': 'javascript',
      '.jsx': 'jsx',
      '.json': 'json',
      '.md': 'markdown',
      '.css': 'css',
      '.html': 'html',
      '.py': 'python',
      '.rs': 'rust',
      '.go': 'go',
      '.sh': 'bash'
    }
    return map[ext]
  }
}
```

      if (err.code !== 'ENOENT') throw err

    return resolved

        await writeFile(tmpPath, content, 'utf-8')
        await rename(tmpPath, safePath)
      } catch (err) {
        // 清理临时文件
        await unlink(tmpPath).catch(() => {})
        throw err
      }
    }

    this.logger.debug(`文件已写入: ${safePath}`)
  }
}

#### 步骤 4：实现文件监听服务

**modules/file/services/file-watcher.ts**： <!-- ✅ Issue#269: 已修复 — 添加防抖速率限制：每个文件路径独立 debounce（默认 100ms），快速批量变更只触发一次回调 -->

```typescript
import chokidar, { FSWatcher } from 'chokidar'
import type { Logger } from '../../../src/main/types/logger'

export interface FileChangeEvent {
  type: 'add' | 'change' | 'unlink'
  path: string
  timestamp: number
}

export interface FileWatcherOptions {
  /** 防抖间隔（毫秒），同一文件在此时间内的多次变更合并为一次回调。默认 100ms */
  debounceMs?: number
}

export class FileWatcherService {
  private watchers = new Map<string, FSWatcher>()
  /** Issue#269: 每个文件路径独立的防抖定时器，防止快速批量变更触发过多回调 */
  private debounceTimers = new Map<string, ReturnType<typeof setTimeout>>()
  /** Issue#269: 防抖缓冲区，暂存被合并的事件，防抖到期后以最后一次事件类型为准 */
  private debounceBuffer = new Map<string, FileChangeEvent>()
  private logger: Logger
  private debounceMs: number

  constructor(logger: Logger, options: FileWatcherOptions = {}) {
    this.logger = logger
    this.debounceMs = options.debounceMs ?? 100
  }

  /**
   * Issue#269: 防抖发射 — 同一文件路径在 debounceMs 内的多次变更只触发一次 onChange。
   * 合并策略：保留最后一次事件的 type 和最新 timestamp，保证回调始终拿到最新状态。
   */
  private debouncedEmit(path: string, event: FileChangeEvent, onChange: (event: FileChangeEvent) => void): void {
    // 清除该路径的旧定时器
    const existingTimer = this.debounceTimers.get(path)
    if (existingTimer) {
      clearTimeout(existingTimer)
    }

    // 更新缓冲区（同一路径只保留最新事件）
    this.debounceBuffer.set(path, event)

    // 设置新定时器
    const timer = setTimeout(() => {
      const buffered = this.debounceBuffer.get(path)
      if (buffered) {
        onChange(buffered)
        this.debounceBuffer.delete(path)
      }
      this.debounceTimers.delete(path)
    }, this.debounceMs)

    this.debounceTimers.set(path, timer)
  }

  watch(dirPath: string, onChange: (event: FileChangeEvent) => void): void {
    if (this.watchers.has(dirPath)) return

    const watcher = chokidar.watch(dirPath, {
      ignoreInitial: true,
      awaitWriteFinish: { stabilityThreshold: 500 }
    })

    // Issue#269: 所有事件通过 debouncedEmit 发射，实现速率限制
    watcher
      .on('add', (path) => {
        this.debouncedEmit(path, { type: 'add', path, timestamp: Date.now() }, onChange)
      })
      .on('change', (path) => {
        this.debouncedEmit(path, { type: 'change', path, timestamp: Date.now() }, onChange)
      })
      .on('unlink', (path) => {
        this.debouncedEmit(path, { type: 'unlink', path, timestamp: Date.now() }, onChange)
      })
      .on('error', (err) => {
        this.logger.error(`文件监听错误: ${dirPath}`, err)
        // 错误发生后尝试重新建立监听
        this.logger.info(`尝试重新监听目录: ${dirPath}`)
        this.retryWatch(dirPath, onChange)
      })

    this.watchers.set(dirPath, watcher)
    this.logger.info(`开始监听目录: ${dirPath}（防抖间隔: ${this.debounceMs}ms）`)
  }

  /** 错误后重试监听（带指数退避，最多重试 3 次） */
  private retryWatch(dirPath: string, onChange: (event: FileChangeEvent) => void, attempt = 1): void {
    const maxRetries = 3
    if (attempt > maxRetries) {
      this.logger.error(`文件监听重试失败，已达最大重试次数 (${maxRetries}): ${dirPath}`)
      return
    }

    const delay = Math.min(1000 * Math.pow(2, attempt - 1), 5000)
    setTimeout(() => {
      try {
        // 清理旧 watcher
        const oldWatcher = this.watchers.get(dirPath)
        if (oldWatcher) {
          oldWatcher.close()
          this.watchers.delete(dirPath)
        }
        // Issue#269: 重试前清理该目录的防抖定时器和缓冲区
        for (const [key, timer] of this.debounceTimers) {
          if (key.startsWith(dirPath)) {
            clearTimeout(timer)
            this.debounceTimers.delete(key)
            this.debounceBuffer.delete(key)
          }
        }
        // 重新建立监听
        this.watch(dirPath, onChange)
        this.logger.info(`目录监听已恢复: ${dirPath} (第 ${attempt} 次重试)`)
      } catch (err) {
        this.logger.error(`重试监听失败 (第 ${attempt} 次): ${dirPath}`, err)
        this.retryWatch(dirPath, onChange, attempt + 1)
      }
    }, delay)
  }

  unwatch(dirPath: string): void {
    const watcher = this.watchers.get(dirPath)
    if (watcher) {
      watcher.close()
      this.watchers.delete(dirPath)
      // Issue#269: 清理该目录下所有文件的防抖定时器
      for (const [key, timer] of this.debounceTimers) {
        if (key.startsWith(dirPath)) {
          clearTimeout(timer)
          this.debounceTimers.delete(key)
          this.debounceBuffer.delete(key)
        }
      }
      this.logger.info(`停止监听目录: ${dirPath}`)
    }
  }

  unwatchAll(): void {
    for (const [, watcher] of this.watchers) {
      watcher.close()
    }
    this.watchers.clear()
    // Issue#269: 清理所有防抖定时器和缓冲区
    for (const timer of this.debounceTimers.values()) {
      clearTimeout(timer)
    }
    this.debounceTimers.clear()
    this.debounceBuffer.clear()
  }
}

/**

export class CodeHighlighter {
  private prism: typeof import('prismjs') | null = null

    this.logger = logger
  }

  /** 延迟加载 Prism.js */
  private async ensurePrism(): Promise<typeof import('prismjs')> {
    if (!this.prism) {
      this.prism = await import('prismjs')
      await this.loadLanguages(CodeHighlighter.DEFAULT_LANGUAGES)
    }
    return this.prism
  }

  /** 加载指定语言语法 */
  private async loadLanguages(languages: string[]): Promise<void> {
    const Prism = await this.ensurePrism()
    for (const lang of languages) {
      if (this.loadedLanguages.has(lang)) continue
      try {
        // Prism.js 语言包通过 require 加载
        // eslint-disable-next-line @typescript-eslint/no-var-requires
        const grammar = require(`prismjs/components/prism-${lang}`)
        if (grammar) {
          Prism.languages[lang] = grammar
          this.loadedLanguages.add(lang)
        }
      } catch {
        this.logger.warn(`Prism.js 语言包加载失败: ${lang}`)
      }
    }
  }

  /**
   * 高亮代码
   * @param code 源代码字符串
   * @param language 编程语言标识
   * @param options 高亮选项
   * @returns 高亮结果，包含带行号的 HTML
   */
  async highlight(code: string, language: string, options: HighlightOptions = {}): Promise<HighlightResult> {
    const {
      showLineNumbers = true,
      startLine = 1,
      highlightedLines = [],
      compact = false
    } = options

    const Prism = await this.ensurePrism()
    const lang = language.toLowerCase()

    if (!this.loadedLanguages.has(lang)) {
      await this.loadLanguages([lang])
    }

    const grammar = Prism.languages[lang]
    const lines = code.split('\n')
    const lineCount = lines.length

    let codeHtml: string
    if (grammar) {
      codeHtml = Prism.highlight(code, grammar, lang)
    } else {
      this.logger.warn(`不支持的语言: ${lang}，返回纯文本`)
      codeHtml = this.escapeHtml(code)
    }

    // 构建带行号的 HTML
    const html = this.buildLineNumbersHtml(
      codeHtml,
      lineCount,
      startLine,
      highlightedLines,
      compact
    )

    return {
      html: showLineNumbers ? html : codeHtml,
      language: grammar ? lang : 'text',
      lineCount,
      codeHtml
    }
  }

  /**
   * 构建带行号的 HTML 结构
   * 结构：
   * <div class="code-highlight">
   *   <div class="code-lines">
   *     <span class="line-number">1</span>
   *     <span class="line-number">2</span>
   *   </div>
   *   <pre class="code-content"><code>...</code></pre>
   * </div>
   */
  private buildLineNumbersHtml(
    codeHtml: string,
    lineCount: number,
    startLine: number,
    highlightedLines: number[],
    compact: boolean
  ): string {
    const highlightedSet = new Set(highlightedLines)
    const hasHighlights = highlightedSet.size > 0

    // 生成行号列
    const lineNumbers: string[] = []
    for (let i = 0; i < lineCount; i++) {
      const lineNum = startLine + i
      const cls = hasHighlights && highlightedSet.has(lineNum)
        ? 'line-number line-highlighted'
        : 'line-number'
      lineNumbers.push(`<span class="${cls}" data-line="${lineNum}">${lineNum}</span>`)
    }

    // 将代码 HTML 按行分割，包裹每行
    const codeLines = this.splitCodeIntoLines(codeHtml)
    const wrappedLines = codeLines.map((line, i) => {
      const lineNum = startLine + i
      const cls = hasHighlights && highlightedSet.has(lineNum)
        ? 'code-line line-highlighted'
        : 'code-line'
      return `<span class="${cls}" data-line="${lineNum}">${line || ' '}</span>`
    })

    const compactClass = compact ? ' compact' : ''

    return [
      `<div class="code-highlight${compactClass}">`,
      `  <div class="code-line-numbers" aria-hidden="true">`,
      ...lineNumbers.map(ln => `    ${ln}`),
      `  </div>`,
      `  <pre class="code-content"><code>`,
      ...wrappedLines.map(ln => `    ${ln}`),
      `  </code></pre>`,
      `</div>`
    ].join('\n')
  }

  /**
   * 将 Prism.js 高亮后的 HTML 按行分割
   * 正确处理跨行的 <span> 标签
   */
  private splitCodeIntoLines(highlightedHtml: string): string[] {
    const rawLines = highlightedHtml.split('\n')
    const result: string[] = []
    let openTags: string[] = []

    for (const rawLine of rawLines) {
      let line = rawLine

      // 如果上一行有未关闭的标签，在当前行开头补上
      if (openTags.length > 0) {
        line = openTags.join('') + line
      }

      // 查找当前行中所有开闭标签
      const openMatches = line.match(/<span[^>]*>/g) || []
      const closeMatches = line.match(/<\/span>/g) || []

      // 计算未关闭的标签
      const openCount = openMatches.length - closeMatches.length
      if (openCount > 0) {
        // 有未关闭的 span，记录并在下一行开头补上
        for (let i = 0; i < openCount; i++) {
          openTags.push(openMatches[openMatches.length - openCount + i] || '<span>')
        }
      } else if (openCount < 0) {
        // 多出的闭合标签，去掉多余的
        line = line.replace(/<\/span>$/, '')
        openTags = []
      } else {
        openTags = []
      }

      // 如果行尾有未闭合标签，补上闭合
      if (openTags.length > 0) {
        line = line + '</span>'.repeat(openTags.length)
      }

      result.push(line)
    }

    return result
  }

  // ─── 能力工厂（单一职责、便于测试） ─────────────────────────

  /**
   * Issue#270: 符号链接路径校验
   * 用于 file_list / file_watch 等接受目录路径的能力，
   * 通过 realpath 解析符号链接后校验是否在白名单目录范围内，
   * 防止攻击者通过符号链接绕过路径限制访问任意目录。
   *
   * @param dirPath  用户传入的目录路径
   * @returns        解析后的真实路径
   */
  private async validateSymlinkPath(dirPath: string): Promise<string> {
    const { resolve, normalize, sep } = await import('path')
    const { realpath } = await import('fs/promises')

    if (dirPath.includes('..')) {
      throw new Error(`[安全] 路径遍历不允许: "${dirPath}"`)
    }

    const resolved = resolve(normalize(dirPath))

    // 白名单校验（如果配置了 allowedRoots）
    const allowedRoots: string[] =
      (this.meta?.settings?.allowedRoots as string[] | undefined)
      ?? (this.context?.config as any)?.get?.('file.allowedRoots', [])
      ?? []

    if (allowedRoots.length > 0) {
      const resolvedRoots = allowedRoots.map((r) => {
        const rr = resolve(r)
        return rr.endsWith(sep) ? rr : rr + sep
      })
      const isAllowed = resolvedRoots.some(
        (root) => resolved === root.slice(0, -1) || resolved.startsWith(root)
      )
      if (!isAllowed) {
        throw new Error(`[安全] 路径不在允许的目录范围内: "${dirPath}"`)
      }

      // 解析符号链接，防止通过 symlink 绕过路径检查
      try {
        const realPath = await realpath(resolved)
        if (realPath !== resolved) {
          const isRealInRoot = resolvedRoots.some(
            (root) => realPath === root.slice(0, -1) || realPath.startsWith(root)
          )
          if (!isRealInRoot) {
            throw new Error(`[安全] 符号链接目标超出允许范围: "${dirPath}" -> "${realPath}"`)
          }
          return realPath
        }
      } catch (err: any) {
        if (err.code !== 'ENOENT') throw err
      }
    }

    return resolved
  }

 *
 * 通道列表：
 *   file:read      — 读取文件内容
 *   file:write     — 写入文件内容
 *   file:list      — 列出目录文件
 *   file:watch     — 监听目录文件变更
 *   file:unwatch   — 取消监听目录
 *   file:highlight — 代码高亮预览
 */
export function registerFileIpcHandlers(
  reader: FileReaderService,
  writer: FileWriterService,
  watcher: FileWatcherService,
  highlighter: CodeHighlighter
): void {
  /** file:read — 读取文件内容 */
  ipcMain.handle('file:read', async (_event, params: { filePath: string; encoding?: BufferEncoding }) => {
    return reader.read(params.filePath, params.encoding)
  })

  /** file:write — 写入文件内容（支持追加模式） */
  ipcMain.handle('file:write', async (_event, params: { filePath: string; content: string; append?: boolean }) => {
    await writer.write(params.filePath, params.content, { append: params.append })
    return { success: true }
  })

  /** file:list — 列出目录内容 */
  ipcMain.handle('file:list', async (_event, params: { dirPath: string }) => {
    const { readdir, stat } = await import('fs/promises')
    const { join } = await import('path')
    const entries = await readdir(params.dirPath, { withFileTypes: true })
    return Promise.all(
      entries.map(async (entry) => {
        const fullPath = join(params.dirPath, entry.name)
        const stats = await stat(fullPath).catch(() => null)
        return {
          name: entry.name,
          path: fullPath,
          isDirectory: entry.isDirectory(),
          size: stats?.size ?? 0,
          modifiedAt: stats?.mtimeMs ?? 0,
        }
      })
    )
  })

  /** file:watch — 启动目录监听，通过事件推送变更 */
  ipcMain.handle('file:watch', async (event, params: { dirPath: string; events?: string[] }) => {
    const watchEvents = params.events ?? ['add', 'change', 'unlink']
    watcher.watch(params.dirPath, (change) => {
      if (watchEvents.includes(change.type)) {
        // 通过 webContents.send 推送变更到渲染进程
        event.sender.send('file:watch-event', change)
      }
    })
    return { watching: true, dirPath: params.dirPath, events: watchEvents }
  })

  /** file:unwatch — 取消目录监听 */
  ipcMain.handle('file:unwatch', async (_event, params: { dirPath: string }) => {
    watcher.unwatch(params.dirPath)
    return { success: true }
  })

  /** file:highlight — 代码高亮预览 */
  ipcMain.handle('file:highlight', async (_event, params: { code: string; language: string }) => {
    return highlighter.highlight(params.code, params.language)
  })
}

/** 移除所有文件操作 IPC 通道处理器（模块停用时调用） */
export function unregisterFileIpcHandlers(): void {
  const channels = ['file:read', 'file:write', 'file:list', 'file:watch', 'file:unwatch', 'file:highlight']
  for (const channel of channels) {
    ipcMain.removeHandler(channel)
  }
}
```

### 16.3 代码规范

- **文件大小限制**：默认 10MB，可配置
- **编码支持**：默认 UTF-8，支持指定编码
- **语言检测**：根据扩展名自动识别编程语言
- **安全检查**：限制可读写的文件扩展名白名单
- **错误处理规范**： <!-- ✅ Issue#267: 已修复 -->
  - **路径校验错误**：`validatePath()` 对路径遍历（`..`）、白名单外目录、符号链接逃逸抛出带 `[安全]` 前缀的 `Error`；`validateExtension()` 对不在白名单中的扩展名抛出明确错误
  - **文件读取错误**：`read()` 中文件不存在（`ENOENT`）、大小超限（带具体 MB 数值）、流式读取失败（stream `error` 事件）均向上抛出，由调用方统一捕获
  - **文件写入错误**：`write()` 原子写入失败时自动清理临时文件（`.tmp`），防止残留；追加写入失败直接向上抛出
  - **文件监听错误**：`file-watcher.ts` 监听 `error` 事件，记录日志后触发指数退避重试（1s → 2s → 4s，最多 3 次），重试前清理旧 watcher 实例
  - **文件监听速率限制**（Issue#269）：`file-watcher.ts` 对每个文件路径独立 debounce 防抖（默认 100ms），快速批量变更（如 git checkout、构建输出）只触发一次回调，避免事件风暴压垮下游处理；`unwatch` / `unwatchAll` / `retryWatch` 均清理防抖定时器和缓冲区
  - **能力层统一错误处理**：`index.ts` 中所有 Capability（file_read / file_write / file_list / file_watch / file_highlight）均使用 `try-catch` 包裹，返回 `{ success: false, error: message, errorCode: 100x }` 结构化错误，errorCode 从 1001 起依次编号
  - **模块停用错误**：`deactivate()` 中 `unwatchAll()` 调用包裹在 `try-catch` 中，确保清理失败不阻塞模块卸载
