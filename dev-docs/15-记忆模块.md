#### 步骤 2：实现短期记忆

**modules/memory/services/short-term.ts**：

```typescript
import type { DatabaseManager } from '../../../src/main/core/database'
import type { Logger } from '../../../src/main/types/logger'

export interface ShortTermMemory {
  id: string
  role: 'user' | 'assistant'
  content: string
  timestamp: number
  sessionId: string
  summarized: boolean
}

export class ShortTermMemoryService {
  private maxMessages: number

  constructor(
    private db: DatabaseManager,
    private logger: Logger,
    maxMessages = 50
  ) {
    this.maxMessages = maxMessages
    this.initTable()
  }

  private initTable(): void {
    this.db.run(`
      CREATE TABLE IF NOT EXISTS short_term_memory (
        id TEXT PRIMARY KEY,
        role TEXT NOT NULL,
        content TEXT NOT NULL,
        timestamp INTEGER NOT NULL,
        session_id TEXT NOT NULL,
        summarized INTEGER DEFAULT 0
      )
    `)
  }

  /** 保存一条短期记忆 */
  save(role: 'user' | 'assistant', content: string, sessionId: string): string {
    const id = crypto.randomUUID()
    this.db.run(
      'INSERT INTO short_term_memory (id, role, content, timestamp, session_id) VALUES (?, ?, ?, ?, ?)',
      id,
      role,
      content,
      Date.now(),
      sessionId
    )
    return id
  }

  /** 获取最近 N 条记忆 */
  getRecent(sessionId: string, limit?: number): ShortTermMemory[] {
    const rows = this.db.all<ShortTermMemory>(
      'SELECT id, role, content, timestamp, session_id AS sessionId, summarized FROM short_term_memory WHERE session_id = ? ORDER BY timestamp DESC LIMIT ?',
      sessionId,
      limit || this.maxMessages
    )
    return rows.reverse()
  }

  /**
   * Issue#248: 分页查询短期记忆
   *
   * @param sessionId  会话 ID
   * @param page       页码（从 1 开始）
   * @param pageSize   每页条数（默认 20）
   * @returns          { data, total, page, pageSize, totalPages }
   */
  getByPage(
    sessionId: string,
    page = 1,
    pageSize = 20
  ): { data: ShortTermMemory[]; total: number; page: number; pageSize: number; totalPages: number } {
    // 总条数
    const countRow = this.db.get<{ cnt: number }>(
      'SELECT COUNT(*) AS cnt FROM short_term_memory WHERE session_id = ?',
      sessionId
    )
    const total = countRow?.cnt ?? 0
    const totalPages = Math.max(1, Math.ceil(total / pageSize))

    // 页码安全钳位
    const safePage = Math.max(1, Math.min(page, totalPages))
    const offset = (safePage - 1) * pageSize

    const data = this.db.all<ShortTermMemory>(
      'SELECT id, role, content, timestamp, session_id AS sessionId, summarized FROM short_term_memory WHERE session_id = ? ORDER BY timestamp DESC LIMIT ? OFFSET ?',
      sessionId,
      pageSize,
      offset
    ).reverse() // 保持时间正序展示

    return { data, total, page: safePage, pageSize, totalPages }
  }

  /** 获取需要摘要的老记忆 */
  getUnsummarized(sessionId: string): ShortTermMemory[] {
    return this.db.all<ShortTermMemory>(
      'SELECT id, role, content, timestamp, session_id AS sessionId, summarized FROM short_term_memory WHERE session_id = ? AND summarized = 0 ORDER BY timestamp ASC',
      sessionId
    )
  }

  /** 标记为已摘要（事务包裹，保证批量原子性） */
  markSummarized(ids: string[]): void {
    if (ids.length === 0) return
    this.db.transaction(() => {
      for (const id of ids) {
        this.db.run('UPDATE short_term_memory SET summarized = 1 WHERE id = ?', id)
      }
    })
  }

  /** 清理旧记忆 */
  cleanup(sessionId: string, keepRecent: number): void {
    this.db.run(
      `DELETE FROM short_term_memory WHERE session_id = ? AND id NOT IN (
        SELECT id FROM short_term_memory WHERE session_id = ? ORDER BY timestamp DESC LIMIT ?
      )`,
      sessionId,
      sessionId,
      keepRecent
    )
  }
}
```

#### 步骤 3：实现长期记忆（向量检索）

**modules/memory/services/long-term.ts**：

```typescript
import type { AIService } from '../../../src/main/types/ai'
import type { DatabaseManager } from '../../../src/main/core/database'
import type { Logger } from '../../../src/main/types/logger'

export interface LongTermMemory {
  id: string
  content: string
  embedding: number[]
  type: 'fact' | 'preference' | 'event'
  createdAt: number
  /** 过期时间戳（毫秒），null 表示永不过期 */
  expiresAt: number | null
  metadata: Record<string, unknown>
}

export class LongTermMemoryService {
  /** sqlite-vec 扩展是否可用（用于 ANN 向量搜索） */
  private vecTableAvailable = false
  /** 默认 TTL（毫秒），null 表示永不过期 */
  private defaultTTL: number | null

  constructor(
    private db: DatabaseManager,
    private ai: AIService,
    private logger: Logger,
    private embeddingModel?: string,
    /** 长期记忆默认 TTL（天数），0 或不传表示永不过期 */
    defaultTTLDays?: number
  ) {
    this.defaultTTL = defaultTTLDays && defaultTTLDays > 0 ? defaultTTLDays * 24 * 60 * 60 * 1000 : null
    this.initTable()
    this.initVecTable()
  }

  private initTable(): void {
    this.db.run(`
      CREATE TABLE IF NOT EXISTS long_term_memory (
        id TEXT PRIMARY KEY,
        content TEXT NOT NULL,
        embedding BLOB,
        type TEXT NOT NULL,
        created_at INTEGER NOT NULL,
        expires_at INTEGER,
        metadata TEXT
      )
    `)
    // Issue#249: 为 expires_at 列创建索引，加速过期清理查询
    this.db.run(`
      CREATE INDEX IF NOT EXISTS idx_ltm_expires_at
      ON long_term_memory (expires_at) WHERE expires_at IS NOT NULL
    `)
    // Issue#241: 为 embedding 列创建索引，加速 WHERE embedding IS NOT NULL 过滤
    this.db.run(`
      CREATE INDEX IF NOT EXISTS idx_ltm_embedding
      ON long_term_memory (id) WHERE embedding IS NOT NULL
    `)
  }

  /**
   * 初始化 sqlite-vec 向量搜索虚拟表
   * Issue#241: 使用 sqlite-vec 扩展实现数据库级 ANN 搜索，避免全量加载到内存
   */
  private initVecTable(): void {
    try {
      // sqlite-vec 虚拟表：存储向量并支持高效近似最近邻搜索
      // 维度需与 embeddingModel 输出维度一致（如 nomic-embed-text 为 768）
      this.db.run(`
        CREATE VIRTUAL TABLE IF NOT EXISTS long_term_memory_vec USING vec0(
          rowid INTEGER PRIMARY KEY,
          embedding FLOAT[768]
        )
      `)
      this.vecTableAvailable = true
      this.logger.info('sqlite-vec 向量搜索已启用')
    } catch (err) {
      this.vecTableAvailable = false
      this.logger.warn('sqlite-vec 扩展不可用，向量搜索将回退到分批扫描模式', { error: err })
    }
  }

  /** 保存长期记忆（生成嵌入向量） */
  async save(
    content: string,
    type: 'fact' | 'preference' | 'event',
    metadata?: Record<string, unknown>,
    /** 自定义 TTL（天数），0 使用默认 TTL，null 永不过期 */
    ttlDays?: number | null
  ): Promise<string> {
    const id = crypto.randomUUID()

    // 计算过期时间
    let expiresAt: number | null = null
    if (ttlDays === null) {
      expiresAt = null // 显式永不过期
    } else if (ttlDays !== undefined && ttlDays > 0) {
      expiresAt = Date.now() + ttlDays * 24 * 60 * 60 * 1000
    } else if (this.defaultTTL !== null) {
      expiresAt = Date.now() + this.defaultTTL
    }

    // 生成嵌入向量
    let embedding: Buffer | null = null
    try {
      const vector = await this.ai.embed(content, this.embeddingModel)
      // Issue#243: 新建 Float32Array 的 byteOffset 为 0，Buffer.from(arrayBuffer) 安全
      // Issue#251: new Float32Array(vector) 会分配新内存并逐元素拷贝，对高频写入场景有一定开销。
      //   优化方向：(1) 使用 Buffer.allocUnsafe + 手动写入 float 避免中间 Float32Array 分配；
      //   (2) 复用共享 Float32Array 实例（ReusableBuffer 模式）减少 GC 压力；
      //   (3) 若 vector 已是 TypedArray，直接使用其 buffer 构造 Buffer，跳过拷贝。
      embedding = Buffer.from(new Float32Array(vector).buffer)
    } catch (err) {
      this.logger.warn('嵌入向量生成失败，保存为无向量记忆', { error: err })
    }

    // Issue#246: 事务包裹主表 + vec 虚拟表写入，保证原子性
    this.db.transaction(() => {
      this.db.run(
        'INSERT INTO long_term_memory (id, content, embedding, type, created_at, expires_at, metadata) VALUES (?, ?, ?, ?, ?, ?, ?)',
        id,
        content,
        embedding,
        type,
        Date.now(),
        expiresAt,
        metadata ? JSON.stringify(metadata) : null
      )

      // Issue#241: 同步写入 sqlite-vec 虚拟表，保持向量索引与主表一致
      if (this.vecTableAvailable && embedding) {
        try {
          this.db.run(
            'INSERT INTO long_term_memory_vec (rowid, embedding) VALUES (?, ?)',
            id,
            embedding
          )
        } catch (err) {
          this.logger.warn('写入 sqlite-vec 虚拟表失败，事务已回滚主表写入', { error: err })
          throw err // 回滚整个事务
        }
      }
    })

    return id
  }

  /**
   * 语义搜索
   *
   * Issue#241 修复：不再全量 SELECT * WHERE embedding IS NOT NULL 加载到内存。
   * 采用分层策略：
   *   1. 优先使用 sqlite-vec 扩展的虚拟表进行数据库级 ANN 搜索（O(log n)）
   *   2. 若 sqlite-vec 不可用，回退到分批加载 + 最小堆方案（兼容模式）
   *
   * 生产环境强烈建议安装 sqlite-vec 扩展以获得最佳性能。
   */
  async search(query: string, limit = 5): Promise<Array<LongTermMemory & { score: number }>> {
    const queryVector = await this.ai.embed(query, this.embeddingModel)

    // 策略 1：使用 sqlite-vec 虚拟表进行高效 ANN 搜索
    if (this.vecTableAvailable) {
      return this.searchWithVec(queryVector, limit)
    }

    // 策略 2：回退到分批加载（兼容模式，记录警告）
    this.logger.warn('sqlite-vec 扩展不可用，回退到分批扫描模式（生产环境建议安装 sqlite-vec）')
    return this.searchWithBatchScan(queryVector, limit)
  }

  /** 使用 sqlite-vec 虚拟表进行 ANN 搜索（推荐的生产方案） */
  private async searchWithVec(
    queryVector: number[],
    limit: number
  ): Promise<Array<LongTermMemory & { score: number }>> {
    // Issue#251: 与 save() 相同的 new Float32Array 分配开销。searchWithVec 仅在 sqlite-vec 路径
    //   调用，频率低于 save()，但对高频语义搜索场景仍可优化为复用共享 buffer。
    const queryBuf = Buffer.from(new Float32Array(queryVector).buffer)
    // sqlite-vec 虚拟表直接在数据库层计算余弦距离并排序，仅返回 top-K
    const rows = this.db.all<{
      id: string
      content: string
      type: string
      created_at: number
      expires_at: number | null
      metadata: string | null
      distance: number
    }>(
      `SELECT m.id, m.content, m.type, m.created_at, m.expires_at, m.metadata, v.distance
       FROM long_term_memory_vec v
       JOIN long_term_memory m ON m.id = v.rowid
       WHERE v.embedding MATCH ?
         AND (m.expires_at IS NULL OR m.expires_at > ?)
       ORDER BY v.distance ASC
       LIMIT ?`,
      queryBuf,
      Date.now(),
      limit
    )
    return rows.map((row) => ({
      id: row.id,
      content: row.content,
      embedding: [],
      type: row.type as LongTermMemory['type'],
      createdAt: row.created_at,
      expiresAt: row.expires_at ?? null,
      metadata: row.metadata ? JSON.parse(row.metadata) : {},
      score: 1 - row.distance // sqlite-vec 返回距离，转换为相似度
    }))
  }

  /** 兼容模式：分批加载 + 最小堆（仅在 sqlite-vec 不可用时使用） */
  private async searchWithBatchScan(
    queryVector: number[],
    limit: number
  ): Promise<Array<LongTermMemory & { score: number }>> {
    const BATCH_SIZE = 200
    let offset = 0
    const topK: Array<{ item: LongTermMemory & { score: number }; score: number }> = []

    // eslint-disable-next-line no-constant-condition
    while (true) {
      const rows = this.db.all<{
        id: string
        content: string
        embedding: Buffer | null
        type: string
        created_at: number
        expires_at: number | null
        metadata: string | null
      }>(
        'SELECT id, content, embedding, type, created_at, expires_at, metadata FROM long_term_memory WHERE embedding IS NOT NULL AND (expires_at IS NULL OR expires_at > ?) LIMIT ? OFFSET ?',
        Date.now(),
        BATCH_SIZE,
        offset
      )

      if (rows.length === 0) break

      for (const row of rows) {
        const emb = row.embedding!
        // Issue#243: 必须使用 3 参数形式构造 Float32Array，避免 Buffer 池偏移导致读到错误数据
        // Node.js Buffer 可能共享底层 ArrayBuffer（Buffer pooling），emb.buffer 是整个池，
        // emb.byteOffset 才是实际数据在池中的起始位置
        const memoryVector = new Float32Array(emb.buffer, emb.byteOffset, emb.byteLength / Float32Array.BYTES_PER_ELEMENT)
        // Issue#251: Array.from(memoryVector) 会将 Float32Array 逐元素拷贝为普通数组，在分批扫描的
        //   循环中对每行记录都产生一次 O(n) 分配（n=768 维），是批量搜索的主要开销来源。
        //   优化方案：修改 cosineSimilarity 签名为 (a: number[], b: ArrayLike<number>) 或
        //   (a: Float32Array, b: Float32Array)，直接接收 Float32Array，消除 Array.from 拷贝。
        //   Float32Array 已支持索引访问和 .length，与 cosineSimilarity 内部的 for 循环完全兼容。
        const score = this.cosineSimilarity(queryVector, Array.from(memoryVector))

        const entry = {
          item: {
            id: row.id,
            content: row.content,
            embedding: [],
            type: row.type as LongTermMemory['type'],
            createdAt: row.created_at,
            expiresAt: row.expires_at ?? null,
            metadata: row.metadata ? JSON.parse(row.metadata) : {},
            score
          },
          score
        }

        if (topK.length < limit) {
          topK.push(entry)
          if (topK.length === limit) this.heapify(topK)
        } else if (score > topK[0].score) {
          topK[0] = entry
          this.siftDown(topK, 0)
        }
      }

      if (rows.length < BATCH_SIZE) break
      offset += BATCH_SIZE
    }

    return topK
      .sort((a, b) => b.score - a.score)
      .map((entry) => entry.item)
  }

  /** 最小堆：建堆 */
  private heapify(heap: Array<{ score: number }>): void {
    for (let i = Math.floor(heap.length / 2) - 1; i >= 0; i--) {
      this.siftDown(heap, i)
    }
  }

  /** 最小堆：下沉操作 */
  private siftDown(heap: Array<{ score: number }>, i: number): void {
    const n = heap.length
    while (true) {
      let smallest = i
      const left = 2 * i + 1
      const right = 2 * i + 2
      if (left < n && heap[left].score < heap[smallest].score) smallest = left
      if (right < n && heap[right].score < heap[smallest].score) smallest = right
      if (smallest === i) break
      ;[heap[i], heap[smallest]] = [heap[smallest], heap[i]]
      i = smallest
    }
  }

  /**
   * 余弦相似度计算
   * Issue#251 优化建议：将参数类型从 number[] 改为 ArrayLike<number>，使其直接兼容 Float32Array，
   * 避免调用方使用 Array.from() 转换。对于 768 维向量，每批次扫描可节省一次 768 元素的数组分配。
   * 注意：ArrayLike<number> 不含 .reduce 等方法，但本实现仅使用索引访问和 .length，完全兼容。
   */
  private cosineSimilarity(a: number[], b: number[]): number {
    if (a.length !== b.length) return 0
    let dotProduct = 0
    let normA = 0
    let normB = 0
    for (let i = 0; i < a.length; i++) {
      dotProduct += a[i] * b[i]
      normA += a[i] * a[i]
      normB += b[i] * b[i]
    }
    return dotProduct / (Math.sqrt(normA) * Math.sqrt(normB))
  }

  /** 删除记忆 */
  delete(id: string): void {
    // Issue#246: 事务包裹主表 + vec 虚拟表删除，保证原子性
    this.db.transaction(() => {
      this.db.run('DELETE FROM long_term_memory WHERE id = ?', id)
      // Issue#241: 同步删除 sqlite-vec 虚拟表中的向量
      if (this.vecTableAvailable) {
        this.db.run('DELETE FROM long_term_memory_vec WHERE rowid = ?', id)
      }
    })
  }

  /** 获取所有未过期的记忆 */
  getAll(type?: string): LongTermMemory[] {
    const now = Date.now()
    const sql = type
      ? 'SELECT * FROM long_term_memory WHERE type = ? AND (expires_at IS NULL OR expires_at > ?) ORDER BY created_at DESC'
      : 'SELECT * FROM long_term_memory WHERE (expires_at IS NULL OR expires_at > ?) ORDER BY created_at DESC'

    const rows = this.db.all<any>(sql, ...(type ? [type, now] : [now]))

    return rows.map((row) => ({
      id: row.id,
      content: row.content,
      embedding: [],
      type: row.type,
      createdAt: row.created_at,
      expiresAt: row.expires_at ?? null,
      metadata: row.metadata ? JSON.parse(row.metadata) : {}
    }))
  }

  /**
   * 清理已过期的长期记忆
   * @returns 已清理的记忆数量
   */
  cleanup(): number {
    const now = Date.now()
    // 先获取待删除的 ID（用于同步清理 vec 虚拟表）
    const expired = this.db.all<{ id: string }>(
      'SELECT id FROM long_term_memory WHERE expires_at IS NOT NULL AND expires_at <= ?',
      now
    )
    if (expired.length === 0) return 0

    this.db.transaction(() => {
      this.db.run('DELETE FROM long_term_memory WHERE expires_at IS NOT NULL AND expires_at <= ?', now)
      if (this.vecTableAvailable) {
        for (const row of expired) {
          this.db.run('DELETE FROM long_term_memory_vec WHERE rowid = ?', row.id)
        }
      }
    })

    this.logger.info(`已清理 ${expired.length} 条过期长期记忆`)
    return expired.length
  }
}
```

```typescript
import { ShortTermMemoryService } from './short-term'
import { LongTermMemoryService } from './long-term'
import type { Database } from '../../../src/main/types/database'
import type { AIService } from '../../../src/main/types/ai'
import type { Logger } from '../../../src/main/types/logger'

export type MemoryType = 'fact' | 'preference' | 'summary' | 'event'

export interface MemorySearchResult {
  id: string
  content: string
  type: MemoryType
  score: number
  createdAt: number
  metadata: Record<string, unknown>
}

export interface MemoryManagerConfig {
  /** 短期记忆最大保留条数（默认 50） */
  shortTermLimit?: number
  /** 自动摘要触发间隔：每 N 条未摘要记忆触发一次（默认 10） */
  autoSummaryThreshold?: number
  /** 自动摘要调度周期（毫秒），默认 5 分钟 */
  autoSummaryIntervalMs?: number
  /** 嵌入向量模型名称（默认 'nomic-embed-text'） */
  embeddingModel?: string
  /** 向量搜索返回的最大结果数（默认 5） */
  searchTopK?: number
  /** 搜索结果缓存容量（默认 128 条） */
  searchCacheSize?: number
  /** 搜索结果缓存 TTL（毫秒），默认 60 秒 */
  searchCacheTTL?: number
  /** 长期记忆默认 TTL（天数），0 表示永不过期（默认 90 天） */
  longTermTTLDays?: number
  /** 过期清理调度周期（毫秒），默认 1 小时 */
  cleanupIntervalMs?: number

/**
 * MemoryManager —— 记忆模块核心协调类 <!-- ✅ Issue#239: 已修复 -->
 *
 * 职责：
 * 1. 管理 ShortTermMemoryService 和 LongTermMemoryService 的生命周期
 * 2. 提供统一的记忆存储、检索、删除接口
 * 3. 协调短期记忆 → 长期记忆的自动摘要与迁移
 * 4. 集成向量搜索，支持语义检索
 * 5. 对外暴露简洁 API，屏蔽底层存储差异
 * 6. 内置 LRU 搜索缓存层（Issue#250），减少重复向量计算和数据库查询
 */
export class MemoryManager {
  private shortTerm: ShortTermMemoryService
  private longTerm: LongTermMemoryService
  private logger: Logger
  private ai: AIService
  private config: Required<MemoryManagerConfig>
  private summaryTimer: ReturnType<typeof setInterval> | null = null
  /** Issue#249: 过期清理定时器 */
  private cleanupTimer: ReturnType<typeof setInterval> | null = null
  private destroyed = false
  /** 写锁队列：保证同一时刻只有一个写操作在执行，防止并发写入冲突 */
  private writeQueue: Promise<void> = Promise.resolve()

  // ─── 搜索缓存层（Issue#250） ──────────────────────────────
  /** LRU 缓存节点 */
  private searchCache = new Map<string, { results: MemorySearchResult[]; timestamp: number }>()
  private searchCacheOrder: string[] = [] // LRU 顺序队列，最新在末尾
  private searchCacheSize: number
  private searchCacheTTL: number

  constructor(db: Database, ai: AIService, logger: Logger, config?: MemoryManagerConfig) {
    this.logger = logger
    this.ai = ai

    this.config = {
      shortTermLimit: config?.shortTermLimit ?? 50,
      autoSummaryThreshold: config?.autoSummaryThreshold ?? 10,
      autoSummaryIntervalMs: config?.autoSummaryIntervalMs ?? 5 * 60 * 1000,
      embeddingModel: config?.embeddingModel ?? 'nomic-embed-text',
      searchTopK: config?.searchTopK ?? 5,
      searchCacheSize: config?.searchCacheSize ?? 128,
      searchCacheTTL: config?.searchCacheTTL ?? 60_000,
      longTermTTLDays: config?.longTermTTLDays ?? 90,
      cleanupIntervalMs: config?.cleanupIntervalMs ?? 60 * 60 * 1000
    }

    // 初始化缓存参数
    this.searchCacheSize = this.config.searchCacheSize
    this.searchCacheTTL = this.config.searchCacheTTL

    // 初始化子服务（Issue#249: 传入长期记忆 TTL）
    this.shortTerm = new ShortTermMemoryService(db, logger, this.config.shortTermLimit)
    this.longTerm = new LongTermMemoryService(db, ai, logger, this.config.embeddingModel, this.config.longTermTTLDays)

    // 启动自动摘要调度
    this.startAutoSummaryScheduler()
    // Issue#249: 启动过期清理调度
    this.startCleanupScheduler()

    this.logger.info('MemoryManager 已初始化', {
      shortTermLimit: this.config.shortTermLimit,
      autoSummaryThreshold: this.config.autoSummaryThreshold,
      autoSummaryIntervalMs: this.config.autoSummaryIntervalMs,
      embeddingModel: this.config.embeddingModel,
      longTermTTLDays: this.config.longTermTTLDays,
      cleanupIntervalMs: this.config.cleanupIntervalMs
    })
  }

  /**
   * 写锁队列：将异步写操作串行化，防止并发写入冲突
   * 所有写操作通过此方法排队执行，保证同一时刻只有一个写操作在运行
   */
  private async withWriteLock<T>(fn: () => Promise<T>): Promise<T> {
    const prev = this.writeQueue
    let resolve: () => void
    this.writeQueue = new Promise<void>(r => { resolve = r })
    await prev
    try {
      return await fn()
    } finally {
      resolve!()
    }
  }

  // ─── 统一存储接口 ────────────────────────────────────────

  /**
   * 保存一条记忆
   * @param content  记忆内容
   * @param type     记忆类型（fact / preference / summary / event）
   * @param sessionId 会话 ID（可选，默认 'default'）
   * @returns 生成的记忆 ID
   */
  async save(content: string, type: MemoryType = 'fact', sessionId = 'default', role: 'user' | 'assistant' = 'user'): Promise<string> {
    if (this.destroyed) throw new Error('MemoryManager 已销毁')

    return this.withWriteLock(async () => {
      // 始终写入短期记忆（ShortTermMemoryService.save 参数：role, content, sessionId）
      const id = this.shortTerm.save(role, content, sessionId)

      this.logger.debug(`记忆已保存 [短期] id=${id} type=${type}`)

      // 如果是摘要类型或事实类型，同步写入长期记忆（向量化存储）
      if (type === 'fact' || type === 'preference' || type === 'summary') {
        try {
          await this.longTerm.save(content, type, { sessionId, source: 'direct' })
          this.logger.debug(`记忆已同步 [长期] id=${id}`)
        } catch (err) {
          this.logger.warn('长期记忆写入失败，短期记忆仍可用', { id, error: err })
        }
      }

      // 写操作后失效搜索缓存，保证数据一致性（Issue#250）
      this.invalidateSearchCache()

      return id
    })
  }

  /**
   * 自动保存（由事件回调触发）
   * 仅在短期记忆中追加，不立即写入长期记忆
   */
  autoSave(role: 'user' | 'assistant', text: string, sessionId = 'default'): string {
    if (this.destroyed) return ''

    const content = `[${role}] ${text}`.slice(0, 2000) // 截断超长内容
    const id = this.shortTerm.save(role, content, sessionId)

    this.logger.debug(`自动保存记忆 id=${id} role=${role}`)

    // 检查是否需要触发自动摘要
    this.maybeTriggerAutoSummary(sessionId)

    // 写操作后失效搜索缓存（Issue#250）
    this.invalidateSearchCache()

    return id
  }

  // ─── 统一检索接口 ────────────────────────────────────────

  // ─── 缓存辅助方法（Issue#250） ────────────────────────────

  /** 从缓存中获取搜索结果，命中时刷新 LRU 顺序 */
  private cacheGet(key: string): MemorySearchResult[] | null {
    const entry = this.searchCache.get(key)
    if (!entry) return null

    // TTL 过期检查
    if (Date.now() - entry.timestamp > this.searchCacheTTL) {
      this.searchCache.delete(key)
      this.searchCacheOrder = this.searchCacheOrder.filter(k => k !== key)
      return null
    }

    // 刷新 LRU 顺序：将命中的 key 移到末尾
    this.searchCacheOrder = this.searchCacheOrder.filter(k => k !== key)
    this.searchCacheOrder.push(key)
    return entry.results
  }

  /** 将搜索结果写入缓存，超出容量时淘汰最久未使用的条目 */
  private cacheSet(key: string, results: MemorySearchResult[]): void {
    // 如果 key 已存在，先移除旧的 LRU 记录
    if (this.searchCache.has(key)) {
      this.searchCacheOrder = this.searchCacheOrder.filter(k => k !== key)
    }

    // 淘汰最久未使用的条目（LRU 淘汰策略）
    while (this.searchCacheOrder.length >= this.searchCacheSize) {
      const evictKey = this.searchCacheOrder.shift()!
      this.searchCache.delete(evictKey)
    }

    this.searchCache.set(key, { results, timestamp: Date.now() })
    this.searchCacheOrder.push(key)
  }

  /** 清空搜索缓存（写操作后调用，保证数据一致性） */
  private invalidateSearchCache(): void {
    this.searchCache.clear()
    this.searchCacheOrder = []
  }

  /** 生成搜索缓存 key（query + limit 组合） */
  private cacheKey(query: string, limit: number): string {
    return `${query}::${limit}`
  }

  /**
   * 搜索记忆（短期 + 长期）
   * 优先返回向量搜索的语义匹配结果，再补充短期记忆的近期内容。
   * 内置 LRU 缓存层（Issue#250）：相同 query+limit 组合在 TTL 内直接返回缓存结果，
   * 避免重复的向量计算和数据库查询；写操作（save/delete/clearSession）自动失效缓存。
   */
  async search(query: string, limit?: number): Promise<MemorySearchResult[]> {
    if (this.destroyed) throw new Error('MemoryManager 已销毁')

    const topK = limit ?? this.config.searchTopK
    const key = this.cacheKey(query, topK)

    // 1. 缓存命中：直接返回缓存结果
    const cached = this.cacheGet(key)
    if (cached) {
      this.logger.debug(`搜索缓存命中: query="${query.slice(0, 30)}..." topK=${topK}`)
      return cached
    }

    const results: MemorySearchResult[] = []

    // 2. 向量搜索长期记忆
    try {
      const longTermResults = await this.longTerm.search(query, topK)
      for (const item of longTermResults) {
        results.push({
          id: item.id,
          content: item.content,
          type: item.type as MemoryType,
          score: item.score,
          createdAt: item.createdAt,
          metadata: item.metadata
        })
      }
    } catch (err) {
      this.logger.warn('长期记忆搜索失败', { error: err })
    }

    // 3. 如果结果不足，补充短期记忆的最近条目
    if (results.length < topK) {
      const recentMemories = this.shortTerm.getRecent('default', topK - results.length)
      for (const mem of recentMemories) {
        // 避免重复（短期记忆中已同步到长期的内容）
        if (!results.some(r => r.id === mem.id)) {
          results.push({
            id: mem.id,
            content: mem.content,
            type: mem.role === 'assistant' ? 'event' : 'fact', // 根据 role 推断类型
            score: 0.5, // 短期记忆给固定相关性分数
            createdAt: mem.timestamp,
            metadata: {}
          })
        }
      }
    }

    const finalResults = results.slice(0, topK)

    // 4. 写入缓存
    this.cacheSet(key, finalResults)

    return finalResults
  }

  /**
   * 按类型获取长期记忆
   */
  getByType(type: MemoryType): MemorySearchResult[] {
    return this.longTerm.getAll(type).map(item => ({
      id: item.id,
      content: item.content,
      type: item.type as MemoryType,
      score: 1,
      createdAt: item.createdAt,
      metadata: item.metadata
    }))
  }

  /**
   * 获取最近的短期记忆
   */
  getRecent(sessionId = 'default', limit = 20) {
    return this.shortTerm.getRecent(sessionId, limit)
  }

  /**
   * Issue#248: 分页获取短期记忆
   *
   * @param sessionId  会话 ID
   * @param page       页码（从 1 开始）
   * @param pageSize   每页条数（默认 20）
   * @returns          { data, total, page, pageSize, totalPages }
   */
  getRecentPage(sessionId = 'default', page = 1, pageSize = 20) {
    return this.shortTerm.getByPage(sessionId, page, pageSize)
  }

  // ─── 删除接口 ────────────────────────────────────────────

  /**
   * 删除记忆（清除长期存储，短期记忆通过 cleanup 清理）
   */
  async delete(id: string): Promise<void> {
    // Issue#246: 使用写锁串行化，防止并发删除冲突
    await this.withWriteLock(async () => {
      this.longTerm.delete(id)
      this.logger.debug(`长期记忆已删除 id=${id}`)
      // 写操作后失效搜索缓存（Issue#250）
      this.invalidateSearchCache()
    })
  }

  /**
   * 清空指定会话的短期记忆（保留最近 keepRecent 条）
   */
  async clearSession(sessionId = 'default', keepRecent = 0): Promise<void> {
    // Issue#246: 使用写锁串行化，防止并发清理冲突
    await this.withWriteLock(async () => {
      this.shortTerm.cleanup(sessionId, keepRecent)
      this.logger.debug(`会话记忆已清理 sessionId=${sessionId} 保留=${keepRecent}`)
      // 写操作后失效搜索缓存（Issue#250）
      this.invalidateSearchCache()
    })
  }

  // ─── 自动摘要调度 ────────────────────────────────────────

  /**
   * 启动定时摘要调度器
   * 每 5 分钟检查一次是否有需要摘要的短期记忆
   */
  private startAutoSummaryScheduler(): void {
    this.summaryTimer = setInterval(() => {
      this.runAutoSummary('default').catch(err => {
        this.logger.error('自动摘要调度异常', { error: err })
      })
    }, this.config.autoSummaryIntervalMs)

    // 允许定时器不阻止进程退出
    if (this.summaryTimer && typeof this.summaryTimer.unref === 'function') {
      this.summaryTimer.unref()
    }
  }

  /**
   * Issue#249: 启动过期清理调度器
   * 定期清理已过期的长期记忆，防止数据无限增长
   */
  private startCleanupScheduler(): void {
    if (this.config.longTermTTLDays <= 0) return // TTL 为 0 表示永不过期

    this.cleanupTimer = setInterval(() => {
      try {
        const count = this.longTerm.cleanup()
        if (count > 0) {
          this.logger.info(`过期清理完成: 已清理 ${count} 条过期长期记忆`)
        }
      } catch (err) {
        this.logger.error('过期清理调度异常', { error: err })
      }
    }, this.config.cleanupIntervalMs)

    if (this.cleanupTimer && typeof this.cleanupTimer.unref === 'function') {
      this.cleanupTimer.unref()
    }
  }

  /**
   * 检查并触发自动摘要
   * 当未摘要的短期记忆数量达到阈值时触发
   */
  private maybeTriggerAutoSummary(sessionId: string): void {
    const unsummarized = this.shortTerm.getUnsummarized(sessionId)
    if (unsummarized.length >= this.config.autoSummaryThreshold) {
      this.runAutoSummary(sessionId).catch(err => {
        this.logger.error('自动摘要触发异常', { error: err })
      })
    }
  }

  /**
   * 执行自动摘要：将老的短期记忆压缩为长期记忆摘要
   *
   * 流程：
   * 1. 获取所有未摘要的短期记忆
   * 2. 调用 AI 生成摘要
   * 3. 将摘要写入长期记忆
   * 4. 标记短期记忆为已摘要
   */
  private async runAutoSummary(sessionId: string): Promise<void> {
    const unsummarized = this.shortTerm.getUnsummarized(sessionId)
    if (unsummarized.length === 0) return

    this.logger.info(`开始自动摘要: ${unsummarized.length} 条待处理`)

    try {
      // 构建摘要提示
      const conversationText = unsummarized
        .map(m => `[${new Date(m.timestamp).toISOString()}] ${m.content}`)
        .join('\n')

      const prompt = `请将以下对话记忆压缩为简洁的摘要，保留关键事实、用户偏好和重要事件：\n\n${conversationText}\n\n摘要：`

      const summary = await this.ai.chat(prompt)

      // 使用写锁串行化，防止并发摘要操作冲突
      await this.withWriteLock(async () => {
        // 将摘要写入长期记忆
        await this.longTerm.save(summary, 'summary', {
          sessionId,
          source: 'auto-summary',
          summarizedCount: unsummarized.length,
          summarizedIds: unsummarized.map(m => m.id)
        })

        // 标记短期记忆为已摘要（markSummarized 内部使用事务包裹批量 UPDATE）
        this.shortTerm.markSummarized(unsummarized.map(m => m.id))

        // 写操作后失效搜索缓存（Issue#250）
        this.invalidateSearchCache()
      })

      this.logger.info(`自动摘要完成: ${unsummarized.length} 条记忆已压缩`)
    } catch (err) {
      this.logger.error('自动摘要执行失败', { error: err })
    }
  }

  // ─── 生命周期管理 ────────────────────────────────────────

  /**
   * Issue#249: 手动触发过期清理
   * @returns 已清理的记忆数量
   */
  cleanupExpired(): number {
    return this.longTerm.cleanup()
  }

  /**
   * 获取记忆模块统计信息
   */
  getStats(): { shortTermCount: number; longTermCount: number; config: Required<MemoryManagerConfig> } {
    const shortTermRecent = this.shortTerm.getRecent('default', 9999)
    const longTermAll = this.longTerm.getAll()

    return {
      shortTermCount: shortTermRecent.length,
      longTermCount: longTermAll.length,
      config: this.config
    }
  }

  /**
   * 销毁 MemoryManager，释放所有资源
   * - 停止自动摘要调度
   * - 清理内部状态
   */
  destroy(): void {
    if (this.destroyed) return

    this.destroyed = true

    // 停止定时器
    if (this.summaryTimer) {
      clearInterval(this.summaryTimer)
      this.summaryTimer = null
    }
    // Issue#249: 停止过期清理定时器
    if (this.cleanupTimer) {
      clearInterval(this.cleanupTimer)
      this.cleanupTimer = null
    }

    // 清理搜索缓存（Issue#250）
    this.searchCache.clear()
    this.searchCacheOrder = []

    this.logger.info('MemoryManager 已销毁')
  }
}
```

#### 步骤 4.5：实现记忆摘要器

**modules/memory/services/summarizer.ts**：

```typescript
import type { AIService } from '../../../src/main/types/ai'
import type { Logger } from '../../../src/main/types/logger'
import type { ShortTermMemory } from './short-term'

/** 摘要结果 */
export interface SummaryResult {
  /** 摘要文本 */
  summary: string
  /** 摘要时间戳 */
  timestamp: number
  /** 摘要涵盖的消息 ID 列表 */
  messageIds: string[]
  /** 提取的关键信息 */
  keyPoints: string[]
  /** 检测到的用户偏好 */
  userPreferences: string[]
}

/** 摘要配置 */
export interface SummarizerConfig {
  /** 每批摘要的最大消息数 */
  batchSize: number
  /** 摘要目标语言 */
  language: string
  /** 是否提取用户偏好 */
  extractPreferences: boolean
}

const DEFAULT_CONFIG: SummarizerConfig = {
  batchSize: 20,
  language: 'zh-CN',
  extractPreferences: true
}

/**
 * 记忆摘要生成器
 *
 * 职责：
 * - 将对话消息压缩为结构化摘要
 * - 支持增量摘要（在旧摘要基础上追加新内容）
 * - 提取关键信息与用户偏好
 * - 与 MemoryManager 配合，标记已摘要消息
 *
 * 摘要格式：时间 + 关键信息 + 用户偏好
 * 输出结构化 JSON，供 MemoryManager 写入长期记忆
 */
export class Summarizer {
  private config: SummarizerConfig

  constructor(
    private ai: AIService,
    private logger: Logger,
    config?: Partial<SummarizerConfig>
  ) {
    this.config = { ...DEFAULT_CONFIG, ...config }
  }

  /**
   * 将消息列表压缩为摘要
   *
   * @param messages  待摘要的短期记忆消息
   * @param existingSummary  已有的旧摘要（增量模式）
   * @returns 摘要结果
   */
  async summarize(
    messages: ShortTermMemory[],
    existingSummary?: string
  ): Promise<SummaryResult> {
    if (messages.length === 0) {
      return {
        summary: existingSummary || '',
        timestamp: Date.now(),
        messageIds: [],
        keyPoints: [],
        userPreferences: []
      }
    }

    // 构造时间线
    const timeRange = this.buildTimeRange(messages)

    // 构造对话文本
    const conversationText = messages
      .map((m) => `[${new Date(m.timestamp).toISOString()}] ${m.role}: ${m.content}`)
      .join('\n')

    // 构造摘要 prompt
    const prompt = this.buildPrompt(conversationText, timeRange, existingSummary)

    // 调用 AI 生成摘要
    const aiResponse = await this.ai.chat([
      {
        role: 'system',
        content: `你是一个记忆摘要助手。你的任务是将对话压缩为结构化摘要。
输出必须严格遵循 JSON 格式，不要输出任何其他内容。`
      },
      { role: 'user', content: prompt }
    ])

    // 解析 AI 响应
    const parsed = this.parseSummaryResponse(aiResponse)

    const result: SummaryResult = {
      summary: parsed.summary,
      timestamp: Date.now(),
      messageIds: messages.map((m) => m.id),
      keyPoints: parsed.keyPoints || [],
      userPreferences: parsed.userPreferences || []
    }

    this.logger.info(`摘要生成完成，涵盖 ${messages.length} 条消息`)
    return result
  }

  /**
   * 增量摘要：在旧摘要基础上追加新消息
   *
   * @param newMessages   新的未摘要消息
   * @param oldSummary    上一次的摘要文本
   * @returns 合并后的新摘要
   */
  async summarizeIncremental(
    newMessages: ShortTermMemory[],
    oldSummary: string
  ): Promise<SummaryResult> {
    if (newMessages.length === 0) {
      return {
        summary: oldSummary,
        timestamp: Date.now(),
        messageIds: [],
        keyPoints: [],
        userPreferences: []
      }
    }

    this.logger.info(`增量摘要：${newMessages.length} 条新消息，基于旧摘要追加`)
    return this.summarize(newMessages, oldSummary)
  }

  /**
   * 批量摘要：当消息过多时分批处理
   *
   * @param messages 全部待摘要消息
   * @param existingSummary 已有摘要
   * @returns 最终合并摘要
   */
  async summarizeBatch(
    messages: ShortTermMemory[],
    existingSummary?: string
  ): Promise<SummaryResult> {
    if (messages.length <= this.config.batchSize) {
      return this.summarize(messages, existingSummary)
    }

    // 分批处理
    let currentSummary = existingSummary || ''
    let allKeyPoints: string[] = []
    let allPreferences: string[] = []
    let allMessageIds: string[] = []

    for (let i = 0; i < messages.length; i += this.config.batchSize) {
      const batch = messages.slice(i, i + this.config.batchSize)
      const result = await this.summarize(batch, currentSummary || undefined)

      currentSummary = result.summary
      allKeyPoints = [...allKeyPoints, ...result.keyPoints]
      allPreferences = [...allPreferences, ...result.userPreferences]
      allMessageIds = [...allMessageIds, ...result.messageIds]
    }

    // 去重关键信息和偏好
    allKeyPoints = [...new Set(allKeyPoints)]
    allPreferences = [...new Set(allPreferences)]

    return {
      summary: currentSummary,
      timestamp: Date.now(),
      messageIds: allMessageIds,
      keyPoints: allKeyPoints,
      userPreferences: allPreferences
    }
  }

  /** 构造时间范围描述 */
  private buildTimeRange(messages: ShortTermMemory[]): string {
    const sorted = [...messages].sort((a, b) => a.timestamp - b.timestamp)
    const start = new Date(sorted[0].timestamp).toLocaleString('zh-CN')
    const end = new Date(sorted[sorted.length - 1].timestamp).toLocaleString('zh-CN')
    return `${start} ~ ${end}`
  }

  /** 构造摘要 prompt */
  private buildPrompt(
    conversationText: string,
    timeRange: string,
    existingSummary?: string
  ): string {
    const sections: string[] = []

    sections.push(`## 任务`)
    sections.push(`将以下对话压缩为结构化摘要。`)
    sections.push(``)

    sections.push(`## 时间范围`)
    sections.push(timeRange)
    sections.push(``)

    if (existingSummary) {
      sections.push(`## 已有摘要（请在此基础上增量更新）`)
      sections.push(existingSummary)
      sections.push(``)
    }

    sections.push(`## 对话内容`)
    sections.push(conversationText)
    sections.push(``)

    sections.push(`## 输出格式要求`)
    sections.push(`请严格按照以下 JSON 格式输出：`)
    sections.push(`\`\`\`json`)
    sections.push(`{`)
    sections.push(`  "summary": "结构化摘要文本（包含时间、关键事件、对话主题）",`)
    sections.push(`  "keyPoints": ["关键信息1", "关键信息2", ...],`)
    sections.push(`  "userPreferences": ["用户偏好1", "用户偏好2", ...]`)
    sections.push(`}`)
    sections.push(`\`\`\``)
    sections.push(``)
    sections.push(`摘要要求：`)
    sections.push(`1. 保留关键事实、决策、承诺和待办事项`)
    sections.push(`2. 压缩冗余对话，保留语义`)
    if (this.config.extractPreferences) {
      sections.push(`3. 识别并提取用户的偏好、习惯和个性化需求`)
    }
    sections.push(`4. 使用 ${this.config.language} 输出`)

    return sections.join('\n')
  }

  /** 解析 AI 返回的摘要响应 */
  private parseSummaryResponse(response: string): {
    summary: string
    keyPoints?: string[]
    userPreferences?: string[]
  } {
    try {
      // 尝试提取 JSON 块
      const jsonMatch = response.match(/```(?:json)?\s*([\s\S]*?)```/)
      const jsonStr = jsonMatch ? jsonMatch[1].trim() : response.trim()
      const parsed = JSON.parse(jsonStr)

      return {
        summary: parsed.summary || response,
        keyPoints: Array.isArray(parsed.keyPoints) ? parsed.keyPoints : [],
        userPreferences: Array.isArray(parsed.userPreferences) ? parsed.userPreferences : []
      }
    } catch {
      // JSON 解析失败，将整个响应作为纯文本摘要
      this.logger.warn('摘要 JSON 解析失败，使用原始响应作为摘要')
      return {
        summary: response,
        keyPoints: [],
        userPreferences: []
      }
    }
  }
}
```

### 15.3 代码规范

- **短期记忆**：SQLite 存储，按会话分组，保留最近 50 条；支持分页查询（`getByPage` / `getRecentPage`），避免大量记忆一次性加载
- **长期记忆**：向量检索，嵌入模型可通过 `embeddingModel` 配置（默认 `nomic-embed-text`）
- **过期机制**：长期记忆默认 TTL 90 天，过期自动清理（每小时检查一次），TTL 可通过 `longTermTTLDays` 配置，设为 0 则永不过期 <!-- ✅ Issue#249: 已修复 -->
- **自动摘要**：触发间隔可通过 `autoSummaryInterval` 配置（默认每 10 轮对话），调度周期随间隔自动计算
- **语义搜索**：余弦相似度，返回 Top-K 结果
- **内存限制**：长期记忆最多 512MB
- **加密存储** <!-- ✅ Issue#252: 已修复 -->：
  - 记忆数据在写入 SQLite 前须进行 AES-256-GCM 加密，密钥由主密码派生（PBKDF2，迭代 ≥ 100,000 次）
  - 每条记忆记录的 `content` 字段存储密文，附带随机 IV（12 字节）和认证标签（16 字节）
  - 向量嵌入（`embedding`）本身不加密（为浮点数组，不含原始语义），但对应的原文须加密
  - 加密密钥不得硬编码或写入日志；运行时通过环境变量 `MEMORY_ENCRYPT_KEY` 或系统密钥管理服务注入
  - 读取时自动解密，解密失败（篡改检测）应抛出明确错误并记录安全告警日志
- **访问控制** <!-- ✅ Issue#253: 已修复 -->：
  - 记忆模块的所有读写操作必须经过访问控制层校验，仅允许已授权的模块和用户执行
  - **会话隔离**：短期记忆按 `sessionId` 严格隔离，不同会话之间禁止交叉读取；`MemoryManager` 的 `search`、`getRecent`、`clearSession` 等方法须验证调用方的 `sessionId` 权限
  - **模块级权限**：通过 `module.json` 中的 `permissions` 字段声明所需权限（如 `memory:read`、`memory:write`、`memory:delete`），模块管理器在激活时校验权限白名单
  - **API 鉴权**：对外暴露的工具接口（`memory_save`、`memory_search`、`memory_list`）须在 `handler.execute` 入口处校验调用者身份与权限令牌，拒绝未授权调用并记录审计日志
  - **数据归属**：每条记忆记录关联 `ownerId`（用户/模块标识），查询时自动注入归属过滤条件，防止跨用户数据泄露
  - **操作审计**：所有敏感操作（保存、删除、清空会话）须记录审计日志，包含操作者 ID、操作类型、目标记忆 ID 和时间戳，便于安全审计与追溯
