```tsx
import React, { Suspense, lazy, Component, type ReactNode, type ErrorInfo } from 'react'
import { Sidebar } from './Sidebar'
import { Header } from './Header'
import { useAppStore } from '../../stores/useAppStore'

// ─── 懒加载页面组件（代码分割，按需加载，减少初始包大小） ────────────
const ChatPage = lazy(() => import('../../pages/ChatPage').then(m => ({ default: m.ChatPage })))
const SettingsPage = lazy(() => import('../../pages/SettingsPage').then(m => ({ default: m.SettingsPage })))
const ModulesPage = lazy(() => import('../../pages/ModulesPage').then(m => ({ default: m.ModulesPage })))

// ─── 加载态 fallback（骨架屏风格，避免布局抖动） ────────────────────
function PageLoading() {
  return (
    <div className="page-loading" role="status" aria-label="页面加载中">
      <div className="page-loading-spinner" />
      <span className="page-loading-text">加载中…</span>
    </div>
  )
}

// ─── 懒加载错误边界（chunk 加载失败时提供重试） ─────────────────────
interface LazyErrorBoundaryState {
  hasError: boolean
  error: Error | null
}

class LazyLoadErrorBoundary extends Component<
  { children: ReactNode; fallback?: ReactNode },
  LazyErrorBoundaryState
> {
  state: LazyErrorBoundaryState = { hasError: false, error: null }

  static getDerivedStateFromError(error: Error) {
    return { hasError: true, error }
  }

  componentDidCatch(error: Error, info: ErrorInfo) {
    console.error('[LazyLoadErrorBoundary] chunk 加载失败:', error, info)
  }

  handleRetry = () => {
    this.setState({ hasError: false, error: null })
  }

  render() {
    if (this.state.hasError) {
      return (
        this.props.fallback ?? (
          <div className="page-error" role="alert">
            <p>页面加载失败，请检查网络后重试</p>
            <button className="btn btn-primary" onClick={this.handleRetry}>
              重试
            </button>
          </div>
        )
      )
    }
    return this.props.children
  }
}

// ─── 主布局 ──────────────────────────────────────────────────────────
export function MainLayout() {
  const activePanel = useAppStore((s) => s.activePanel)

  const renderPage = () => {
    switch (activePanel) {
      case 'settings':
        return <SettingsPage />
      case 'modules':
        return <ModulesPage />
      case 'chat':
      default:
        return <ChatPage />
    }
  }

  return (
    <div className="main-layout">
      <Sidebar />
      <div className="main-content">
        <Header />
        {/* <!-- ✅ Issue#195: 已修复 --> */}
        <main className="page-container" aria-label="主内容区">
          <LazyLoadErrorBoundary>
            <Suspense fallback={<PageLoading />}>
              {renderPage()}
            </Suspense>
          </LazyLoadErrorBoundary>
        </main>
      </div>
    </div>
  )
}
```

#### 步骤 4：实现 IPC Hook

**src/renderer/src/hooks/useIpc.ts**：

```typescript
import { useCallback, useEffect, useRef } from 'react'

/** 调用主进程 IPC */
export function useIpcInvoke() {
  return useCallback(async <T = unknown>(channel: string, ...args: unknown[]): Promise<T> => {
    return window.electronAPI.invoke(channel, ...args)
  }, [])
}

/** 监听主进程推送 */
<!-- ✅ Issue#192: 已修复 — cleanup 使用精确的 unsubscribe() 而非 removeAllListeners -->
export function useIpcOn(channel: string, callback: (...args: unknown[]) => void) {
  const callbackRef = useRef(callback)
  callbackRef.current = callback

  // 存储精确的 unsubscribe 引用，用于 cleanup 时只移除当前注册的单个监听器
  // ⚠️ 禁止使用 window.electronAPI.removeAllListeners(channel) — 它会误杀同 channel 的其他监听器
  const unsubscribeRef = useRef<(() => void) | null>(null)

  useEffect(() => {
    // window.electronAPI.on() 内部注册 ipcRenderer.on(channel, wrappedListener)，
    // 返回 () => ipcRenderer.removeListener(channel, wrappedListener)
    // 该 cleanup 只精确移除本次注册的监听器，不影响同 channel 其他监听者
    const unsubscribe = window.electronAPI.on(channel, (...args: unknown[]) => {
      callbackRef.current(...args)
    })
    unsubscribeRef.current = unsubscribe

    return () => {
      unsubscribe()
      unsubscribeRef.current = null
    }
  }, [channel])
}
```

```typescript
import { useEffect, useCallback } from 'react'
import { useAppStore } from '../stores/useAppStore'

// ─── 常量 ───────────────────────────────────────────────────

const STORAGE_KEY = 'app-theme'

/** 主题切换顺序：light → dark → system → light */
const THEME_CYCLE: Array<'light' | 'dark' | 'system'> = ['light', 'dark', 'system']

// ─── 辅助函数 ───────────────────────────────────────────────

/** 从 localStorage 读取已保存的主题偏好，不存在则返回 'system' */
function loadSavedTheme(): 'light' | 'dark' | 'system' {
  try {
    const saved = localStorage.getItem(STORAGE_KEY)
    if (saved === 'light' || saved === 'dark' || saved === 'system') {
      return saved
    }
  } catch {
    // localStorage 不可用时静默降级
  }
  return 'system'
}

/** 将主题偏好持久化到 localStorage */
function persistTheme(theme: 'light' | 'dark' | 'system'): void {
  try {
    localStorage.setItem(STORAGE_KEY, theme)
  } catch {
    // 静默降级
  }
}

/** 检测系统是否偏好深色模式 */
function getSystemPrefersDark(): boolean {
  return window.matchMedia('(prefers-color-scheme: dark)').matches
}

/** 根据主题设置解析实际应应用的 data-theme 值 */
function resolveTheme(theme: 'light' | 'dark' | 'system'): 'light' | 'dark' {
  if (theme === 'system') {
    return getSystemPrefersDark() ? 'dark' : 'light'
  }
  return theme
}

// ─── Hook ───────────────────────────────────────────────────

export function useTheme() {
  const theme = useAppStore((s) => s.theme)
  const storeSetTheme = useAppStore((s) => s.setTheme)

  // ── 初始化：从 localStorage 恢复主题偏好 ──────────────────
  useEffect(() => {
    const saved = loadSavedTheme()
    if (saved !== theme) {
      storeSetTheme(saved)
    }
    // 仅在挂载时执行一次
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [])

  // ── 主题切换：light → dark → system → light ───────────────
  const toggleTheme = useCallback(() => {
    const currentIndex = THEME_CYCLE.indexOf(theme)
    const nextIndex = (currentIndex + 1) % THEME_CYCLE.length
    const next = THEME_CYCLE[nextIndex]
    storeSetTheme(next)
    persistTheme(next)
  }, [theme, storeSetTheme])

  // ── 设置主题并持久化 ─────────────────────────────────────
  const setTheme = useCallback(
    (newTheme: 'light' | 'dark' | 'system') => {
      storeSetTheme(newTheme)
      persistTheme(newTheme)
    },
    [storeSetTheme]
  )

  // ── 将主题应用到 document.documentElement ────────────────
  useEffect(() => {
    const root = document.documentElement
    const resolved = resolveTheme(theme)

    // 设置 data-theme 属性，CSS 变量通过 [data-theme] 选择器自动响应
    root.setAttribute('data-theme', resolved)

    // 如果是 system 模式，监听系统偏好变化
    if (theme === 'system') {
      const mediaQuery = window.matchMedia('(prefers-color-scheme: dark)')
      const handleChange = () => {
        root.setAttribute('data-theme', getSystemPrefersDark() ? 'dark' : 'light')
      }
      mediaQuery.addEventListener('change', handleChange)
      return () => mediaQuery.removeEventListener('change', handleChange)
    }
  }, [theme])

  return { theme, setTheme, toggleTheme }
}
```

```tsx
  const dialogRef = useRef<HTMLDialogElement>(null)

  useEffect(() => {
    const dialog = dialogRef.current
    if (!dialog) return
    if (open && !dialog.open) {
      dialog.showModal()
    } else if (!open && dialog.open) {
      dialog.close()
    }

    dialog.addEventListener('close', handleClose)

        <div className="modal-header">

        </div>
      )}
      <div className="modal-body">{children}</div>
```

```tsx
interface ToastItem {
  /** 唯一标识 */
  id: string
  /** 提示类型 */
  type: 'info' | 'success' | 'warning' | 'error'
  /** 提示内容 */
  message: string
  /** 自动关闭时间（ms），0 表示不自动关闭 */
  duration: number
}

/** Toast 显示选项 */
interface ToastOptions {
  /** 提示类型，默认 'info' */
  type?: ToastItem['type']
  /** 自动关闭时间（ms），默认 3000，设为 0 则不自动关闭 */
  autoClose?: number
}

// 全局 Toast 管理
let toastListeners: ((toasts: ToastItem[]) => void)[] = []
let toastList: ToastItem[] = []

/**
 * 显示一条 Toast 提示
 * @param message 提示内容
 * @param options 配置项（type / autoClose）
 */
export function showToast(message: string, options: ToastOptions = {}) {
  const { type = 'info', autoClose = 3000 } = options
  const id = crypto.randomUUID()
  const item: ToastItem = { id, type, message, duration: autoClose }
  toastList = [...toastList, item]
  toastListeners.forEach((fn) => fn(toastList))

  if (autoClose > 0) {
    setTimeout(() => {
      toastList = toastList.filter((t) => t.id !== id)
      toastListeners.forEach((fn) => fn(toastList))
    }, autoClose)
  }
}

export function ToastContainer() {
  const [toasts, setToasts] = useState<ToastItem[]>(toastList)

  useEffect(() => {
    toastListeners.push(setToasts)
    return () => {
      toastListeners = toastListeners.filter((fn) => fn !== setToasts)
    }
  }, [])

  if (toasts.length === 0) return null

  return (
    <div className="toast-container" role="alert" aria-live="polite">
      {toasts.map((t) => (
        <div key={t.id} className={`toast toast-${t.type}`}>
          {t.message}
        </div>
      ))}
    </div>
  )
}
```

```tsx
import React, { Component, type ErrorInfo, type ReactNode } from 'react'

interface ErrorBoundaryProps {
  children: ReactNode
  fallback?: ReactNode
}

interface ErrorBoundaryState {
  hasError: boolean
  error: Error | null
}

/**
 * 全局错误边界
 * 捕获子组件树中的渲染错误，展示降级 UI 而非白屏崩溃
 */
export class ErrorBoundary extends Component<ErrorBoundaryProps, ErrorBoundaryState> {
  constructor(props: ErrorBoundaryProps) {
    super(props)
    this.state = { hasError: false, error: null }
  }

  static getDerivedStateFromError(error: Error): ErrorBoundaryState {
    return { hasError: true, error }
  }

  componentDidCatch(error: Error, errorInfo: ErrorInfo): void {
    console.error('[ErrorBoundary] 捕获渲染错误:', error, errorInfo)
    // 可在此上报错误到主进程或远程日志服务
    window.electronAPI?.send('error:report', {
      message: error.message,
      stack: error.stack,
      componentStack: errorInfo.componentStack
    })
  }

  handleReset = (): void => {
    this.setState({ hasError: false, error: null })
  }

  render() {
    if (this.state.hasError) {
      if (this.props.fallback) {
        return this.props.fallback
      }

      return (
        <div className="error-boundary" role="alert">
          <div className="error-boundary-icon">⚠️</div>
          <h2 className="error-boundary-title">应用遇到了错误</h2>
          <p className="error-boundary-message">
            {this.state.error?.message ?? '未知错误'}
          </p>
          <div className="error-boundary-actions">
            <button className="btn btn-primary" onClick={this.handleReset}>
              重试
            </button>
            <button
              className="btn btn-secondary"
              onClick={() => window.location.reload()}
            >
              刷新页面
            </button>
          </div>
        </div>
      )
    }

    return this.props.children
  }
}
```

```tsx
import { useModulesStore } from '../stores/useModulesStore'

export function ModulesPage() {
  const { modules, toggleModule, selectModule, selectedModuleId } = useModulesStore()

  useEffect(() => {
    // 从主进程加载模块列表
    window.electronAPI.invoke('modules:list').then((list) => {
      if (Array.isArray(list)) {
        useModulesStore.getState().setModules(list)
      }
    }).catch(console.error)
  }, [])

  return (
    <div className="modules-page">
      <h2>已安装模块</h2>
      <ul className="module-list">
        {modules.map((m) => (
          <li
            key={m.id}
            className={`module-item ${selectedModuleId === m.id ? 'selected' : ''}`}
            onClick={() => selectModule(m.id)}
          >
            <div className="module-header">
              <span className="module-icon">{m.icon ?? '📦'}</span>
              <span className="module-name">{m.name}</span>
              <span className="module-version">v{m.version}</span>
            </div>
            <p className="module-desc">{m.description}</p>
            <label className="module-toggle" onClick={(e) => e.stopPropagation()}>
              <input
                type="checkbox"
                checked={m.enabled}
                onChange={() => toggleModule(m.id)}
              />
              启用
            </label>
          </li>
        ))}
      </ul>
    </div>
  )
}
```

### 12.3 代码规范

- **组件拆分**：按功能模块组织，通用组件放在 `components/common/`
- **样式方案**：CSS 变量 + CSS Modules，避免内联样式
- **主题切换**：通过 `data-theme` 属性切换，CSS 变量自动响应；`useTheme` Hook 自动处理 `prefers-color-scheme`
- **路由**：简单场景用状态切换，复杂场景可用 React Router
- **懒加载**：页面组件使用 `React.lazy` + `Suspense` 按需加载，外层包裹 `LazyLoadErrorBoundary` 处理 chunk 加载失败并提供重试
- **无障碍**：`<main>` 标签添加 `aria-label`，导航使用 `aria-label`，按钮使用语义化 `aria-*` 属性
- **IPC 类型安全**：`window.electronAPI` 通过 `electron.d.ts` 声明全局类型，避免 `any`
- **焦点管理**：Modal 打开时自动聚焦，关闭时恢复焦点到触发元素；路由切换时焦点移到主内容区；提供 Skip Link 和 aria-live 路由通知
- **图标与状态消息**：统一使用 `utils/statusMessages.ts` 中的 `Msg.*`、`STATUS_ICONS.*`、`EMPTY_ICONS.*`，禁止各页面自行定义。详见 [30-图标与状态消息开发规范](./30-图标与状态消息开发规范.md)
