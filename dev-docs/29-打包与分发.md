```nsi
; NSIS 自定义安装逻辑 — 安装前检查、目录选择、快捷方式、首次启动

; ========== 安装前：检查旧版本进程并关闭 ==========
!macro customInstall
  ; 再次确认旧版本进程已退出（防止用户在欢迎页期间重新打开）
  ; 先尝试中文进程名，失败后 fallback 到英文名
```

```typescript
// scripts/notarize.ts

import { notarize } from '@electron/notarize';
import type { AfterPackContext } from 'electron-builder';
import path from 'node:path';

export default async function notarizing(context: AfterPackContext): Promise<void> {
  const { electronPlatformName, appOutDir } = context;

  if (electronPlatformName !== 'darwin') {
    return;
  }

  const appName = context.packager.appInfo.productFilename;
  const appPath = path.join(appOutDir, `${appName}.app`);

  const appleId = process.env.APPLE_ID;
  const appleIdPassword = process.env.APPLE_ID_PASSWORD;
  const teamId = process.env.APPLE_TEAM_ID;

  if (!appleId || !appleIdPassword || !teamId) {
    console.warn('⚠️  跳过公证：缺少 APPLE_ID / APPLE_ID_PASSWORD / APPLE_TEAM_ID');
    return;
  }

  console.log(`🍎 正在公证 ${appName}...`);

  await notarize({
    appBundleId: 'com.smartdesktop.assistant',
    appPath,
    appleId,
    appleIdPassword,
    teamId,
    tool: 'notarytool',
  });

  console.log(`✅ 公证完成: ${appName}`);
}
```

#### 步骤 6：代码签名工具

```typescript
// scripts/sign.ts

import { execSync } from 'node:child_process';
import fs from 'node:fs';
import os from 'node:os';
import path from 'node:path';

interface SignOptions {
  path: string;
  hash?: string;
  isNest?: boolean;
}

// 时间戳服务器列表，按优先级排列，主服务器失败时自动尝试备用服务器
const TIMESTAMP_SERVERS = [
  'http://timestamp.digicert.com',
  'http://tsa.starfieldtech.com',
  'http://timestamp.comodoca.com',
];

/**
 * Windows 代码签名
 * 支持：
 * 1. EV 证书（硬件令牌）
 * 2. 标准证书（PFX 文件）
 * 3. Azure Key Vault 远程签名
 */
export default async function sign(configuration: SignOptions): Promise<void> {
  const { path: filePath } = configuration;

  // 方式1: Azure Key Vault 签名（推荐用于 CI）
  if (process.env.AZURE_KEY_VAULT_URL) {
    return signWithAzure(filePath);
  }

  // 方式2: signtool 签名（本地 PFX）
  if (process.env.CSC_LINK) {
    return signWithPfx(filePath);
  }

  // 方式3: 跳过签名（开发环境）
  console.warn('⚠️  跳过代码签名：未配置签名证书');
}

function signWithTimestampServers(
  buildArgs: (tsaUrl: string) => string[],
): void {
  for (const tsaUrl of TIMESTAMP_SERVERS) {
    try {
      execSync(buildArgs(tsaUrl).join(' '), { stdio: 'inherit' });
      return; // 签名成功，直接返回
    } catch (err) {
      console.warn(`⚠️  时间戳服务器 ${tsaUrl} 签名失败，尝试下一个...`);
    }
  }
  throw new Error(
    `所有时间戳服务器均不可用: ${TIMESTAMP_SERVERS.join(', ')}`,
  );
}

function signWithAzure(filePath: string): void {
  const vaultUrl = process.env.AZURE_KEY_VAULT_URL!;
  const certName = process.env.AZURE_CERT_NAME!;
  const clientId = process.env.AZURE_CLIENT_ID!;
  const clientSecret = process.env.AZURE_CLIENT_SECRET!;
  const tenantId = process.env.AZURE_TENANT_ID!;

  // 使用 AzureSignTool（带时间戳服务器容错）
  signWithTimestampServers((tsaUrl) => [
    'AzureSignTool sign',
    `-kvu "${vaultUrl}"`,
    `-kvc "${certName}"`,
    `-kvi "${clientId}"`,
    `-kvs "${clientSecret}"`,
    `-kvt "${tenantId}"`,
    `-tr "${tsaUrl}"`,
    `-td sha256`,
    `-d "智能桌面助手"`,
    `"${filePath}"`,
  ]);
}

function signWithPfx(filePath: string): void {
  const pfxPath = process.env.CSC_LINK!;
  const password = process.env.CSC_KEY_PASSWORD!;

  // CSC_LINK 为 PFX 证书文件路径（CI 中已通过 base64 -d 解码为文件）
  if (!fs.existsSync(pfxPath)) {
    throw new Error(`PFX 证书文件不存在: ${pfxPath}`);
  }

  try {
    // 使用 signtool 签名（带时间戳服务器容错）
    signWithTimestampServers((tsaUrl) => [
      'signtool sign',
      '/fd SHA256',
      `/tr ${tsaUrl}`,
      '/td SHA256',
      `/f "${pfxPath}"`,
      `/p "${password}"`,
      '/d "智能桌面助手"',
      `"${filePath}"`,
    ]);
  } finally {
    // 清理 CI 中生成的临时证书文件
    if (pfxPath.startsWith(os.tmpdir())) {
      fs.unlinkSync(pfxPath);
    }
  }
}
```

```typescript
// ⚠️ 版本兼容性说明：
// 1. 本模块需要 electron-updater v6.x 或更高版本（package.json 中配置为 "^6.3.0"）
// 2. `UpdateInfo` 类型从 electron-updater v6 开始从主包导出，v5.x 中需要从 'electron-updater/out/Types' 导入
// 3. 如果使用 v5.x，需注意以下差异：
//    - `type UpdateInfo` 导入路径不同，且部分字段名称可能变化（如 `updateInfo` 的结构）
//    - `autoUpdater.checkForUpdates()` 的返回值结构在 v6 中有所调整
//    - 建议统一升级到 v6.x 以获得完整 TypeScript 支持和最新特性

import { autoUpdater, type UpdateInfo } from 'electron-updater';
import { app, BrowserWindow, dialog } from 'electron';
import { EventEmitter } from 'node:events';

export interface UpdaterConfig {
  /** 是否启用自动检查更新 */
  autoCheck: boolean;
  /** 检查更新间隔 (ms)，默认 4 小时 */
  checkInterval: number;
  /** 是否允许自动下载 */
  autoDownload: boolean;
  /** 更新通道 */
  channel: 'latest' | 'beta' | 'alpha';
  /** GitHub token (私有仓库) */
  token?: string;
}

export type UpdateStatus =
  | 'idle'
  | 'checking'
  | 'available'
  | 'not-available'
  | 'downloading'
  | 'downloaded'
  | 'error';

export class AutoUpdater extends EventEmitter {
  private config: UpdaterConfig;
  private checkTimer: ReturnType<typeof setInterval> | null = null;
  private mainWindow: BrowserWindow | null = null;
  private status: UpdateStatus = 'idle';

  constructor(config: UpdaterConfig) {
    super();
    this.config = config;
    this.configureUpdater();
  }

  private configureUpdater(): void {
    autoUpdater.autoDownload = this.config.autoDownload;
    autoUpdater.autoInstallOnAppQuit = true;
    autoUpdater.channel = this.config.channel;

    if (this.config.token) {
      autoUpdater.requestHeaders = {
        Authorization: `token ${this.config.token}`,
      };
    }

    // 事件绑定
    autoUpdater.on('checking-for-update', () => {
      this.setStatus('checking');
    });

    autoUpdater.on('update-available', (info: UpdateInfo) => {
      this.setStatus('available');
      this.emit('update:available', info);
      this.notifyUser('update-available', info);
    });

    autoUpdater.on('update-not-available', (info: UpdateInfo) => {
      this.setStatus('not-available');
      this.emit('update:not-available', info);
    });

    autoUpdater.on('download-progress', (progress) => {
      this.setStatus('downloading');
      this.emit('update:progress', {
        percent: progress.percent,
        bytesPerSecond: progress.bytesPerSecond,
        total: progress.total,
        transferred: progress.transferred,
      });

      // 通知渲染进程更新进度
      this.mainWindow?.webContents.send('update:progress', progress);
    });

    autoUpdater.on('update-downloaded', (info: UpdateInfo) => {
      this.setStatus('downloaded');
      this.emit('update:downloaded', info);
      this.notifyUser('update-downloaded', info);
    });

    autoUpdater.on('error', (error: Error) => {
      this.setStatus('error');
      this.emit('update:error', error);
      console.error('AutoUpdater error:', error);
    });
  }

  /** 设置主窗口引用 */
  setMainWindow(window: BrowserWindow): void {
    this.mainWindow = window;
  }

  /** 启动自动检查 */
  start(): void {
    if (this.config.autoCheck) {
      // 首次检查延迟 30 秒
      setTimeout(() => {
        void this.checkForUpdates();
      }, 30_000);

      // 定期检查
      this.checkTimer = setInterval(() => {
        void this.checkForUpdates();
      }, this.config.checkInterval);
    }
  }

  /** 停止自动检查 */
  stop(): void {
    if (this.checkTimer) {
      clearInterval(this.checkTimer);
      this.checkTimer = null;
    }
  }

  /** 手动检查更新（指数退避重试，最多 3 次） */
  async checkForUpdates(): Promise<void> {
    const maxRetries = 3;
    for (let attempt = 0; attempt <= maxRetries; attempt++) {
      try {
        await autoUpdater.checkForUpdates();
        return; // 成功则立即返回
      } catch (error) {
        const isNetworkError = this.isNetworkError(error);
        console.error(
          `检查更新失败 (第 ${attempt + 1}/${maxRetries + 1} 次)` +
            (isNetworkError ? '，网络连接异常' : '') + ':',
          error,
        );
        if (attempt < maxRetries) {
          const delay = Math.min(1000 * Math.pow(2, attempt), 30_000);
          console.log(`将在 ${delay / 1000}s 后重试...`);
          await this.sleep(delay);
        }
      }
    }
    this.emit('update:error', new Error('检查更新在多次重试后仍然失败'));
  }

  /** 下载更新（指数退避重试，最多 3 次） */
  async downloadUpdate(): Promise<void> {
    const maxRetries = 3;
    for (let attempt = 0; attempt <= maxRetries; attempt++) {
      try {
        await autoUpdater.downloadUpdate();
        return; // 成功则立即返回
      } catch (error) {
        const isNetworkError = this.isNetworkError(error);
        console.error(
          `下载更新失败 (第 ${attempt + 1}/${maxRetries + 1} 次)` +
            (isNetworkError ? '，网络连接异常' : '') + ':',
          error,
        );
        if (attempt < maxRetries) {
          const delay = Math.min(1000 * Math.pow(2, attempt), 30_000);
          console.log(`将在 ${delay / 1000}s 后重试...`);
          await this.sleep(delay);
        }
      }
    }
    this.emit('update:error', new Error('下载更新在多次重试后仍然失败'));
  }

  /** 判断是否为网络相关错误 */
  private isNetworkError(error: unknown): boolean {
    const msg = error instanceof Error ? error.message : String(error);
    const networkPatterns = [
      'ENOTFOUND', 'ECONNREFUSED', 'ECONNRESET', 'ETIMEDOUT',
      'ENETUNREACH', 'EHOSTUNREACH', 'network', 'fetch failed',
      'ERR_NETWORK', 'ERR_INTERNET_DISCONNECTED',
    ];
    return networkPatterns.some((p) => msg.toUpperCase().includes(p.toUpperCase()));
  }

  /** 延迟工具方法 */
  private sleep(ms: number): Promise<void> {
    return new Promise((resolve) => setTimeout(resolve, ms));
  }

  /** 安装更新并重启 */
  quitAndInstall(): void {
    autoUpdater.quitAndInstall(false, true);
  }

  /** 获取当前状态 */
  getStatus(): UpdateStatus {
    return this.status;
  }

  private setStatus(status: UpdateStatus): void {
    this.status = status;
    this.emit('status:changed', status);
  }

  private async notifyUser(type: string, info: UpdateInfo): Promise<void> {
    if (!this.mainWindow) return;

    if (type === 'update-available') {
      const result = await dialog.showMessageBox(this.mainWindow, {
        type: 'info',
        title: '发现新版本',
        message: `新版本 ${info.version} 可用`,
        detail: '是否立即下载更新？',
        buttons: ['立即下载', '稍后再说'],
        defaultId: 0,
        cancelId: 1,
      });

      if (result.response === 0) {
        void this.downloadUpdate();
      }
    }

    if (type === 'update-downloaded') {
      const result = await dialog.showMessageBox(this.mainWindow, {
        type: 'info',
        title: '更新已就绪',
        message: `版本 ${info.version} 已下载完成`,
        detail: '是否立即重启以应用更新？',
        buttons: ['立即重启', '稍后提醒'],
        defaultId: 0,
        cancelId: 1,
      });

      if (result.response === 0) {
        this.quitAndInstall();
      }
    }
  }
}
```

#### 步骤 7.1：增量（差分）更新方案

> `electron-updater` 内置了基于 blockmap 的增量更新支持，无需额外安装依赖。
> 增量更新仅下载新旧版本之间的差异数据，可将更新包体积降低 60%-90%，显著减少用户等待时间和带宽消耗。

##### 原理说明

`electron-updater` 在构建时会自动生成 `.blockmap` 文件（随安装包一同发布），更新时：

1. 客户端下载新版 blockmap（几十 KB）
2. 与本地已安装版本的 blockmap 进行逐块对比
3. 仅下载发生变化的 block 差异包
4. 在本地合并差异，生成完整安装包

##### 配置方式

增量更新在 `electron-updater` 中**默认启用**，无需额外配置。但需确保以下条件：

```typescript
// src/main/updater/AutoUpdater.ts（补充配置）

// 以下配置已内置支持增量更新，无需额外代码：
// 1. autoUpdater.autoDownload = true 时自动使用 differential 下载
// 2. electron-builder 构建时自动生成 .blockmap 文件
// 3. publish 配置中 provider: 'github' 会同时上传 blockmap 到 Release

// 如需强制全量下载（跳过增量），可设置：
// autoUpdater.disableDifferentialDownload = true;

// 验证 blockmap 是否生成：检查构建产物目录中是否存在以下文件：
// - latest.yml（包含 sha512 和文件列表）
// - *.blockmap（如 SmartDesktopAssistant-1.0.0-x64.nsis.7z.blockmap）
```

**electron-builder.yml 中确认 blockmap 生成**（已默认开启，此处显式声明以确保可追溯）：

```yaml
# electron-builder.yml（补充）

# 差分更新：构建时自动生成 blockmap 文件
# electron-builder v23+ 默认开启，此处显式声明确保行为一致
# 如需禁用（减小构建产物体积），设为 false
compression: normal  # 使用 normal 而非 maximum，blockmap 匹配更稳定
```

##### 构建产物中的增量更新文件

构建完成后，`dist/` 目录中应包含以下增量更新相关文件：

```
dist/
├── SmartDesktopAssistant-1.0.0-x64.exe          # 完整安装包
├── SmartDesktopAssistant-1.0.0-x64.exe.blockmap  # blockmap 文件（增量更新依据）
├── SmartDesktopAssistant-1.0.0-x64.nsis.7z       # NSIS 压缩包
├── SmartDesktopAssistant-1.0.0-x64.nsis.7z.blockmap
├── latest.yml                                     # 更新清单（含 sha512、文件大小、下载 URL）
└── ...
```

GitHub Actions CI 中已通过 `upload-artifact` 包含 `*.blockmap` 和 `*.yml` 文件，确保 Release 发布时自动附带。

##### 增量更新的限制条件

```
⚠️ 以下场景会自动降级为全量下载（用户无感知）：

1. 首次安装（无旧版本 blockmap 可对比）
2. 跨大版本升级（如 v1.x → vv2.x，blockmap 差异过大）
3. 安装路径变更或文件被手动修改（本地 blockmap 与实际不一致）
4. blockmap 文件损坏或下载失败
5. 用户手动设置了 disableDifferentialDownload = true
6. NSIS 安装器在某些 Windows 版本上的兼容性问题
   （electron-updater v6 已大幅改善，但 Win7 仍有概率回退全量）
7. Linux AppImage 格式天然支持增量更新；deb/rpm 不支持（始终全量）

💡 性能参考（以 100MB 安装包为例）：
- 全量下载：~100MB
- 增量更新（小版本）：~5-15MB
- 增量更新（大版本）：~30-60MB
```

#### 步骤 8：构建脚本

```typescript
// scripts/build.ts

import { build, type CliOptions } from 'electron-builder';
import { readFileSync } from 'node:fs';
import path from 'node:path';

interface BuildOptions {
  platform?: 'win' | 'mac' | 'linux';
  arch?: 'x64' | 'arm64' | 'universal';
  publish?: 'always' | 'never' | 'onTag';
  config?: string;
}

const PLATFORM_CONFIGS: Record<string, string> = {
  win: 'electron-builder.win.yml',
  mac: 'electron-builder.mac.yml',
  linux: 'electron-builder.linux.yml',
};

async function main(): Promise<void> {
  const options = parseArgs();
  const platform = options.platform ?? detectPlatform();
  const arch = options.arch ?? 'x64';

  console.log(`🔨 开始构建 ${platform}-${arch}...`);
  console.log(`📦 版本: ${getPackageVersion()}`);

  const configPath = options.config ?? PLATFORM_CONFIGS[platform];
  const configFullPath = path.resolve(process.cwd(), configPath);

  const buildOptions: CliOptions = {
    targets: getBuildTargets(platform, arch),
    config: configFullPath,
    publish: options.publish ?? 'never',
    projectDir: process.cwd(),
  };

  try {
    const result = await build(buildOptions);
    console.log('✅ 构建成功！');
    console.log('📄 产物:');
    for (const artifact of result) {
      if (typeof artifact === 'string') {
        console.log(`  - ${artifact}`);
      } else if ('file' in artifact) {
        console.log(`  - ${artifact.file} (${formatBytes(artifact.size)})`);
      }
    }
  } catch (error) {
    console.error('❌ 构建失败:', error);
    process.exit(1);
  }
}

function parseArgs(): BuildOptions {
  const args = process.argv.slice(2);
  const options: BuildOptions = {};

  for (let i = 0; i < args.length; i++) {
    switch (args[i]) {
      case '--platform':
      case '-p':
        options.platform = args[++i] as BuildOptions['platform'];
        break;
      case '--arch':
      case '-a':
        options.arch = args[++i] as BuildOptions['arch'];
        break;
      case '--publish':
        options.publish = args[++i] as BuildOptions['publish'];
        break;
      case '--config':
      case '-c':
        options.config = args[++i];
        break;
    }
  }

  return options;
}

function detectPlatform(): 'win' | 'mac' | 'linux' {
  switch (process.platform) {
    case 'win32':
      return 'win';
    case 'darwin':
      return 'mac';
    default:
      return 'linux';
  }
}

function getBuildTargets(platform: string, arch: string): Map<string, string[]> {
  const targets = new Map<string, string[]>();

  switch (platform) {
    case 'win':
      targets.set('nsis', [arch]);
      break;
    case 'mac':
      targets.set('dmg', [arch]);
      targets.set('zip', [arch]);
      break;
    case 'linux':
      targets.set('AppImage', [arch]);
      targets.set('deb', [arch]);
      targets.set('rpm', [arch]);
      break;
  }

  return targets;
}

function getPackageVersion(): string {
  const pkg = JSON.parse(readFileSync(path.join(__dirname, '../package.json'), 'utf-8'));
  return pkg.version;
}

function formatBytes(bytes: number): string {
  if (bytes < 1024) return `${bytes} B`;
  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
  return `${(bytes / 1024 / 1024).toFixed(1)} MB`;
}

main();
```

#### 步骤 9：GitHub Actions CI/CD

> **设计说明**：`build.yml` 和 `release.yml` 合并为单一工作流文件 `build.yml`，解决原有两个独立 workflow 在 `tags: ['v*']` 触发条件重叠导致的 CI 时序问题。`release` job 通过 `needs: build` 确保所有平台构建完成后再执行，`publish` job 通过 `needs: release` 确保 Release 创建完成后再发布更新。所有构建产物通过 `actions/download-artifact` 在 workflow 内部安全传递，不再依赖跨 workflow 的 artifact 访问。

```yaml
# .github/workflows/build.yml
# 构建 + 发布统一工作流
# - PR / push main → 仅构建验证（不触发 release/publish）
# - push tag v* → 构建 → 创建 Release → 发布更新

name: Build & Release

on:
  push:
    branches: [main]
    tags: ['v*']
  pull_request:
    branches: [main]

permissions:
  contents: write

jobs:
  build:
    name: Build (${{ matrix.os }})
    runs-on: ${{ matrix.os }}
    strategy:
      fail-fast: false
      matrix:
        include:
          - os: ubuntu-latest
            platform: linux
            arch: x64
          - os: macos-latest
            platform: mac
            arch: x64
          - os: macos-latest
            platform: mac
            arch: arm64
          - os: windows-latest
            platform: win
            arch: x64

    steps:
      - uses: actions/checkout@v4

      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: 'npm'

      - name: Install dependencies
        run: npm ci

      - name: Build renderer
        run: npm run build:renderer

      - name: Build main
        run: npm run build:main

      - name: Run tests
        run: npm run test:unit
        if: github.event_name == 'pull_request'

      # Windows 签名
      - name: Setup Windows signing
        if: matrix.platform == 'win' && startsWith(github.ref, 'refs/tags/')
        shell: bash
        run: |
          echo "${{ secrets.WIN_CSC_LINK }}" | base64 -d > cert.pfx
          echo "CSC_LINK=cert.pfx" >> $GITHUB_ENV
          echo "CSC_KEY_PASSWORD=${{ secrets.WIN_CSC_KEY_PASSWORD }}" >> $GITHUB_ENV

      # macOS 签名
      - name: Setup macOS signing
        if: matrix.platform == 'mac' && startsWith(github.ref, 'refs/tags/')
        run: |
          KEYCHAIN_PATH=$RUNNER_TEMP/build.keychain
          KEYCHAIN_PASSWORD=${{ secrets.MAC_KEYCHAIN_PASSWORD }}

          security create-keychain -p "$KEYCHAIN_PASSWORD" "$KEYCHAIN_PATH"
          security set-keychain-settings -lut 21600 "$KEYCHAIN_PATH"
          security unlock-keychain -p "$KEYCHAIN_PASSWORD" "$KEYCHAIN_PATH"

          echo "${{ secrets.MAC_CERT_P12 }}" | base64 -d > $RUNNER_TEMP/cert.p12
          security import $RUNNER_TEMP/cert.p12 -k "$KEYCHAIN_PATH" \
            -P "${{ secrets.MAC_CERT_PASSWORD }}" -T /usr/bin/codesign
          security set-key-partition-list -S apple-tool:,apple: \
            -k "$KEYCHAIN_PASSWORD" "$KEYCHAIN_PATH"

          echo "CSC_NAME=${{ secrets.MAC_CSC_NAME }}" >> $GITHUB_ENV
          echo "KEYCHAIN_PATH=$KEYCHAIN_PATH" >> $GITHUB_ENV

      - name: Build (${{ matrix.platform }}-${{ matrix.arch }})
        run: npx tsx scripts/build.ts --platform ${{ matrix.platform }} --arch ${{ matrix.arch }}
        env:
          GH_TOKEN: ${{ secrets.GITHUB_TOKEN }}
          APPLE_ID: ${{ secrets.APPLE_ID }}
          APPLE_ID_PASSWORD: ${{ secrets.APPLE_ID_PASSWORD }}
          APPLE_TEAM_ID: ${{ secrets.APPLE_TEAM_ID }}

      - name: Upload artifacts
        uses: actions/upload-artifact@v4
        with:
          name: build-${{ matrix.platform }}-${{ matrix.arch }}
          path: |
            dist/*.exe
            dist/*.dmg
            dist/*.zip
            dist/*.AppImage
            dist/*.deb
            dist/*.rpm
            dist/*.yml
            dist/*.blockmap
          retention-days: 7

  # ── 以下 job 仅在 tag 推送时触发 ──────────────────────────────

  release:
    name: Create Release
    needs: build
    if: startsWith(github.ref, 'refs/tags/v')
    runs-on: ubuntu-latest
    outputs:
      version: ${{ steps.version.outputs.version }}

    steps:
      - uses: actions/checkout@v4

      - name: Extract version
        id: version
        run: echo "version=${GITHUB_REF_NAME#v}" >> $GITHUB_OUTPUT

      # 下载所有平台的构建产物（此时 build job 已全部完成）
      - name: Download all build artifacts
        uses: actions/download-artifact@v4
        with:
          path: artifacts/
          pattern: build-*

      # 将嵌套目录中的产物平铺到 release-files/ 目录
      - name: Flatten artifacts for release
        run: |
          mkdir -p release-files
          find artifacts/ -type f \( \
            -name '*.exe' -o -name '*.dmg' -o -name '*.zip' -o \
            -name '*.AppImage' -o -name '*.deb' -o -name '*.rpm' -o \
            -name '*.yml' -o -name '*.blockmap' \
          \) -exec cp {} release-files/ \;
          echo "=== Release files ==="
          ls -lh release-files/

      - name: Create GitHub Release
        uses: softprops/action-gh-release@v2
        with:
          name: v${{ steps.version.outputs.version }}
          draft: false
          prerelease: ${{ contains(steps.version.outputs.version, '-') }}
          generate_release_notes: true
          files: release-files/*
        env:
          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}

  publish:
    name: Publish Update
    needs: release
    if: startsWith(github.ref, 'refs/tags/v')
    runs-on: ubuntu-latest

    steps:
      - uses: actions/checkout@v4

      - name: Download all build artifacts
        uses: actions/download-artifact@v4
        with:
          path: artifacts/
          pattern: build-*

      - name: Flatten artifacts
        run: |
          mkdir -p dist
          find artifacts/ -type f \( \
            -name '*.exe' -o -name '*.dmg' -o -name '*.zip' -o \
            -name '*.AppImage' -o -name '*.deb' -o -name '*.rpm' -o \
            -name '*.yml' -o -name '*.blockmap' \
          \) -exec cp {} dist/ \;

      - name: Generate checksums
        run: npx tsx scripts/checksum.ts
        env:
          GH_TOKEN: ${{ secrets.GITHUB_TOKEN }}
          VERSION: ${{ needs.release.outputs.version }}
```

#### 步骤 10：校验和生成脚本

```typescript
// scripts/checksum.ts

import { execSync } from 'node:child_process';
import crypto from 'node:crypto';
import fs from 'node:fs';
import path from 'node:path';

interface ReleaseInfo {
  version: string;
  tag: string;
  notes: string;
  artifacts: string[];
}

async function main(): Promise<void> {
  const version = process.env.VERSION;
  if (!version) {
    console.error('❌ VERSION 环境变量未设置');
    process.exit(1);
  }

  const releaseInfo = collectReleaseInfo(version);

  console.log(`🚀 发布版本: ${releaseInfo.version}`);
  console.log(`📎 Tag: ${releaseInfo.tag}`);
  console.log(`📦 产物数量: ${releaseInfo.artifacts.length}`);

  // 验证产物完整性
  validateArtifacts(releaseInfo);

  // 生成校验和
  const checksums = generateChecksums(releaseInfo.artifacts);

  // 写入校验和文件
  const checksumPath = path.join('dist', 'checksums.txt');
  fs.writeFileSync(checksumPath, checksums);

  console.log('✅ 发布准备完成');
  console.log(`📄 校验和: ${checksumPath}`);
}

function collectReleaseInfo(version: string): ReleaseInfo {
  const distDir = path.resolve('dist');
  const artifacts: string[] = [];

  if (fs.existsSync(distDir)) {
    const files = fs.readdirSync(distDir);
    for (const file of files) {
      if (/\.(exe|dmg|zip|AppImage|deb|rpm)$/.test(file)) {
        artifacts.push(path.join(distDir, file));
      }
    }
  }

  return {
    version,
    tag: `v${version}`,
    notes: extractChangelog(version),
    artifacts,
  };
}

function validateArtifacts(info: ReleaseInfo): void {
  if (info.artifacts.length === 0) {
    console.error('❌ 没有找到构建产物');
    process.exit(1);
  }

  // 验证每个产物大小 > 1MB（排除空文件）
  for (const artifact of info.artifacts) {
    const stat = fs.statSync(artifact);
    if (stat.size < 1024 * 1024) {
      console.warn(`⚠️  产物可能异常（文件过小）: ${path.basename(artifact)} (${stat.size} bytes)`);
    }
  }
}

function generateChecksums(artifacts: string[]): string {
  const lines: string[] = [];

  for (const artifact of artifacts) {
    const content = fs.readFileSync(artifact);
    const hash = crypto.createHash('sha256').update(content).digest('hex');
    lines.push(`${hash}  ${path.basename(artifact)}`);
  }

  return lines.join('\n') + '\n';
}

function extractChangelog(version: string): string {
  const changelogPath = 'CHANGELOG.md';
  if (!fs.existsSync(changelogPath)) {
    return `Release v${version}`;
  }

  const content = fs.readFileSync(changelogPath, 'utf-8');
  const versionRegex = new RegExp(`## \\[${version}\\][\\s\\S]*?(?=## \\[|$)`);
  const match = content.match(versionRegex);

  return match ? match[0].trim() : `Release v${version}`;
}

main().catch((error) => {
  console.error('发布失败:', error);
  process.exit(1);
});
```

### 29.3 代码规范

1. **构建配置分离**：主配置 `electron-builder.yml` + 平台专用配置 `electron-builder.{platform}.yml`，通过 `--config` 参数指定
2. **代码签名安全**：签名证书通过 CI 环境变量注入，禁止将证书文件提交到仓库；EV 证书使用 Azure Key Vault 远程签名
3. **macOS 公证**：所有 macOS 发布版本必须经过 Apple 公证，使用 `notarytool` 工具，`afterSign` 钩子自动触发。公证脚本 `notarize.ts` 使用 TypeScript 编写，通过 `npx tsx` 在运行时即时编译执行（`afterSign: npx tsx scripts/notarize.ts`），需确保 `tsx` 已在 `devDependencies` 中声明
4. **自动更新策略**：默认自动检查（4小时间隔），下载完成后弹窗提示用户确认重启，支持 `latest`/`beta`/`alpha` 通道
5. **NSIS 安装器**：支持中英文双语，自定义安装路径，安装前检测并关闭运行中的实例，创建桌面和开始菜单快捷方式
6. **ASAR 打包**：应用代码打包为 ASAR，原生模块（better-sqlite3）使用 `asarUnpack` 排除
7. **产物命名**：`${productName}-${version}-${arch}.${ext}` 格式，便于用户识别版本和架构
8. **校验和验证**：发布时自动生成 SHA-256 校验和文件 `checksums.txt`，用户可验证下载完整性
9. **CI/CD 流程**：PR 仅构建验证，tag 推送触发自动发布；构建矩阵覆盖 Win/Mac/Linux + x64/arm64
10. **版本管理**：遵循语义化版本，预发布版本（含 `-`）自动标记为 prerelease，CHANGELOG.md 自动提取发布说明
11. **资源管理**：大模型文件（.gguf）默认不打包进安装包，通过 `extraResources` 配置或首次启动时下载
12. **构建产物清理**：CI 构建产物保留 7 天，正式发布产物永久保留于 GitHub Releases

### 29.4 版本回滚与紧急修复策略

#### 29.4.1 GitHub Release 旧版本保留

- 所有正式发布版本（`vX.Y.Z` tag）的构建产物**永久保留**在 GitHub Releases 中，不做删除
- 每个 Release 包含完整的安装包（`.exe`/`.dmg`/`.AppImage`/`.deb`/`.rpm`）和 `checksums.txt` 校验和文件
- Release 页面按版本号倒序排列，用户可手动下载任意历史版本
- GitHub API 可查询所有版本：`GET /repos/{owner}/{repo}/releases`，用于自动化回滚脚本

#### 29.4.2 回滚版本发布流程

当确认当前版本存在严重问题需要回滚时，执行以下流程：

1. **评估影响**：确认受影响的平台（Windows/macOS/Linux）和严重程度
2. **选择回滚版本**：确定回滚目标版本（通常是上一个稳定版本）
3. **发布回滚 Release**：
   - 在 GitHub Releases 中找到目标稳定版本
   - 编辑该 Release，添加醒目标签说明（如 `⚠️ RECOMMENDED - Rollback target for vX.Y.Z issues`）
   - 更新 Release Notes 说明回滚原因和已知问题
4. **暂停自动更新推送**：
   - 通过修改 `latest.yml` / `latest-mac.yml` / `latest-linux.yml` 指向回滚目标版本
   - 或在 CI 中临时禁用 `publish` job，阻止新版本通过 `electron-updater` 推送
5. **通知用户**：
   - 在 Release Notes、项目 README、社区公告中说明回滚原因
   - 提供手动下载旧版本的直接链接

#### 29.4.3 `autoUpdater` 的 `allowDowngrade` 配置

`electron-updater` 默认不允许降级更新（即从高版本更新到低版本）。回滚场景需要显式启用：

```typescript
// src/main/updater.ts
import { autoUpdater } from 'electron-updater';

// 允许降级更新（回滚场景启用）
autoUpdater.allowDowngrade = true;

// 回滚完成后建议恢复为 false，防止后续意外降级
// autoUpdater.allowDowngrade = false;
```

**使用场景与注意事项**：

| 场景 | `allowDowngrade` 值 | 说明 |
|------|---------------------|------|
| 正常运行 | `false`（默认） | 防止意外降级 |
| 回滚发布 | `true` | 允许用户自动回退到旧版本 |
| 回滚完成 | 恢复 `false` | 确保后续更新正常升级 |

**推荐实现方式**：通过远程配置或环境变量控制，而非硬编码：

```typescript
// 通过远程配置动态控制
const config = await fetchRemoteConfig();
autoUpdater.allowDowngrade = config.allowDowngrade ?? false;
```

#### 29.4.4 紧急修复版本（Hotfix）快速发布流程

当发现严重 Bug 需要在最短时间内发布修复版本时：

1. **创建 Hotfix 分支**：
   ```bash
   git checkout -b hotfix/v1.0.1 v1.0.0  # 从问题版本创建
   ```

2. **修复代码并提交**：
   ```bash
   git commit -m "fix: 描述修复内容"
   ```

3. **版本号递增**：
   ```bash
   npm version patch  # 1.0.0 → 1.0.1
   ```

4. **推送并触发自动发布**：
   ```bash
   git push origin hotfix/v1.0.1
   git tag v1.0.1
   git push origin v1.0.1  # 触发 CI/CD 自动构建和发布
   ```

5. **同步到主分支**：
   ```bash
   git checkout main
   git merge hotfix/v1.0.1
   git push origin main
   ```

**Hotfix 流程优化**：

- **CI 加速**：Hotfix 分支可跳过部分非关键测试（需在 CI 配置中预先定义 `hotfix` 快速通道）
- **签名复用**：复用主分支的签名证书和公证流程，不额外等待
- **增量更新**：`electron-updater` 的 `blockmap` 支持差量更新，Hotfix 版本仅下载变更部分
- **回滚验证**：Hotfix 发布后，立即在各平台验证自动更新是否正确推送，确认修复生效
