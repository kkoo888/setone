#### 步骤 2：实现定时调度器 (`CronScheduler.ts`)

```typescript
// modules/proactive/CronScheduler.ts

import * as cron from 'node-cron';
import type { CronJobConfig } from './types';

/**
 * 定时调度器
 * 基于 node-cron 封装，管理所有定时任务
 */
export class CronScheduler {
  private readonly jobs: Map<string, cron.ScheduledTask> = new Map();
  private readonly configs: Map<string, CronJobConfig> = new Map();

  /**
   * 注册定时任务
   */
  register(config: CronJobConfig, callback: () => void | Promise<void>): void {
    // 如果已存在同名任务，先注销
    this.unregister(config.id);

    if (!cron.validate(config.expression)) {
      throw new Error(`无效的 Cron 表达式: ${config.expression}`);
    }

    const task = cron.schedule(config.expression, async () => {
      if (!config.enabled) return;
      try {
        await callback();
      } catch (error) {
        console.error(
          `[CronScheduler] 任务 "${config.name}" 执行失败:`,
          error
        );
      }
    });

    this.jobs.set(config.id, task);
    this.configs.set(config.id, { ...config });
  }

  /**
   * 注销定时任务
   */
  unregister(id: string): boolean {
    const task = this.jobs.get(id);
    if (task) {
      task.stop();
      this.jobs.delete(id);
      this.configs.delete(id);
      return true;
    }
    return false;
  }

  /**
   * 启用/禁用任务
   * 禁用时直接停止 cron 任务，避免回调仍被触发造成不必要开销
   */
  setEnabled(id: string, enabled: boolean): boolean {
    const config = this.configs.get(id);
    const task = this.jobs.get(id);
    if (config && task) {
      config.enabled = enabled;
      if (enabled) {
        task.start();
      } else {
        task.stop();
      }
      return true;
    }
    return false;
  }

  /**
   * 获取任务配置
   */
  getConfig(id: string): CronJobConfig | undefined {
    const config = this.configs.get(id);
    return config ? { ...config } : undefined;
  }

  /**
   * 获取所有任务
   */
  getAllConfigs(): CronJobConfig[] {
    return Array.from(this.configs.values()).map((c) => ({ ...c }));
  }

  /**
   * 停止所有任务
   */
  stopAll(): void {
    for (const task of this.jobs.values()) {
      task.stop();
    }
  }

  /**
   * 销毁调度器
   */
  destroy(): void {
    this.stopAll();
    this.jobs.clear();
    this.configs.clear();
  }
}
```

```typescript
// modules/proactive/WeatherService.ts

import type { ModuleContext } from '../types';
import type {
  WeatherData,
  WeatherQueryParams,
  WeatherForecast,
} from './types';

/**
 * 天气服务
 * 使用 wttr.in API 查询天气信息
 */
export class WeatherService {
  private readonly context: ModuleContext;
  private readonly baseUrl = 'https://wttr.in';
  private cache: Map<string, { data: WeatherData; expireAt: number }> = new Map();
  private readonly cacheTtlMs = 10 * 60 * 1000; // 10 分钟缓存

  constructor(context: ModuleContext) {
    this.context = context;
  }

  /**
   * 查询当前天气
   */
  async getCurrentWeather(params: WeatherQueryParams): Promise<WeatherData> {
    const cacheKey = `current:${params.city}`;
    const cached = this.getFromCache(cacheKey);
    if (cached) return cached;

    try {
      const url = `${this.baseUrl}/${encodeURIComponent(params.city)}?format=j1`;
      const response = await fetch(url, {
        headers: { 'User-Agent': 'SmartDesktopAssistant/1.0' },
        signal: AbortSignal.timeout(10000),
      });

      if (!response.ok) {
        throw new Error(`天气查询失败: HTTP ${response.status}`);
      }

      const data = (await response.json()) as WttrResponse;
      const weather = this.parseWeatherData(data, params.city);

      this.setCache(cacheKey, weather);

      this.context.logger.info(
        `[WeatherService] ${params.city}: ${weather.temperature}°C, ${weather.description}`
      );

      return weather;
    } catch (error) {
      this.context.logger.error(
        `[WeatherService] 查询天气失败: ${error instanceof Error ? error.message : String(error)}`
      );
      throw error;
    }
  }

  /**
   * 获取天气预报
   */
  async getForecast(
    city: string,
    days = 3
  ): Promise<WeatherForecast[]> {
    const cacheKey = `forecast:${city}:${days}`;
    const cached = this.forecastCache.get(cacheKey);
    if (cached && cached.expireAt > Date.now()) {
      return cached.data;
    }

    try {
      const url = `${this.baseUrl}/${encodeURIComponent(city)}?format=j1`;
      const response = await fetch(url, {
        headers: { 'User-Agent': 'SmartDesktopAssistant/1.0' },
        signal: AbortSignal.timeout(10000),
      });

      if (!response.ok) {
        throw new Error(`天气预报查询失败: HTTP ${response.status}`);
      }

      const data = (await response.json()) as WttrResponse;
      const forecast = this.parseForecast(data, days);

      this.forecastCache.set(cacheKey, {
        data: forecast,
        expireAt: Date.now() + this.cacheTtlMs,
      });

      return forecast;
    } catch (error) {
      this.context.logger.error(
        `[WeatherService] 查询天气预报失败: ${error instanceof Error ? error.message : String(error)}`
      );
      throw error;
    }
  }

  /**
   * 生成天气播报文本
   */
  generateBroadcast(weather: WeatherData, forecast?: WeatherForecast[]): string {
    const lines: string[] = [];

    lines.push(`📍 ${weather.city} 天气播报`);
    lines.push(`🌡️ 温度: ${weather.temperature}°C（体感 ${weather.feelsLike}°C）`);
    lines.push(`☁️ 天气: ${weather.description}`);
    lines.push(`💧 湿度: ${weather.humidity}%`);
    lines.push(`💨 风速: ${weather.windSpeed} km/h ${weather.windDirection}`);

    // 温度建议
    if (weather.temperature < 5) {
      lines.push('🧥 天气寒冷，注意保暖！');
    } else if (weather.temperature > 35) {
      lines.push('🥵 高温天气，注意防暑降温！');
    }

    // 降水提醒
    if (weather.description.includes('雨') || weather.description.includes('rain')) {
      lines.push('🌧️ 有雨，出门记得带伞！');
    }

    // 预报信息
    if (forecast && forecast.length > 0) {
      lines.push('');
      lines.push('📅 未来天气:');
      for (const day of forecast) {
        lines.push(
          `  ${day.date}: ${day.minTemp}~${day.maxTemp}°C, ${day.description}` +
            (day.chanceOfRain > 30 ? ` (🌧️ ${day.chanceOfRain}%)` : '')
        );
      }
    }

    return lines.join('\n');
  }

  private forecastCache: Map<
    string,
    { data: WeatherForecast[]; expireAt: number }
  > = new Map();

  /**
   * 解析 wttr.in 响应为天气数据
   */
  private parseWeatherData(data: WttrResponse, city: string): WeatherData {
    const current = data.current_condition?.[0];
    if (!current) {
      throw new Error('天气数据格式异常');
    }

    return {
      city,
      temperature: parseInt(current.temp_C, 10),
      feelsLike: parseInt(current.FeelsLikeC, 10),
      description: current.weatherDesc?.[0]?.value ?? '未知',
      icon: current.weatherCode ?? '',
      humidity: parseInt(current.humidity, 10),
      windSpeed: parseInt(current.windspeedKmph, 10),
      windDirection: current.winddir16Point ?? '',
      visibility: parseInt(current.visibility, 10),
      updatedAt: Date.now(),
    };
  }

  /**
   * 解析天气预报
   */
  private parseForecast(data: WttrResponse, days: number): WeatherForecast[] {
    const forecasts: WeatherForecast[] = [];
    const weatherList = data.weather ?? [];

    for (let i = 0; i < Math.min(days, weatherList.length); i++) {
      const day = weatherList[i];
      if (!day) continue;

      forecasts.push({
        date: day.date,
        maxTemp: parseInt(day.maxtempC, 10),
        minTemp: parseInt(day.mintempC, 10),
        description: day.hourly?.[4]?.weatherDesc?.[0]?.value ?? '未知',
        chanceOfRain: parseInt(day.hourly?.[4]?.chanceofrain ?? '0', 10),
      });
    }

    return forecasts;
  }

  private getFromCache(key: string): WeatherData | null {
    const cached = this.cache.get(key);
    if (cached && cached.expireAt > Date.now()) {
      return cached.data;
    }
    this.cache.delete(key);
    return null;
  }

  private setCache(key: string, data: WeatherData): void {
    this.cache.set(key, { data, expireAt: Date.now() + this.cacheTtlMs });
  }
}

/**
 * wttr.in API 响应类型
 */
interface WttrResponse {
  current_condition?: Array<{
    temp_C: string;
    FeelsLikeC: string;
    weatherDesc?: Array<{ value: string }>;
    weatherCode?: string;
    humidity: string;
    windspeedKmph: string;
    winddir16Point?: string;
    visibility: string;
  }>;
  weather?: Array<{
    date: string;
    maxtempC: string;
    mintempC: string;
    hourly?: Array<{
      weatherDesc?: Array<{ value: string }>;
      chanceofrain?: string;
    }>;
  }>;
}
```

#### 步骤 4：实现提醒服务 (`ReminderService.ts`)

```typescript
// modules/proactive/ReminderService.ts

import { randomUUID } from 'node:crypto';
import type { ModuleContext } from '../types';
import type {
  Reminder,
  ReminderSetParams,
  ReminderTypeValue,
} from './types';
import { ReminderType, ReminderPriority } from './types';
import { CronScheduler } from './CronScheduler';

/**
 * 提醒服务
 * 管理各种类型的定时提醒
 */
export class ReminderService {
  private readonly context: ModuleContext;
  private readonly scheduler: CronScheduler;
  private readonly reminders: Map<string, Reminder> = new Map();

  constructor(context: ModuleContext, scheduler: CronScheduler) {
    this.context = context;
    this.scheduler = scheduler;
  }

  /**
   * 创建提醒
   */
  async createReminder(params: ReminderSetParams): Promise<Reminder> {
    const id = randomUUID();
    const type = params.type ?? ReminderType.ONCE;
    const priority = params.priority ?? ReminderPriority.MEDIUM;

    const reminder: Reminder = {
      id,
      title: params.title,
      message: params.message,
      type,
      priority,
      cronExpression: params.cronExpression,
      intervalMs: params.intervalMs,
      targetTime: this.resolveTargetTime(params.targetTime),
      enabled: true,
      createdAt: Date.now(),
    };

    // 注册调度任务
    this.scheduleReminder(reminder);
    this.reminders.set(id, reminder);

    this.context.logger.info(
      `[ReminderService] 提醒已创建: "${reminder.title}" (${type})`
    );

    return reminder;
  }

  /**
   * 删除提醒
   */
  async deleteReminder(id: string): Promise<boolean> {
    const reminder = this.reminders.get(id);
    if (!reminder) return false;

    this.scheduler.unregister(`reminder_${id}`);

    // 清理一次性提醒的 timer，防止删除后仍触发
    if (this.onceTimers.has(id)) {
      clearTimeout(this.onceTimers.get(id));
      this.onceTimers.delete(id);
    }

    this.reminders.delete(id);

    this.context.logger.info(
      `[ReminderService] 提醒已删除: "${reminder.title}"`
    );
    return true;
  }

  /**
   * 启用/禁用提醒
   */
  toggleReminder(id: string, enabled: boolean): boolean {
    const reminder = this.reminders.get(id);
    if (!reminder) return false;

    reminder.enabled = enabled;
    this.scheduler.setEnabled(`reminder_${id}`, enabled);
    return true;
  }

  /**
   * 获取所有提醒
   */
  getAllReminders(): Reminder[] {
    return Array.from(this.reminders.values());
  }

  /**
   * 获取提醒
   */
  getReminder(id: string): Reminder | undefined {
    return this.reminders.get(id);
  }

  /**
   * 触发提醒（发送通知）
   */
  private async triggerReminder(reminder: Reminder): Promise<void> {
    reminder.lastTriggeredAt = Date.now();

    // 通过通知模块发送提醒
    const notifier = this.context.getModule('notification');
    if (notifier) {
      await notifier.execute('notify', {
        title: `⏰ ${reminder.title}`,
        body: reminder.message,
        priority: reminder.priority,
      });
    }

    this.context.logger.info(
      `[ReminderService] 提醒触发: "${reminder.title}"`
    );

    // 一次性提醒触发后自动禁用
    if (reminder.type === ReminderType.ONCE) {
      reminder.enabled = false;
    }
  }

  /**
   * 调度提醒任务
   */
  private scheduleReminder(reminder: Reminder): void {
    const cronId = `reminder_${reminder.id}`;

    switch (reminder.type) {
      case ReminderType.ONCE: {
        if (!reminder.targetTime) {
          throw new Error('一次性提醒需要指定目标时间');
        }
        const delay = reminder.targetTime - Date.now();
        if (delay <= 0) {
          // 已过期，立即触发
          this.triggerReminder(reminder);
          return;
        }
        // 使用 setTimeout 处理一次性提醒
        const timerId = setTimeout(() => {
          this.triggerReminder(reminder);
        }, delay);
        // 存储 timer ID 以便取消
        this.onceTimers.set(reminder.id, timerId);
        break;
      }

      case ReminderType.DAILY: {
        const expression = reminder.cronExpression ?? '0 9 * * *'; // 默认每天 9:00
        this.scheduler.register(
          { id: cronId, expression, name: reminder.title, enabled: reminder.enabled },
          () => this.triggerReminder(reminder)
        );
        break;
      }

      case ReminderType.WEEKLY: {
        const expression = reminder.cronExpression ?? '0 9 * * 1'; // 默认每周一 9:00
        this.scheduler.register(
          { id: cronId, expression, name: reminder.title, enabled: reminder.enabled },
          () => this.triggerReminder(reminder)
        );
        break;
      }

      case ReminderType.INTERVAL: {
        const intervalMs = reminder.intervalMs ?? 60 * 60 * 1000; // 默认 1 小时
        // 使用 Cron 的秒级精度近似间隔
        const minutes = Math.max(1, Math.round(intervalMs / 60000));
        const expression = `*/${minutes} * * * *`;
        this.scheduler.register(
          { id: cronId, expression, name: reminder.title, enabled: reminder.enabled },
          () => this.triggerReminder(reminder)
        );
        break;
      }
    }
  }

  private onceTimers: Map<string, NodeJS.Timeout> = new Map();

  /**
   * 解析目标时间
   */
  private resolveTargetTime(
    target: number | string | undefined
  ): number | undefined {
    if (target === undefined) return undefined;
    if (typeof target === 'number') return target;

    // 简单的自然语言时间解析
    const now = Date.now();
    const lower = target.toLowerCase();

    if (lower.includes('明天')) {
      const tomorrow = new Date();
      tomorrow.setDate(tomorrow.getDate() + 1);
      const hourMatch = lower.match(/(\d{1,2})\s*[点时:：]\s*(\d{1,2})?/);
      if (hourMatch) {
        tomorrow.setHours(parseInt(hourMatch[1], 10), parseInt(hourMatch[2] ?? '0', 10), 0, 0);
      } else {
        tomorrow.setHours(9, 0, 0, 0);
      }
      return tomorrow.getTime();
    }

    if (lower.includes('后天')) {
      const dayAfter = new Date();
      dayAfter.setDate(dayAfter.getDate() + 2);
      const hourMatch = lower.match(/(\d{1,2})\s*[点时:：]\s*(\d{1,2})?/);
      if (hourMatch) {
        dayAfter.setHours(parseInt(hourMatch[1], 10), parseInt(hourMatch[2] ?? '0', 10), 0, 0);
      } else {
        dayAfter.setHours(9, 0, 0, 0);
      }
      return dayAfter.getTime();
    }

    // 分钟后
    const minMatch = lower.match(/(\d+)\s*分钟/);
    if (minMatch) {
      return now + parseInt(minMatch[1], 10) * 60000;
    }

    // 小时后
    const hourMatch = lower.match(/(\d+)\s*小时/);
    if (hourMatch) {
      return now + parseInt(hourMatch[1], 10) * 3600000;
    }

    // 尝试解析为时间戳
    const parsed = parseInt(target, 10);
    if (!isNaN(parsed) && parsed > 1000000000000) {
      return parsed;
    }

    return undefined;
  }
}
```

```typescript
    if (!config) {
      throw new Error(`未知的健康关怀类型: ${type}`);

    if (updates.enabled !== undefined) {
      config.enabled = updates.enabled;
    }
    if (updates.cronExpression !== undefined) {
      config.cronExpression = updates.cronExpression;
    }
    if (updates.message !== undefined) {
      config.message = updates.message;

    this.scheduler.unregister(`health_${type}`);
    if (config.enabled) {
      this.scheduleHealthCare(config);
    }

    this.context.logger.info(
      `[HealthCareService] ${type} 配置已更新: enabled=${config.enabled}`
    );

    return config;

   * 调度健康关怀任务

      {
        id: `health_${config.type}`,
        expression: config.cronExpression,
        name: `健康关怀: ${config.type}`,
        enabled: config.enabled,
      },
      () => this.triggerHealthCare(config)
    );
  }

   * 触发健康关怀通知
   */
  private async triggerHealthCare(config: HealthCareConfig): Promise<void> {
    // 检查是否在工作时间内
    if (!this.isInWorkHours(config.workHours)) {
      return;
    }

    const notifier = this.context.getModule('notification');
    if (notifier) {
      await notifier.execute('notify', {
        title: '💚 健康关怀',
        body: config.message,
        priority: 'low',
      });
    }

    this.context.logger.info(
      `[HealthCareService] 健康关怀触发: ${config.type}`
    );
  }

  /**
   * 检查当前是否在工作时间
   */
  private isInWorkHours(workHours: { start: string; end: string }): boolean {
    const now = new Date();
    const currentTime = now.getHours() * 100 + now.getMinutes();

    const [startH, startM] = workHours.start.split(':').map(Number);
    const [endH, endM] = workHours.end.split(':').map(Number);

    const startTime = startH * 100 + startM;
    const endTime = endH * 100 + endM;

    // 跨天场景：如 start: '22:00', end: '02:00'
    if (startTime > endTime) {
      return currentTime >= startTime || currentTime <= endTime;
    }
    return currentTime >= startTime && currentTime <= endTime;
  }
}
```

```typescript
import type { ModuleContext } from '../types';
import type { EmotionResult, EmotionTypeValue } from './types';
import { EmotionType } from './types';

/**
 * 情绪感知器
 * 基于用户对话内容分析情绪状态，提供关怀建议
 */
export class EmotionDetector {
  private readonly context: ModuleContext;
  private recentEmotions: EmotionResult[] = [];
  private readonly maxHistory = 50;

  constructor(context: ModuleContext) {
    this.context = context;
  }

  /**
   * 分析文本情绪
   */
  async analyze(text: string): Promise<EmotionResult> {
    try {
      const result = await this.callLLMForEmotion(text);
      this.addToHistory(result);
      return result;
    } catch (error) {
      this.context.logger.warn(
        `[EmotionDetector] 情绪分析失败，使用规则回退: ${error instanceof Error ? error.message : String(error)}`
      );
      const fallback = this.ruleBasedAnalysis(text);
      this.addToHistory(fallback);
      return fallback;
    }
  }

  /**
   * 获取最近的情绪趋势
   */
  getRecentTrend(count = 5): EmotionResult[] {
    return this.recentEmotions.slice(-count);
  }

  /**
   * 获取主要情绪（最近 N 次中出现最多的）
   */
  getDominantEmotion(count = 10): EmotionTypeValue | null {
    const recent = this.recentEmotions.slice(-count);
    if (recent.length === 0) return null;

    const counts = new Map<EmotionTypeValue, number>();
    for (const r of recent) {
      counts.set(r.emotion, (counts.get(r.emotion) ?? 0) + 1);
    }

    let maxCount = 0;
    let dominant: EmotionTypeValue | null = null;
    for (const [emotion, cnt] of counts) {
      if (cnt > maxCount) {
        maxCount = cnt;
        dominant = emotion;
      }
    }

    return dominant;
  }

  /**
   * 调用 LLM 分析情绪
   */
  private async callLLMForEmotion(text: string): Promise<EmotionResult> {
    const controller = new AbortController();
    const timeoutId = setTimeout(() => controller.abort(), 30000); // 30秒超时保护

    try {
      const response = await this.context.llm.chat({
        model: this.context.config.defaultModel,
        messages: [
          {
            role: 'system',
            content: `你是一个情绪分析专家。分析用户文本中的情绪，输出 JSON：
{
  "emotion": "happy|sad|angry|anxious|tired|neutral|excited|frustrated",
  "confidence": 0.0-1.0,
  "suggestion": "基于情绪的关怀建议，简短温馨"
}
只输出 JSON，不要其他内容。`,
          },
          { role: 'user', content: text },
        ],
        format: 'json',
        stream: false,
        signal: controller.signal,
      });

      const parsed = JSON.parse(response.message.content) as {
        emotion: string;
        confidence: number;
        suggestion: string;
      };

      const validEmotions: EmotionTypeValue[] = Object.values(EmotionType);
      const emotion = validEmotions.includes(parsed.emotion as EmotionTypeValue)
        ? (parsed.emotion as EmotionTypeValue)
        : EmotionType.NEUTRAL;

      return {
        emotion,
        confidence: Math.max(0, Math.min(1, parsed.confidence)),
        suggestion: parsed.suggestion,
        analyzedAt: Date.now(),
      };
    } finally {
      clearTimeout(timeoutId);
    }
  }

  /**
   * 基于规则的回退情绪分析
   */
  private ruleBasedAnalysis(text: string): EmotionResult {
    const lower = text.toLowerCase();

    const patterns: Array<{
      keywords: string[];
      emotion: EmotionTypeValue;
      suggestion: string;
    }> = [
      {
        keywords: ['开心', '高兴', '太好了', '哈哈', 'happy', 'great', 'awesome', '棒'],

      {
        keywords: ['累', '疲倦', '困', 'tired', 'exhausted', '没精神'],

    if (this.recentEmotions.length > this.maxHistory) {
      this.recentEmotions = this.recentEmotions.slice(-this.maxHistory);
```

### 23.3 代码规范

1. **定时调度**：所有定时任务通过 `CronScheduler` 统一管理，禁止直接使用 `setInterval`
2. **缓存策略**：天气数据缓存 10 分钟，避免频繁请求外部 API
3. **错误容忍**：单个服务失败不影响其他服务运行，使用 try-catch 隔离
4. **超时保护**：所有外部 HTTP 请求使用 `AbortSignal.timeout()` 设置超时
5. **工作时间**：健康关怀仅在工作时间触发，通过 `isInWorkHours()` 检查
6. **情绪分析**：优先使用 LLM 分析，失败时回退到规则引擎
7. **资源清理**：`deactivate()` 中停止所有定时任务和清理资源
8. **自然语言时间**：提醒时间支持自然语言描述（如"明天早上9点"、"30分钟后"）
9. **通知解耦**：通过 `ModuleContext.getModule('notification')` 发送通知，不直接依赖通知模块
10. **历史限制**：情绪历史记录限制为 50 条，防止内存膨胀
