#### 步骤 2：实现重试策略 (`RetryPolicy.ts`)

```typescript
// modules/task/RetryPolicy.ts

import type { RetryConfig } from './types';

/**
 * 默认重试配置
 */
export const DEFAULT_RETRY_CONFIG: RetryConfig = {
  maxRetries: 3,
  baseDelayMs: 1000,
  maxDelayMs: 30000,
  backoffMultiplier: 2,
  retryableErrors: [
    'ECONNRESET',
    'ETIMEDOUT',
    'ECONNREFUSED',
    'RATE_LIMIT',
    'TIMEOUT',
    'NETWORK_ERROR',
  ],
};

/**
 * 重试策略管理器
 */
export class RetryPolicy {
  private readonly config: RetryConfig;

  constructor(config: Partial<RetryConfig> = {}) {
    this.config = { ...DEFAULT_RETRY_CONFIG, ...config };
  }

  /**
   * 判断错误是否可重试
   */
  isRetryable(error: Error): boolean {
    const errorCode = (error as NodeJS.ErrnoException).code ?? '';
    const errorMessage = error.message.toUpperCase();

    return this.config.retryableErrors.some(
      (retryable) =>
        errorCode.toUpperCase().includes(retryable) ||
        errorMessage.includes(retryable)
    );
  }

  /**
   * 计算退避延迟（毫秒），使用指数退避 + 随机抖动
   */
  calculateDelay(retryCount: number): number {
    const exponentialDelay =
      this.config.baseDelayMs *
      Math.pow(this.config.backoffMultiplier, retryCount);
    const cappedDelay = Math.min(exponentialDelay, this.config.maxDelayMs);
    // 添加 0-25% 随机抖动防止雷群效应
    const jitter = cappedDelay * Math.random() * 0.25;
    return Math.floor(cappedDelay + jitter);
  }

  /**
   * 获取最大重试次数
   */
  getMaxRetries(): number {
    return this.config.maxRetries;
  }

  /**
   * 执行带重试的异步操作
   */
  async execute<T>(
    operation: () => Promise<T>,
    onRetry?: (error: Error, attempt: number, delay: number) => void
  ): Promise<T> {
    let lastError: Error | undefined;

    for (let attempt = 0; attempt <= this.config.maxRetries; attempt++) {
      try {
        return await operation();
      } catch (error) {
        lastError = error instanceof Error ? error : new Error(String(error));

        if (attempt >= this.config.maxRetries || !this.isRetryable(lastError)) {
          throw lastError;
        }

        const delay = this.calculateDelay(attempt);
        onRetry?.(lastError, attempt + 1, delay);
        await this.sleep(delay);
      }
    }

    throw lastError ?? new Error('Unknown retry error');
  }

  private sleep(ms: number): Promise<void> {
    return new Promise((resolve) => setTimeout(resolve, ms));
  }
}
```

```typescript
// modules/task/TaskPlanner.ts

import type { ModuleContext } from '../types';
import type { Task, TaskStep, PlanResponse } from './types';
import { TaskStatus, StepStatus } from './types';

/**
 * 任务规划器
 * 使用 LLM 将复杂指令拆解为可执行的步骤序列
 */
export class TaskPlanner {
  private readonly context: ModuleContext;

  constructor(context: ModuleContext) {
    this.context = context;
  }

  /**
   * 为任务生成执行计划
   */
  async plan(task: Task): Promise<Task> {
    task.status = TaskStatus.PLANNING;
    task.updatedAt = Date.now();

    try {
      const availableTools = await this.getAvailableTools();
      const planResponse = await this.callLLMForPlan(task.instruction, availableTools);
      task.steps = this.buildStepsFromPlan(planResponse, task.id);
      task.status = TaskStatus.PENDING;
      task.updatedAt = Date.now();

      this.context.logger.info(
        `[TaskPlanner] 任务 "${task.name}" 已规划 ${task.steps.length} 个步骤`
      );

      return task;
    } catch (error) {
      task.status = TaskStatus.FAILED;
      task.error = error instanceof Error ? error.message : String(error);
      task.updatedAt = Date.now();
      throw error;
    }
  }

  /**
   * 获取可用工具列表
   */
  private async getAvailableTools(): Promise<string[]> {
    const router = this.context.getModule('tools');
    if (!router) {
      this.context.logger.warn('[TaskPlanner] 工具路由模块未加载，使用默认工具列表');
      return [];
    }
    // 调用工具路由模块获取可用工具列表
    const tools = await router.listTools();
    return tools.map(t => t.name);
  }

  /**
   * 调用 LLM 进行任务规划
   */
  private async callLLMForPlan(
    instruction: string,
    availableTools: string[]
  ): Promise<PlanResponse> {
    const systemPrompt = this.buildSystemPrompt(availableTools);

    const response = await this.context.llm.chat({
      model: this.context.config.defaultModel,
      messages: [
        { role: 'system', content: systemPrompt },
        { role: 'user', content: instruction },
      ],
      format: 'json',
      stream: false,
    });

    const parsed = this.parsePlanResponse(response.message.content);
    return parsed;
  }

  /**
   * 构建系统提示词
   */
  private buildSystemPrompt(availableTools: string[]): string {
    const toolList =
      availableTools.length > 0
        ? availableTools.join(', ')
        : 'general_chat, web_search, file_read, file_write, code_execute';

    return `你是一个任务规划专家。你的职责是将用户的复杂指令拆解为清晰、可执行的步骤。

可用工具：${toolList}

请严格按以下 JSON 格式输出：
{
  "summary": "任务的简短摘要",
  "steps": [
    {
      "name": "步骤名称",
      "description": "步骤详细描述",
      "toolName": "使用的工具名称",
      "params": { "key": "value" },
      "dependsOn": ["依赖的步骤序号，从0开始"]
    }
  ]
}

规则：
1. 每个步骤必须是原子操作，不能再拆分
2. 工具名称必须是可用工具列表中的一个
3. dependsOn 表示前置依赖步骤的序号数组，无依赖则为空数组
4. 步骤应当按逻辑顺序排列
5. 如果任务简单不需要工具，steps 可以为空数组
6. params 中的值应为具体参数，不要留占位符`;
  }

  /**
   * 解析 LLM 返回的计划
   */
  private parsePlanResponse(content: string): PlanResponse {
    try {
      const jsonMatch = content.match(/\{[\s\S]*\}/);
      if (!jsonMatch) {
        throw new Error('LLM 返回中未找到 JSON 结构');
      }
      const parsed: unknown = JSON.parse(jsonMatch[0]);

      if (!this.isValidPlanResponse(parsed)) {
        throw new Error('LLM 返回的计划格式不正确');
      }

      return parsed;
    } catch (error) {
      throw new Error(
        `解析计划失败: ${error instanceof Error ? error.message : String(error)}`
      );
    }
  }

  /**
   * 类型守卫：验证计划响应结构
   */
  private isValidPlanResponse(data: unknown): data is PlanResponse {
    if (typeof data !== 'object' || data === null) return false;
    const obj = data as Record<string, unknown>;
    if (typeof obj.summary !== 'string') return false;
    if (!Array.isArray(obj.steps)) return false;

    return obj.steps.every((step: unknown) => {
      if (typeof step !== 'object' || step === null) return false;
      const s = step as Record<string, unknown>;
      return (
        typeof s.name === 'string' &&
        typeof s.description === 'string' &&
        typeof s.toolName === 'string' &&
        typeof s.params === 'object' &&
        s.params !== null &&
        Array.isArray(s.dependsOn)
      );
    });
  }

  /**
   * 将计划转换为任务步骤
   */
  private buildStepsFromPlan(plan: PlanResponse, taskId: string): TaskStep[] {
    return plan.steps.map((step, index) => ({
      id: `${taskId}_step_${index}`,
      index,
      name: step.name,
      description: step.description,
      toolName: step.toolName,
      params: step.params,
      status: StepStatus.PENDING,
      retryCount: 0,
      maxRetries: 3,
      dependsOn: step.dependsOn.map((dep) => `${taskId}_step_${dep}`),
    }));
  }
}
```

#### 步骤 4：实现任务执行器 (`TaskExecutor.ts`)

```typescript
// modules/task/TaskExecutor.ts

import type { ModuleContext } from '../types';
import type { Task, TaskStep, TaskStatusResult } from './types';
import { TaskStatus, StepStatus } from './types';
import { RetryPolicy } from './RetryPolicy';

/**
 * 任务执行器
 * 按依赖关系拓扑排序执行步骤，支持并行与重试
 */
export class TaskExecutor {
  private readonly context: ModuleContext;
  private readonly retryPolicy: RetryPolicy;
  private runningTasks: Map<string, AbortController> = new Map();

  constructor(context: ModuleContext, retryPolicy?: RetryPolicy) {
    this.context = context;
    this.retryPolicy = retryPolicy ?? new RetryPolicy();
  }

  /**
   * 执行任务
   */
  async execute(task: Task, resumeFromFailure = false): Promise<TaskStatusResult> {
    task.status = TaskStatus.EXECUTING;
    task.updatedAt = Date.now();

    const abortController = new AbortController();
    this.runningTasks.set(task.id, abortController);

    try {
      if (resumeFromFailure) {
        this.resetFailedSteps(task);
      }

      await this.executeSteps(task, abortController.signal);

      const allCompleted = task.steps.every(
        (s) => s.status === StepStatus.COMPLETED || s.status === StepStatus.SKIPPED
      );

      task.status = allCompleted ? TaskStatus.COMPLETED : TaskStatus.FAILED;
      task.completedAt = Date.now();
      task.updatedAt = Date.now();

      if (allCompleted) {
        task.result = task.steps.map((s) => s.result);
      }

      this.context.logger.info(
        `[TaskExecutor] 任务 "${task.name}" 执行${allCompleted ? '成功' : '失败'}`
      );
    } catch (error) {
      task.status = TaskStatus.FAILED;
      task.error = error instanceof Error ? error.message : String(error);
      task.updatedAt = Date.now();
    } finally {
      this.runningTasks.delete(task.id);
    }

    return this.buildStatusResult(task);
  }

  /**
   * 暂停任务
   */
  pause(taskId: string): void {
    const controller = this.runningTasks.get(taskId);
    if (controller) {
      controller.abort();
      this.context.logger.info(`[TaskExecutor] 任务 ${taskId} 已暂停`);
    }
  }

  /**
   * 重置失败步骤为待执行状态
   */
  private resetFailedSteps(task: Task): void {
    for (const step of task.steps) {
      if (step.status === StepStatus.FAILED) {
        step.status = StepStatus.PENDING;
        step.error = undefined;
        step.retryCount = 0;
      }
    }
  }

  /**
   * 按拓扑顺序执行步骤
   */
  private async executeSteps(task: Task, signal: AbortSignal): Promise<void> {
    const completed = new Set<string>();
    const stepsMap = new Map<string, TaskStep>();

    for (const step of task.steps) {
      stepsMap.set(step.id, step);
    }

    while (completed.size < task.steps.length) {
      if (signal.aborted) {
        task.status = TaskStatus.PAUSED;
        return;
      }

      // 找出所有可执行的步骤（依赖已完成 + 状态为 pending）
      const readySteps = task.steps.filter(
        (step) =>
          step.status === StepStatus.PENDING &&
          step.dependsOn.every((depId) => completed.has(depId))
      );

      if (readySteps.length === 0) {
        // 没有可执行的步骤且未全部完成 → 存在循环依赖或全部失败
        const hasRunning = task.steps.some(
          (s) => s.status === StepStatus.RUNNING || s.status === StepStatus.RETRYING
        );
        if (!hasRunning) {
          throw new Error('无可用步骤可执行，可能存在循环依赖');
        }
        await this.sleep(100);
        continue;
      }

      // 并行执行所有就绪步骤
      const results = await Promise.allSettled(
        readySteps.map((step) => this.executeSingleStep(step, signal))
      );

      for (let i = 0; i < readySteps.length; i++) {
        const step = readySteps[i];
        const result = results[i];

        if (result.status === 'fulfilled') {
          step.status = StepStatus.COMPLETED;
          step.result = result.value;
          step.completedAt = Date.now();
          completed.add(step.id);
        } else {
          step.status = StepStatus.FAILED;
          step.error =
            result.reason instanceof Error
              ? result.reason.message
              : String(result.reason);
          step.completedAt = Date.now();
          // 步骤失败不阻塞其他独立分支
          completed.add(step.id);
        }
      }

      task.updatedAt = Date.now();
    }
  }

  /**
   * 执行单个步骤（含重试）
   */
  private async executeSingleStep(
    step: TaskStep,
    signal: AbortSignal
  ): Promise<unknown> {
    step.status = StepStatus.RUNNING;
    step.startedAt = Date.now();

    return this.retryPolicy.execute(
      async () => {
        if (signal.aborted) {
          throw new Error('ABORTED');
        }

        this.context.logger.info(
          `[TaskExecutor] 执行步骤: ${step.name} (工具: ${step.toolName})`
        );

        // 通过工具路由模块调用对应工具
        const router = this.context.getModule('tools');
        if (!router) {
          throw new Error('工具路由模块未加载');
        }

        const result = await router.execute(step.toolName, step.params);
        return result;
      },
      (error, attempt, delay) => {
        step.status = StepStatus.RETRYING;
        step.retryCount = attempt;
        this.context.logger.warn(
          `[TaskExecutor] 步骤 "${step.name}" 第 ${attempt} 次重试，` +
            `等待 ${delay}ms: ${error.message}`
        );
      }
    );
  }

  /**
   * 构建状态查询结果
   */
  buildStatusResult(task: Task): TaskStatusResult {
    const completedSteps = task.steps.filter(
      (s) => s.status === StepStatus.COMPLETED || s.status === StepStatus.SKIPPED
    ).length;
    const failedSteps = task.steps.filter(
      (s) => s.status === StepStatus.FAILED
    ).length;

    return {
      taskId: task.id,
      name: task.name,
      status: task.status,
      totalSteps: task.steps.length,
      completedSteps,
      failedSteps,
      progress:
        task.steps.length > 0
          ? Math.round((completedSteps / task.steps.length) * 100)
          : 0,
      steps: task.steps.map((s) => ({
        id: s.id,
        name: s.name,
        status: s.status,
        error: s.error,
      })),
      createdAt: task.createdAt,
      updatedAt: task.updatedAt,
    };
  }

  private sleep(ms: number): Promise<void> {
    return new Promise((resolve) => setTimeout(resolve, ms));
  }
}
```

```typescript
// modules/task/TaskStore.ts

import * as fs from 'node:fs/promises';
import * as path from 'node:path';
import type { Task } from './types';

/**
 * 任务持久化存储
 * 使用 JSON 文件存储任务状态，支持内存缓存
 */
export class TaskStore {
  private readonly storageDir: string;
  private cache: Map<string, Task> = new Map();
  private initialized = false;

  constructor(storageDir: string) {
    this.storageDir = storageDir;
  }

  /**
   * 初始化存储目录
   */
  async init(): Promise<void> {
    if (this.initialized) return;
    await fs.mkdir(this.storageDir, { recursive: true });
    await this.loadFromDisk();
    this.initialized = true;
  }

  /**
   * 保存任务
   */
  async save(task: Task): Promise<void> {
    this.cache.set(task.id, { ...task });
    await this.persistToDisk(task);
  }

  /**
   * 获取任务
   */
  get(taskId: string): Task | undefined {
    const task = this.cache.get(taskId);
    return task ? { ...task } : undefined;
  }

  /**
   * 获取所有任务
   */
  getAll(): Task[] {
    return Array.from(this.cache.values()).map((t) => ({ ...t }));
  }

  /**
   * 删除任务
   */
  async delete(taskId: string): Promise<boolean> {
    const existed = this.cache.delete(taskId);
    if (existed) {
      const filePath = this.getFilePath(taskId);
      try {
        await fs.unlink(filePath);
      } catch {
        // 文件不存在忽略
      }
    }
    return existed;
  }

  /**
   * 按状态筛选任务
   */
  getByStatus(status: Task['status']): Task[] {
    return this.getAll().filter((t) => t.status === status);
  }

  /**
   * 持久化单个任务到磁盘
   */
  private async persistToDisk(task: Task): Promise<void> {
    const filePath = this.getFilePath(task.id);
    await fs.writeFile(filePath, JSON.stringify(task, null, 2), 'utf-8');
  }

  /**
   * 从磁盘加载所有任务
   */
  private async loadFromDisk(): Promise<void> {
    try {
      const files = await fs.readdir(this.storageDir);
      const jsonFiles = files.filter((f) => f.endsWith('.json'));

      for (const file of jsonFiles) {
        try {
          const content = await fs.readFile(
            path.join(this.storageDir, file),
            'utf-8'
          );
          const task: Task = JSON.parse(content);
          this.cache.set(task.id, task);
        } catch {
          // 跳过损坏的文件
        }
      }
    } catch {
      // 目录不存在忽略
    }
  }

  /**
   * 获取任务文件路径
   */
  private getFilePath(taskId: string): string {
    return path.join(this.storageDir, `${taskId}.json`);
  }
}
```

```typescript
// modules/task/TaskModule.ts

import { randomUUID } from 'node:crypto';
import * as path from 'node:path';
import type { Module, ModuleContext, Capability } from '../types';
import type {
  Task,
  TaskCreateParams,
  TaskExecuteParams,
  TaskStatusResult,
} from './types';
import { TaskStatus } from './types';
import { TaskPlanner } from './TaskPlanner';
import { TaskExecutor } from './TaskExecutor';
import { TaskStore } from './TaskStore';
import { RetryPolicy } from './RetryPolicy';

/**
 * 任务规划模块
 * 提供复杂任务拆解、多步执行、状态追踪能力
 */
export class TaskModule implements Module {
  readonly id = 'task';
  readonly version = '1.0.0';
  readonly description = '复杂任务拆解与多步执行模块';

  private context!: ModuleContext;
  private planner!: TaskPlanner;
  private executor!: TaskExecutor;
  private store!: TaskStore;

  /**
   * 声明提供的能力
   */
  getCapabilities(): Capability[] {
    return [
      {
        type: 'tool',
        name: 'task_create',
        description: '创建新任务，自动拆解为可执行步骤',
        priority: 10,
        moduleId: 'task',
        params: {
          name: '任务名称',
          instruction: '用户指令',
          metadata: '可选元数据',
        },
      },
      {
        type: 'tool',
        name: 'task_execute',
        description: '执行已创建的任务',
        priority: 10,
        moduleId: 'task',
        params: {
          taskId: '任务 ID',
          resumeFromFailure: '是否从失败处恢复',
        },
      },
      {
        type: 'tool',
        name: 'task_status',
        description: '查询任务执行状态',
        priority: 5,
        moduleId: 'task',
        params: {
          taskId: '任务 ID',
        },
      },
    ];
  }

  /**
   * 模块初始化
   */
  async activate(context: ModuleContext): Promise<void> {
    this.context = context;

    const storageDir = path.join(context.config.dataDir, 'tasks');
    this.store = new TaskStore(storageDir);
    await this.store.init();

    const retryPolicy = new RetryPolicy({
      maxRetries: context.config.taskMaxRetries ?? 3,
      baseDelayMs: 1000,
    });

    this.planner = new TaskPlanner(context);
    this.executor = new TaskExecutor(context, retryPolicy);

    context.logger.info('[TaskModule] 模块初始化完成');
  }

  /**
   * 处理能力调用
   */
  async execute(capabilityName: string, params: unknown): Promise<unknown> {
    switch (capabilityName) {
      case 'task_create':
        return this.handleCreate(params as TaskCreateParams);
      case 'task_execute':
        return this.handleExecute(params as TaskExecuteParams);
      case 'task_status':
        return this.handleStatus(
          (params as { taskId: string }).taskId
        );
      default:
        throw new Error(`[TaskModule] 未知能力: ${capabilityName}`);
    }
  }

  /**
   * 模块销毁
   */
  async deactivate(): Promise<void> {
    this.context.logger.info('[TaskModule] 模块销毁');
  }

  /**
   * 处理任务创建
   */
  private async handleCreate(params: TaskCreateParams): Promise<Task> {
    const task: Task = {
      id: randomUUID(),
      name: params.name,
      instruction: params.instruction,
      status: TaskStatus.PENDING,
      steps: [],
      createdAt: Date.now(),
      updatedAt: Date.now(),
      metadata: params.metadata ?? {},
    };

    await this.store.save(task);

    // 自动进行任务规划
    const plannedTask = await this.planner.plan(task);
    await this.store.save(plannedTask);

    this.context.logger.info(
      `[TaskModule] 任务已创建: ${plannedTask.name} (${plannedTask.id})`
    );

    return plannedTask;
  }

  /**
   * 处理任务执行
   */
  private async handleExecute(params: TaskExecuteParams): Promise<TaskStatusResult> {
    const task = this.store.get(params.taskId);
    if (!task) {
      throw new Error(`任务不存在: ${params.taskId}`);
    }

    if (task.status === TaskStatus.EXECUTING) {
      throw new Error(`任务正在执行中: ${params.taskId}`);
    }

    const result = await this.executor.execute(task, params.resumeFromFailure);
    await this.store.save(task);

    return result;
  }

  /**
   * 处理状态查询
   */
  private handleStatus(taskId: string): TaskStatusResult {
    const task = this.store.get(taskId);
    if (!task) {
      throw new Error(`任务不存在: ${taskId}`);
    }

    return this.executor.buildStatusResult(task);
  }
}
```

### 21.3 代码规范

1. **类型安全**：所有接口字段使用 `readonly` 修饰不可变属性，状态枚举使用 `as const` 断言
2. **错误处理**：每个异步操作必须有 try-catch，错误信息包含上下文（模块名、任务 ID、步骤名）
3. **重试策略**：使用指数退避 + 随机抖动，仅对可重试错误进行重试
4. **并发控制**：步骤按拓扑排序并行执行，使用 `Promise.allSettled` 避免单点失败
5. **状态管理**：任务状态变更必须同步持久化，内存缓存与磁盘存储保持一致
6. **日志规范**：使用 `[TaskModule]` 前缀，包含关键操作和错误信息
7. **类型守卫**：LLM 返回数据必须通过类型守卫验证后使用
8. **模块解耦**：通过 `ModuleContext.getModule()` 获取依赖模块，不直接 import
9. **取消支持**：使用 `AbortController` 支持任务暂停/取消
10. **ID 生成**：使用 `crypto.randomUUID()` 生成唯一标识
