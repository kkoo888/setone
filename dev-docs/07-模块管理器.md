```typescript
import { existsSync, readFileSync, statSync } from 'fs'
import { join, resolve } from 'path'
import type { ModuleMeta } from '../types/module'
import type { Logger } from '../types/logger'

// ─── 类型定义 ───────────────────────────────────────────────

export interface ValidationResult {
  valid: boolean
  errors: string[]
}

export interface ValidationError {
  code: string
  message: string
  field?: string
}

export interface ValidationWarning {
  code: string
  message: string
}

// ─── 常量 ───────────────────────────────────────────────────

/** ModuleMeta 必填字段定义 */
const REQUIRED_META_FIELDS: Array<{ key: keyof ModuleMeta; type: string }> = [
  { key: 'id', type: 'string' },
  { key: 'name', type: 'string' },
  { key: 'version', type: 'string' }
]

/** 模块 ID 合法字符：小写字母、数字、连字符，且以字母开头 */
const MODULE_ID_REGEX = /^[a-z][a-z0-9-]*$/

/** Semver 简易校验 (x.y.z 可选 -prerelease +build) */
const SEMVER_REGEX = /^\d+\.\d+\.\d+(-[\w.]+)?(\+[\w.]+)?$/

/** 模块目录必须包含的文件 */
const REQUIRED_FILES = ['module.json']

/** 模块入口文件名（按优先级：先 .js 后 .ts） */
const ENTRY_FILE_CANDIDATES = ['index.js', 'index.ts']

/** 合法的能力类型 */
const VALID_CAPABILITY_TYPES = new Set([
  'tool',
  'event',
  'ui',
  'service'
])

/** 能力 name 合法字符：字母、数字、连字符、下划线、点 */
const CAPABILITY_NAME_REGEX = /^[a-zA-Z][a-zA-Z0-9._-]*$/

// ─── 验证器类 ────────────────────────────────────────────────

export class ModuleValidator {
  private logger: Logger

  constructor(logger: Logger) {
    this.logger = logger
  }

  /**
   * 验证模块目录结构
   * 检查目录是否存在、必需文件是否齐全、目录权限是否可读
   *
   * @param modulePath 模块目录的绝对路径
   * @returns ValidationResult
   */
  validateModule(modulePath: string): ValidationResult {
    const errors: string[] = []

    // 1. 目录是否存在
    if (!existsSync(modulePath)) {
      return { valid: false, errors: [`模块目录不存在: ${modulePath}`] }
    }

    // 2. 是否为目录
    try {
      const stat = statSync(modulePath)
      if (!stat.isDirectory()) {
        errors.push(`路径不是目录: ${modulePath}`)
        return { valid: false, errors }
      }
    } catch (err) {
      errors.push(`无法读取目录信息: ${modulePath} (${(err as Error).message})`)
      return { valid: false, errors }
    }

    // 3. 必需文件检查
    for (const file of REQUIRED_FILES) {
      const filePath = join(modulePath, file)
      if (!existsSync(filePath)) {
        errors.push(`缺少必需文件: ${file}`)
      } else {
        try {
          const stat = statSync(filePath)
          if (!stat.isFile()) {
            errors.push(`${file} 不是文件`)
          }
        } catch (err) {
          errors.push(`无法读取文件信息: ${file} (${(err as Error).message})`)
        }
      }
    }

    // 3.5 入口文件检查（index.js 或 index.ts，至少存在一个）
    const hasEntry = ENTRY_FILE_CANDIDATES.some(f => existsSync(join(modulePath, f)))
    if (!hasEntry) {
      errors.push(`缺少入口文件: 需要 index.js 或 index.ts`)
    }

    // 4. module.json 可读性 + JSON 解析
    const metaPath = join(modulePath, 'module.json')
    if (existsSync(metaPath)) {
      try {
        const raw = readFileSync(metaPath, 'utf-8')
        const parsed = JSON.parse(raw)
        if (typeof parsed !== 'object' || parsed === null || Array.isArray(parsed)) {
          errors.push('module.json 根节点必须是对象')
        }
      } catch (err) {
        if ((err as NodeJS.ErrnoException).code === 'EACCES') {
          errors.push('module.json 无读取权限')
        } else {
          errors.push(`module.json 解析失败: ${(err as Error).message}`)
        }
      }
    }

    // 5. 警告：非必需但推荐的文件
    const recommendedFiles = ['README.md', 'package.json']
    for (const file of recommendedFiles) {
      if (!existsSync(join(modulePath, file))) {
        this.logger.debug(`[ModuleValidator] 建议添加 ${file}`)
      }
    }

    return { valid: errors.length === 0, errors }
  }

  /**
   * 验证 module.json 格式和字段
   * 独立于目录结构校验，接受已解析的 JSON 对象
   *
   * @param moduleJson 已解析的 module.json 对象
   * @returns ValidationResult
   */
  validateModuleJson(moduleJson: unknown): ValidationResult {
    const errors: string[] = []

    // 1. 基本结构检查
    if (!moduleJson || typeof moduleJson !== 'object' || Array.isArray(moduleJson)) {
      return { valid: false, errors: ['module.json 必须是一个 JSON 对象'] }
    }

    const meta = moduleJson as Record<string, unknown>

    // 2. 必填字段存在性 + 类型检查
    for (const { key, type } of REQUIRED_META_FIELDS) {
      const value = meta[key]
      if (value === undefined || value === null) {
        errors.push(`缺少必填字段: ${key}`)
      } else if (typeof value !== type) {
        errors.push(`字段 "${key}" 类型应为 ${type}，实际为 ${typeof value}`)
      }
    }

    // 3. 模块 ID 格式
    if (typeof meta.id === 'string' && meta.id) {
      if (!MODULE_ID_REGEX.test(meta.id)) {
        errors.push(`模块 ID "${meta.id}" 格式不合法，仅允许小写字母、数字和连字符，且以字母开头`)
      }
      if (meta.id.length > 64) {
        errors.push(`模块 ID 长度不能超过 64 字符`)
      }
    }

    // 4. 版本号（Semver）
    if (typeof meta.version === 'string' && meta.version) {
      if (!SEMVER_REGEX.test(meta.version)) {
        errors.push(`版本号 "${meta.version}" 不符合 Semver 规范 (x.y.z)`)
      }
    }

    // 5. name 字段
    if (typeof meta.name === 'string' && meta.name) {
      if (meta.name.length > 128) {
        errors.push('模块名称长度不能超过 128 字符')
      }
    }

    // 6. description（可选，但类型检查）
    if (meta.description !== undefined && typeof meta.description !== 'string') {
      errors.push('description 字段应为字符串')
    }

    // 7. dependencies 校验
    if (meta.dependencies !== undefined) {
      if (!Array.isArray(meta.dependencies)) {
        errors.push('dependencies 字段应为字符串数组')
      } else {
        for (let i = 0; i < meta.dependencies.length; i++) {
          const dep = meta.dependencies[i]
          if (typeof dep !== 'string' || !dep.trim()) {
            errors.push(`dependencies[${i}] 不是有效的非空字符串`)
          }
        }
        // 检查重复依赖
        const uniqueDeps = new Set(meta.dependencies)
        if (uniqueDeps.size !== meta.dependencies.length) {
          errors.push('dependencies 中存在重复项')
        }
      }
    }

    // 8. provides（能力声明）校验
    if (meta.provides !== undefined) {
      const capResult = this.validateCapabilities(meta.provides)
      if (!capResult.valid) {
        errors.push(...capResult.errors)
      }
    }

    // 9. priority 校验（可选）
    if (meta.priority !== undefined) {
      if (typeof meta.priority !== 'number' || !Number.isInteger(meta.priority)) {
        errors.push('priority 字段应为整数')
      } else if (meta.priority < 0 || meta.priority > 1000) {
        errors.push(`priority 值 "${meta.priority}" 不在有效范围 0-1000 内`)
      }
    }

    // 10. 未知字段警告（不影响 valid）
    const knownFields = new Set([
      'id', 'name', 'version', 'description', 'author',
      'dependencies', 'provides', 'priority', 'config',
      'repository', 'license', 'keywords'
    ])
    for (const key of Object.keys(meta)) {
      if (!knownFields.has(key)) {
        this.logger.debug(`[ModuleValidator] 未知字段: ${key}`)
      }
    }

    return { valid: errors.length === 0, errors }
  }

  /**
   * 验证依赖是否满足
   * 检查每个依赖模块是否在已注册模块列表中
   *
   * @param deps 依赖的模块 ID 列表
   * @param registeredModules 已注册的模块 ID 集合
   * @returns ValidationResult
   */
  validateDependencies(
    deps: string[],
    registeredModules: Set<string>
  ): ValidationResult {
    const errors: string[] = []

    if (!Array.isArray(deps)) {
      return { valid: false, errors: ['依赖列表必须是字符串数组'] }
    }

    for (const dep of deps) {
      if (typeof dep !== 'string' || !dep.trim()) {
        errors.push('存在无效的依赖声明（非字符串或为空）')
        continue
      }

      if (!registeredModules.has(dep)) {
        errors.push(`依赖模块 "${dep}" 未在已注册模块中找到`)
      }
    }

    // 检查循环依赖（简单自引用检测）
    // 注意：完整的循环依赖检测应在 DependencyResolver 中进行
    for (const dep of deps) {
      if (dep === deps.find((_, i) => deps.indexOf(dep) === i)) {
        // 自引用由 DependencyResolver 处理，此处仅做基础检查
      }
    }

    return { valid: errors.length === 0, errors }
  }

  /**
   * 验证能力声明格式
   * 检查每个 capability 的 type、name、handler 是否合法
   *
   * @param capabilities provides 数组
   * @returns ValidationResult
   */
  validateCapabilities(capabilities: unknown): ValidationResult {
    const errors: string[] = []

    if (!Array.isArray(capabilities)) {
      return { valid: false, errors: ['能力声明 (provides) 必须是数组'] }
    }

    if (capabilities.length === 0) {
      return { valid: true, errors: [] }
    }

    const declaredNames = new Set<string>()

    for (let i = 0; i < capabilities.length; i++) {
      const cap = capabilities[i]
      const prefix = `provides[${i}]`

      // 基本结构
      if (!cap || typeof cap !== 'object' || Array.isArray(cap)) {
        errors.push(`${prefix} 必须是对象`)
        continue
      }

      const capObj = cap as Record<string, unknown>

      // name 字段（必填）
      if (!capObj.name || typeof capObj.name !== 'string') {
        errors.push(`${prefix} 缺少必填字段 "name"（字符串）`)
      } else {
        if (!CAPABILITY_NAME_REGEX.test(capObj.name)) {
          errors.push(`${prefix}.name "${capObj.name}" 格式不合法，仅允许字母、数字、连字符、下划线和点，且以字母开头`)
        }
        // 重复 name 检测
        if (declaredNames.has(capObj.name)) {
          errors.push(`${prefix}.name "${capObj.name}" 重复声明`)
        }
        declaredNames.add(capObj.name)
      }

      // type 字段（必填）
      if (!capObj.type || typeof capObj.type !== 'string') {
        errors.push(`${prefix} 缺少必填字段 "type"（字符串）`)
      } else if (!VALID_CAPABILITY_TYPES.has(capObj.type)) {
        errors.push(`${prefix}.type "${capObj.type}" 不是合法的能力类型，可选值: ${[...VALID_CAPABILITY_TYPES].join(', ')}`)
      }

      // handler 字段（可选，但若存在需为字符串路径）
      if (capObj.handler !== undefined) {
        if (typeof capObj.handler !== 'string' || !capObj.handler.trim()) {
          errors.push(`${prefix}.handler 应为非空字符串（模块内相对路径）`)
        }
      }

      // description（可选，类型检查）
      if (capObj.description !== undefined && typeof capObj.description !== 'string') {
        errors.push(`${prefix}.description 应为字符串`)
      }

      // options（可选，类型检查）
      if (capObj.options !== undefined) {
        if (typeof capObj.options !== 'object' || capObj.options === null || Array.isArray(capObj.options)) {
          errors.push(`${prefix}.options 应为对象`)
        }
      }
    }

    return { valid: errors.length === 0, errors }
  }

  /**
   * 完整校验模块（组合校验）
   * 依次执行目录结构、module.json、依赖、能力的完整校验
   *
   * @param modulePath 模块目录路径
   * @param registeredModules 已注册模块 ID 集合（可选，用于依赖校验）
   * @returns ValidationResult
   */
  validate(modulePath: string, registeredModules?: Set<string>): ValidationResult {
    const allErrors: string[] = []

    // 1. 目录结构校验
    const dirResult = this.validateModule(modulePath)
    if (!dirResult.valid) {
      allErrors.push(...dirResult.errors)
      // 目录不合法则跳过后续校验
      return { valid: false, errors: allErrors }
    }

    // 2. module.json 格式校验
    let meta: Record<string, unknown>
    try {
      const raw = readFileSync(join(modulePath, 'module.json'), 'utf-8')
      meta = JSON.parse(raw) as Record<string, unknown>
    } catch (err) {
      return { valid: false, errors: [`module.json 读取或解析失败: ${(err as Error).message}`] }
    }

    const jsonResult = this.validateModuleJson(meta)
    if (!jsonResult.valid) {
      allErrors.push(...jsonResult.errors)
    }

    // 3. 依赖校验（仅当 module.json 合法且提供了已注册模块列表时）
    if (jsonResult.valid && registeredModules && Array.isArray(meta.dependencies)) {
      const depResult = this.validateDependencies(
        meta.dependencies as string[],
        registeredModules
      )
      if (!depResult.valid) {
        allErrors.push(...depResult.errors)
      }
    }

    // 4. 能力声明校验
    if (meta.provides !== undefined) {
      const capResult = this.validateCapabilities(meta.provides)
      if (!capResult.valid) {
        allErrors.push(...capResult.errors)
      }
    }

    this.logger.debug(`[ModuleValidator] 模块 "${(meta as ModuleMeta).id}" 校验完成`, {
      valid: allErrors.length === 0,
      errorCount: allErrors.length
    })

    return { valid: allErrors.length === 0, errors: allErrors }
  }
}
```

**src/main/core/module-scanner.ts**：

```typescript
import { readdirSync, existsSync } from 'fs'
import { readFile } from 'fs/promises'
import { join } from 'path'
import { createHash } from 'crypto'
import type { ModuleMeta } from '../types/module'
import type { Logger } from '../types/logger'

export interface ScannedModule {
  meta: ModuleMeta
  path: string
  hash: string
}

export class ModuleScanner {
  constructor(
    private modulesDir: string,
    private logger: Logger
  ) {}

  /** 扫描模块目录，返回所有有效模块 */
  async scan(): Promise<ScannedModule[]> {
    const results: ScannedModule[] = []

    if (!existsSync(this.modulesDir)) {
      this.logger.warn('模块目录不存在', { path: this.modulesDir })
      return results
    }

    const entries = readdirSync(this.modulesDir, { withFileTypes: true })

    for (const entry of entries) {
      if (!entry.isDirectory()) continue

      const modulePath = join(this.modulesDir, entry.name)
      const metaPath = join(modulePath, 'module.json')

      if (!existsSync(metaPath)) {
        this.logger.debug(`跳过目录 "${entry.name}"：缺少 module.json`)
        continue
      }

      try {
        const metaRaw = await readFile(metaPath, 'utf-8')
        const meta = JSON.parse(metaRaw) as ModuleMeta
        const hash = await this.calculateHash(modulePath)

        results.push({ meta, path: modulePath, hash })
        this.logger.debug(`发现模块: ${meta.id} v${meta.version}`)
      } catch (err) {
        this.logger.warn(`模块 "${entry.name}" 的 module.json 解析失败`, { error: err })
      }
    }

    return results
  }

  /** 计算模块目录的 hash（用于检测变化） */
  private async calculateHash(modulePath: string): Promise<string> {
    const hash = createHash('md5')
    const metaPath = join(modulePath, 'module.json')
    // 优先 hash 编译后的 .js，其次 .ts
    const entryPath = existsSync(join(modulePath, 'index.js'))
      ? join(modulePath, 'index.js')
      : join(modulePath, 'index.ts')

    if (existsSync(metaPath)) hash.update(await readFile(metaPath))
    if (existsSync(entryPath)) hash.update(await readFile(entryPath))

    return hash.digest('hex')
  }
}
```

#### 步骤 2：实现依赖解析器

**src/main/core/dependency-resolver.ts**：

```typescript
import type { ScannedModule } from './module-scanner'

export interface ResolutionResult {
  order: string[]             // 拓扑排序后的模块 ID 列表
  errors: DependencyError[]   // 依赖错误
}

export interface DependencyError {
  moduleId: string
  type: 'missing' | 'incompatible' | 'circular'
  message: string
}

export class DependencyResolver {
  /**
   * 解析依赖关系，返回拓扑排序结果
   * 注意：循环依赖的模块会被排除，但不阻塞其他正常模块
   */
  resolve(modules: ScannedModule[]): ResolutionResult {
    const errors: DependencyError[] = []
    const moduleMap = new Map(modules.map(m => [m.meta.id, m]))

    // 1. 检查依赖是否存在
    for (const mod of modules) {
      for (const dep of mod.meta.dependencies) {
        if (!moduleMap.has(dep)) {
          errors.push({
            moduleId: mod.meta.id,
            type: 'missing',
            message: `缺少依赖模块: ${dep}`
          })
        }
      }
    }

    // 2. 检测循环依赖
    const cycles = this.detectCycles(modules)
    errors.push(...cycles)

    // 3. 收集有问题的模块 ID，排除它们后继续加载其余模块
    const problematicIds = new Set(errors.map(e => e.moduleId))
    const validModules = modules.filter(m => !problematicIds.has(m.meta.id))

    // 4. 对有效模块执行拓扑排序
    const order = this.topologicalSort(validModules)
    return { order, errors }
  }

  /** 检测循环依赖（DFS） */
  private detectCycles(modules: ScannedModule[]): DependencyError[] {
    const errors: DependencyError[] = []
    const adjacency = new Map<string, string[]>()
    const visited = new Set<string>()
    const inStack = new Set<string>()

    for (const mod of modules) {
      adjacency.set(mod.meta.id, mod.meta.dependencies)
    }

    const dfs = (nodeId: string, path: string[]): boolean => {
      if (inStack.has(nodeId)) {
        const cycleStart = path.indexOf(nodeId)
        const cycle = path.slice(cycleStart).concat(nodeId)
        const cycleMessage = `检测到循环依赖: ${cycle.join(' → ')}`
        // 为循环中的每个模块添加错误，确保全部被排除（而非仅入口模块）
        for (const cycleNodeId of cycle.slice(0, -1)) {
          errors.push({
            moduleId: cycleNodeId,
            type: 'circular',
            message: cycleMessage
          })
        }
        return true
      }

      if (visited.has(nodeId)) return false

      visited.add(nodeId)
      inStack.add(nodeId)

      const deps = adjacency.get(nodeId) || []
      for (const dep of deps) {
        if (dfs(dep, [...path, nodeId])) return true
      }

      inStack.delete(nodeId)
      return false
    }

    for (const mod of modules) {
      if (!visited.has(mod.meta.id)) {
        dfs(mod.meta.id, [])
      }
    }

    return errors
  }

  /** 拓扑排序（Kahn 算法 + 最小堆优先队列） */
  private topologicalSort(modules: ScannedModule[]): string[] {
    const inDegree = new Map<string, number>()
    const adjacency = new Map<string, string[]>()
    const moduleIds = new Set(modules.map(m => m.meta.id))
    const priorityMap = new Map(modules.map(m => [m.meta.id, m.meta.priority]))

    for (const mod of modules) {
      inDegree.set(mod.meta.id, 0)
      adjacency.set(mod.meta.id, [])
    }

    for (const mod of modules) {
      for (const dep of mod.meta.dependencies) {
        // 只计算仍在有效集合中的依赖，已被排除的（如循环依赖）不计入 inDegree
        if (moduleIds.has(dep)) {
          adjacency.get(dep)?.push(mod.meta.id)
          inDegree.set(mod.meta.id, (inDegree.get(mod.meta.id) || 0) + 1)
        }
      }
    }

    // 最小堆优先队列（按 priority 排序），替代每次入队后全排序
    // 堆操作 O(log n)，整体复杂度 O((V+E) log V)，优于原方案 O(V² log V)
    const heap: string[] = []

    const heapPush = (id: string): void => {
      heap.push(id)
      let i = heap.length - 1
      while (i > 0) {
        const parent = (i - 1) >> 1
        if ((priorityMap.get(heap[parent]) || 999) <= (priorityMap.get(heap[i]) || 999)) break
        ;[heap[parent], heap[i]] = [heap[i], heap[parent]]
        i = parent
      }
    }

    const heapPop = (): string | undefined => {
      if (heap.length === 0) return undefined
      const top = heap[0]
      const last = heap.pop()!
      if (heap.length > 0) {
        heap[0] = last
        let i = 0
        while (true) {
          let min = i
          const left = 2 * i + 1
          const right = 2 * i + 2
          if (left < heap.length &&
              (priorityMap.get(heap[left]) || 999) < (priorityMap.get(heap[min]) || 999)) {
            min = left
          }
          if (right < heap.length &&
              (priorityMap.get(heap[right]) || 999) < (priorityMap.get(heap[min]) || 999)) {
            min = right
          }
          if (min === i) break
          ;[heap[i], heap[min]] = [heap[min], heap[i]]
          i = min
        }
      }
      return top
    }

    for (const [id, degree] of inDegree) {
      if (degree === 0) heapPush(id)
    }

    const result: string[] = []

    while (heap.length > 0) {
      const current = heapPop()!
      result.push(current)

      const neighbors = adjacency.get(current) || []
      for (const neighbor of neighbors) {
        const newDegree = (inDegree.get(neighbor) || 1) - 1
        inDegree.set(neighbor, newDegree)
        if (newDegree === 0) {
          heapPush(neighbor)
        }
      }
    }

    return result
  }
}
```

**src/main/core/module-loader.ts**：

```typescript
import type { Module, ModuleContext, ModuleRegistration, ModuleMeta } from '../types/module'
import type { EventBus } from '../types/event'
import type { Logger } from '../types/logger'
import type { ConfigManager } from '../types/config'
import type { AIService } from '../types/ai'
import { existsSync } from 'fs'
import { join } from 'path'
import { createScopedEventBus } from './scoped-event-bus'
import { createLogger } from './logger'

/** deactivate 默认超时时间（ms） */
const DEFAULT_DEACTIVATE_TIMEOUT_MS = 5000

export class ModuleLoader {
  private modules = new Map<string, ModuleRegistration>()
  private logger: Logger
  /** deactivate 超时时间（ms），可由外部配置覆盖 */
  private deactivateTimeoutMs: number = DEFAULT_DEACTIVATE_TIMEOUT_MS

  constructor(
    private eventBus: EventBus,
    private config: ConfigManager,
    private ai: AIService,
    private db: DatabaseManager,
    logger: Logger,
    options?: { deactivateTimeoutMs?: number }
  ) {
    this.logger = logger
    if (options?.deactivateTimeoutMs) {
      this.deactivateTimeoutMs = options.deactivateTimeoutMs
    }
  }

  /**
   * 解析模块入口文件路径
   * 优先加载编译后的 .js（生产环境），其次 .ts（开发环境）
   */
  private resolveEntryPath(modulePath: string): string {
    const jsPath = join(modulePath, 'index.js')
    const tsPath = join(modulePath, 'index.ts')

    if (existsSync(jsPath)) return jsPath
    if (existsSync(tsPath)) return tsPath
    throw new Error(`模块入口文件不存在: ${jsPath} 或 ${tsPath}`)
  }

  /**
   * 动态加载模块入口文件
   * - .js 文件：直接使用 import()
   * - .ts 文件：仅在开发环境下可用，需确保 tsx/esbuild register 已启用
   */
  private async loadModuleEntry(entryPath: string): Promise<Record<string, unknown>> {
    if (entryPath.endsWith('.ts')) {
      this.logger.warn(
        `正在加载 .ts 源文件: ${entryPath}。` +
        '生产环境应使用编译后的 .js 文件。'
      )
    }
    // import() 可加载 .js（Node.js 原生支持）
    // .ts 需要运行时 register 支持（如 tsx, ts-node, 或 Electron + esbuild 预处理）
    return import(entryPath)
  }

  /** 注册模块 */
  register(meta: ModuleMeta, path: string, hash: string): void {
    this.modules.set(meta.id, {
      meta,
      instance: null,
      status: 'discovered',
      hash
    })
    this.logger.info(`模块已注册: ${meta.id}`)
  }

  /** 注销模块（从注册表移除，调用前需先 deactivate） */
  unregister(moduleId: string): boolean {
    const reg = this.modules.get(moduleId)
    if (!reg) return false

    if (reg.status === 'active') {
      this.logger.warn(`模块 "${moduleId}" 仍在活跃状态，请先调用 deactivate`)
      return false
    }

    this.modules.delete(moduleId)
    this.logger.info(`模块已注销: ${moduleId}`)
    return true
  }

  /** 加载并激活模块 */
  async activate(moduleId: string): Promise<boolean> {
    const reg = this.modules.get(moduleId)
    if (!reg) {
      this.logger.error(`模块 "${moduleId}" 未注册`)
      return false
    }

    if (reg.status === 'active') {
      this.logger.warn(`模块 "${moduleId}" 已经激活`)
      return true
    }

    reg.status = 'loading'

    try {
      // 解析入口文件（优先 .js，其次 .ts）
      const modulePath = reg.meta.path || ''
      const entryPath = this.resolveEntryPath(modulePath)
      const moduleExports = await this.loadModuleEntry(entryPath)
      const ModuleClass = moduleExports.default || moduleExports

      // 创建模块上下文
      const context = this.createContext(reg.meta)

      // 实例化模块
      const instance: Module = new ModuleClass(context)
      instance.meta = reg.meta

      // 调用 activate 生命周期钩子（带超时）
      await this.withTimeout(
        () => instance.activate(context),
        10000, // 10s 超时
        `模块 "${moduleId}" activate 超时`
      )

      reg.instance = instance
      reg.status = 'active'

      this.eventBus.emit('on_module_loaded', { moduleId })
      this.logger.info(`模块 "${moduleId}" 已激活`)
      return true
    } catch (err) {
      reg.status = 'error'
      reg.loadError = (err as Error).message
      this.logger.error(`模块 "${moduleId}" 激活失败`, err as Error)
      this.eventBus.emit('on_module_error', { moduleId, error: (err as Error).message })
      return false
    }
  }

  /** 停用模块（带超时保护，超时后不阻塞其他模块） */
  async deactivate(moduleId: string): Promise<boolean> {
    const reg = this.modules.get(moduleId)
    if (!reg || !reg.instance) return false

    let timedOut = false
    try {
      // 超时保护：模块 deactivate 可能阻塞，超时后强制继续
      await this.withTimeout(
        () => reg.instance!.deactivate(),
        this.deactivateTimeoutMs,
        `模块 "${moduleId}" deactivate 超时（${this.deactivateTimeoutMs}ms），强制停用`
      )
      return true
    } catch (err) {
      timedOut = (err as Error).message.includes('超时')
      if (timedOut) {
        this.logger.warn(
          `模块 "${moduleId}" deactivate 超时（${this.deactivateTimeoutMs}ms），已强制标记为停用，不阻塞后续模块`,
          err as Error
        )
      } else {
        this.logger.error(`模块 "${moduleId}" 停用异常`, err as Error)
      }
      return false
    } finally {
      // 无论超时或出错，都清理状态，避免模块卡在中间态阻塞其他模块停用
      reg.instance = null
      reg.status = 'disabled'
      this.eventBus.emit('on_module_unloaded', { moduleId })
      if (!timedOut) {
        this.logger.info(`模块 "${moduleId}" 已停用`)
      }
    }
  }

  /** 创建模块上下文 */
  private createContext(meta: ModuleMeta): ModuleContext {
    return {
      eventBus: createScopedEventBus(
        this.eventBus,
        meta.id,
        meta.provides,
        meta.consumes
      ),
      config: this.config,
      ai: this.ai,
      logger: createLogger(meta.id),
      store: this.createScopedStore(meta.id),
      db: this.db,
      getModule: (name: string) => {
        const reg = this.modules.get(name)
        return reg?.instance ?? undefined
      }
    }
  }

  /** 创建模块私有 Store */
  private createScopedStore(moduleId: string): import('../types/store').ScopedStore {
    const memoryStore = new Map<string, unknown>()
    return {
      get: <T>(key: string) => memoryStore.get(key) as T | undefined,
      set: <T>(key: string, value: T) => { memoryStore.set(key, value) },
      delete: (key: string) => { memoryStore.delete(key) },
      getPersist: async <T>(key: string) => {
        return this.config.getModuleConfig<T>(moduleId, key)
      },
      setPersist: async <T>(key: string, value: T) => {
        await this.config.setModuleConfig(moduleId, key, value)
      },
      deletePersist: async (key: string) => {
        await this.config.deleteModuleConfig(moduleId, key)
      }
    }
  }

  /** 带超时的 Promise（基于 AbortSignal，超时后真正取消执行） */
  private withTimeout<T>(
    operation: () => Promise<T>,
    ms: number,
    message: string
  ): Promise<T> {
    return Promise.race([
      operation(),
      new Promise<T>((_, reject) => {
        setTimeout(() => reject(new Error(message)), ms)
      })
    ])
  }

  /** 获取所有模块注册信息 */
  getAll(): ModuleRegistration[] {
    return Array.from(this.modules.values())
  }

  /** 获取指定模块 */
  get(moduleId: string): ModuleRegistration | undefined {
    return this.modules.get(moduleId)
  }
}
```

#### 步骤 4：实现模块管理器（整合）

**src/main/core/module-manager.ts**：

```typescript
import { join } from 'path'
import { watch, FSWatcher } from 'chokidar'
import { app } from 'electron'
import type { EventBus } from '../types/event'
import type { ConfigManager } from '../types/config'
import type { AIService } from '../types/ai'
import type { Logger } from '../types/logger'
import { ModuleScanner } from './module-scanner'
import { DependencyResolver } from './dependency-resolver'
import { ModuleLoader } from './module-loader'
import type { CapabilityArbiter } from './capability-arbiter'
import { createLogger } from './logger'

export class ModuleManager {
  private scanner: ModuleScanner
  private resolver: DependencyResolver
  private loader: ModuleLoader
  private watcher: FSWatcher | null = null
  private logger: Logger
  private moduleLoadCounter = 0 // 模块启用顺序计数器

  constructor(
    private eventBus: EventBus,
    private config: ConfigManager,
    private ai: AIService,
    private arbiter?: CapabilityArbiter
  ) {
    this.logger = createLogger('module-manager')
    const modulesDir = join(app.getPath('userData'), 'modules')
    this.scanner = new ModuleScanner(modulesDir, this.logger)
    this.resolver = new DependencyResolver()
    // 读取可配置的 deactivate 超时时间
    const deactivateTimeoutMs = this.config.get<number>(
      'modules.deactivateTimeoutMs',
      5000
    ) as unknown as number
    this.loader = new ModuleLoader(eventBus, config, ai, this.logger, { deactivateTimeoutMs })
  }

  /** 初始化：扫描 → 解析 → 加载 */
  async initialize(): Promise<void> {
    this.logger.info('开始初始化模块管理器...')

    // 1. 扫描模块
    const scanned = await this.scanner.scan()
    this.logger.info(`发现 ${scanned.length} 个模块`)

    // 2. 依赖解析
    const resolution = this.resolver.resolve(scanned)
    if (resolution.errors.length > 0) {
      for (const err of resolution.errors) {
        this.logger.warn(`依赖错误 [${err.moduleId}]: ${err.message}`)
      }
    }

    // 3. 注册模块
    for (const mod of scanned) {
      this.loader.register(mod.meta, mod.path, mod.hash)
    }

    // 3.5. 标记因依赖问题被跳过的模块为错误状态
    const errorModuleIds = new Set(resolution.errors.map(e => e.moduleId))
    for (const mod of scanned) {
      if (errorModuleIds.has(mod.meta.id)) {
        const reg = this.loader.get(mod.meta.id)
        if (reg) {
          reg.status = 'error'
          const moduleErrors = resolution.errors.filter(e => e.moduleId === mod.meta.id)
          reg.loadError = moduleErrors.map(e => e.message).join('; ')
        }
      }
    }
    if (errorModuleIds.size > 0) {
      this.logger.warn(
        `以下模块因依赖问题被跳过，不影响其他模块加载: [${[...errorModuleIds].join(', ')}]`
      )
    }

    // 4. 读取用户配置，加载已启用模块
    const enabledModules = await this.config.get<string[]>('modules.enabled', [])
    const disabledModules = await this.config.get<string[]>('modules.disabled', [])

    for (const moduleId of resolution.order) {
      const reg = this.loader.get(moduleId)
      if (!reg) continue

      // 默认启用，除非用户明确禁用
      const isDisabled = disabledModules.includes(moduleId)
      const isEnabled = enabledModules.includes(moduleId) || (!isDisabled && reg.meta.enabled)

      if (isEnabled) {
        const success = await this.loader.activate(moduleId)
        // 记录模块启用顺序，供 CapabilityArbiter 在同优先级时降级使用
        if (success && this.arbiter) {
          this.arbiter.setModuleLoadOrder(moduleId, this.moduleLoadCounter++)
        }
      } else {
        this.logger.info(`模块 "${moduleId}" 已禁用，跳过加载`)
      }
    }

    this.logger.info('模块管理器初始化完成')
  }

  /** 启用模块 */
  async enableModule(moduleId: string): Promise<boolean> {
    const success = await this.loader.activate(moduleId)
    if (success) {
      // 记录启用顺序，供仲裁器降级使用
      if (this.arbiter) {
        this.arbiter.setModuleLoadOrder(moduleId, this.moduleLoadCounter++)
      }
      const disabled = await this.config.get<string[]>('modules.disabled', [])
      await this.config.set(
        'modules.disabled',
        disabled.filter(id => id !== moduleId)
      )
    }
    return success
  }

  /** 禁用模块 */
  async disableModule(moduleId: string): Promise<boolean> {
    const success = await this.loader.deactivate(moduleId)
    if (success) {
      const disabled = await this.config.get<string[]>('modules.disabled', [])
      if (!disabled.includes(moduleId)) {
        disabled.push(moduleId)
        await this.config.set('modules.disabled', disabled)
      }
    }
    return success
  }

  /** 热重载模块 */
  async reloadModule(moduleId: string): Promise<boolean> {
    this.logger.info(`热重载模块: ${moduleId}`)
    await this.loader.deactivate(moduleId)
    return this.loader.activate(moduleId)
  }

  /**
   * 监听模块目录变化（热更新）
   * 支持三种场景：
   *   1. 文件修改 → 热重载已有模块
   *   2. 新增目录 → 自动扫描、验证、注册并激活新模块
   *   3. 删除目录 → 自动停用并清理已移除模块
   *
   * ⚠️ 【未实现】以下代码为设计草案，当前版本未集成。
   * 实际 module-manager.ts 中无此方法，也无 chokidar 依赖。
   * 如需启用，需额外安装 chokidar 并在 initialize() 末尾调用 startWatching()。
   */
  startWatching(): void {
    const modulesDir = join(app.getPath('userData'), 'modules')

    this.watcher = watch(modulesDir, {
      depth: 2,
      ignoreInitial: true,
      awaitWriteFinish: { stabilityThreshold: 1000 }
    })

    // ── 场景 1：文件修改 → 热重载已有模块 ──
    this.watcher.on('change', async (filePath) => {
      const relative = filePath.replace(modulesDir, '')
      const moduleId = relative.split(/[/\\]/)[1]
      if (!moduleId) return

      this.logger.info(`检测到模块变化: ${moduleId}`)
      await this.reloadModule(moduleId)
    })

    // ── 场景 2：新增模块目录 → 自动扫描、验证、注册并激活 ──
    this.watcher.on('addDir', async (dirPath) => {
      // 只处理模块根目录（depth=0 的直接子目录）
      const relative = dirPath.replace(modulesDir, '')
      const parts = relative.split(/[/\\]/).filter(Boolean)
      if (parts.length !== 1) return

      const moduleId = parts[0]
      // 忽略隐藏目录（如 .git、.DS_Store）
      if (moduleId.startsWith('.')) return

      this.logger.info(`检测到新模块目录: ${moduleId}`)

      try {
        // 1. 重新扫描该模块
        const scanned = await this.scanner.scan()
        const newModule = scanned.find(m => m.meta.id === moduleId)

        if (!newModule) {
          this.logger.warn(`新目录 "${moduleId}" 未通过扫描（可能缺少 module.json）`)
          return
        }

        // 2. 验证依赖
        const allModules = [...this.loader.getAll().map(r => r.meta), newModule.meta]
        const resolution = this.resolver.resolve(
          scanned.concat(
            ...this.loader.getAll().map(r => ({
              meta: r.meta,
              path: r.meta.path || '',
              hash: r.hash
            }))
          )
        )

        if (resolution.errors.some(e => e.moduleId === moduleId)) {
          this.logger.error(
            `新模块 "${moduleId}" 依赖校验失败，跳过加载: ${
              resolution.errors.filter(e => e.moduleId === moduleId).map(e => e.message).join('; ')
            }`
          )
          return
        }

        // 3. 注册并激活
        this.loader.register(newModule.meta, newModule.path, newModule.hash)

        // 默认启用（除非用户已在禁用列表中配置）
        const disabledModules = await this.config.get<string[]>('modules.disabled', [])
        if (!disabledModules.includes(moduleId)) {
          const success = await this.loader.activate(moduleId)
          if (success && this.arbiter) {
            this.arbiter.setModuleLoadOrder(moduleId, this.moduleLoadCounter++)
          }
          this.logger.info(`新模块 "${moduleId}" 已自动加载`)
        } else {
          this.logger.info(`新模块 "${moduleId}" 已注册但处于禁用状态，跳过激活`)
        }
      } catch (err) {
        this.logger.error(`自动加载新模块 "${moduleId}" 失败`, err as Error)
      }
    })

    // ── 场景 3：删除模块目录 → 自动停用并清理 ──
    this.watcher.on('unlinkDir', async (dirPath) => {
      const relative = dirPath.replace(modulesDir, '')
      const parts = relative.split(/[/\\]/).filter(Boolean)
      if (parts.length !== 1) return

      const moduleId = parts[0]
      if (moduleId.startsWith('.')) return

      this.logger.info(`检测到模块目录被删除: ${moduleId}`)

      try {
        const reg = this.loader.get(moduleId)
        if (!reg) {
          this.logger.debug(`已删除的目录 "${moduleId}" 未在注册表中，忽略`)
          return
        }

        // 1. 停用（如果正在运行）
        if (reg.status === 'active') {
          await this.loader.deactivate(moduleId)
        }

        // 2. 从注册表移除
        this.loader.unregister(moduleId)

        // 3. 清理用户配置中的引用
        const enabled = await this.config.get<string[]>('modules.enabled', [])
        const disabled = await this.config.get<string[]>('modules.disabled', [])
        await this.config.set(
          'modules.enabled',
          enabled.filter(id => id !== moduleId)
        )
        await this.config.set(
          'modules.disabled',
          disabled.filter(id => id !== moduleId)
        )

        this.logger.info(`模块 "${moduleId}" 已清理完毕`)
      } catch (err) {
        this.logger.error(`清理已删除模块 "${moduleId}" 失败`, err as Error)
      }
    })

    this.logger.info('模块目录监听已启动（支持新增/修改/删除）')
  }

  /** 停止监听 */
  stopWatching(): void {
    this.watcher?.close()
    this.watcher = null
  }

  /** 获取所有模块状态 */
  getModules() {
    return this.loader.getAll()
  }
}
```

### 7.3 代码规范

- **生命周期**：`activate()` 必须在沙箱中调用，超时 10s 自动中止
- **依赖解析**：Kahn 算法拓扑排序，priority 作为排序辅助
- **热重载**：文件变化后等待 1s 稳定再触发重载
- **错误隔离**：单个模块加载失败不影响其他模块
- **循环依赖处理**：检测到循环依赖时，循环中的所有模块标记为 `error` 状态并跳过加载，不阻塞其他正常模块；依赖被排除模块的普通模块仍正常参与拓扑排序（排除的依赖不计入 inDegree）
