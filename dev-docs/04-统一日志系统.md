#### 步骤 1：实现 Logger

**src/main/core/logger.ts**：

```typescript
import { app } from 'electron'
import { join } from 'path'
import { mkdir } from 'fs/promises'
import type { Logger, LogLevel } from '../types/logger'
import { FileTransport, ConsoleTransport, type LogTransport } from './log-transport'

const LOG_LEVELS: Record<LogLevel, number> = {
  debug: 0,
  info: 1,
  warn: 2,
  error: 3
}

export class AppLogger implements Logger {
  private level: LogLevel
  private transports: LogTransport[] = []
  private moduleId: string
  private logDir: string
  private errorTransport?: FileTransport
  private initialized: Promise<void>

  constructor(moduleId: string, level: LogLevel = 'info') {
    this.moduleId = moduleId
    this.level = level
    // 注意：app.getPath() 必须在 app.ready 之后调用
    // 推荐在 app.whenReady().then(...) 或 BrowserWindow 创建后再创建 Logger 实例
    this.logDir = join(app.getPath('userData'), 'logs')

    // 异步初始化：确保日志目录存在后再添加传输器
    this.initialized = this.init()
  }

  private async init(): Promise<void> {
    try {
      await mkdir(this.logDir, { recursive: true })
    } catch (err) {
      // 目录已存在时忽略，其他错误打印警告
      if ((err as NodeJS.ErrnoException).code !== 'EEXIST') {
        console.error('[Logger] 创建日志目录失败:', err)
      }
    }

    // 默认传输器
    this.transports.push(new ConsoleTransport(this.moduleId))
    this.transports.push(new FileTransport(this.moduleId, this.logDir))

    // error 级别额外写入 error.log（方便快速排查）
    this.errorTransport = new FileTransport('error', this.logDir)
  }

  setLevel(level: LogLevel): void {
    this.level = level
  }

  debug(message: string, meta?: object): void {
    this.log('debug', message, undefined, meta)
  }

  info(message: string, meta?: object): void {
    this.log('info', message, undefined, meta)
  }

  warn(message: string, meta?: object): void {
    this.log('warn', message, undefined, meta)
  }

  error(message: string, error?: Error, meta?: object): void {
    this.log('error', message, error, meta)
  }

  private async log(level: LogLevel, message: string, error?: Error, meta?: object): Promise<void> {
    // 等待异步初始化完成（日志目录创建 + 传输器注册）
    await this.initialized

    if (LOG_LEVELS[level] < LOG_LEVELS[this.level]) return

    const entry = {
      timestamp: new Date().toISOString(),
      level,
      moduleId: this.moduleId,
      message,
      error: error ? { name: error.name, message: error.message, stack: error.stack } : undefined,
      meta
    }

    for (const transport of this.transports) {
      transport.write(entry)
    }

    // error 级别额外写入 error.log
    if (level === 'error' && this.errorTransport) {
      this.errorTransport.write(entry)
    }
  }
}

/** 创建模块级别的 Logger */
export function createLogger(moduleId: string, level?: LogLevel): Logger {
  return new AppLogger(moduleId, level)
}

/**
 * 动态日志级别配置说明：
 *
 * 1. 创建时指定级别：const logger = createLogger('myModule', 'debug')
 * 2. 运行时动态修改：logger.setLevel('warn')  // 只输出 warn 和 error
 * 3. 可通过配置管理器联动：
 *    config.onChange('logLevel', (level) => logger.setLevel(level as LogLevel))
 *
 * 级别从低到高：debug → info → warn → error
 * 设置某级别后，仅输出该级别及以上的日志
 */
```

#### 步骤 2：实现传输器

**src/main/core/log-transport.ts**：

```typescript
import { appendFile } from 'fs/promises'
import { join } from 'path'
import { LogRotationManager } from './log-rotation'

export interface LogEntry {
  timestamp: string
  level: string
  moduleId: string
  message: string
  error?: { name: string; message: string; stack?: string }
  meta?: object
}

export interface LogTransport {
  write(entry: LogEntry): void | Promise<void>
}

/** 控制台传输器 */
export class ConsoleTransport implements LogTransport {
  constructor(private moduleId: string) {}

  write(entry: LogEntry): void {
    const prefix = `[${entry.timestamp}] [${entry.level.toUpperCase()}] [${this.moduleId}]`
    const msg = `${prefix} ${entry.message}`

    switch (entry.level) {
      case 'debug':
        console.debug(msg, entry.meta || '')
        break
      case 'info':
        console.info(msg, entry.meta || '')
        break
      case 'warn':
        console.warn(msg, entry.meta || '')
        break
      case 'error':
        console.error(msg, entry.error || '', entry.meta || '')
        break
    }
  }
}

/** 文件传输器（委托 LogRotationManager 处理轮转） */
export class FileTransport implements LogTransport {
  private filePath: string
  private rotation: LogRotationManager
  private writeQueue: Promise<void> = Promise.resolve()

  constructor(moduleId: string, logDir: string) {
    this.filePath = join(logDir, `${moduleId}.log`)
    this.rotation = new LogRotationManager()
  }

  write(entry: LogEntry): void {
    // 使用 Promise 链串行写入，避免并发写入导致文件损坏
    this.writeQueue = this.writeQueue.then(() => this.doWrite(entry)).catch((err) => {
      // 写入失败时记录错误但不中断后续日志写入
      console.error(`[LogTransport] 写入失败 [${this.filePath}]:`, err)
      // 备用方案：尝试写入控制台，确保日志不丢失
      console.warn(`[LogTransport] 降级输出: ${JSON.stringify(entry)}`)
    })
  }

  private async doWrite(entry: LogEntry): Promise<void> {
    await this.rotation.rotateIfNeeded(this.filePath)
    const line = JSON.stringify(entry) + '\n'
    await appendFile(this.filePath, line, 'utf-8')
  }
}
```

#### 步骤 2.5：实现日志轮转策略（独立模块）

**src/main/core/log-rotation.ts**：

```typescript
import { access, stat, rename, unlink } from 'fs/promises'

/** 轮转配置 */
export interface RotationConfig {
  /** 单文件最大字节数（默认 10MB） */
  maxSizeBytes: number
  /** 最大保留文件数（默认 5） */
  maxFiles: number
  /** 是否压缩归档（默认 false，未来扩展） */
  compress: boolean
}

const DEFAULT_CONFIG: RotationConfig = {
  maxSizeBytes: 10 * 1024 * 1024,
  maxFiles: 5,
  compress: false
}

/**
 * 日志轮转管理器
 * 独立于 FileTransport，可被其他传输器复用
 */
export class LogRotationManager {
  private config: RotationConfig
  private rotating = false

  constructor(config: Partial<RotationConfig> = {}) {
    this.config = { ...DEFAULT_CONFIG, ...config }
  }

  /**
   * 检查是否需要轮转（不含执行）
   * @param filePath 当前日志文件路径
   * @returns true 表示文件已超过 maxSizeBytes
   */
  async checkRotation(filePath: string): Promise<boolean> {
    try {
      await access(filePath)
    } catch {
      return false
    }
    const stats = await stat(filePath)
    return stats.size >= this.config.maxSizeBytes
  }

  /**
   * 检查并执行轮转
   * @param filePath 当前日志文件路径
   * @returns 是否执行了轮转
   */
  async rotateIfNeeded(filePath: string): Promise<boolean> {
    if (this.rotating) return false

    const needsRotation = await this.checkRotation(filePath)
    if (!needsRotation) return false

    try {
      this.rotating = true
      await this.rotate(filePath)
      return true
    } catch (err) {
      // 轮转失败不应阻塞日志写入，记录错误后继续
      console.error('[LogRotation] 轮转失败，日志将继续追加到当前文件:', err)
      // 可选：通过事件通知上层监控系统
      // this.emit('rotation-error', { filePath, error: err })
      return false
    } finally {
      this.rotating = false
    }
  }

  /**
   * 执行文件轮转
   * 策略：从最老的文件开始，依次重命名或删除
   * 修复了中间序号文件不存在时导致日志丢失的bug：
   *   即使中间某个 .N 文件缺失，也不会阻断后续文件的重命名
   */
  async rotate(filePath: string): Promise<void> {
    // 先删除最老的文件
    const oldestFile = `${filePath}.${this.config.maxFiles}`
    try {
      await access(oldestFile)
      await unlink(oldestFile)
    } catch {
      // 文件不存在，跳过
    }

    // 从次老的开始，依次重命名（序号 +1）
    // 即使中间某个文件不存在（如 .log.2 缺失），也不影响 .log.3 → .log.4
    for (let i = this.config.maxFiles - 1; i >= 1; i--) {
      const from = `${filePath}.${i}`
      const to = `${filePath}.${i + 1}`
      try {
        await access(from)
        await rename(from, to)
      } catch {
        // 文件不存在，跳过（不会阻断后续文件）
      }
    }

    // 当前文件 → .1
    await rename(filePath, `${filePath}.1`)

    // TODO: 日志压缩归档（未来扩展）
    // 当 this.config.compress 为 true 时，轮转后的旧日志文件应使用 zlib.gzip 压缩：
    // 1. 读取重命名后的文件内容
    // 2. 使用 zlib.gzip() 压缩
    // 3. 写入 .gz 文件并删除原文件
    // 4. cleanupOldFiles 需同时清理 .log 和 .log.gz 文件
    //
    // 实现示例：
    // import { gzip } from 'zlib/promises'
    // import { readFile, writeFile, unlink } from 'fs/promises'
    // const data = await readFile(rotatedPath)
    // const compressed = await gzip(data)
    // await writeFile(rotatedPath + '.gz', compressed)
    // await unlink(rotatedPath)
  }

  /**
   * 获取所有日志文件列表（当前文件 + 已轮转文件）
   * @param filePath 当前日志文件路径
   * @returns 按序号从新到老排列的文件路径列表
   */
  async getLogFiles(filePath: string): Promise<string[]> {
    const files: string[] = []

    // 当前文件（无后缀）
    try {
      await access(filePath)
      files.push(filePath)
    } catch {
      // 当前文件不存在
    }

    // 已轮转文件：.1, .2, .3, ...
    for (let i = 1; i <= this.config.maxFiles; i++) {
      const rotated = `${filePath}.${i}`
      try {
        await access(rotated)
        files.push(rotated)
      } catch {
        break
      }
    }

    return files
  }

  /**
   * 获取轮转状态信息（用于监控）
   */
  async getRotationInfo(filePath: string): Promise<{
    currentSize: number
    fileCount: number
    needsRotation: boolean
  }> {
    let currentSize = 0
    let fileCount = 0

    try {
      const stats = await stat(filePath)
      currentSize = stats.size
      fileCount = 1
    } catch {
      // 当前文件不存在
    }

    for (let i = 1; i <= this.config.maxFiles; i++) {
      try {
        await access(`${filePath}.${i}`)
        fileCount++
      } catch {
        break
      }
    }

    return {
      currentSize,
      fileCount,
      needsRotation: currentSize >= this.config.maxSizeBytes
    }
  }
}
```

```typescript
import { describe, it, expect, vi, beforeEach } from 'vitest'

// Mock electron app
vi.mock('electron', () => ({

    vi.clearAllMocks()

  })

    expect(consoleSpy).toHaveBeenCalled()

    consoleSpy.mockRestore()
  })

  it('error 包含错误堆栈', () => {
    const logger = new AppLogger('test', 'debug')
    const consoleSpy = vi.spyOn(console, 'error').mockImplementation(() => {})

    logger.error('failed', new Error('boom'))
    expect(consoleSpy).toHaveBeenCalled()

    consoleSpy.mockRestore()
  })

  it('error 级别日志同时写入主日志和 error.log', async () => {
    const logger = new AppLogger('test-module', 'debug')
    // 等待异步初始化完成（日志目录创建 + 传输器注册）
    await logger['initialized']

    // Mock FileTransport.write 以捕获写入调用
    const mainTransport = logger['transports'].find(
      (t: unknown) => t.constructor.name === 'FileTransport' && (t as { filePath: string }).filePath.includes('test-module.log')
    ) as { write: ReturnType<typeof vi.fn> } | undefined
    const errorTransport = logger['errorTransport'] as unknown as { write: ReturnType<typeof vi.fn> }

    const mainWriteSpy = mainTransport ? vi.spyOn(mainTransport, 'write') : vi.fn()
    const errorWriteSpy = errorTransport ? vi.spyOn(errorTransport, 'write') : vi.fn()

    logger.error('something broke', new Error('boom'))

    // 等待异步写入队列
    await new Promise(r => setTimeout(r, 50))

    // error 应同时写入主日志和 error.log
    expect(mainWriteSpy).toHaveBeenCalled()
    expect(errorWriteSpy).toHaveBeenCalled()

    // 验证 errorTransport 接收到的条目包含正确的错误信息
    const writtenEntry = errorWriteSpy.mock.calls[0]?.[0] as { level: string; message: string } | undefined
    expect(writtenEntry?.level).toBe('error')
    expect(writtenEntry?.message).toBe('something broke')

    mainWriteSpy.mockRestore()
    errorWriteSpy.mockRestore()
  })

  it('info/warn 级别日志不写入 error.log', async () => {
    const logger = new AppLogger('test-module', 'debug')
    await logger['initialized']

    const errorTransport = logger['errorTransport'] as unknown as { write: ReturnType<typeof vi.fn> }
    const errorWriteSpy = vi.spyOn(errorTransport, 'write')

    logger.info('just info')
    logger.warn('just warn')

    await new Promise(r => setTimeout(r, 50))

    // 非 error 级别不应写入 error.log
    expect(errorWriteSpy).not.toHaveBeenCalled()

    errorWriteSpy.mockRestore()
  })
```

### 4.3 代码规范

- **日志格式**：JSON 格式存储，方便后续解析和查询
- **级别使用**：`debug` 调试信息、`info` 正常流程、`warn` 警告但可恢复、`error` 错误需处理
- **级别动态配置**：通过 `Logger.setLevel(level)` 方法可在运行时动态调整日志级别，无需重启应用。适用于调试时临时开启 `debug` 级别，或生产环境降低日志量
- **轮转策略**：单文件 10MB，保留最近 5 个文件
- **轮转优化**：`rotateIfNeeded` 采用写入时检查而非每次 `stat`，通过内部 `rotating` 标志避免并发轮转。高频写入场景下建议增加时间间隔检查（如 10 秒内只检查一次文件大小），减少 `stat` 系统调用开销
- **日志压缩/归档**：`RotationConfig.compress` 字段预留了压缩支持（当前默认 `false`）。未来可通过集成 `zlib` 对轮转后的 `.N` 文件进行 gzip 压缩，生成 `.N.gz` 归档文件，进一步节省磁盘空间
- **错误日志**：`error.log` 单独汇总一份，方便快速排查
  - 实现方式：`AppLogger` 内部维护独立的 `errorTransport`（`FileTransport` 实例）
  - 写入逻辑：`error` 级别日志同时写入模块主日志文件（如 `test-module.log`）和 `error.log`
  - 非 error 级别日志不写入 `error.log`
  - 使用异步 IO（`fs/promises.appendFile`），通过 Promise 链串行写入避免并发损坏
  - `error.log` 独立轮转，与各模块主日志互不影响
