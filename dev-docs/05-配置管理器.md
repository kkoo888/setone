#### 步骤 1：实现配置管理器

**src/main/core/config-manager.ts**：

```typescript
import { app } from 'electron'
import { join } from 'path'
import { readFileSync, writeFileSync, existsSync, mkdirSync } from 'fs'
import { writeFile } from 'fs/promises'
import { EventEmitter } from 'events'
import type { ConfigManager as IConfigManager } from '../types/config'

export class ConfigManagerImpl implements IConfigManager {
  private configPath: string
  private moduleConfigDir: string
  private cache: Map<string, unknown> = new Map()
  private moduleCache: Map<string, Map<string, unknown>> = new Map()
  private changeEmitter = new EventEmitter()

  constructor() {
    const userData = app.getPath('userData')
    this.configPath = join(userData, 'config', 'global.json')
    this.moduleConfigDir = join(userData, 'config', 'modules')
    this.ensureDirectories()
    this.loadFromDisk()
  }

  private ensureDirectories(): void {
    const configDir = join(app.getPath('userData'), 'config')
    mkdirSync(configDir, { recursive: true })
    mkdirSync(this.moduleConfigDir, { recursive: true })
  }

  private loadFromDisk(): void {
    try {
      if (existsSync(this.configPath)) {
        const data = JSON.parse(readFileSync(this.configPath, 'utf-8'))
        for (const [key, value] of Object.entries(data)) {
          this.cache.set(key, value)
        }
      }
    } catch (err) {
      console.error('[ConfigManager] 加载配置失败:', err)
    }
  }

  private async saveToDisk(): Promise<void> {
    try {
      const data = Object.fromEntries(this.cache)
      await writeFile(this.configPath, JSON.stringify(data, null, 2), 'utf-8')
    } catch (err) {
      console.error('[ConfigManager] 保存配置失败:', err)
    }
  }

  async get<T = unknown>(key: string, defaultValue?: T): Promise<T> {
    const keys = key.split('.')
    let current: unknown = Object.fromEntries(this.cache)

    for (const k of keys) {
      if (current && typeof current === 'object' && k in current) {
        current = (current as Record<string, unknown>)[k]
      } else {
        return defaultValue as T
      }
    }

    return current as T
  }

  async set<T = unknown>(key: string, value: T): Promise<void> {
    const keys = key.split('.')
    const root = Object.fromEntries(this.cache)

    let current: Record<string, unknown> = root
    for (let i = 0; i < keys.length - 1; i++) {
      const k = keys[i]
      if (!current[k] || typeof current[k] !== 'object') {
        current[k] = {}
      }
      current = current[k] as Record<string, unknown>
    }

    current[keys[keys.length - 1]] = value

    // 更新缓存
    this.cache.clear()
    for (const [k, v] of Object.entries(root)) {
      this.cache.set(k, v)
    }

    await this.saveToDisk()
    this.changeEmitter.emit('change', key, value)
  }

  async delete(key: string): Promise<void> {
    const keys = key.split('.')

    if (keys.length === 1) {
      // 顶层键：直接删除
      this.cache.delete(key)
    } else {
      // 嵌套键：遍历到父对象，删除目标属性，清理空父对象
      const root = Object.fromEntries(this.cache)
      let current: Record<string, unknown> = root

      // 遍历到倒数第二层（父对象）
      for (let i = 0; i < keys.length - 1; i++) {
        const k = keys[i]
        if (!current[k] || typeof current[k] !== 'object') {
          // 路径不存在，无需删除
          return
        }
        current = current[k] as Record<string, unknown>
      }

      // 删除目标属性
      delete current[keys[keys.length - 1]]

      // 清理空的父对象（从内向外逐层检查）
      const cleanEmptyParents = (
        obj: Record<string, unknown>,
        path: string[],
        depth: number
      ): boolean => {
        if (depth >= path.length - 1) return Object.keys(obj).length === 0
        const child = obj[path[depth]]
        if (!child || typeof child !== 'object') return false
        const shouldRemove = cleanEmptyParents(child as Record<string, unknown>, path, depth + 1)
        if (shouldRemove) {
          delete obj[path[depth]]
        }
        return Object.keys(obj).length === 0
      }
      cleanEmptyParents(root, keys, 0)

      // 回写缓存
      this.cache.clear()
      for (const [k, v] of Object.entries(root)) {
        this.cache.set(k, v)
      }
    }

    await this.saveToDisk()
    this.changeEmitter.emit('change', key, undefined)
  }

  // 模块配置（独立文件存储）
  async getModuleConfig<T = unknown>(moduleId: string, key: string, defaultValue?: T): Promise<T> {
    if (!this.moduleCache.has(moduleId)) {
      this.loadModuleConfig(moduleId)
    }

    const keys = key.split('.')
    let current: unknown = Object.fromEntries(this.moduleCache.get(moduleId) || new Map())

    for (const k of keys) {
      if (current && typeof current === 'object' && k in current) {
        current = (current as Record<string, unknown>)[k]
      } else {
        return defaultValue as T
      }
    }

    return current as T
  }

  async setModuleConfig<T = unknown>(moduleId: string, key: string, value: T): Promise<void> {
    if (!this.moduleCache.has(moduleId)) {
      this.moduleCache.set(moduleId, new Map())
    }

    const cache = this.moduleCache.get(moduleId)!
    cache.set(key, value)
    this.saveModuleConfig(moduleId)
    this.changeEmitter.emit('change', `module.${moduleId}.${key}`, value)
    this.changeEmitter.emit(`module:${moduleId}:${key}`, value)
  }

  async deleteModuleConfig(moduleId: string, key: string): Promise<void> {
    if (!this.moduleCache.has(moduleId)) {
      this.loadModuleConfig(moduleId)
    }

    const cache = this.moduleCache.get(moduleId)
    if (!cache) return

    cache.delete(key)
    this.saveModuleConfig(moduleId)
    this.changeEmitter.emit('change', `module.${moduleId}.${key}`, undefined)
    this.changeEmitter.emit(`module:${moduleId}:${key}`, undefined)
  }

  private loadModuleConfig(moduleId: string): void {
    const filePath = join(this.moduleConfigDir, `${moduleId}.json`)
    const cache = new Map<string, unknown>()

    try {
      if (existsSync(filePath)) {
        const data = JSON.parse(readFileSync(filePath, 'utf-8'))
        for (const [key, value] of Object.entries(data)) {
          cache.set(key, value)
        }
      }
    } catch (err) {
      console.error(`[ConfigManager] 加载模块 "${moduleId}" 配置失败:`, err)
    }

    this.moduleCache.set(moduleId, cache)
  }

  private saveModuleConfig(moduleId: string): void {
    const filePath = join(this.moduleConfigDir, `${moduleId}.json`)
    const cache = this.moduleCache.get(moduleId)
    if (!cache) return

    try {
      const data = Object.fromEntries(cache)
      writeFileSync(filePath, JSON.stringify(data, null, 2), 'utf-8')
    } catch (err) {
      console.error(`[ConfigManager] 保存模块 "${moduleId}" 配置失败:`, err)
    }
  }

  onChange(callback: (key: string, value: unknown) => void): () => void {
    this.changeEmitter.on('change', callback)
    return () => {
      this.changeEmitter.off('change', callback)
    }
  }
}
```

#### 步骤 2：编写测试

**tests/unit/config-manager.test.ts**：

```typescript
import { describe, it, expect, vi, beforeEach } from 'vitest'

// Mock electron
vi.mock('electron', () => ({
  app: { getPath: () => '/tmp/test-config' }
}))

describe('ConfigManager', () => {
  it('get 返回默认值当键不存在', async () => {
    const { ConfigManagerImpl } = await import('../../src/main/core/config-manager')
    const config = new ConfigManagerImpl()
    const value = await config.get('nonexistent', 'default')
    expect(value).toBe('default')
  })

  it('set/get 基本读写', async () => {
    const { ConfigManagerImpl } = await import('../../src/main/core/config-manager')
    const config = new ConfigManagerImpl()
    await config.set('test.key', 'hello')
    const value = await config.get('test.key')
    expect(value).toBe('hello')
  })

  it('delete 支持嵌套键', async () => {
    const { ConfigManagerImpl } = await import('../../src/main/core/config-manager')
    const config = new ConfigManagerImpl()
    await config.set('a.b.c', 'value')
    await config.set('a.b.d', 'keep')
    await config.delete('a.b.c')
    expect(await config.get('a.b.c')).toBeUndefined()
    expect(await config.get('a.b.d')).toBe('keep')
  })

  it('delete 嵌套键后清理空父对象', async () => {
    const { ConfigManagerImpl } = await import('../../src/main/core/config-manager')
    const config = new ConfigManagerImpl()
    await config.set('x.y.z', 'only')
    await config.delete('x.y.z')
    expect(await config.get('x')).toBeUndefined()
  })

  it('onChange 触发回调', async () => {
    const { ConfigManagerImpl } = await import('../../src/main/core/config-manager')
    const config = new ConfigManagerImpl()
    const callback = vi.fn()

    config.onChange(callback)
    await config.set('key', 'value')

    expect(callback).toHaveBeenCalledWith('key', 'value')
  })

  it('onChange 监听模块配置变更', async () => {
    const { ConfigManagerImpl } = await import('../../src/main/core/config-manager')
    const config = new ConfigManagerImpl()
    const callback = vi.fn()

    config.onChange(callback)
    await config.setModuleConfig('myModule', 'apiKey', 'abc123')

    expect(callback).toHaveBeenCalledWith('module.myModule.apiKey', 'abc123')
  })
})
```

### 5.3 代码规范

- **配置格式**：JSON 文件存储，便于人工查看和编辑
- **模块隔离**：每个模块独立配置文件 `modules/{moduleId}.json`
- **缓存策略**：内存缓存 + 异步写盘，避免频繁 IO
- **变更通知**：通过 EventEmitter 通知配置变更，支持监听和取消
- **默认值**：`get()` 支持 `defaultValue` 参数，避免 `undefined` 传播
