```typescript
/** 模块元数据（来自 module.json） */
export interface ModuleMeta {
  id: string
  name: string
  version: string
  description: string
  author: string
  enabled: boolean
  dependencies: string[]
  hostVersion: string
  priority: number
  resourceLimits: ResourceLimits
  provides: CapabilityDeclaration[]
  consumes: ConsumesDeclaration[]
  settings: Record<string, unknown>
}

/** 资源限制 */
export interface ResourceLimits {
  maxMemoryMB: number
  maxCpuPercent: number
}

/** 能力声明（module.json 中的 provides） */
export interface CapabilityDeclaration {
  type: 'tool' | 'event' | 'ui' | 'service'
  name: string
  description: string
  priority?: number
}

/** 消费声明（module.json 中的 consumes） */
export interface ConsumesDeclaration {
  type: 'event'
  name: string
  from?: string
}

/** 模块接口 */
export interface Module {
  id: string
  meta: ModuleMeta

  /** 激活模块 */
  activate(context: ModuleContext): Promise<void>

  /** 停用模块 */
  deactivate(): Promise<void>

  /** 获取能力列表 */
  getCapabilities(): Capability[]

  /** 获取 UI 组件（可选） */
  getUI?(): React.ComponentType

  /** 获取模块状态（可选） */
  getStatus?(): ModuleStatusReport
}

/**
 * 模块入口文件规范：
 * - 统一命名为 index.ts（位于模块根目录）
 * - 默认导出 Module 接口实现
 * - 模块元数据通过同目录 module.json 声明
 */

/**
 * 模块运行时值（纯数据，由框架注入，模块只读使用）
 * 包含框架级路径、默认值等静态运行时参数，与配置管理逻辑无关。
 */
export interface ModuleRuntimeValues {
  /** 应用数据目录（如 Electron app.getPath('userData')） */
  dataDir: string
  /** 应用安装目录（如 process.resourcesPath） */
  appDir: string
  /** 默认 LLM 模型名称 */
  defaultModel: string
  /** 默认城市（用于天气等模块） */
  defaultCity: string
  /** 额外技能搜索路径 */
  skillSearchPaths: string[]
  /** 任务最大重试次数 */
  taskMaxRetries: number
}

/**
 * 模块运行时配置（组合式设计：运行时值 + 配置管理器）
 *
 * 采用交叉类型组合，而非继承，职责分离如下：
 * - ModuleRuntimeValues：框架注入的静态运行时参数（只读）
 * - ConfigManager：动态配置的读写、监听能力（可调用）
 *
 * 外部 API 保持扁平访问，对模块无感知变化：
 *   context.config.dataDir          // 运行时值（直接访问）
 *   context.config.get('key')       // 配置管理（直接调用）
 */
export type ModuleRuntimeConfig = ModuleRuntimeValues & ConfigManager

/** 模块上下文（注入给模块的服务） */
export interface ModuleContext {
  eventBus: ScopedEventBus
  config: ModuleRuntimeConfig
  ai: AIService
  /** LLM 快捷访问（等价于 ai.chat，但支持指定模型） */
  llm: {
    chat(messages: ChatMessage[], options?: ChatOptions): Promise<ChatResponse>
  }
  logger: Logger
  store: ScopedStore
  /** 数据库实例（SQLite） */
  db: DatabaseManager

  /** 获取其他已注册模块实例 */
  getModule(name: string): Module | undefined
}

/**
 * 模块运行时状态报告（主进程模块管理器内部使用）
 * 包含模块 ID、当前状态、消息及资源占用信息
 */
export interface ModuleStatusReport {
  id: string
  state: ModuleStatus
  message?: string
  resourceUsage?: {
    memoryMB: number
    cpuPercent: number
  }
}

/** 模块注册信息 */
export interface ModuleRegistration {
  meta: ModuleMeta
  instance: Module | null
  status: ModuleRegistrationStatus
  loadError?: string
  hash: string
}

/**
 * 模块注册状态（主进程模块管理器使用）
 * 用于模块生命周期管理（6个状态），与 ModuleStatus 枚举互补：
 * - ModuleRegistrationStatus：模块发现/加载/激活的生命周期（主进程侧，关注"模块从哪来、能不能用"）
 * - ModuleStatus：模块运行时 UI 展示状态（渲染进程侧，关注"模块现在在干什么"）
 * 两者通过模块管理器的 listModules() 方法桥接转换。
 *
 * 状态映射关系：
 * - discovered → IDLE（已发现但未加载）
 * - loading → LOADING（加载中）
 * - active → RUNNING 或 STOPPED（激活后由运行状态决定）
 * - error → ERROR
 * - disabled → DISABLED
 * - incompatible → 不映射到 ModuleStatus（直接标记为不可用）
 */
export type ModuleRegistrationStatus =
  | 'discovered'    // 已发现，未加载
  | 'loading'       // 正在加载
  | 'active'        // 已激活
  | 'error'         // 加载出错
  | 'disabled'      // 用户禁用
  | 'incompatible'  // 版本不兼容

/**
 * 模块状态枚举（统一版）
 * 合并主进程生命周期状态与渲染进程 UI 展示状态
 * - 版块 7（模块管理器）和版块 25（设置面板）共用此枚举
 * - 与 ModuleRegistrationStatus 互补：本枚举关注"模块现在在干什么"（渲染进程侧）
 *   而 ModuleRegistrationStatus 关注"模块从哪来、能不能用"（主进程侧）
 * - 模块管理器的 listModules() 方法负责从 ModuleRegistrationStatus 转换到本枚举
 *
 * 状态映射关系（ModuleRegistrationStatus → ModuleStatus）：
 * - discovered → IDLE（已发现但未加载）
 * - loading → LOADING（加载中）
 * - active → RUNNING 或 STOPPED（激活后由运行状态决定）
 * - error → ERROR
 * - disabled → DISABLED
 * - incompatible → 不映射到本枚举（直接标记为不可用）
 */
export enum ModuleStatus {
  /** 空闲（已加载未激活） */
  IDLE = 'idle',
  /** 运行中 */
  RUNNING = 'running',
  /** 已停止 */
  STOPPED = 'stopped',
  /** 错误 */
  ERROR = 'error',
  /** 加载中 */
  LOADING = 'loading',
  /** 已禁用 */
  DISABLED = 'disabled',
}

// 资源相关类型统一定义在 src/main/core/types/performance.ts
// 此处通过共享类型模块 re-export，避免重复定义
export type {
  ModuleResourceDeclaration,
  ModuleResourceState,
  SystemResourceSnapshot,
} from './performance'

/**
 * 模块信息（统一版）
 * 合并版块 2 的 ModuleMeta（主进程元数据）与版块 25 的 ModuleInfo（渲染进程展示）
 * 供版块 11（Zustand 状态管理）、版块 25（设置面板）等所有消费者共用
 */
export interface ModuleInfo {
  /** 模块唯一标识 */
  readonly id: string;
  /** 模块名称 */
  readonly name: string;
  /** 模块描述 */
  readonly description: string;
  /** 模块版本（Semver） */
  readonly version: string;
  /** 作者 */
  readonly author: string;
  /** 当前状态 */
  readonly status: ModuleStatus;
  /** 是否启用 */
  readonly enabled: boolean;
  /** 资源声明 */
  readonly resources: ModuleResourceDeclaration;
  /** 模块图标（emoji 或 URL） */
  readonly icon: string;
  /** 依赖的模块 ID 列表 */
  readonly dependencies: readonly string[];
  /** 宿主版本要求 */
  readonly hostVersion: string;
  /** 加载优先级 (0-1000) */
  readonly priority: number;
  /** 资源限制 */
  readonly resourceLimits: ResourceLimits;
  /** 提供的能力声明 */
  readonly provides: readonly CapabilityDeclaration[];
  /** 消费声明 */
  readonly consumes: readonly ConsumesDeclaration[];
  /** 模块配置 */
  readonly settings: Record<string, unknown>;
  /** 最后更新时间 */
  readonly lastUpdated: string;
  /** 当前资源使用（运行时，可选） */
  readonly currentUsage?: ModuleResourceState;
}
```

```typescript
/** 事件处理器 */
export type EventHandler<T = unknown> = (data: T) => void | Promise<void>

/** 全局事件总线接口 */
export interface EventBus {
  on<K extends keyof EventMap>(event: K, handler: EventHandler<EventMap[K]>): void
  on(event: string, handler: EventHandler): void  // 向后兼容：允许任意字符串事件
  off<K extends keyof EventMap>(event: K, handler: EventHandler<EventMap[K]>): void
  off(event: string, handler: EventHandler): void  // 向后兼容
  emit<K extends keyof EventMap>(event: K, data: EventMap[K]): void
  emit(event: string, data?: unknown): void  // 向后兼容
  once<K extends keyof EventMap>(event: K, handler: EventHandler<EventMap[K]>): void
  once(event: string, handler: EventHandler): void  // 向后兼容
  removeAllListeners(event?: string): void
}

/** 限定范围的事件总线（模块使用） */
export interface ScopedEventBus {
  on<K extends keyof EventMap>(event: K, handler: EventHandler<EventMap[K]>): void
  on(event: string, handler: EventHandler): void  // 向后兼容
  off<K extends keyof EventMap>(event: K, handler: EventHandler<EventMap[K]>): void
  off(event: string, handler: EventHandler): void  // 向后兼容
  emit<K extends keyof EventMap>(event: K, data: EventMap[K]): void
  emit(event: string, data?: unknown): void  // 向后兼容
}

/** 事件定义（用于类型约束） */
export interface EventMap {
  // 系统事件
  /** 内部错误事件，事件处理器抛出异常时自动触发 */
  _error: { event: string; error: unknown }
  on_module_loaded: { moduleId: string }
  on_module_unloaded: { moduleId: string }
  on_module_error: { moduleId: string; error: string }

  // 对话事件
  on_user_input: { text: string; timestamp: number }
  on_ai_response: { text: string; moduleId?: string }
  on_intent_resolved: { intent: string; confidence: number; moduleId: string }

  // 屏幕事件
  on_screen_change: { analysis: string; timestamp: number }
  on_vision_toggle: { enabled: boolean; mode?: string }
  on_vision_started: { mode: string }
  on_vision_stopped: void // 无数据载荷
  on_perception_request: { question?: string }
  on_perception_result: { analysis?: string; error?: string }

  // 记忆事件
  on_new_memory: { content: string; type: 'short' | 'long' }

  // 窗口事件
  on_toggle_window: void // 无数据载荷

  // 关怀事件
  on_reminder_triggered: { id: string; message: string }
  on_health_check: void // 无数据载荷

  // 内部事件（由 EventBus 自身 emit，不声明在 module.json provides 中）
  _error: { event: string; error: unknown }
}
```

```typescript
import type { ModuleRegistration } from '../../main/types/module'
import type { ChatMessage, ChatResponse, ChatOptions, ChatChunk } from '../../main/types/ai'
import type { LogLevel } from '../../main/types/logger'

/** 屏幕源信息（对应 desktopCapturer.getSources 返回值） */
export interface ScreenSource {

export interface IPCChannels {
  // 模块管理
  'module:list': { request: void; response: ModuleRegistration[] }
  'module:enable': { request: { moduleId: string }; response: boolean }
  'module:disable': { request: { moduleId: string }; response: boolean }
  'module:reload': { request: { moduleId: string }; response: boolean }

  // chatStream 使用事件式流式通信：invoke 发起请求，on 接收 chunks
  // 通过 requestId 区分并发流，channel 命名：ai:chatStream:chunk:${requestId}
  'ai:chatStream': { request: { requestId: string; messages: ChatMessage[]; options?: ChatOptions }; response: void }
  // 以下为动态 channel，TypeScript 无法静态枚举，运行时拼接 requestId：
  // 'ai:chatStream:chunk:${requestId}'  → ChatChunk
  // 'ai:chatStream:done:${requestId}'   → void
  // 'ai:chatStream:error:${requestId}'  → { error: string }

  // 屏幕
  'screen:sources': { request: void; response: ScreenSource[] }
  'vision:toggle': { request: { enabled: boolean; mode?: string }; response: boolean }

  // 日志
  'log:query': { request: { moduleId?: string; level?: LogLevel }; response: LogEntry[] }
}
```

```typescript
export type DeepPartial<T> = {
  [P in keyof T]?: T[P] extends object ? DeepPartial<T[P]> : T[P]
}
```

#### 步骤 10：创建渲染进程类型声明

**src/renderer/src/types/electron.d.ts**：

```typescript
/**
 * Electron API 类型声明
 * 与 src/preload/index.ts 中 contextBridge.exposeInMainWorld 暴露的 API 对应
 * 类型与 src/shared/types/ipc.ts 中的 IPCChannels 定义严格对应
 */

/** IPC 通道名称联合类型（与 src/shared/types/ipc.ts 对应） */
type IPCChannel = keyof IPCChannels

/** 类型安全的 invoke 方法签名 */
type TypedInvoke = {
  <C extends keyof IPCChannels>(
    channel: C,
    ...args: IPCChannels[C]['request'] extends void ? [] : [IPCChannels[C]['request']]
  ): Promise<IPCChannels[C]['response']>
  /** 兜底：允许自定义通道（向后兼容） */
  (channel: string, ...args: unknown[]): Promise<unknown>
}

/** 类型安全的 send 方法签名（无返回值的单向通信） */
type TypedSend = {
  <C extends keyof IPCChannels>(
    channel: C,
    ...args: IPCChannels[C]['request'] extends void ? [] : [IPCChannels[C]['request']]
  ): void
  (channel: string, ...args: unknown[]): void
}

/** 类型安全的 on 监听签名（用于接收主进程推送） */
type TypedOn = {
  <C extends keyof IPCChannels>(
    channel: C,
    callback: (data: IPCChannels[C]['response']) => void
  ): () => void
  (channel: string, callback: (...args: unknown[]) => void): () => void
}

/** 预加载脚本暴露的 Electron API */
interface ElectronAPI {
  /** 当前平台标识（由主进程通过 process.platform 注入） */
  platform: NodeJS.Platform

  /**
   * 双向 IPC 调用（渲染进程 → 主进程，等待返回值）
   * @example
   * const sources = await window.electronAPI.invoke('screen:sources')
   * const result = await window.electronAPI.invoke('config:get', { key: 'ai.model' })
   */
  invoke: TypedInvoke

  /**
   * 单向 IPC 发送（渲染进程 → 主进程，不等待返回值）
   * @example
   * window.electronAPI.invoke('ai:chatStream', { requestId: crypto.randomUUID(), messages })
   */
  send: TypedSend

  /**
   * 监听主进程推送的事件
   * @returns 取消监听的清理函数
   * @example
   * const unsubscribe = window.electronAPI.on(`ai:chatStream:chunk:${requestId}`, (chunk) => { ... })
   * // 组件卸载时取消监听
   * unsubscribe()
   */
  on: TypedOn

  /**
   * 移除指定通道的所有监听器
   * @param channel 事件通道名
   */
  removeAllListeners: (channel: string) => void
}
```

### 2.3 代码规范

- **类型文件**：每个领域一个文件，避免单文件过大
- **导出方式**：统一使用 `export interface` / `export type`，最后通过 `index.ts` 统一导出
- **命名规范**：接口名用 PascalCase，不加 `I` 前缀
- **可选属性**：明确标注 `?`，避免隐式 `undefined`
- **泛型**：合理使用泛型提高复用性，如 `ConfigManager.get<T>()`
