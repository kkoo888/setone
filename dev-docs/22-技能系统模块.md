#### 步骤 2：实现 SKILL.md 解析器 (`SkillMdParser.ts`)

```typescript
// modules/skill/SkillMdParser.ts

import type { SkillMetadata } from './types';

/**
 * SKILL.md 解析器
 * 解析 YAML front matter + Markdown 内容
 */
export class SkillMdParser {
  /**
   * 解析 SKILL.md 文件内容
   */
  parse(content: string, filePath: string, skillDir: string): {
    metadata: SkillMetadata;
    body: string;
  } {
    const { frontMatter, body } = this.extractFrontMatter(content);
    const metadata = this.parseFrontMatter(frontMatter, filePath, skillDir);

    return { metadata, body };
  }

  /**
   * 提取 YAML front matter
   */
  private extractFrontMatter(content: string): {
    frontMatter: string;
    body: string;
  } {
    const match = content.match(/^---\s*\n([\s\S]*?)\n---\s*\n([\s\S]*)$/);

    if (!match) {
      // 无 front matter，尝试从内容提取
      return { frontMatter: '', body: content };
    }

    return {
      frontMatter: match[1],
      body: match[2],
    };
  }

  /**
   * 解析 YAML front matter 为元数据
   * 简易 YAML 解析，避免引入外部依赖
   */
  private parseFrontMatter(
    yaml: string,
    filePath: string,
    skillDir: string
  ): SkillMetadata {
    const fields = this.simpleYamlParse(yaml);

    const name = fields.name ?? this.inferNameFromPath(skillDir);
    const description = fields.description ?? '';

    if (!name) {
      throw new Error(`SKILL.md 缺少必要字段 name: ${filePath}`);
    }

    return {
      name,
      description,
      version: fields.version ?? '1.0.0',
      author: fields.author,
      skillDir,
      skillMdPath: filePath,
      permissions: this.parseArrayField(fields.permissions),
      dependencies: this.parseArrayField(fields.dependencies),
      tags: this.parseArrayField(fields.tags),
    };
  }

  /**
   * 简易 YAML 键值对解析（支持单行值、数组、多行字符串）
   */
  private simpleYamlParse(yaml: string): Record<string, string | string[]> {
    const result: Record<string, string | string[]> = {};
    const lines = yaml.split('\n');
    let i = 0;

    while (i < lines.length) {
      const line = lines[i];
      const trimmed = line.trim();
      i++;

      // 跳过空行和注释
      if (!trimmed || trimmed.startsWith('#')) continue;

      // 跳过纯数组行（属于上一个 key 的续行，已在下方处理）
      if (trimmed.startsWith('- ') && !trimmed.includes(':')) continue;

      const colonIndex = trimmed.indexOf(':');
      if (colonIndex === -1) continue;

      const key = trimmed.slice(0, colonIndex).trim();
      let value = trimmed.slice(colonIndex + 1).trim();

      // --- 情况 1：值为 '|' 或 '>'，进入多行字符串模式 ---
      if (value === '|' || value === '>') {
        const blockLines: string[] = [];
        // 计算缩进基准（下一个非空行的缩进）
        while (i < lines.length) {
          const nextLine = lines[i];
          const nextTrimmed = nextLine.trim();
          // 空行保留（对 | 模式有意义）
          if (nextLine.trim() === '') {
            blockLines.push('');
            i++;
            continue;
          }
          // 如果缩进小于等于 key 行缩进，说明块结束
          const keyIndent = line.length - line.trimStart().length;
          const nextIndent = nextLine.length - nextLine.trimStart().length;
          if (nextIndent <= keyIndent) break;
          // 去掉基准缩进（取第一个内容行的缩进作为基准）
          const baseIndent = blockLines.length === 0
            ? nextIndent
            : (lines[i - 1] && lines[i - 1].trim() !== ''
                ? lines[i - 1].length - lines[i - 1].trimStart().length
                : nextIndent);
          blockLines.push(nextLine.slice(Math.min(nextIndent, baseIndent)));
          i++;
        }
        // '|' 保留换行，'>' 将换行折叠为空格
        if (value === '>') {
          result[key] = blockLines
            .join(' ')
            .replace(/\s+/g, ' ')
            .trim();
        } else {
          result[key] = blockLines.join('\n').trimEnd();
        }
        continue;
      }

      // --- 情况 2：值为空，检查下一行是否是数组 ---
      if (value === '') {
        const arrayItems: string[] = [];
        while (i < lines.length) {
          const nextLine = lines[i];
          const nextTrimmed = nextLine.trim();
          if (!nextTrimmed || nextTrimmed.startsWith('#')) {
            i++;
            continue;
          }
          if (nextTrimmed.startsWith('- ')) {
            let item = nextTrimmed.slice(2).trim();
            // 去除引号
            if (
              (item.startsWith('"') && item.endsWith('"')) ||
              (item.startsWith("'") && item.endsWith("'"))
            ) {
              item = item.slice(1, -1);
            }
            arrayItems.push(item);
            i++;
          } else {
            break; // 非数组行，结束
          }
        }
        if (arrayItems.length > 0) {
          result[key] = arrayItems;
          continue;
        }
      }

      // --- 情况 3：普通单行值 ---
      // 去除引号
      if (
        (value.startsWith('"') && value.endsWith('"')) ||
        (value.startsWith("'") && value.endsWith("'"))
      ) {
        value = value.slice(1, -1);
      }

      result[key] = value;
    }

    return result;
  }

  /**
   * 解析数组字段（YAML 列表格式或逗号分隔）
   */
  private parseArrayField(value: string | string[] | undefined): string[] | undefined {
    if (!value) return undefined;

    // 如果 simpleYamlParse 已经解析为数组，直接返回
    if (Array.isArray(value)) return value;

    // 支持 [a, b, c] 格式
    if (value.startsWith('[') && value.endsWith(']')) {
      return value
        .slice(1, -1)
        .split(',')
        .map((s) => s.trim().replace(/['"]/g, ''))
        .filter(Boolean);
    }

    // 支持逗号分隔
    return value
      .split(',')
      .map((s) => s.trim())
      .filter(Boolean);
  }

  /**
   * 从路径推断技能名称
   */
  private inferNameFromPath(skillDir: string): string {
    const parts = skillDir.split(/[/\\]/);
    return parts[parts.length - 1] ?? 'unknown';
  }
}
```

```typescript
// modules/skill/SkillDiscovery.ts

import * as fs from 'node:fs/promises';
import * as path from 'node:path';
import type { SkillMetadata } from './types';
import { SkillMdParser } from './SkillMdParser';

/**
 * 技能发现器
 * 扫描指定目录，发现所有包含 SKILL.md 的技能
 */
export class SkillDiscovery {
  private readonly searchPaths: readonly string[];
  private readonly parser: SkillMdParser;

  constructor(searchPaths: readonly string[]) {
    this.searchPaths = searchPaths;
    this.parser = new SkillMdParser();
  }

  /**
   * 扫描所有搜索路径，发现技能
   */
  async discover(): Promise<SkillMetadata[]> {
    const allSkills: SkillMetadata[] = [];

    for (const searchPath of this.searchPaths) {
      try {
        const skills = await this.scanDirectory(searchPath);
        allSkills.push(...skills);
      } catch (error) {
        // 搜索路径不存在时跳过
        if ((error as NodeJS.ErrnoException).code !== 'ENOENT') {
          throw error;
        }
      }
    }

    // 按名称去重（先发现的优先）
    const seen = new Set<string>();
    return allSkills.filter((skill) => {
      if (seen.has(skill.name)) return false;
      seen.add(skill.name);
      return true;
    });
  }

  /**
   * 扫描单个目录
   */
  private async scanDirectory(dirPath: string): Promise<SkillMetadata[]> {
    const skills: SkillMetadata[] = [];

    try {
      await fs.access(dirPath);
    } catch {
      return skills;
    }

    const entries = await fs.readdir(dirPath, { withFileTypes: true });

    for (const entry of entries) {
      if (!entry.isDirectory()) continue;

      const skillDir = path.join(dirPath, entry.name);
      const skillMdPath = path.join(skillDir, 'SKILL.md');

      try {
        await fs.access(skillMdPath);
        const content = await fs.readFile(skillMdPath, 'utf-8');
        const { metadata } = this.parser.parse(content, skillMdPath, skillDir);
        skills.push(metadata);
      } catch {
        // 跳过无 SKILL.md 或解析失败的目录
      }
    }

    return skills;
  }

  /**
   * 单独发现一个技能目录
   */
  async discoverOne(skillDir: string): Promise<SkillMetadata | null> {
    const skillMdPath = path.join(skillDir, 'SKILL.md');

    try {
      const content = await fs.readFile(skillMdPath, 'utf-8');
      const { metadata } = this.parser.parse(content, skillMdPath, skillDir);
      return metadata;
    } catch {
      return null;
    }
  }
}
```

```typescript
// modules/skill/SkillLoader.ts

import * as fs from 'node:fs/promises';
import * as path from 'node:path';
import type { ModuleContext } from '../types';
import type { Skill, SkillMetadata, SkillLevelType } from './types';
import { SkillLevel, SkillState } from './types';
import { SkillMdParser } from './SkillMdParser';

/**
 * 技能加载器
 * 实现三级渐进加载机制：
 * Level 1 (META): 仅加载元数据（名称、描述、版本）
 * Level 2 (DESCRIPTION): 加载完整 SKILL.md 内容
 * Level 3 (FULL): 加载执行脚本和依赖
 */
export class SkillLoader {
  private readonly context: ModuleContext;
  private readonly parser: SkillMdParser;

  constructor(context: ModuleContext) {
    this.context = context;
    this.parser = new SkillMdParser();
  }

  /**
   * 加载技能到指定级别
   */
  async load(metadata: SkillMetadata, targetLevel: SkillLevelType): Promise<Skill> {
    const skill: Skill = {
      ...metadata,
      level: SkillLevel.META,
      state: SkillState.DISCOVERED,
    };

    try {
      // Level 1: META（已在 discovery 阶段完成）
      skill.level = SkillLevel.META;

      if (targetLevel === SkillLevel.META) {
        skill.state = SkillState.LOADED;
        skill.loadedAt = Date.now();
        return skill;
      }

      // Level 2: DESCRIPTION
      await this.loadDescription(skill);
      skill.level = SkillLevel.DESCRIPTION;

      if (targetLevel === SkillLevel.DESCRIPTION) {
        skill.state = SkillState.LOADED;
        skill.loadedAt = Date.now();
        return skill;
      }

      // Level 3: FULL
      await this.loadFull(skill);
      skill.level = SkillLevel.FULL;
      skill.state = SkillState.ACTIVE;
      skill.loadedAt = Date.now();

      this.context.logger.info(`[SkillLoader] 技能 "${skill.name}" 完全加载完成`);
    } catch (error) {
      skill.state = SkillState.ERROR;
      skill.error = error instanceof Error ? error.message : String(error);
      this.context.logger.error(
        `[SkillLoader] 加载技能 "${skill.name}" 失败: ${skill.error}`
      );
    }

    return skill;
  }

  /**
   * 卸载技能
   */
  async unload(skill: Skill): Promise<void> {
    skill.state = SkillState.UNLOADED;
    skill.level = SkillLevel.META;
    skill.content = undefined;
    skill.scriptPath = undefined;
    skill.entryPoint = undefined;

    this.context.logger.info(`[SkillLoader] 技能 "${skill.name}" 已卸载`);
  }

  /**
   * Level 2: 加载 SKILL.md 完整内容
   */
  private async loadDescription(skill: Skill): Promise<void> {
    const content = await fs.readFile(skill.skillMdPath, 'utf-8');
    const { metadata, body } = this.parser.parse(
      content,
      skill.skillMdPath,
      skill.skillDir
    );

    // 更新元数据（可能解析出更多信息）
    Object.assign(skill, {
      description: metadata.description || skill.description,
      permissions: metadata.permissions ?? skill.permissions,
      dependencies: metadata.dependencies ?? skill.dependencies,
      tags: metadata.tags ?? skill.tags,
    });

    skill.content = body;
  }

  /**
   * Level 3: 加载执行脚本和依赖
   */
  private async loadFull(skill: Skill): Promise<void> {
    // 检查依赖
    if (skill.dependencies && skill.dependencies.length > 0) {
      await this.checkDependencies(skill);
    }

    // 查找执行脚本
    const scriptPath = await this.findEntryPoint(skill.skillDir);
    if (scriptPath) {
      skill.scriptPath = scriptPath;
      skill.entryPoint = await this.detectEntryPoint(scriptPath);
    }
  }

  /**
   * 检查技能依赖
   */
  private async checkDependencies(skill: Skill): Promise<void> {
    if (!skill.dependencies) return;

    const registry = this.context.getModule?.('skill') as
      | { hasSkill: (name: string) => boolean }
      | undefined;

    if (!registry) return;

    for (const dep of skill.dependencies) {
      if (!registry.hasSkill(dep)) {
        this.context.logger.warn(
          `[SkillLoader] 技能 "${skill.name}" 依赖 "${dep}" 未加载`
        );
      }
    }
  }

  /**
   * 查找技能入口脚本
   * 优先级：index.ts > index.js > main.ts > main.js
   */
  private async findEntryPoint(skillDir: string): Promise<string | null> {
    const candidates = ['index.ts', 'index.js', 'main.ts', 'main.js', 'run.sh'];

    for (const candidate of candidates) {
      const fullPath = path.join(skillDir, candidate);
      try {
        await fs.access(fullPath);
        return fullPath;
      } catch {
        continue;
      }
    }

    return null;
  }

  /**
   * 检测入口点（从脚本中提取导出的主函数）
   */
  private async detectEntryPoint(scriptPath: string): Promise<string> {
    const content = await fs.readFile(scriptPath, 'utf-8');

    // 检查默认导出
    if (content.includes('export default')) return 'default';
    if (content.includes('export async function run')) return 'run';
    if (content.includes('export async function main')) return 'main';
    if (content.includes('module.exports')) return 'module.exports';

    return 'default';
  }
}
```

#### 步骤 5：实现技能执行器 (`SkillExecutor.ts`)

```typescript
// modules/skill/SkillExecutor.ts

import { Worker } from 'node:worker_threads';
import * as path from 'node:path';
import type { ModuleContext } from '../types';
import type { Skill, SkillRunParams, SkillRunResult } from './types';
import { SkillLevel, SkillState } from './types';

/**
 * 技能执行器
 * 在隔离环境中执行技能脚本，支持超时和错误捕获
 */
export class SkillExecutor {
  private readonly context: ModuleContext;
  private readonly defaultTimeoutMs: number;

  constructor(context: ModuleContext, defaultTimeoutMs = 30000) {
    this.context = context;
    this.defaultTimeoutMs = defaultTimeoutMs;
  }

  /**
   * 执行技能
   */
  async run(skill: Skill, params: SkillRunParams): Promise<SkillRunResult> {
    const startTime = Date.now();
    const timeoutMs = params.timeoutMs ?? this.defaultTimeoutMs;

    // 确保技能已完全加载
    if (skill.level !== SkillLevel.FULL) {
      throw new Error(
        `技能 "${skill.name}" 未完全加载（当前级别: ${skill.level}）`
      );
    }

    if (skill.state !== SkillState.ACTIVE) {
      throw new Error(
        `技能 "${skill.name}" 未激活（当前状态: ${skill.state}）`
      );
    }

    try {
      const output = await this.executeWithTimeout(skill, params.params, timeoutMs);

      skill.lastExecutedAt = Date.now();

      return {
        success: true,
        output,
        durationMs: Date.now() - startTime,
      };
    } catch (error) {
      const errorMessage =
        error instanceof Error ? error.message : String(error);

      return {
        success: false,
        error: errorMessage,
        durationMs: Date.now() - startTime,
      };
    }
  }

  /**
   * 带超时的执行
   */
  private async executeWithTimeout(
    skill: Skill,
    params: Record<string, unknown> | undefined,
    timeoutMs: number
  ): Promise<unknown> {
    if (!skill.scriptPath) {
      // 无执行脚本，返回 SKILL.md 内容作为结果
      return { type: 'content', content: skill.content };
    }

    const ext = path.extname(skill.scriptPath);

    if (ext === '.sh') {
      return this.executeShellScript(skill.scriptPath, params, timeoutMs);
    }

    if (ext === '.ts' || ext === '.js') {
      return this.executeInWorker(skill.scriptPath, skill.entryPoint ?? 'default', params, timeoutMs);
    }

    throw new Error(`不支持的脚本类型: ${ext}`);
  }

  /**
   * 在 Worker 线程中执行 JS/TS 脚本
   */
  private executeInWorker(
    scriptPath: string,
    entryPoint: string,
    params: Record<string, unknown> | undefined,
    timeoutMs: number
  ): Promise<unknown> {
    return new Promise((resolve, reject) => {
      const worker = new Worker(
        this.buildWorkerScript(scriptPath, entryPoint, params),
        { eval: true }
      );

      const timer = setTimeout(() => {
        worker.terminate();
        reject(new Error(`技能执行超时 (${timeoutMs}ms)`));
      }, timeoutMs);

      worker.on('message', (result: { success: boolean; data?: unknown; error?: string }) => {
        clearTimeout(timer);
        if (result.success) {
          resolve(result.data);
        } else {
          reject(new Error(result.error ?? 'Unknown worker error'));
        }
      });

      worker.on('error', (error) => {
        clearTimeout(timer);
        reject(error);
      });

      worker.on('exit', (code) => {
        clearTimeout(timer);
        if (code !== 0) {
          reject(new Error(`Worker 退出，代码: ${code}`));
        }
      });
    });
  }

  /**
   * 构建 Worker 脚本
   */
  private buildWorkerScript(
    scriptPath: string,
    entryPoint: string,
    params: Record<string, unknown> | undefined
  ): string {
    return `
      const parentPort = process.parentPort;
      if (!parentPort) {
        throw new Error('process.parentPort 不可用，当前进程可能不是 utilityProcess');
      }
      const script = require(${JSON.stringify(scriptPath)});

      async function main() {
        try {
          let fn;
          if ('${entryPoint}' === 'default') {
            fn = script.default ?? script;
          } else {
            fn = script['${entryPoint}'];
          }

          if (typeof fn !== 'function') {
            throw new Error('入口点不是函数: ${entryPoint}');
          }

          const result = await fn(${JSON.stringify(params ?? {})});
          parentPort.postMessage({ success: true, data: result });
        } catch (error) {
          parentPort.postMessage({
            success: false,
            error: error instanceof Error ? error.message : String(error)
          });
        }
      }

      main();
    `;
  }

  /**
   * 执行 Shell 脚本
   */
  private async executeShellScript(
    scriptPath: string,
    params: Record<string, unknown> | undefined,
    timeoutMs: number
  ): Promise<unknown> {
    const { execFile } = await import('node:child_process');
    const { promisify } = await import('node:util');
    const execFileAsync = promisify(execFile);

    const env: Record<string, string> = { ...process.env as Record<string, string> };
    if (params) {
      for (const [key, value] of Object.entries(params)) {
        env[`SKILL_${key.toUpperCase()}`] = String(value);
      }
    }

    const { stdout, stderr } = await execFileAsync('bash', [scriptPath], {
      env,
      timeout: timeoutMs,
    });

    return { stdout: stdout.trim(), stderr: stderr.trim() };
  }
}
```

```typescript
// modules/skill/SkillRegistry.ts

import type { Skill, SkillLevelType } from './types';

/**
 * 技能注册表
 * 管理所有已发现和已加载的技能
 */
export class SkillRegistry {
  private readonly skills: Map<string, Skill> = new Map();

  /**
   * 注册技能
   */
  register(skill: Skill): void {
    this.skills.set(skill.name, skill);
  }

  /**
   * 注销技能
   */
  unregister(name: string): boolean {
    return this.skills.delete(name);
  }

  /**
   * 获取技能
   */
  get(name: string): Skill | undefined {
    return this.skills.get(name);
  }

  /**
   * 检查技能是否存在
   */
  has(name: string): boolean {
    return this.skills.has(name);
  }

  /**
   * 获取所有技能
   */
  getAll(): Skill[] {
    return Array.from(this.skills.values());
  }

  /**
   * 按状态筛选
   */
  getByState(state: Skill['state']): Skill[] {
    return this.getAll().filter((s) => s.state === state);
  }

  /**
   * 按加载级别筛选
   */
  getByLevel(level: SkillLevelType): Skill[] {
    return this.getAll().filter((s) => s.level === level);
  }

  /**
   * 按标签搜索
   */
  searchByTag(tag: string): Skill[] {
    return this.getAll().filter((s) => s.tags?.includes(tag));
  }

  /**
   * 按关键词搜索（名称 + 描述）
   */
  search(query: string): Skill[] {
    const lowerQuery = query.toLowerCase();
    return this.getAll().filter(
      (s) =>
        s.name.toLowerCase().includes(lowerQuery) ||
        s.description.toLowerCase().includes(lowerQuery) ||
        s.tags?.some((t) => t.toLowerCase().includes(lowerQuery))
    );
  }

  /**
   * 获取统计信息
   */
  getStats(): {
    total: number;
    byState: Record<string, number>;
    byLevel: Record<string, number>;
  } {
    const byState: Record<string, number> = {};
    const byLevel: Record<string, number> = {};

    for (const skill of this.skills.values()) {
      byState[skill.state] = (byState[skill.state] ?? 0) + 1;
      byLevel[skill.level] = (byLevel[skill.level] ?? 0) + 1;
    }

    return {
      total: this.skills.size,
      byState,
      byLevel,
    };
  }
}
```

#### 步骤 7：实现模块主类 (`SkillModule.ts`)

```typescript
// modules/skill/SkillModule.ts

import * as path from 'node:path';
import type { Module, ModuleContext, Capability } from '../types';
import type {
  Skill,
  SkillMetadata,
  SkillLevelType,
  SkillRunParams,
  SkillRunResult,
  SkillManageParams,
} from './types';
import { SkillLevel, SkillState, SkillManageAction } from './types';
import { SkillDiscovery } from './SkillDiscovery';
import { SkillLoader } from './SkillLoader';
import { SkillExecutor } from './SkillExecutor';
import { SkillRegistry } from './SkillRegistry';

/**
 * 技能系统模块
 * 提供插件式技能扩展、SKILL.md 解析、三级加载机制
 */
export class SkillModule implements Module {
  readonly id = 'skill';
  readonly version = '1.0.0';
  readonly description = '插件式技能扩展系统';

  private context!: ModuleContext;
  private discovery!: SkillDiscovery;
  private loader!: SkillLoader;
  private executor!: SkillExecutor;
  private registry!: SkillRegistry;

  getCapabilities(): Capability[] {
    return [
      {
        type: 'tool',
        name: 'skill_load',
        description: '加载指定技能',
        priority: 10,
        moduleId: 'skill',
        params: {
          skillName: '技能名称',
          level: '加载级别 (meta/description/full)',
        },
      },
      {
        type: 'tool',
        name: 'skill_run',
        description: '执行已加载的技能',
        priority: 10,
        moduleId: 'skill',
        params: {
          skillName: '技能名称',
          params: '传入参数',
          timeoutMs: '超时时间',
        },
      },
      {
        type: 'tool',
        name: 'skill_manage',
        description: '管理技能（列表、信息、加载、卸载、搜索）',
        priority: 5,
        moduleId: 'skill',
        params: {
          action: '操作类型',
          skillName: '技能名称',
          query: '搜索关键词',
        },
      },
    ];
  }

  async activate(context: ModuleContext): Promise<void> {
    this.context = context;

    const searchPaths = this.getSearchPaths();
    this.discovery = new SkillDiscovery(searchPaths);
    this.loader = new SkillLoader(context);
    this.executor = new SkillExecutor(context);
    this.registry = new SkillRegistry();

    // 自动发现技能（仅加载元数据级别）
    await this.autoDiscover();

    context.logger.info(
      `[SkillModule] 初始化完成，发现 ${this.registry.getAll().length} 个技能`
    );
  }

  async execute(capabilityName: string, params: unknown): Promise<unknown> {
    switch (capabilityName) {
      case 'skill_load':
        return this.handleLoad(
          (params as { skillName: string; level?: SkillLevelType })
        );
      case 'skill_run':
        return this.handleRun(params as SkillRunParams);
      case 'skill_manage':
        return this.handleManage(params as SkillManageParams);
      default:
        throw new Error(`[SkillModule] 未知能力: ${capabilityName}`);
    }
  }

  async deactivate(): Promise<void> {
    // 卸载所有已加载的技能
    for (const skill of this.registry.getAll()) {
      if (skill.state === SkillState.ACTIVE) {
        await this.loader.unload(skill);
      }
    }
    this.context.logger.info('[SkillModule] 模块销毁');
  }

  /**
   * 检查技能是否存在
   */
  hasSkill(name: string): boolean {
    return this.registry.has(name);
  }

  /**
   * 获取搜索路径
   */
  private getSearchPaths(): string[] {
    const paths: string[] = [];

    // 内置技能目录
    paths.push(path.join(this.context.config.appDir, 'skills'));

    // 用户自定义技能目录
    paths.push(path.join(this.context.config.dataDir, 'skills'));

    // 配置中的额外路径
    const extraPaths = this.context.config.skillSearchPaths;
    if (Array.isArray(extraPaths)) {
      paths.push(...extraPaths);
    }

    return paths;
  }

  /**
   * 自动发现并注册技能
   */
  private async autoDiscover(): Promise<void> {
    const discovered = await this.discovery.discover();

    for (const metadata of discovered) {
      const skill: Skill = {
        ...metadata,
        level: SkillLevel.META,
        state: SkillState.DISCOVERED,
      };
      this.registry.register(skill);
    }
  }

  /**
   * 处理技能加载
   */
  private async handleLoad(params: {
    skillName: string;
    level?: SkillLevelType;
  }): Promise<Skill> {
    const existing = this.registry.get(params.skillName);

    if (existing && existing.state === SkillState.ACTIVE) {
      this.context.logger.info(`[SkillModule] 技能 "${params.skillName}" 已加载`);
      return existing;
    }

    let metadata: SkillMetadata;

    if (existing) {
      metadata = existing;
    } else {
      // 尝试发现新技能
      const discovered = await this.discovery.discover();
      const found = discovered.find((s) => s.name === params.skillName);
      if (!found) {
        throw new Error(`技能不存在: ${params.skillName}`);
      }
      metadata = found;
    }

    const targetLevel = params.level ?? SkillLevel.FULL;
    const skill = await this.loader.load(metadata, targetLevel);
    this.registry.register(skill);

    return skill;
  }

  /**
   * 处理技能执行
   */
  private async handleRun(params: SkillRunParams): Promise<SkillRunResult> {
    const skill = this.registry.get(params.skillName);

    if (!skill) {
      throw new Error(`技能不存在: ${params.skillName}`);
    }

    if (skill.level !== SkillLevel.FULL) {
      // 自动升级到完全加载
      await this.handleLoad({
        skillName: params.skillName,
        level: SkillLevel.FULL,
      });
    }

    const loadedSkill = this.registry.get(params.skillName)!;
    return this.executor.run(loadedSkill, params);
  }

  /**
   * 处理技能管理
   */
  private async handleManage(params: SkillManageParams): Promise<unknown> {
    switch (params.action) {
      case SkillManageAction.LIST:
        return this.registry.getAll().map((s) => ({
          name: s.name,
          description: s.description,
          version: s.version,
          level: s.level,
          state: s.state,
        }));

      case SkillManageAction.INFO: {
        if (!params.skillName) throw new Error('缺少 skillName 参数');
        const skill = this.registry.get(params.skillName);
        if (!skill) throw new Error(`技能不存在: ${params.skillName}`);
        return skill;
      }

      case SkillManageAction.LOAD: {
        if (!params.skillName) throw new Error('缺少 skillName 参数');
        return this.handleLoad({
          skillName: params.skillName,
          level: params.level,
        });
      }

      case SkillManageAction.UNLOAD: {
        if (!params.skillName) throw new Error('缺少 skillName 参数');
        const skill = this.registry.get(params.skillName);
        if (!skill) throw new Error(`技能不存在: ${params.skillName}`);
        await this.loader.unload(skill);
        this.registry.unregister(params.skillName);
        return { success: true };
      }

      case SkillManageAction.RELOAD: {
        if (!params.skillName) throw new Error('缺少 skillName 参数');
        const existing = this.registry.get(params.skillName);
        if (existing) {
          await this.loader.unload(existing);
        }
        return this.handleLoad({
          skillName: params.skillName,
          level: SkillLevel.FULL,
        });
      }

      case SkillManageAction.SEARCH: {
        if (!params.query) throw new Error('缺少 query 参数');
        return this.registry.search(params.query);
      }

      default:
        throw new Error(`未知管理操作: ${params.action}`);
    }
  }
}
```

### 22.3 代码规范

1. **三级加载**：严格遵循 META → DESCRIPTION → FULL 渐进加载，避免不必要的资源消耗
2. **类型守卫**：SKILL.md 解析结果必须验证必填字段（name、description）
3. **隔离执行**：技能脚本在 Worker 线程中执行，防止崩溃影响主进程
4. **超时保护**：所有执行操作必须有超时机制，默认 30 秒
5. **依赖检查**：加载 FULL 级别时检查依赖技能是否已加载
6. **搜索去重**：多路径发现时按名称去重，先发现的优先
7. **安全沙箱**：Worker 执行环境限制访问敏感模块
8. **错误恢复**：加载失败的技能标记为 ERROR 状态，不阻塞其他技能
9. **延迟加载**：默认仅加载 META 级别，按需升级
10. **日志追踪**：关键操作（加载、执行、卸载）记录日志
