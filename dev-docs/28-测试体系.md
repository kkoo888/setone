import { describe, it, expect, vi, beforeEach } from 'vitest';
import { EventBus } from '@main/core/event-bus/EventBus';

describe('EventBus', () => {
  let bus: EventBus;

  beforeEach(() => {
    bus = new EventBus();
  });

  describe('发布/订阅', () => {
    it('应正确触发订阅回调', () => {
      const handler = vi.fn();
      bus.on('test:event', handler);
      bus.emit('test:event', { data: 'hello' });

      expect(handler).toHaveBeenCalledOnce();
      expect(handler).toHaveBeenCalledWith({ data: 'hello' });
    });

    it('应支持多个订阅者', () => {
      const handler1 = vi.fn();
      const handler2 = vi.fn();

      bus.on('test:event', handler1);
      bus.on('test:event', handler2);
      bus.emit('test:event', 'payload');

      expect(handler1).toHaveBeenCalledWith('payload');
      expect(handler2).toHaveBeenCalledWith('payload');
    });

    it('应支持取消订阅', () => {
      const handler = vi.fn();
      const unsubscribe = bus.on('test:event', handler);

      unsubscribe();
      bus.emit('test:event', 'payload');

      expect(handler).not.toHaveBeenCalled();
    });
  });

  describe('类型安全', () => {
    interface TestEvents {
      'user:login': { userId: string };
      'user:logout': { userId: string; reason: string };
    }

    it('类型化事件总线应正确工作', () => {
      const typedBus = new EventBus<TestEvents>();
      const handler = vi.fn();

      typedBus.on('user:login', handler);
      typedBus.emit('user:login', { userId: '123' });

      expect(handler).toHaveBeenCalledWith({ userId: '123' });
    });
  });

  describe('错误处理', () => {
    it('单个处理器错误不应影响其他处理器', () => {
      const goodHandler = vi.fn();
      const badHandler = vi.fn(() => {
        throw new Error('handler error');
      });

      bus.on('test:event', badHandler);
      bus.on('test:event', goodHandler);

      expect(() => bus.emit('test:event', 'data')).not.toThrow();
      expect(goodHandler).toHaveBeenCalled();
    });
  });

  describe('once', () => {
    it('应只触发一次', () => {
      const handler = vi.fn();
      bus.once('test:event', handler);

      bus.emit('test:event', 'first');
      bus.emit('test:event', 'second');

      expect(handler).toHaveBeenCalledOnce();
    });
  });
});

  let pluginLoader: PluginLoader;

        key TEXT NOT NULL UNIQUE,
        value TEXT,
        created_at TEXT DEFAULT (datetime('now'))
      )
    `);

    // 验证表存在
    const tables = db
      .prepare("SELECT name FROM sqlite_master WHERE type='table' AND name='plugin_data'")
      .get() as { name: string } | undefined;

    expect(tables?.name).toBe('plugin_data');

    // 验证数据完整性
    const integrity = db.pragma('integrity_check') as Array<{ integrity_check: string }>;
    expect(integrity[0].integrity_check).toBe('ok');
  });

  it('通过事件总线触发的数据写入应正确持久化', () => {
    db.exec(`
      CREATE TABLE IF NOT EXISTS events (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        type TEXT NOT NULL,
        payload TEXT
      )
    `);

    const insertStmt = db.prepare('INSERT INTO events (type, payload) VALUES (?, ?)');

    // 监听事件并写入数据库
    eventBus.on('data:write', (payload: { type: string; data: unknown }) => {
      insertStmt.run(payload.type, JSON.stringify(payload.data));
    });

    // 触发事件
    eventBus.emit('data:write', { type: 'user-action', data: { action: 'click' } });
    eventBus.emit('data:write', { type: 'config-change', data: { key: 'theme' } });

    // 验证数据已写入
    const rows = db.prepare('SELECT * FROM events').all() as Array<{
      type: string;
      payload: string;
    }>;

    expect(rows).toHaveLength(2);
    expect(rows[0].type).toBe('user-action');
    expect(JSON.parse(rows[1].payload)).toEqual({ key: 'theme' });
  });
});

function handleGenerate(req: http.IncomingMessage, res: http.ServerResponse): void {
  let body = '';
  req.on('data', (chunk) => (body += chunk));
  req.on('end', () => {
    const parsed = JSON.parse(body);
    const prompt: string = parsed.prompt ?? '';
    const stream: boolean = parsed.stream !== false; // 默认流式

    res.setHeader('Content-Type', 'application/x-ndjson');

    if (stream) {
      // 流式响应：逐 token 发送
      const tokens = generateTokens(prompt);
      let i = 0;

      const sendNext = () => {
        if (i < tokens.length) {
          const chunk = JSON.stringify({
            model: parsed.model ?? 'llama3.2:latest',
            created_at: new Date().toISOString(),
            response: tokens[i],
            done: false,
          });
          res.write(chunk + '\n');
          i++;
          setTimeout(sendNext, 5); // 5ms 间隔模拟流式输出
        } else {
          // 最终 chunk
          const final = JSON.stringify({
            model: parsed.model ?? 'llama3.2:latest',
            created_at: new Date().toISOString(),
            response: '',
            done: true,
            total_duration: 1_000_000_000,
            load_duration: 500_000_000,
            prompt_eval_count: prompt.length,
            eval_count: tokens.length,
          });
          res.write(final + '\n');
          res.end();
        }
      };

      sendNext();
    } else {
      // 非流式：一次性返回完整响应
      const tokens = generateTokens(prompt);
      const result = JSON.stringify({
        model: parsed.model ?? 'llama3.2:latest',
        created_at: new Date().toISOString(),
        response: tokens.join(''),
        done: true,
        total_duration: 1_000_000_000,
        load_duration: 500_000_000,
        prompt_eval_count: prompt.length,
        eval_count: tokens.length,
      });
      res.end(result);
    }
  });
}

### 28.3 代码规范

1. **测试文件命名**：`{模块名}.test.ts` 或 `{模块名}.spec.ts`，与被测文件同目录或放在 `tests/` 对应子目录
2. **测试结构**：使用 `describe` 按功能分组，`it` 描述具体行为（"应..."），禁止 `test.todo` 在 CI 中存在
3. **覆盖率要求**：行覆盖率、函数覆盖率、分支覆盖率均 ≥ 80%，新增代码必须 ≥ 90%
4. **Mock 原则**：单元测试 mock 所有外部依赖；集成测试仅 mock 外部服务（Ollama、文件系统）；E2E 使用真实进程
5. **测试隔离**：每个测试用例独立，不依赖执行顺序，不共享可变状态
6. **断言规范**：每个测试至少一个有意义的断言，禁止空测试；使用 `expect` 语义化断言而非手动 `if`
7. **异步测试**：使用 `async/await`，禁止回调风格；超时使用 `vi.useFakeTimers()` 或配置 `testTimeout`
8. **E2E 选择器**：使用 `data-testid` 属性，禁止依赖 CSS 类名或文本内容（易变）
9. **契约测试**：所有第三方插件必须通过契约测试才能上架，契约测试由平台提供模板
10. **CI 规范**：PR 必须通过所有测试才能合并，覆盖率报告自动上传，E2E 失败时自动上传截图和 trace
