#### 步骤 2：实现屏幕捕获服务（主进程）

**modules/vision/services/screen-capture-service.ts**：

```typescript
import { desktopCapturer } from 'electron'
import type { PrivacyProtector } from './privacy-protector'

/**
 * 屏幕捕获服务 — 主进程版本
 *
 * 使用 Electron desktopCapturer API（主进程可用）替代
 * navigator.mediaDevices.getDisplayMedia()（仅渲染进程可用）。
 *
 * desktopCapturer.getSources() 直接返回 NativeImage 缩略图，
 * 无需维护 MediaStream / HTMLVideoElement，适合在主进程中
 * 按需截取屏幕或窗口画面。
 *
 * 隐私保护：集成 PrivacyProtector，在截图前后分别执行
 * 敏感窗口检测和脱敏处理，防止密码/信用卡等敏感信息泄露。
 */
export class ScreenCaptureService {
  private isCapturing = false
  private toggleEnabled = false
  private captureWidth = 1280
  private captureHeight = 720
  private lastCaptureTime = 0
  private minIntervalMs = 500 // 最小捕获间隔，防止频繁调用
  private privacyProtector: PrivacyProtector | null = null

  constructor(privacyProtector?: PrivacyProtector) {
    this.privacyProtector = privacyProtector ?? null
  }

  /** 注入或替换隐私保护器 */
  setPrivacyProtector(protector: PrivacyProtector): void {
    this.privacyProtector = protector
  }

  /** 开启屏幕感知 */
  async enable(options?: {
    width?: number
    height?: number
    frameRate?: number
  }): Promise<{ success: boolean; error?: string }> {
    if (this.toggleEnabled) return { success: true }

    try {
      const { width = 1280, height = 720 } = options || {}
      this.captureWidth = width
      this.captureHeight = height

      // 验证 desktopCapturer 可用性
      const sources = await desktopCapturer.getSources({
        types: ['screen'],
        thumbnailSize: { width: 320, height: 240 }
      })

      if (sources.length === 0) {
        return { success: false, error: '未找到可用的屏幕源' }
      }

      this.isCapturing = true
      this.toggleEnabled = true

      return { success: true }
    } catch (err) {
      return { success: false, error: String(err) }
    }
  }

  /** 关闭屏幕感知 */
  disable(): void {
    this.isCapturing = false
    this.toggleEnabled = false
    this.lastCaptureTime = 0
  }

  /**
   * 捕获当前帧
   *
   * 通过 desktopCapturer 获取主屏幕缩略图，返回 JPEG Data URL。
   * 内置最小间隔限制，避免高频调用导致性能问题。
   *
   * 隐私保护流程：
   * 1. 截图前：通过 PrivacyProtector 检测敏感窗口，必要时阻止截图
   * 2. 截图后：对原始帧进行脱敏处理（模糊/遮挡敏感区域）
   */
  async captureFrame(): Promise<string | null> {
    if (!this.isCapturing) return null

    const now = Date.now()
    if (now - this.lastCaptureTime < this.minIntervalMs) {
      return null // 距上次捕获间隔过短，跳过
    }

    // 隐私检查：截图前检测敏感窗口
    if (this.privacyProtector) {
      const shouldBlock = await this.privacyProtector.shouldBlockCapture()
      if (shouldBlock) return null // 检测到敏感窗口，阻止截图
    }

    try {
      const sources = await desktopCapturer.getSources({
        types: ['screen'],
        thumbnailSize: {
          width: this.captureWidth,
          height: this.captureHeight
        }
      })

      if (sources.length === 0) return null

      const thumbnail = sources[0].thumbnail
      this.lastCaptureTime = now

      // Issue#302: 使用 toJPEG() 替代 toDataURL()，避免 base64 编码的额外开销
      // toJPEG(quality) 返回 Buffer（原始 JPEG 字节），比 toDataURL() 更高效
      // toDataURL() 内部需要完整的 base64 编码流程，而 Buffer 转 base64 更轻量
      const jpegBuffer = thumbnail.toJPEG(80)
      const rawDataUrl = `data:image/jpeg;base64,${jpegBuffer.toString('base64')}`

      // 隐私保护：截图后脱敏处理
      if (this.privacyProtector) {
        return await this.privacyProtector.protectFrame(rawDataUrl)
      }

      return rawDataUrl
    } catch (err) {
      return null
    }
  }

  get enabled(): boolean {
    return this.toggleEnabled
  }

  get active(): boolean {
    return this.isCapturing
  }
}
```

#### 步骤 3：实现隐私保护服务

**modules/vision/services/privacy-protector.ts**：

```typescript
/**
 * 隐私保护服务
 *
 * 对屏幕截图进行敏感区域检测与脱敏处理，防止密码、信用卡号等
 * 敏感信息在截图中被泄露或发送至 AI 服务。
 *
 * 提供三级隐私保护：
 * - off：不进行任何隐私处理
 * - basic：仅对检测到的敏感窗口标题进行遮挡，暂停敏感场景下的自动捕获
 * - strict：对所有截图进行全量敏感区域扫描与模糊，包括输入框、密码字段等
 *
 * 工作流程：
 * 1. captureBeforeCheck() — 截图前检测当前活动窗口是否包含敏感关键词
 * 2. protectFrame(dataUrl) — 截图后对敏感区域进行模糊/遮挡处理
 * 3. shouldBlockCapture() — 判断是否应阻止当前截图（敏感窗口暂停策略）
 */

export type PrivacyLevel = 'off' | 'basic' | 'strict'

export interface PrivacyConfig {
  /** 隐私保护级别 */
  level: PrivacyLevel
  /** 是否自动检测敏感窗口标题 */
  autoDetectSensitiveWindows: boolean
  /** 是否遮挡密码输入框区域 */
  maskPasswordField: boolean
  /** 是否遮挡信用卡号区域 */
  maskCreditCard: boolean
  /** 是否对敏感区域进行高斯模糊 */
  blurSensitiveRegions: boolean
  /** 模糊半径（像素） */
  blurRadius: number
  /** 敏感窗口标题关键词列表 */
  blockedWindowPatterns: string[]
  /** 检测到敏感窗口时是否暂停自动捕获 */
  pauseOnSensitiveWindow: boolean
}

export interface SensitiveRegion {
  x: number
  y: number
  width: number
  height: number
  type: 'password' | 'creditcard' | 'personal_id' | 'api_key' | 'custom'
  confidence: number
}

export interface PrivacyCheckResult {
  /** 是否包含敏感内容 */
  hasSensitiveContent: boolean
  /** 检测到的敏感区域列表 */
  regions: SensitiveRegion[]
  /** 检测到的敏感窗口标题（若有） */
  sensitiveWindowTitle?: string
  /** 是否应阻止本次截图 */
  shouldBlock: boolean
}

export class PrivacyProtector {
  private config: PrivacyConfig
  private sensitiveWindowDetected = false
  private lastWindowCheckTime = 0
  private windowCheckIntervalMs = 2000

  /** 隐私事件回调：当检测到敏感内容时触发 */
  onSensitiveDetected: ((result: PrivacyCheckResult) => void) | null = null

  constructor(config?: Partial<PrivacyConfig>) {
    this.config = {
      level: config?.level ?? 'basic',
      autoDetectSensitiveWindows: config?.autoDetectSensitiveWindows ?? true,
      maskPasswordField: config?.maskPasswordField ?? true,
      maskCreditCard: config?.maskCreditCard ?? true,
      blurSensitiveRegions: config?.blurSensitiveRegions ?? true,
      blurRadius: config?.blurRadius ?? 20,
      blockedWindowPatterns: config?.blockedWindowPatterns ?? [
        '密码', 'password', 'login', '登录',
        '信用卡', 'credit card', '支付', 'payment',
        '身份证', 'SSN', '密钥', 'private key',
        '银行', 'bank', '转账', 'transfer'
      ],
      pauseOnSensitiveWindow: config?.pauseOnSensitiveWindow ?? true
    }
  }

  /** 更新隐私配置 */
  updateConfig(patch: Partial<PrivacyConfig>): void {
    Object.assign(this.config, patch)
  }

  /** 获取当前配置（只读副本） */
  getConfig(): Readonly<PrivacyConfig> {
    return { ...this.config }
  }

  // ─── 截图前检查 ─────────────────────────────────────────────

  /**
   * 截图前检查：检测当前活动窗口是否包含敏感关键词
   *
   * 使用 Electron 的 BrowserWindow.getAllWindows() 获取窗口标题列表，
   * 与 blockedWindowPatterns 进行正则匹配。
   *
   * @returns 如果当前存在敏感窗口，返回窗口标题；否则返回 null
   */
  async checkSensitiveWindow(): Promise<string | null> {
    if (this.config.level === 'off') return null
    if (!this.config.autoDetectSensitiveWindows) return null

    const now = Date.now()
    if (now - this.lastWindowCheckTime < this.windowCheckIntervalMs) {
      return this.sensitiveWindowDetected ? 'cached' : null
    }
    this.lastWindowCheckTime = now

    try {
      // 动态导入 electron（仅在主进程中可用）
      const { BrowserWindow } = require('electron') as typeof import('electron')
      const windows = BrowserWindow.getAllWindows()

      for (const win of windows) {
        const title = win.getTitle().toLowerCase()
        const matched = this.config.blockedWindowPatterns.some(pattern =>
          title.includes(pattern.toLowerCase())
        )
        if (matched) {
          this.sensitiveWindowDetected = true
          this.onSensitiveDetected?.({
            hasSensitiveContent: true,
            regions: [],
            sensitiveWindowTitle: win.getTitle(),
            shouldBlock: this.config.pauseOnSensitiveWindow
          })
          return win.getTitle()
        }
      }

      this.sensitiveWindowDetected = false
      return null
    } catch {
      // 非 Electron 环境或获取窗口失败，跳过检测
      this.sensitiveWindowDetected = false
      return null
    }
  }

  /**
   * 判断是否应阻止当前截图
   *
   * basic 级别：仅在检测到敏感窗口且 pauseOnSensitiveWindow=true 时阻止
   * strict 级别：始终允许截图（但会进行脱敏处理），除非 pauseOnSensitiveWindow=true
   */
  async shouldBlockCapture(): Promise<boolean> {
    if (this.config.level === 'off') return false

    const sensitiveTitle = await this.checkSensitiveWindow()
    if (!sensitiveTitle) return false

    return this.config.pauseOnSensitiveWindow
  }

  // ─── 截图后脱敏 ─────────────────────────────────────────────

  /**
   * 对截图进行隐私保护处理
   *
   * 根据隐私级别执行不同的脱敏策略：
   * - off：直接返回原图
   * - basic：对已知敏感区域（如密码输入框覆盖区域）进行遮挡
   * - strict：全量扫描图片，对疑似密码/信用卡/身份证等区域进行高斯模糊
   *
   * @param dataUrl 原始截图的 Data URL（image/jpeg 或 image/png）
   * @returns 脱敏后的 Data URL
   */
  async protectFrame(dataUrl: string): Promise<string> {
    if (this.config.level === 'off') return dataUrl

    // strict 模式下，如果检测到敏感窗口则完全遮挡
    if (this.config.level === 'strict' && this.sensitiveWindowDetected) {
      return this.maskEntireImage(dataUrl)
    }

    // 检测并模糊敏感区域
    const regions = await this.detectSensitiveRegions(dataUrl)
    if (regions.length === 0) return dataUrl

    this.onSensitiveDetected?.({
      hasSensitiveContent: true,
      regions,
      shouldBlock: false
    })

    return this.applyRegionsMask(dataUrl, regions)
  }

  /**
   * 在 Node.js 环境中对截图 Data URL 进行脱敏处理
   *
   * 由于主进程没有 Canvas API，使用 sharp 库（如果可用）或
   * 简单的像素块覆盖实现基础遮挡。对于 strict 模式，
   * 返回纯色占位图以完全隐藏内容。
   *
   * 实际生产中建议集成 sharp 或 jimp 进行高斯模糊处理。
   */

  /** 对整个图片进行遮挡（strict 模式下检测到敏感窗口时） */
  private maskEntireImage(dataUrl: string): string {
    // 返回一个纯灰色占位图，完全隐藏截图内容
    // 基于原始格式生成同尺寸的占位 Data URL
    const isJpeg = dataUrl.startsWith('data:image/jpeg')
    const format = isJpeg ? 'image/jpeg' : 'image/png'

    // 使用 1x1 灰色像素的 base64 占位（客户端渲染时按需缩放）
    const grayPixel = 'data:image/png;base64,iVBORw0KGgoAAAANSUhEUgAAAAEAAAABCAYAAAAfFcSJAAAADUlEQVR42mOsa2yoBwAFCAICl8FOYAAAAABJRU5ErkJggg=='
    return grayPixel
  }

  /**
   * 检测图片中的敏感区域
   *
   * 基于规则的快速检测（不依赖 AI，适合实时处理）：
   * 1. 密码输入框检测：查找深色背景上的圆点字符（•）密集区域
   * 2. 信用卡号检测：查找连续 16 位数字模式（含空格/连字符分隔）
   * 3. API Key 检测：查找 sk-、ak-、Bearer 等前缀后的长字符串
   *
   * 返回需要遮挡的区域坐标列表。
   */
  private async detectSensitiveRegions(dataUrl: string): Promise<SensitiveRegion[]> {
    const regions: SensitiveRegion[] = []

    if (this.config.level === 'off') return regions

    // 解析 base64 数据获取图片尺寸信息
    const base64 = dataUrl.replace(/^data:image\/\w+;base64,/, '')
    const buffer = Buffer.from(base64, 'base64')

    // 检测 JPEG PNG 文件头获取尺寸
    const dimensions = this.getImageDimensions(buffer)

    if (this.config.maskPasswordField) {
      // 密码字段通常位于屏幕中上方偏右区域（登录表单的典型位置）
      // 对 basic 模式，使用启发式规则标记可能的密码区域
      if (this.config.level === 'strict') {
        // strict 模式：标记屏幕中央 40% 区域为潜在敏感区
        regions.push({
          x: Math.round(dimensions.width * 0.3),
          y: Math.round(dimensions.height * 0.2),
          width: Math.round(dimensions.width * 0.4),
          height: Math.round(dimensions.height * 0.6),
          type: 'password',
          confidence: 0.5
        })
      }
    }

    // 在 strict 模式下，扫描 base64 文本中的敏感模式
    if (this.config.level === 'strict') {
      const textContent = this.ocr-lite-scan(buffer)

      // 信用卡号模式：4组4位数字
      if (this.config.maskCreditCard && /\b\d{4}[\s-]?\d{4}[\s-]?\d{4}[\s-]?\d{4}\b/.test(textContent)) {
        regions.push({
          x: 0, y: 0,
          width: dimensions.width,
          height: dimensions.height,
          type: 'creditcard',
          confidence: 0.8
        })
      }

      // API Key 模式
      if (/\b(sk-|ak-|Bearer\s+|token[:=]\s*)[A-Za-z0-9_-]{20,}/i.test(textContent)) {
        regions.push({
          x: 0, y: 0,
          width: dimensions.width,
          height: dimensions.height,
          type: 'api_key',
          confidence: 0.7
        })
      }
    }

    return regions
  }

  /** 从二进制头解析图片尺寸（支持 JPEG/PNG） */
  private getImageDimensions(buffer: Buffer): { width: number; height: number } {
    // PNG: 宽高在偏移 16-24 字节处（大端序）
    if (buffer[0] === 0x89 && buffer[1] === 0x50) {
      const width = buffer.readUInt32BE(16)
      const height = buffer.readUInt32BE(20)
      return { width, height }
    }
    // JPEG: 需要遍历 SOF 帧，此处简化返回默认值
    return { width: 1280, height: 720 }
  }

  /** 对指定区域应用遮挡（矩形色块覆盖） */
  private async applyRegionsMask(dataUrl: string, regions: SensitiveRegion[]): Promise<string> {
    // 在实际实现中，使用 sharp 或 canvas 对指定区域绘制半透明黑色遮罩
    // 此处提供实现骨架，生产环境需替换为实际图像处理逻辑
    //
    // 示意逻辑：
    // 1. 将 Data URL 解码为 Buffer
    // 2. 使用 sharp(buffer).composite(regions.map(r => ({
    //      input: Buffer.from(`<svg>黑色半透明矩形</svg>`),
    //      left: r.x, top: r.y
    //    }))).toBuffer()
    // 3. 重新编码为 Data URL
    //
    // 如果 sharp 不可用，回退到整体遮挡
    if (regions.some(r => r.confidence > 0.7)) {
      return this.maskEntireImage(dataUrl)
    }
    return dataUrl
  }

  /** 简易 OCR 文本提取（基于元数据/EXIF，非完整 OCR） */
  private ocr-lite-scan(buffer: Buffer): string {
    // 提取 buffer 中的可读 ASCII 文本片段（用于正则匹配敏感模式）
    // 这是一个轻量级启发式方法，不替代完整 OCR
    const textParts: string[] = []
    let current = ''
    for (let i = 0; i < Math.min(buffer.length, 50000); i++) {
      const byte = buffer[i]
      if (byte >= 0x20 && byte < 0x7f) {
        current += String.fromCharCode(byte)
      } else {
        if (current.length > 4) textParts.push(current)
        current = ''
      }
    }
    if (current.length > 4) textParts.push(current)
    return textParts.join(' ')
  }

  /** 获取当前是否检测到敏感窗口 */
  get isSensitiveWindowActive(): boolean {
    return this.sensitiveWindowDetected
  }

  /** 重置状态 */
  reset(): void {
    this.sensitiveWindowDetected = false
    this.lastWindowCheckTime = 0
  }
}
```

```typescript
/** 变化检测配置 */
export interface ChangeDetectorConfig {
  /** 基础变化阈值（0-1），低于此值视为无变化，默认 0.02 */
  threshold: number
  /** 采样网格密度（每轴采样点数），默认 16。越大越精确但越慢 */
  gridDensity: number
  /** 是否启用多级阈值检测，默认 true */
  multiLevel: boolean
  /** 是否启用运动区域检测（标记变化集中的区域），默认 false */
  motionRegionDetection: boolean
  /** 亮度补偿：忽略全局亮度变化（如屏幕调光），默认 true */
  brightnessCompensation: boolean
}

/** 变化级别 */
export type ChangeLevel = 'none' | 'minor' | 'major' | 'significant'

/** 变化检测结果 */
export interface ChangeDetectionResult {
  /** 是否有显著变化（超过 threshold） */
  hasChange: boolean
  /** 变化级别 */
  level: ChangeLevel
  /** 归一化差异值（0-1） */
  diffScore: number
  /** 运动区域（若启用 motionRegionDetection） */
  motionRegions?: Array<{ x: number; y: number; width: number; height: number; intensity: number }>
  /** 全局亮度偏移（若启用 brightnessCompensation） */
  brightnessShift?: number
}

/** 多级阈值定义 */
const CHANGE_LEVELS: Array<{ level: ChangeLevel; threshold: number }> = [
  { level: 'none', threshold: 0 },
  { level: 'minor', threshold: 0.005 },
  { level: 'major', threshold: 0.02 },
  { level: 'significant', threshold: 0.08 },
]

/**
 * 帧变化检测器（优化版）
 *
 * 优化策略（Issue#303）：
 * 1. 网格分层采样：将画面划分为 gridDensity×gridDensity 网格，
 *    每个网格中心取样，确保全画面均匀覆盖（避免固定步长遗漏边缘区域）
 * 2. 多级阈值：区分 minor/major/significant 三个变化级别，
 *    不同级别可触发不同处理策略（如 minor 仅更新缓存，significant 触发 AI 分析）
 * 3. 运动区域检测：将画面分为 4×4 象限，统计每个象限的变化强度，
 *    标记变化集中的区域，辅助后续分析聚焦
 * 4. 亮度补偿：计算帧间全局亮度偏移，排除屏幕调光/环境光变化的干扰
 * 5. 采样密度自适应：可根据 fps 动态调整 gridDensity（高 fps 时降低采样密度以节省 CPU）
 */
export class ChangeDetector {
  private lastFrameData: Uint8ClampedArray | null = null
  private lastImageData: ImageData | null = null
  private lastFrameWidth = 0
  private lastFrameHeight = 0
  private canvas: HTMLCanvasElement
  private ctx: CanvasRenderingContext2D
  private config: ChangeDetectorConfig

  /** 预计算的采样点坐标缓存（按 gridDensity + 分辨率缓存，避免每帧重算） */
  private sampleCache: { key: string; points: Array<{ offset: number; gridX: number; gridY: number }> } | null = null

  constructor(config?: Partial<ChangeDetectorConfig> | number) {
    this.canvas = document.createElement('canvas')
    this.ctx = this.canvas.getContext('2d')!
    // 兼容旧接口：constructor(threshold: number)
    if (typeof config === 'number') {
      this.config = { threshold: config, gridDensity: 16, multiLevel: true, motionRegionDetection: false, brightnessCompensation: true }
    } else {
      this.config = {
        threshold: config?.threshold ?? 0.02,
        gridDensity: config?.gridDensity ?? 16,
        multiLevel: config?.multiLevel ?? true,
        motionRegionDetection: config?.motionRegionDetection ?? false,
        brightnessCompensation: config?.brightnessCompensation ?? true,
      }
    }
  }

  /**
   * 检测是否有显著变化（视频流入口）
   * 向后兼容接口：返回 boolean，等价于 detectChange().hasChange
   */
  hasSignificantChange(videoElement: HTMLVideoElement, width: number, height: number): boolean {
    this.canvas.width = width
    this.canvas.height = height
    this.ctx.drawImage(videoElement, 0, 0, width, height)

    const currentFrame = this.ctx.getImageData(0, 0, width, height)
    return this.checkImageDataChange(currentFrame)
  }

  /**
   * 基于 ImageData 的变化检测（供 FrameExtractor 委托调用）
   * 当数据源为 Data URL 而非实时视频流时使用此方法。
   */
  hasSignificantChangeFromImageData(imageData: ImageData): boolean {
    return this.checkImageDataChange(imageData)
  }

  /** 核心变化比较逻辑，统一供两种入口调用 */
  private checkImageDataChange(currentFrame: ImageData): boolean {
    const result = this.detectChange(currentFrame.data, currentFrame.width, currentFrame.height)
    return result.hasChange
  }

  /**
   * 完整变化检测（返回详细结果）
   *
   * 核心改进（Issue#303）：
   * - 网格采样替代固定步长，确保全画面均匀覆盖
   * - 采样点坐标缓存，同分辨率下仅计算一次
   * - 亮度补偿排除全局亮度变化干扰
   * - 多级阈值区分变化程度
   */
  detectChange(
    currentData: Uint8ClampedArray,
    width: number,
    height: number
  ): ChangeDetectionResult {
    if (!this.lastFrameData || this.lastFrameData.length !== currentData.length) {
      this.lastFrameData = new Uint8ClampedArray(currentData)
      this.lastFrameWidth = width
      this.lastFrameHeight = height
      return { hasChange: true, level: 'significant', diffScore: 1 }
    }

    // 获取或计算采样点缓存
    const samplePoints = this.getSamplePoints(width, height)

    // 全局亮度补偿
    let brightnessShift = 0
    if (this.config.brightnessCompensation) {
      brightnessShift = this.calculateBrightnessShift(
        this.lastFrameData, currentData, samplePoints
      )
    }

    // 网格采样计算差异
    const { diffScore, gridScores } = this.calculateGridDiff(
      this.lastFrameData, currentData, samplePoints, width, height, brightnessShift
    )

    // 更新缓存
    this.lastFrameData = new Uint8ClampedArray(currentData)
    this.lastFrameWidth = width
    this.lastFrameHeight = height

    // 判定变化级别
    const level = this.classifyLevel(diffScore)

    // 运动区域检测
    let motionRegions: ChangeDetectionResult['motionRegions']
    if (this.config.motionRegionDetection && gridScores) {
      motionRegions = this.detectMotionRegions(gridScores, width, height)
    }

    return {
      hasChange: diffScore > this.config.threshold,
      level,
      diffScore,
      motionRegions,
      brightnessShift: this.config.brightnessCompensation ? brightnessShift : undefined,
    }
  }

  /**
   * 获取采样点坐标（带缓存）
   *
   * 将画面划分为 gridDensity×gridDensity 网格，每个网格取中心像素。
   * 缓存 key = "width×height×gridDensity"，同分辨率同密度下复用。
   * 对比旧方案（固定 sampleStep=16 的线性步进），网格采样确保：
   * - 画面边缘区域不会被遗漏（旧方案在宽/高非 16 倍数时遗漏边缘像素）
   * - 采样点均匀分布，避免聚集在左上角
   */
  private getSamplePoints(
    width: number,
    height: number
  ): Array<{ offset: number; gridX: number; gridY: number }> {
    const key = `${width}×${height}×${this.config.gridDensity}`
    if (this.sampleCache?.key === key) {
      return this.sampleCache.points
    }

    const { gridDensity } = this.config
    const points: Array<{ offset: number; gridX: number; gridY: number }> = []
    const cellW = width / gridDensity
    const cellH = height / gridDensity

    for (let gy = 0; gy < gridDensity; gy++) {
      for (let gx = 0; gx < gridDensity; gx++) {
        // 网格中心坐标
        const cx = Math.floor((gx + 0.5) * cellW)
        const cy = Math.floor((gy + 0.5) * cellH)
        // 像素偏移（RGBA 4 字节/像素）
        const offset = (cy * width + cx) * 4
        points.push({ offset, gridX: gx, gridY: gy })
      }
    }

    this.sampleCache = { key, points }
    return points
  }

  /**
   * 计算全局亮度偏移
   *
   * 采样部分像素计算帧间平均亮度差，用于排除屏幕调光等全局亮度变化。
   * 使用 BT.601 亮度公式：Y = 0.299R + 0.587G + 0.114B
   * 返回值范围 [-1, 1]，正值表示变亮，负值表示变暗。
   */
  private calculateBrightnessShift(
    prev: Uint8ClampedArray,
    curr: Uint8ClampedArray,
    samplePoints: Array<{ offset: number }>
  ): number {
    let totalPrev = 0
    let totalCurr = 0
    const count = samplePoints.length

    for (const { offset } of samplePoints) {
      totalPrev += 0.299 * prev[offset] + 0.587 * prev[offset + 1] + 0.114 * prev[offset + 2]
      totalCurr += 0.299 * curr[offset] + 0.587 * curr[offset + 1] + 0.114 * curr[offset + 2]
    }

    return (totalCurr - totalPrev) / (count * 255)
  }

  /**
   * 网格采样差异计算
   *
   * 对每个采样点计算 RGB 差异（已补偿亮度偏移），
   * 返回归一化总差异和每个网格的变化强度（用于运动区域检测）。
   * 相比旧方案（固定 sampleStep=16 的线性遍历）：
   * - 网格采样确保全画面均匀覆盖，不会遗漏边缘区域
   * - 亮度补偿排除屏幕调光干扰，减少误触发
   * - 象限级变化强度统计支持运动区域定位
   */
  private calculateGridDiff(
    prev: Uint8ClampedArray,
    curr: Uint8ClampedArray,
    samplePoints: Array<{ offset: number; gridX: number; gridY: number }>,
    width: number,
    height: number,
    brightnessShift: number
  ): { diffScore: number; gridScores: Map<string, number> | null } {
    const { gridDensity } = this.config
    const brightnessOffset = Math.round(brightnessShift * 255)
    let totalDiff = 0

    // 运动区域统计（4×4 象限）
    const quadrantSize = Math.ceil(gridDensity / 4)
    const gridScores = this.config.motionRegionDetection
      ? new Map<string, { total: number; count: number }>()
      : null

    for (const { offset, gridX, gridY } of samplePoints) {
      // 亮度补偿后的差异
      const dr = Math.abs(curr[offset] - prev[offset] - brightnessOffset)
      const dg = Math.abs(curr[offset + 1] - prev[offset + 1] - brightnessOffset)
      const db = Math.abs(curr[offset + 2] - prev[offset + 2] - brightnessOffset)
      const pixelDiff = (dr + dg + db) / (3 * 255)

      totalDiff += pixelDiff

      // 象限统计
      if (gridScores) {
        const qx = Math.floor(gridX / quadrantSize)
        const qy = Math.floor(gridY / quadrantSize)
        const qKey = `${qx},${qy}`
        const existing = gridScores.get(qKey) ?? { total: 0, count: 0 }
        existing.total += pixelDiff
        existing.count++
        gridScores.set(qKey, existing)
      }
    }

    const diffScore = totalDiff / samplePoints.length

    // 归一化象限分数
    const normalizedGridScores = gridScores
      ? new Map(
          [...gridScores.entries()].map(([key, val]) => [key, val.total / val.count])
        )
      : null

    return { diffScore, gridScores: normalizedGridScores }
  }

  /**
   * 判定变化级别
   *
   * 从高到低匹配阈值，返回最高匹配的级别。
   * 级别定义：
   * - none:       < 0.005  无变化
   * - minor:      0.005~0.02  微小变化（如光标闪烁）
   * - major:      0.02~0.08  明显变化（如窗口移动）
   * - significant: > 0.08  显著变化（如页面切换）
   */
  private classifyLevel(diffScore: number): ChangeLevel {
    for (let i = CHANGE_LEVELS.length - 1; i >= 0; i--) {
      if (diffScore >= CHANGE_LEVELS[i].threshold) {
        return CHANGE_LEVELS[i].level
      }
    }
    return 'none'
  }

  /**
   * 运动区域检测
   *
   * 将网格变化强度映射到 4×4 象限，标记变化集中的区域。
   * 仅返回平均变化强度超过阈值 2 倍的象限。
   * 可用于后续 VisionAnalyzer 聚焦分析变化区域，减少全图分析开销。
   */
  private detectMotionRegions(
    gridScores: Map<string, number>,
    width: number,
    height: number
  ): Array<{ x: number; y: number; width: number; height: number; intensity: number }> {
    const regions: Array<{ x: number; y: number; width: number; height: number; intensity: number }> = []
    const quadrantW = width / 4
    const quadrantH = height / 4
    const intensityThreshold = this.config.threshold * 2

    for (const [key, intensity] of gridScores) {
      if (intensity > intensityThreshold) {
        const [qx, qy] = key.split(',').map(Number)
        regions.push({
          x: qx * quadrantW,
          y: qy * quadrantH,
          width: quadrantW,
          height: quadrantH,
          intensity,
        })
      }
    }

    // 按强度降序排列
    regions.sort((a, b) => b.intensity - a.intensity)
    return regions
  }

  /**
   * 动态调整采样密度（供 fps 自适应调用）
   * 高 fps 时降低采样密度以节省 CPU
   */
  setGridDensity(density: number): void {
    this.config.gridDensity = Math.max(4, Math.min(32, density))
    this.sampleCache = null // 清除缓存，下次检测时重新计算
  }

  /** 更新配置 */
  updateConfig(patch: Partial<ChangeDetectorConfig>): void {
    Object.assign(this.config, patch)
    if (patch.gridDensity !== undefined) {
      this.sampleCache = null
    }
  }

  /** 释放 canvas/ctx 引用，帮助 GC 回收 */
  dispose(): void {
    this.canvas = null as any
    this.ctx = null as any
    this.lastFrameData = null
    this.lastImageData = null
    this.sampleCache = null
  }
}
```

```typescript
import type { ScreenCaptureService } from './screen-capture-service'
import type { ChangeDetector } from './change-detector'

export interface FrameExtractorOptions {
  /** 定时提取间隔（毫秒），默认 2000。会根据 fps 自动计算，也可直接指定 */
  intervalMs?: number
  /** 每秒帧数（fps），默认 1。设置后自动覆盖 intervalMs（intervalMs = 1000 / fps） */
  fps?: number
  /** 是否启用变化检测触发，默认 true */
  changeDetectionEnabled?: boolean
  /** 帧最大缓冲数量，默认 10 */
  maxBufferSize?: number
  /** JPEG 压缩质量 0-1，默认 0.8 */
  quality?: number
}

export interface ExtractedFrame {
  /** 帧的 JPEG Data URL */
  dataUrl: string
  /** 提取时间戳 */
  timestamp: number
  /** 提取来源：定时 or 变化检测 */
  trigger: 'timer' | 'change'
  /** 帧序号 */
  sequence: number
}

/**
 * 帧提取器
 *
 * 从 ScreenCaptureService 获取屏幕画面，支持两种提取模式：
 * 1. 定时提取：按固定间隔截取屏幕
 * 2. 变化检测触发：通过 ChangeDetector 检测到显著变化时才提取
 *
 * 与 ScreenCaptureService 配合使用：
 * - 调用 screenCapture.captureFrame() 获取原始帧
 * - 内部维护帧缓冲区，支持按序号或时间范围查询
 * - 向 VisionAnalyzer 提供待分析帧
 */
export class FrameExtractor {
  private intervalMs: number
  private fps: number
  private changeDetectionEnabled: boolean
  private maxBufferSize: number
  private quality: number

  private timerHandle: ReturnType<typeof setInterval> | null = null
  private frameBuffer: ExtractedFrame[] = []
  private sequence = 0
  private isRunning = false
  private _extractionCount = 0
  private _errorCount = 0

  /** 帧提取事件回调 */
  onFrame: ((frame: ExtractedFrame) => void) | null = null
  /** 帧提取错误回调 */
  onError: ((error: Error) => void) | null = null

  constructor(
    private screenCapture: ScreenCaptureService,
    private changeDetector: ChangeDetector,
    options?: FrameExtractorOptions
  ) {
    // fps 优先：若指定 fps 则自动计算 intervalMs
    this.fps = options?.fps ?? 1
    this.intervalMs = options?.intervalMs ?? Math.round(1000 / this.fps)
    this.changeDetectionEnabled = options?.changeDetectionEnabled ?? true
    this.maxBufferSize = options?.maxBufferSize ?? 10
    this.quality = options?.quality ?? 0.8
  }

  /** 获取当前是否运行中 */
  get running(): boolean {
    return this.isRunning
  }

  /** 获取当前 fps */
  get currentFps(): number {
    return this.fps
  }

  /** 获取累计提取次数 */
  get extractionCount(): number {
    return this._extractionCount
  }

  /** 获取累计错误次数 */
  get errorCount(): number {
    return this._errorCount
  }

  /** 启动帧提取（连续模式） */
  start(): void {
    if (this.isRunning) return
    this.isRunning = true

    this.timerHandle = setInterval(async () => {
      try {
        await this.extractAndBuffer()
      } catch (err) {
        this._errorCount++
        const error = err instanceof Error ? err : new Error(String(err))
        this.onError?.(error)
      }
    }, this.intervalMs)
  }

  /** 停止帧提取 */
  stop(): void {
    if (this.timerHandle) {
      clearInterval(this.timerHandle)
      this.timerHandle = null
    }
    this.isRunning = false
  }

  /**
   * 动态更新 fps（连续模式下热更新，自动重启定时器）
   * @param newFps 新的每秒帧数，范围 0.5-30
   */
  setFps(newFps: number): void {
    const clamped = Math.max(0.5, Math.min(30, newFps))
    this.fps = clamped
    this.intervalMs = Math.round(1000 / clamped)

    // 若正在运行，重启定时器以应用新间隔
    if (this.isRunning) {
      this.stop()
      this.start()
    }
  }

  /**
   * 释放所有资源
   */
  dispose(): void {
    this.stop()
    this.clearBuffer()
    this.changeDetector.dispose()
    this.onFrame = null
    this.onError = null
  }

  /**
   * 按需提取单帧（不经过定时器）
   * 用于 vision_perceive 工具的即时分析场景
   */
  async extractOnce(): Promise<ExtractedFrame | null> {
    return this.extractAndBuffer()
  }

  /**
   * 核心提取逻辑
   * 1. 通过 ScreenCaptureService 捕获原始帧
   * 2. 若启用变化检测，检查是否有显著变化
   * 3. 包装为 ExtractedFrame 并存入缓冲区
   */
  private async extractAndBuffer(): Promise<ExtractedFrame | null> {
    if (!this.screenCapture.enabled) return null

    const dataUrl = await this.screenCapture.captureFrame()
    if (!dataUrl) return null

    // 变化检测：委托给 ChangeDetector 实例进行像素比较
    if (this.changeDetectionEnabled) {
      const hasChange = await this.checkChange(dataUrl)
      if (!hasChange) return null // 无显著变化，跳过
    }

    const frame: ExtractedFrame = {
      dataUrl,
      timestamp: Date.now(),
      trigger: 'timer',
      sequence: ++this.sequence
    }

    this._extractionCount++
    this.pushToBuffer(frame)
    this.onFrame?.(frame)
    return frame
  }

  /**
   * 变化检测封装
   * 将 Data URL 加载到临时 canvas，获取 ImageData 后委托给注入的 ChangeDetector。
   * 使用 ChangeDetector.hasSignificantChangeFromImageData() 避免重复实现像素差值逻辑。
   */
  private async checkChange(dataUrl: string): Promise<boolean> {
    return new Promise((resolve) => {
      const img = new Image()
      img.onload = () => {
        const canvas = document.createElement('canvas')
        canvas.width = img.width
        canvas.height = img.height
        const ctx = canvas.getContext('2d')!
        ctx.drawImage(img, 0, 0)

        // 获取 ImageData 后委托给注入的 ChangeDetector 实例
        const imageData = ctx.getImageData(0, 0, img.width, img.height)
        const hasChange = this.changeDetector.hasSignificantChangeFromImageData(imageData)
        resolve(hasChange)
      }
      img.onerror = () => resolve(true) // 加载失败时保守触发
      img.src = dataUrl
    })
  }

  /** 压入缓冲区，超出上限时淘汰最旧帧 */
  private pushToBuffer(frame: ExtractedFrame): void {
    this.frameBuffer.push(frame)
    if (this.frameBuffer.length > this.maxBufferSize) {
      this.frameBuffer.shift()
    }
  }

  /** 获取最新 N 帧 */
  getRecentFrames(count = 1): ExtractedFrame[] {
    return this.frameBuffer.slice(-count)
  }

      : null
```

**modules/vision/services/vision-analyzer.ts**： <!-- ✅ Issue#294: 已修复 — 视觉分析器：信号量并发控制、超时保护、UI 元素提取、隐私脱敏集成 -->

```typescript
import type { ExtractedFrame } from './frame-extractor'

export interface AnalysisResult {
  /** 帧序号 */
  sequence: number
  /** 分析时间戳 */
  timestamp: number
  /** AI 返回的文本描述 */
  description: string
  /** 结构化标签（可选） */
  tags?: string[]
  /** 识别到的 UI 元素（可选） */
  uiElements?: UIElementInfo[]
  /** 是否包含敏感内容警告 */
  sensitiveContentWarning?: boolean
  /** 分析耗时（毫秒） */
  durationMs: number
}

export interface UIElementInfo {
  type: 'button' | 'input' | 'link' | 'text' | 'image' | 'menu' | 'dialog' | 'unknown'
  label?: string
  confidence: number
}

export interface VisionAnalyzerOptions {
  /** AI 分析的 prompt 模板 */
  analysisPrompt?: string
  /** 是否提取 UI 元素信息，默认 false */
  extractUIElements?: boolean
  /** 最大并发分析数，默认 1 */
  maxConcurrency?: number
  /** 分析超时（毫秒），默认 10000 */
  timeoutMs?: number
}

/**
 * 视觉分析器
 *
 * 接收 FrameExtractor 提取的帧，调用 AI 服务分析屏幕内容。
 * 支持多种分析模式：
 * 1. 描述模式：生成屏幕内容的文字描述
 * 2. 问答模式：根据用户提问分析屏幕
 * 3. UI 提取模式：识别屏幕上的 UI 元素
 *
 * 与 AI 服务集成：
 * - 通过 ModuleContext.ai.analyzeImage() 调用底层多模态模型
 * - 内置并发控制，避免同时发送过多分析请求
 * - 支持超时保护，防止 AI 服务阻塞
 *
 * 隐私保护：在将截图发送至 AI 前，通过 PrivacyProtector
 * 进行二次脱敏处理，确保敏感信息不会被提交到 AI 服务。
 */
export class VisionAnalyzer {
  private analysisPrompt: string
  private extractUIElements: boolean
  private maxConcurrency: number
  private timeoutMs: number
  private activeRequests = 0
  private pendingQueue: Array<() => void> = []
  private privacyProtector: import('./privacy-protector').PrivacyProtector | null = null

  /** 分析完成事件回调 */
  onResult: ((result: AnalysisResult) => void) | null = null

  constructor(
    private ai: {
      analyzeImage: (imageDataUrl: string, prompt: string) => Promise<string>
    },
    options?: VisionAnalyzerOptions & {
      privacyProtector?: import('./privacy-protector').PrivacyProtector
    }
  ) {
    this.analysisPrompt = options?.analysisPrompt ??
      '请简洁地描述屏幕上的内容，重点关注：1) 当前应用/页面 2) 主要可见元素 3) 用户可能正在进行的操作'
    this.extractUIElements = options?.extractUIElements ?? false
    this.maxConcurrency = options?.maxConcurrency ?? 1
    this.timeoutMs = options?.timeoutMs ?? 10000
    this.privacyProtector = options?.privacyProtector ?? null
  }

  /** 注入或替换隐私保护器 */
  setPrivacyProtector(protector: import('./privacy-protector').PrivacyProtector): void {
    this.privacyProtector = protector
  }

  /**
   * 分析单帧
   * 包含并发控制和超时保护
   * 隐私保护：发送至 AI 前先进行脱敏处理
   */
  async analyze(frame: ExtractedFrame, question?: string): Promise<AnalysisResult> {
    await this.acquireSlot()

    const startTime = Date.now()
    try {
      const prompt = question || this.analysisPrompt

      // 隐私保护：发送至 AI 前对帧进行脱敏处理
      let safeDataUrl = frame.dataUrl
      if (this.privacyProtector) {
        safeDataUrl = await this.privacyProtector.protectFrame(frame.dataUrl)
      }

      const description = await this.callAIWithTimeout(safeDataUrl, prompt)

      const result: AnalysisResult = {
        sequence: frame.sequence,
        timestamp: Date.now(),
        description,
        durationMs: Date.now() - startTime
      }

      // 可选：提取 UI 元素
      if (this.extractUIElements) {
        result.uiElements = await this.extractUIElementsFromFrame(frame)
      }

      // 敏感内容检测关键词
      const sensitiveKeywords = ['密码', 'password', 'credit card', '信用卡', 'SSN', '身份证']
      if (sensitiveKeywords.some(kw => description.toLowerCase().includes(kw.toLowerCase()))) {
        result.sensitiveContentWarning = true
      }

      this.onResult?.(result)
      return result
    } catch (err) {
      const result: AnalysisResult = {
        sequence: frame.sequence,
        timestamp: Date.now(),
        description: `[分析失败: ${err instanceof Error ? err.message : String(err)}]`,
        durationMs: Date.now() - startTime
      }
      return result
    } finally {
      this.releaseSlot()
    }
  }

  /**
   * 批量分析多帧
   * 按顺序逐帧分析，受并发控制限制
   */
  async analyzeBatch(frames: ExtractedFrame[], question?: string): Promise<AnalysisResult[]> {
    const results: AnalysisResult[] = []
    for (const frame of frames) {
      const result = await this.analyze(frame, question)
      results.push(result)
    }
    return results
  }

  /**
   * 分析最新帧并返回结果
   * 便捷方法，适用于 vision_perceive 工具
   */
  async analyzeLatest(
    getLatestFrame: () => ExtractedFrame | null,
    question?: string
  ): Promise<AnalysisResult | null> {
    const frame = getLatestFrame()
    if (!frame) return null
    return this.analyze(frame, question)
  }

  /** 调用 AI 服务（带超时保护） */
  private callAIWithTimeout(imageDataUrl: string, prompt: string): Promise<string> {
    return new Promise<string>((resolve, reject) => {
      const timer = setTimeout(() => {
        reject(new Error(`AI 分析超时（${this.timeoutMs}ms）`))
      }, this.timeoutMs)

      this.ai.analyzeImage(imageDataUrl, prompt)
        .then((result) => {
          clearTimeout(timer)
          resolve(result)
        })
        .catch((err) => {
          clearTimeout(timer)
          reject(err)
        })
    })
  }

  /** 提取 UI 元素信息（二次 AI 调用） */
  private async extractUIElementsFromFrame(frame: ExtractedFrame): Promise<UIElementInfo[]> {
    try {
      const prompt = '请识别屏幕上的所有可交互 UI 元素（按钮、输入框、链接、菜单等），以 JSON 数组格式返回，每个元素包含 type、label、confidence 字段。'
      const raw = await this.callAIWithTimeout(frame.dataUrl, prompt)

      // 尝试解析 JSON，失败时返回空数组
      const parsed = JSON.parse(raw)
      if (Array.isArray(parsed)) {
        return parsed.filter((item: any) =>
          item && typeof item.type === 'string' && typeof item.confidence === 'number'
        )
      }
      return []
    } catch {
      return []
    }
  }

  /** 获取并发槽位（信号量模式） */
  private acquireSlot(): Promise<void> {
    if (this.activeRequests < this.maxConcurrency) {
      this.activeRequests++
      return Promise.resolve()
    }

    return new Promise<void>((resolve) => {
      this.pendingQueue.push(() => {
        this.activeRequests++
        resolve()
      })
    })
  }

  /** 释放并发槽位 */
  private releaseSlot(): void {
    this.activeRequests--
    if (this.pendingQueue.length > 0) {
      const next = this.pendingQueue.shift()!
      next()
    }
  }

  /** 当前活跃请求数 */
  get activeCount(): number {
    return this.activeRequests
  }

  /** 等待队列长度 */
  get queueLength(): number {
    return this.pendingQueue.length
  }

  /** 销毁 */
  dispose(): void {
    this.pendingQueue = []
    this.onResult = null
  }
}
```

```typescript
import type { Module, ModuleContext, Capability } from '../../src/main/types/module'
import { ScreenCaptureService } from './services/screen-capture-service'
import { ChangeDetector } from './services/change-detector'
import { FrameExtractor } from './services/frame-extractor'
import { VisionAnalyzer } from './services/vision-analyzer'
import { PrivacyProtector, type PrivacyLevel } from './services/privacy-protector'

export default class VisionModule implements Module {
  id = 'vision'
  meta!: import('../../src/main/types/module').ModuleMeta

  private screenCapture: ScreenCaptureService
  private changeDetector: ChangeDetector
  private frameExtractor!: FrameExtractor
  private visionAnalyzer!: VisionAnalyzer
  private privacyProtector!: PrivacyProtector
  private context!: ModuleContext
  private mode: 'off' | 'ondemand' | 'continuous' = 'off'
  private fps: number = 1

  constructor() {
    // PrivacyProtector 将在 activate() 中根据配置初始化
    this.screenCapture = new ScreenCaptureService()
    this.changeDetector = new ChangeDetector(0.02)
  }

  async activate(context: ModuleContext): Promise<void> {
    this.context = context

    // 从配置读取 fps（默认 1），优先级：config > store > 默认值
    this.fps = context.config.getSync?.('vision.fps', 1) ?? 1

    // 初始化隐私保护器（从配置读取隐私级别和选项）
    const privacyLevel: PrivacyLevel = context.config.getSync?.('vision.privacy.level', 'basic') ?? 'basic'
    const privacyConfig = context.config.getSync?.('vision.privacy', {}) ?? {}
    this.privacyProtector = new PrivacyProtector({
      level: privacyLevel,
      ...privacyConfig
    })

    // 隐私事件监听：记录敏感内容检测日志
    this.privacyProtector.onSensitiveDetected = (result) => {
      context.logger.warn('检测到敏感内容', {
        sensitiveWindowTitle: result.sensitiveWindowTitle,
        regionCount: result.regions.length,
        blocked: result.shouldBlock
      })
    }

    // 将 PrivacyProtector 注入 ScreenCaptureService
    this.screenCapture.setPrivacyProtector(this.privacyProtector)

    // 初始化帧提取器（使用 fps 配置）
    this.frameExtractor = new FrameExtractor(
      this.screenCapture,
      this.changeDetector,
      {
        fps: this.fps,
        changeDetectionEnabled: true,
        maxBufferSize: 10
      }
    )

    // 初始化视觉分析器（传入隐私保护器，确保发送至 AI 前脱敏）
    this.visionAnalyzer = new VisionAnalyzer(
      { analyzeImage: (img, prompt) => context.ai.analyzeImage(img, prompt) },
      {
        extractUIElements: false,
        maxConcurrency: 1,
        timeoutMs: 10000,
        privacyProtector: this.privacyProtector
      }
    )

    // 帧提取完成 → 自动触发分析（持续模式）
    this.frameExtractor.onFrame = async (frame) => {
      if (this.mode === 'continuous') {
        try {
          const result = await this.visionAnalyzer.analyze(frame)
          context.eventBus.emit('on_screen_change', {
            analysis: result.description,
            timestamp: frame.timestamp
          })
        } catch (err) {
          context.logger.warn('连续模式帧分析失败', { error: err })
        }
      }
    }

    // 帧提取错误回调
    this.frameExtractor.onError = (error) => {
      context.logger.warn('帧提取错误', { error: error.message })
    }

    // 监听开关事件
    // ✅ Issue#305: 已修复 — mode 使用类型守卫验证，避免 as any 类型断言
    const VALID_MODES = new Set(['off', 'ondemand', 'continuous'] as const)
    type VisionMode = 'off' | 'ondemand' | 'continuous'
    const isValidMode = (m: string): m is VisionMode => VALID_MODES.has(m as VisionMode)

    context.eventBus.on('on_vision_toggle', async (data: { enabled: boolean; mode?: string }) => {
      if (data.enabled) {
        const resolvedMode: VisionMode = data.mode && isValidMode(data.mode) ? data.mode : 'ondemand'
        await this.startCapture(resolvedMode)
      } else {
        this.stopCapture()
      }
    })

    // 监听 fps 变化事件（来自 UI store 的 setFps 调用）
    context.eventBus.on('on_vision_fps_change', (data: { fps: number }) => {
      this.setFps(data.fps)
    })

    // 监听感知请求事件：按需触发屏幕分析并返回结果
    context.eventBus.on('on_perception_request', async (data: { question?: string }) => {
      try {
        if (!this.screenCapture.enabled) {
          await this.screenCapture.enable({ width: 1280, height: 720 })
        }

        const frame = await this.frameExtractor.extractOnce()
        if (!frame) {
          context.eventBus.emit('on_perception_result', { error: '无法捕获屏幕画面' })
          return
        }

        const result = await this.visionAnalyzer.analyze(frame, data.question)
        context.eventBus.emit('on_perception_result', { analysis: result.description })
      } catch (err) {
        context.logger.warn('感知请求处理失败', { error: err })
        context.eventBus.emit('on_perception_result', { error: String(err) })
      }
    })

    // 恢复上次状态
    const savedToggle = await context.config.get('vision.enabled', false)
    if (savedToggle) {
      await this.startCapture('continuous')
    }

    context.logger.info('视觉感知模块已激活', { fps: this.fps })
  }

  /**
   * 动态更新 fps
   * 连续模式运行中可热更新捕获频率
   */
  private setFps(fps: number): void {
    const clamped = Math.max(0.5, Math.min(30, fps))
    this.fps = clamped
    this.frameExtractor.setFps(clamped)
    this.context.config.set?.('vision.fps', clamped)
    this.context.logger.info('视觉感知 fps 已更新', { fps: clamped })
  }

  private async startCapture(mode: 'off' | 'ondemand' | 'continuous'): Promise<void> {
    const result = await this.screenCapture.enable({ width: 1280, height: 720, frameRate: this.fps })
    if (!result.success) {
      this.context.logger.error('屏幕捕获启动失败', undefined, { error: result.error })
      return
    }

    this.mode = mode

    // 启动帧提取（持续模式下使用定时器，按需模式仅在请求时提取）
    if (this.mode === 'continuous') {
      this.frameExtractor.start()
      this.context.logger.info('连续模式已启动', { fps: this.fps, intervalMs: Math.round(1000 / this.fps) })
    }

    this.context.eventBus.emit('on_vision_started', { mode, fps: this.fps })
  }

  private stopCapture(): void {
    const wasRunning = this.frameExtractor.running
    this.frameExtractor.stop()
    this.frameExtractor.clearBuffer()
    this.screenCapture.disable()
    this.mode = 'off'

    if (wasRunning) {
      this.context.logger.info('连续模式已停止', {
        extractionCount: this.frameExtractor.extractionCount,
        errorCount: this.frameExtractor.errorCount
      })
    }

    this.context.eventBus.emit('on_vision_stopped')
  }

  async deactivate(): Promise<void> {
    this.stopCapture()
    this.frameExtractor.dispose()
    this.visionAnalyzer.dispose()
    this.changeDetector.dispose()   // 释放 canvas/ctx 引用，帮助 GC 回收
    this.context.logger.info('视觉感知模块已停用')
  }

  getCapabilities(): Capability[] {
    return [
      {
        type: 'tool',
        name: 'vision_perceive',
        description: '按需分析屏幕内容',
        priority: 10,
        moduleId: this.id,
        handler: {
          execute: async (params) => {
            // 按需模式：先确保捕获已启用
            if (!this.screenCapture.enabled) {
              await this.screenCapture.enable({ width: 1280, height: 720 })
            }

            // 提取单帧并分析
            const frame = await this.frameExtractor.extractOnce()
            if (!frame) return { error: '无法捕获屏幕画面' }

            const question = (params as { question?: string }).question
            const result = await this.visionAnalyzer.analyze(frame, question)

            // 按需分析完成后 emit on_screen_change 事件
            context.eventBus.emit('on_screen_change', {
              analysis: result.description,
              timestamp: frame.timestamp
            })

            return {
              analysis: result.description,
              sequence: result.sequence,
              durationMs: result.durationMs,
              sensitiveContentWarning: result.sensitiveContentWarning
            }
          }
        }
      },
      {
        type: 'tool',
        name: 'vision_toggle',
        description: '开启/关闭屏幕感知，切换视觉模式（off/ondemand/continuous）',
        priority: 10,
        moduleId: this.id,
        handler: {
          execute: async (params) => {
            const { enabled, mode: newMode } = params as { enabled: boolean; mode?: 'off' | 'ondemand' | 'continuous' }
            const resolvedMode = enabled ? (newMode || 'ondemand') : 'off'

            if (enabled) {
              await this.startCapture(newMode || 'ondemand')
            } else {
              this.stopCapture()
            }

            // 同步状态到 useVisionStore（通过事件总线通知渲染进程）
            this.context.eventBus.emit('on_vision_toggle', {
              enabled: this.mode !== 'off',
              mode: this.mode
            })

            // 持久化开关状态
            await this.context.config.set?.('vision.enabled', this.mode !== 'off')

            return { enabled: this.mode !== 'off', mode: this.mode }
          }
        }
      }
    ]
  }
}
```

### 19.3 代码规范

- **开关控制**：默认关闭，不消耗资源
- **帧率控制**：`frameRate: { ideal: 2 }`，不需要 30fps
- **变化检测**：<!-- ✅ Issue#303: 已修复 --> 网格分层采样（gridDensity×gridDensity 均匀覆盖），多级阈值（minor/major/significant），亮度补偿排除屏幕调光干扰，运动区域检测辅助聚焦分析，无变化跳过 AI 推理
- **资源清理**：关闭开关时立即停止流、销毁 canvas
- **权限处理**：首次使用需系统授权
